# 本章小结

1. **信号干扰**：从压制式到欺骗式，从窄带到宽带，干扰攻击是最常见的卫星通信威胁
2. **窃听威胁**：卫星广播特性使得窃听天然可行，加密是主要防护手段
3. **调制解调安全**：调制识别是攻击的前提，安全调制技术可以增加攻击难度
4. **加密与密钥管理**：多层加密体系是卫星通信安全的基础，密钥管理是关键挑战
5. **链路加密**：上行、下行、星间链路各有不同的加密需求和方案
6. **抗干扰技术**：扩频、跳频、调零天线、FEC 等多层抗干扰措施
7. **信号分析与侦察**：SDR 和数字信号处理技术降低了信号分析的门槛

#### 4. 卫星信号识别与分析流程

```
卫星信号识别分析工作流：

1. 信号检测（Detection）
   │
   ├── 宽频谱扫描 → 发现未知信号
   ├── 已知频率列表比对 → 确认目标信号
   └── 长期监测 → 发现信号规律（如卫星过境）

2. 参数测量（Parameter Measurement）
   │
   ├── 中心频率 → 使用频谱仪/inspectrum
   ├── 信号带宽 → 测量-3dB带宽
   ├── 符号率 → 观察频谱主瓣宽度/循环自相关
   ├── 调制方式 → 星座图/眼图分析
   └── 极化方式 → 旋转天线测量信号强度变化

3. 调制识别（Modulation Classification）
   │
   ├── 模拟调制：AM / FM / PM
   ├── 数字调制：ASK / FSK / BPSK / QPSK / QAM
   ├── 扩频信号：DSSS（直序扩频）/ FHSS（跳频）
   └── OFDM信号：子载波数量、循环前缀

4. 信号解调（Demodulation）
   │
   ├── 载波恢复（Costas环、PLL）
   ├── 定时恢复（Gardner算法、MM算法）
   ├── 均衡（自适应均衡器）
   └── 判决输出比特流

5. 协议分析（Protocol Analysis）
   │
   ├── 帧同步（查找前导码/同步字）
   ├── 帧结构分析（帧长、字段划分）
   ├── 编码识别（卷积码、Turbo码、LDPC）
   ├── 加扰识别（扰码多项式）
   └── 数据解析（遥测、通信数据）

6. 内容解读（Content Decoding）
   │
   ├── 遥测数据解析（传感器读数、状态）
   ├── 通信数据解析（语音、数据、图像）
   └── 信令分析（控制指令、状态查询）
```

### 卫星信号侦察与分析 Python 脚本

```python
#!/usr/bin/env python3
"""
卫星信号频谱监测与分析脚本
功能：频谱扫描、信号检测、记录
"""

import numpy as np
import matplotlib.pyplot as plt
from rtlsdr import RtlSdr
import time
import csv
from datetime import datetime

class SatelliteSpectrumAnalyzer:
    def __init__(self, sample_rate=2.048e6, gain=40, ppm=0):
        self.sdr = RtlSdr()
        self.sdr.sample_rate = sample_rate
        self.sdr.gain = gain
        self.sdr.freq_correction = ppm
        
    def scan_frequency_range(self, start_freq, end_freq, step_freq):
        """扫描指定频率范围，返回各频点功率"""
        freqs = np.arange(start_freq, end_freq, step_freq)
        powers = []
        
        print(f"[*] 扫描范围: {start_freq/1e6:.2f} MHz ~ {end_freq/1e6:.2f} MHz")
        print(f"[*] 步进: {step_freq/1e3:.1f} kHz, 共 {len(freqs)} 个频点")
        
        for i, freq in enumerate(freqs):
            self.sdr.center_freq = freq
            time.sleep(0.01)  # 等待PLL稳定
            
            # 读取样本
            samples = self.sdr.read_samples(256*1024)
            
            # 计算PSD（功率谱密度）
            psd = np.abs(np.fft.fftshift(np.fft.fft(samples)))**2
            psd_db = 10 * np.log10(psd)
            
            # 取中心区域的平均功率（避免滤波器边缘）
            center_power = np.mean(psd_db[len(psd_db)//4 : 3*len(psd_db)//4])
            powers.append(center_power)
            
            if i % 10 == 0:
                print(f"  进度: {i}/{len(freqs)} ({100*i/len(freqs):.1f}%)")
        
        return freqs, np.array(powers)
    
    def detect_signals(self, freqs, powers, threshold_db=10):
        """检测高于噪声基底的信号"""
        # 估计噪声基底（中位数）
        noise_floor = np.median(powers)
        threshold = noise_floor + threshold_db
        
        # 查找超过阈值的频点
        signal_indices = np.where(powers > threshold)[0]
        
        # 分组相邻频点为一个信号
        signals = []
        if len(signal_indices) > 0:
            current_group = [signal_indices[0]]
            for i in range(1, len(signal_indices)):
                if signal_indices[i] == signal_indices[i-1] + 1:
                    current_group.append(signal_indices[i])
                else:
                    # 记录一个信号
                    start_idx = current_group[0]
                    end_idx = current_group[-1]
                    peak_idx = start_idx + np.argmax(powers[start_idx:end_idx+1])
                    signals.append({
                        'freq_mhz': freqs[peak_idx] / 1e6,
                        'bandwidth_khz': (freqs[end_idx] - freqs[start_idx]) / 1e3,
                        'power_db': powers[peak_idx],
                    })
                    current_group = [signal_indices[i]]
            
            # 最后一组
            if current_group:
                start_idx = current_group[0]
                end_idx = current_group[-1]
                peak_idx = start_idx + np.argmax(powers[start_idx:end_idx+1])
                signals.append({
                    'freq_mhz': freqs[peak_idx] / 1e6,
                    'bandwidth_khz': (freqs[end_idx] - freqs[start_idx]) / 1e3,
                    'power_db': powers[peak_idx],
                })
        
        return signals, noise_floor
    
    def plot_spectrum(self, freqs, powers, signals, title="Spectrum Analyzer"):
        """绘制频谱图"""
        plt.figure(figsize=(12, 6))
        plt.plot(freqs/1e6, powers, 'b-', linewidth=0.5)
        
        # 标记检测到的信号
        for sig in signals:
            plt.axvline(x=sig['freq_mhz'], color='r', linestyle='--', alpha=0.5)
            plt.text(sig['freq_mhz'], sig['power_db'] + 2, 
                    f"{sig['freq_mhz']:.3f} MHz", 
                    ha='center', fontsize=8, color='red')
        
        plt.xlabel('Frequency (MHz)')
        plt.ylabel('Power (dB)')
        plt.title(title)
        plt.grid(True, alpha=0.3)
        plt.tight_layout()
        plt.savefig('spectrum_analysis.png', dpi=150)
        print(f"[*] 频谱图已保存为 spectrum_analysis.png")
        plt.close()
    
    def save_scan_result(self, freqs, powers, filename):
        """保存扫描结果到CSV"""
        with open(filename, 'w', newline='') as f:
            writer = csv.writer(f)
            writer.writerow(['Frequency (Hz)', 'Power (dB)'])
            for freq, power in zip(freqs, powers):
                writer.writerow([freq, power])
        print(f"[*] 扫描结果已保存到 {filename}")
    
    def close(self):
        self.sdr.close()

# 使用示例
if __name__ == '__main__':
    analyzer = SatelliteSpectrumAnalyzer(gain=40)
    
    try:
        # 扫描 L 波段（GPS/伽利略/北斗）
        freqs, powers = analyzer.scan_frequency_range(
            start_freq=1550e6,
            end_freq=1610e6,
            step_freq=200e3
        )
        
        # 检测信号
        signals, noise_floor = analyzer.detect_signals(freqs, powers, threshold_db=6)
        
        print(f"\n[*] 噪声基底: {noise_floor:.2f} dB")
        print(f"[*] 检测到 {len(signals)} 个信号:")
        for i, sig in enumerate(signals, 1):
            print(f"  信号{i}: {sig['freq_mhz']:.3f} MHz, "
                  f"带宽: {sig['bandwidth_khz']:.1f} kHz, "
                  f"功率: {sig['power_db']:.2f} dB")
        
        # 绘图
        analyzer.plot_spectrum(freqs, powers, signals, 
                              "L-Band Spectrum (1550-1610 MHz)")
        
        # 保存结果
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        analyzer.save_scan_result(freqs, powers, f"spectrum_scan_{timestamp}.csv")
        
    finally:
        analyzer.close()
```
