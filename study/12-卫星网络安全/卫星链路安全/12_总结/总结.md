# 总结

卫星链路安全是整个卫星系统中最具挑战性的安全领域。由于卫星链路的开放性（上行链路任何人都能发射，下行链路任何人都能接收），安全防护必须从物理层到应用层构建完整的防护体系。

关键防护原则：
1. **上行认证** - 所有上行指令必须经过加密认证
2. **下行加密** - 所有下行数据必须加密保护
3. **扩频抗干扰** - 使用DSSS/FHSS增强抗干扰能力
4. **防重放** - 序列号+时间戳防止指令重放
5. **密钥管理** - 多层次密钥体系，定期更新
6. **实时监测** - 射频、加密、流量多维监测
7. **自适应防御** - 根据威胁等级动态调整防护策略

```python
#!/usr/bin/env python3
"""
卫星链路预算计算工具
功能：计算上下行链路的链路余量、误码率等参数
"""

import math

class SatelliteLinkBudget:
    def __init__(self):
        # 物理常数
        self.c = 3e8  # 光速 m/s
        self.k = 1.38e-23  # 玻尔兹曼常数 J/K
        
    def free_space_loss(self, frequency_hz, distance_m):
        """计算自由空间损耗 (dB)"""
        wavelength = self.c / frequency_hz
        fspl = (4 * math.pi * distance_m / wavelength) ** 2
        return 10 * math.log10(fspl)
    
    def eirp(self, tx_power_dbm, tx_gain_db):
        """计算等效全向辐射功率 EIRP (dBm)"""
        return tx_power_dbm + tx_gain_db
    
    def received_power(self, eirp_dbm, fspl_db, rx_gain_db, other_loss_db=0):
        """计算接收功率 (dBm)"""
        return eirp_dbm - fspl_db + rx_gain_db - other_loss_db
    
    def noise_power(self, bandwidth_hz, noise_temp_k):
        """计算噪声功率 (dBm)"""
        noise_watts = self.k * noise_temp_k * bandwidth_hz
        return 10 * math.log10(noise_watts) + 30  # 转换为dBm
    
    def snr(self, rx_power_dbm, noise_power_dbm):
        """计算信噪比 SNR (dB)"""
        return rx_power_dbm - noise_power_dbm
    
    def link_margin(self, snr_db, required_snr_db):
        """计算链路余量 (dB)"""
        return snr_db - required_snr_db
    
    def calculate_uplink(self, params):
        """计算上行链路预算
        
        params: dict, 包含以下参数：
            - frequency: 上行频率 (Hz)
            - distance: 卫星距离 (m)
            - tx_power: 地面站发射功率 (dBm)
            - tx_gain: 地面站天线增益 (dBi)
            - rx_gain: 卫星天线增益 (dBi)
            - bandwidth: 信号带宽 (Hz)
            - noise_temp: 卫星接收机噪声温度 (K)
            - other_loss: 其他损耗 (dB)
            - required_snr: 所需信噪比 (dB)
        """
        f = params['frequency']
        d = params['distance']
        
        # 自由空间损耗
        fspl = self.free_space_loss(f, d)
        
        # EIRP
        eirp = self.eirp(params['tx_power'], params['tx_gain'])
        
        # 接收功率
        rx_power = self.received_power(
            eirp, fspl, params['rx_gain'], 
            params.get('other_loss', 0)
        )
        
        # 噪声功率
        noise = self.noise_power(params['bandwidth'], params['noise_temp'])
        
        # 信噪比
        snr = self.snr(rx_power, noise)
        
        # 链路余量
        margin = self.link_margin(snr, params['required_snr'])
        
        return {
            'frequency_hz': f,
            'distance_km': d / 1000,
            'fspl_db': fspl,
            'eirp_dbm': eirp,
            'rx_power_dbm': rx_power,
            'noise_power_dbm': noise,
            'snr_db': snr,
            'link_margin_db': margin,
            'status': 'OK' if margin > 0 else 'MARGIN_FAIL'
        }
    
    def calculate_downlink(self, params):
        """计算下行链路预算（参数类似上行）"""
        return self.calculate_uplink(params)  # 计算公式相同
    
    def print_report(self, link_name, result):
        """打印链路预算报告"""
        print(f"\n{'='*50}")
        print(f"  {link_name} 链路预算报告")
        print(f"{'='*50}")
        print(f"  频率:         {result['frequency_hz']/1e6:.2f} MHz")
        print(f"  距离:         {result['distance_km']:.2f} km")
        print(f"  自由空间损耗:  {result['fspl_db']:.2f} dB")
        print(f"  EIRP:         {result['eirp_dbm']:.2f} dBm")
        print(f"  接收功率:      {result['rx_power_dbm']:.2f} dBm")
        print(f"  噪声功率:      {result['noise_power_dbm']:.2f} dBm")
        print(f"  信噪比:        {result['snr_db']:.2f} dB")
        print(f"  链路余量:      {result['link_margin_db']:.2f} dB")
        print(f"  状态:          {'✓ 充足' if result['link_margin_db'] > 3 else '✗ 不足'}")
        print(f"{'='*50}")

# 使用示例
if __name__ == '__main__':
    calc = SatelliteLinkBudget()
    
    # GEO卫星上行链路示例（Ku波段）
    uplink_params = {
        'frequency': 14.25e9,       # 14.25 GHz (Ku波段上行)
        'distance': 35786e3,        # GEO轨道高度 ~35,786 km

    }
    
    uplink_result = calc.calculate_uplink(uplink_params)
    calc.print_report("GEO 上行 (Ku波段)", uplink_result)
    
    # GEO卫星下行链路示例

        'tx_gain': 35,              # 卫星天线增益 35dBi

```

协议标准        用途                应用场景
