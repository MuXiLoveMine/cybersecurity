# 总结

卫星地面站是整个卫星系统中安全防护的核心节点。其安全不仅关系到卫星资产的安全，更关系到依赖卫星服务的关键基础设施的稳定运行。

地面站安全防护需要从网络、射频、物理、供应链等多个维度构建纵深防御体系。关键原则包括：网络分段与隔离、严格的身份认证与访问控制、端到端的加密与签名、实时安全监控与告警、以及完善的应急响应机制。

Viasat KA-SAT攻击事件等真实案例提醒我们，地面站安全不是理论问题而是现实威胁。建议结合本文中的架构分析和安全配置示例，在实际地面站环境中进行安全评估和加固。

```bash
#!/usr/bin/env python3
"""
地面站射频频谱监测系统
监控指定频段的信号质量，异常告警
"""

import numpy as np
from rtlsdr import RtlSdr
import time
import smtplib
from email.mime.text import MIMEText

class RFSpectrumMonitor:
    def __init__(self, center_freq, bandwidth, threshold_db):
        self.sdr = RtlSdr()
        self.sdr.sample_rate = 2.048e6
        self.sdr.center_freq = center_freq
        self.sdr.gain = 'auto'
        self.center_freq = center_freq
        self.bandwidth = bandwidth
        self.threshold_db = threshold_db
        self.baseline_power = None
        self.alert_sent = False
    
    def measure_power(self):
        """测量当前频段平均功率"""
        samples = self.sdr.read_samples(1024*1024)
        psd = np.abs(np.fft.fftshift(np.fft.fft(samples)))**2
        psd_db = 10 * np.log10(psd)
        # 取中心频段的平均功率
        center = len(psd_db) // 2
        bw_samples = int(len(psd_db) * self.bandwidth / self.sdr.sample_rate)
        band_power = np.mean(psd_db[center-bw_samples//2:center+bw_samples//2])
        return band_power
    
    def calibrate_baseline(self, duration=60):
        """校准基线功率"""
        print("[*] 正在校准基线功率...")
        powers = []
        start = time.time()
        while time.time() - start < duration:
            powers.append(self.measure_power())
            time.sleep(1)
        self.baseline_power = np.mean(powers)
        print(f"[*] 基线功率: {self.baseline_power:.2f} dB")
        print(f"[*] 告警阈值: {self.baseline_power + self.threshold_db:.2f} dB")
    
    def monitor(self):
        """持续监测"""
        print("[*] 开始频谱监测...")
        while True:
            current_power = self.measure_power()
            
            # 检测异常功率升高（可能是干扰）
            if self.baseline_power and current_power > self.baseline_power + self.threshold_db:
                if not self.alert_sent:
                    print(f"[!] 告警：信号功率异常升高！")
                    print(f"    当前功率: {current_power:.2f} dB")
                    print(f"    基线功率: {self.baseline_power:.2f} dB")
                    print(f"    差值: {current_power - self.baseline_power:.2f} dB")
                    self.send_alert(current_power)
                    self.alert_sent = True
            else:
                self.alert_sent = False
            
            time.sleep(5)
```

```bash
echo "[4] 防火墙状态"
echo "------------------------------------------"
if command -v ufw &> /dev/null; then
    echo "UFW状态: $(ufw status | head -1)"
elif command -v iptables &> /dev/null; then
    echo "iptables规则数: $(iptables -L -n | grep -c '^[A-Z]')"
    echo "INPUT策略: $(iptables -L INPUT -n | head -1 | grep -oP 'policy \K\w+')"
fi
echo ""

# 5. 开放端口
echo "[5] 开放端口检查"
echo "------------------------------------------"
echo "监听的TCP端口:"
ss -tlnp | grep -v State | awk '{print "  " $4 " - " $7}' | head -20
echo ""
echo "监听的UDP端口:"
ss -ulnp | grep -v State | awk '{print "  " $4 " - " $7}' | head -10
echo ""

# 6. 文件系统安全
echo "[6] 文件系统安全"
echo "------------------------------------------"
echo "SUID文件数量: $(find / -perm -4000 -type f 2>/dev/null | wc -l)"
echo "SGID文件数量: $(find / -perm -2000 -type f 2>/dev/null | wc -l)"
echo "/tmp目录权限: $(stat -c %a /tmp)"
echo ""

# 7. 审计日志
echo "[7] 审计日志状态"
echo "------------------------------------------"
echo "auditd状态: $(systemctl is-active auditd 2>/dev/null || echo '未安装')"
echo "syslog状态: $(systemctl is-active rsyslog 2>/dev/null || echo '未安装')"
echo "日志文件列表（最近修改）:"
find /var/log -name "*.log" -type f -mtime -7 | head -10 | while read f; do
    echo "  $f ($(stat -c %s $f) bytes)"
done
echo ""

# 8. 数据库安全（如果有）
echo "[8] 数据库安全检查"
echo "------------------------------------------"
if command -v psql &> /dev/null; then
    echo "PostgreSQL版本: $(psql --version)"
    echo "服务状态: $(systemctl is-active postgresql 2>/dev/null || echo '未运行')"
fi
echo ""

echo "=========================================="
echo "  审计完成"
echo "=========================================="
```
