### 6.2 指令安全系统

```python
# 卫星指令安全系统
class SatelliteCommandSecurity:
    """卫星指令安全控制"""

    # 指令安全等级
    COMMAND_LEVELS = {
        1: {'name': '常规遥测请求', 'auth': 'single', 'audit': True},
        2: {'name': '配置更新', 'auth': 'single', 'audit': True},
        3: {'name': '姿态控制', 'auth': 'dual', 'audit': True},
        4: {'name': '轨道机动', 'auth': 'dual+review', 'audit': True},
        5: {'name': '应急指令', 'auth': 'dual+override', 'audit': True},
        6: {'name': '自毁/退役', 'auth': 'triple+review', 'audit': True},
    }

    def __init__(self):
        self.command_queue = []
        self.pending_authorizations = {}
        self.executed_commands = []
        self.command_counter = 0

    def submit_command(self, command: dict,
                       operator_id: str) -> dict:
        """提交指令"""
        cmd_level = command.get('security_level', 1)
        auth_req = self.COMMAND_LEVELS[cmd_level]

        # 创建待授权指令
        pending_cmd = {
            'id': f'CMD-{self.command_counter:06d}',
            'command': command,
            'submitter': operator_id,
            'submit_time': datetime.now().isoformat(),
            'security_level': cmd_level,
            'auth_required': auth_req['auth'],
            'authorizations': [],
            'status': 'pending',
        }

        self.command_counter += 1
        self.pending_authorizations[pending_cmd['id']] = pending_cmd

        logger.info(f"指令提交: {pending_cmd['id']} "
                    f"(级别: {cmd_level}, "
                    f"操作员: {operator_id})")

        return pending_cmd

    def authorize_command(self, cmd_id: str,
                          authorizer_id: str) -> dict:
        """授权指令"""
        if cmd_id not in self.pending_authorizations:
            return {'error': '指令不存在'}

        cmd = self.pending_authorizations[cmd_id]

        # 检查授权者不是提交者
        if authorizer_id == cmd['submitter']:
            return {'error': '不能自授权'}

        # 记录授权
        cmd['authorizations'].append({
            'authorizer': authorizer_id,
            'time': datetime.now().isoformat(),
        })

        # 检查是否满足授权要求
        auth_type = cmd['auth_required']
        auth_count = len(cmd['authorizations'])

        if auth_type == 'single' and auth_count >= 1:
            return self._execute_command(cmd)
        elif auth_type == 'dual' and auth_count >= 2:
            return self._execute_command(cmd)
        elif auth_type == 'dual+review' and auth_count >= 2:
            # 需要额外审查
            return {'status': 'needs_review', 'command': cmd}
        elif auth_type == 'triple+review' and auth_count >= 3:
            return {'status': 'needs_review', 'command': cmd}

        return {
            'status': 'pending',
            'authorizations': auth_count,
            'required': auth_type,
        }

    def _execute_command(self, cmd: dict) -> dict:
        """执行指令"""
        # 加密签名
        signed_cmd = self._sign_command(cmd)

        # 防回滚检查
        if self._check_rollback(cmd):
            return {'error': '指令版本回滚被拒绝'}

        # 发送指令
        cmd['status'] = 'executed'
        cmd['execute_time'] = datetime.now().isoformat()
        self.executed_commands.append(cmd)

        logger.info(f"指令执行: {cmd['id']}")

        return {'status': 'executed', 'command_id': cmd['id']}

    def _sign_command(self, cmd: dict) -> bytes:
        """对指令进行数字签名"""
        # 使用ECDSA-P256签名
        # 实际实现使用HSM
        cmd_data = json.dumps(cmd['command'], sort_keys=True).encode()
        # signature = hsm.sign(cmd_data)
        return cmd_data  # 简化

    def _check_rollback(self, cmd: dict) -> bool:
        """检查防回滚"""
        # 检查指令序列号是否大于上次执行的
        return False
```
