class GroundStationSecurityZone:

            'trust_level': 0,
            'allowed_protocols': ['HTTPS', 'VPN'],
            'monitoring': 'full IDS/IPS + DDoS防护',
            'access_control': '默认拒绝所有入站',
            'logging': '完整审计日志',
        },
        'dmz': {
            'name': 'DMZ区域',
            'trust_level': 1,
            'allowed_protocols': ['HTTPS', 'SMTP', 'DNS'],
            'monitoring': '应用层检测 + WAF',
            'access_control': '仅允许特定服务端口',
            'logging': '完整审计日志',
        },
        'operations': {
            'name': '操作区域(OT网络)',
            'trust_level': 3,
            'allowed_protocols': ['SSH', 'SNMPv3', '内部协议'],
            'monitoring': '深度包检测 + 行为分析',
            'access_control': '白名单访问 + 双因素认证',
            'logging': '实时告警 + 审计',
        },
        'critical': {
            'name': '关键控制区域',
            'trust_level': 5,
            'allowed_protocols': ['自定义协议', '专用加密通道'],
            'monitoring': '深度包检测 + 实时异常检测',
            'access_control': '严格白名单 + 物理访问控制',
            'logging': '实时告警 + 完整记录',
        },
    }

    INTER_ZONE_RULES = {
        ('external', 'dmz'): {
            'allowed': [443, 25, 53, 1194],  # HTTPS, SMTP, DNS, VPN
            'action': 'ALLOW + IDS',
        },
        ('external', 'operations'): {
            'allowed': [],
            'action': 'DENY ALL',
        },
        ('dmz', 'operations'): {
            'allowed': [22, 443],  # SSH, HTTPS
            'action': 'ALLOW + APPLICATION_FILTER',
        },
        ('operations', 'critical'): {
            'allowed': [],  # 通过单向数据二极管
            'action': 'ONE_WAY_DIODE',
            'note': '关键区域使用单向数据流，防止反向入侵',
        },
    }
