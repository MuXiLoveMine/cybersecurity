### 11.2 合规自动化流水线

```yaml
# CI/CD 合规检查流水线示例
stages:
  - iac_scan        # IaC安全扫描
  - policy_check     # 策略合规检查
  - deploy          # 部署
  - post_audit      # 部署后审计

iac_scan:
  stage: iac_scan
  script:
    # Checkov扫描Terraform
    - checkov -d terraform/ --framework terraform
    # Trivy扫描
    - trivy config terraform/
    # 输出合规报告
    - checkov -d terraform/ --output json --report

policy_check:
  stage: policy_check
  script:
    # OPA策略检查
    - opa eval -d policies/ -i terraform/plan.json "data.terraform.deny"
    # 等保基线检查
    - inspec exec compliance/ --input-file attributes.yml

post_audit:
  stage: post_audit
  script:
    # Prowler合规扫描
    - prowler aws --compliance cis_2.0_aws
    # 生成等保对照报告
    - python3 generate_compliance_report.py
    # 推送至GRC平台
    - curl -X POST https://grc.internal/api/v1/evidence -H "Authorization: Bearer $TOKEN" -d @report.json
```

#### 4.2 GRC平台数据导入模板

```python
#!/usr/bin/env python3
"""
GRC平台数据批量导入工具
支持资产、风险、合规数据的批量导入
"""
import csv
import json
import requests
from datetime import datetime

class GRCDataImporter:
    def __init__(self, grc_url, api_key):
        self.grc_url = grc_url.rstrip('/')
        self.api_key = api_key
        self.headers = {
            'Authorization': f'Bearer {api_key}',
            'Content-Type': 'application/json'
        }
    
    def import_assets(self, csv_file):
        """
        从CSV批量导入资产
        CSV格式：名称,类型,描述,价值,所有者,位置,状态
        """
        assets = []
        with open(csv_file, 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            for row in reader:
                asset = {
                    "name": row['名称'],
                    "asset_type": row['类型'],
                    "description": row['描述'],
                    "value": int(row['价值']) if row['价值'] else 1,
                    "owner": row['所有者'],
                    "location": row['位置'],
                    "status": row.get('状态', 'active')
                }
                assets.append(asset)
                
                # 调用API创建资产（示例）
                # response = requests.post(
                #     f"{self.grc_url}/api/assets",
                #     headers=self.headers,
                #     json=asset
                # )
        
        print(f"共读取 {len(assets)} 条资产记录")
        return assets
    
    def import_risks(self, csv_file):
        """
        从CSV批量导入风险
        CSV格式：风险名称,资产,威胁,脆弱性,可能性,影响,现有措施,责任人,状态
        """
        risks = []
        with open(csv_file, 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            for row in reader:
                likelihood = int(row['可能性'])
                impact = int(row['影响'])
                risk_value = likelihood * impact
                
                # 风险等级判定
                if risk_value >= 20:
                    risk_level = '极高'
                elif risk_value >= 13:
                    risk_level = '高'
                elif risk_value >= 6:
                    risk_level = '中'
                else:
                    risk_level = '低'
                
                risk = {
                    "name": row['风险名称'],
                    "asset": row['资产'],
                    "threat": row['威胁'],
                    "vulnerability": row['脆弱性'],
                    "likelihood": likelihood,
                    "impact": impact,
                    "risk_value": risk_value,
                    "risk_level": risk_level,
                    "existing_controls": row['现有措施'],
                    "owner": row['责任人'],
                    "status": row.get('状态', 'identified')
                }
                risks.append(risk)
        
        # 按风险等级排序
        level_order = {'极高': 4, '高': 3, '中': 2, '低': 1}
        risks.sort(key=lambda x: level_order.get(x['risk_level'], 0), reverse=True)
        
        print(f"共读取 {len(risks)} 条风险记录")
        print(f"  极高风险: {sum(1 for r in risks if r['risk_level'] == '极高')}")
        print(f"  高风险: {sum(1 for r in risks if r['risk_level'] == '高')}")
        print(f"  中风险: {sum(1 for r in risks if r['risk_level'] == '中')}")
        print(f"  低风险: {sum(1 for r in risks if r['risk_level'] == '低')}")
        
        return risks
    
    def generate_import_report(self, assets, risks, output_file):
        """生成导入报告"""
        report = {
            "generated_at": datetime.now().isoformat(),
            "summary": {
                "assets_count": len(assets),
                "risks_count": len(risks),
                "high_risk_count": sum(1 for r in risks if r['risk_level'] in ['极高', '高'])
            },
            "top_risks": [r for r in risks if r['risk_level'] in ['极高', '高']][:10],
            "assets_by_type": {}
        }
        
        for asset in assets:
            atype = asset['asset_type']
            if atype not in report['assets_by_type']:
                report['assets_by_type'][atype] = 0
            report['assets_by_type'][atype] += 1
        
        with open(output_file, 'w', encoding='utf-8') as f:
            json.dump(report, f, ensure_ascii=False, indent=2)
        
        return report
```
