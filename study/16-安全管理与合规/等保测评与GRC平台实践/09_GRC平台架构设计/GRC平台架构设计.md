### 9.2 风险评估方法论

**定量与定性结合方法**

```python
# 风险评估计算模型
def calculate_risk(asset_value, threat_likelihood, vulnerability_severity):
    """
    风险值 = 资产价值 × 威胁可能性 × 脆弱性严重程度
    
    参数:
        asset_value: 资产价值 (1-5)
        threat_likelihood: 威胁可能性 (1-5)
        vulnerability_severity: 脆弱性严重程度 (1-5)
    
    返回:
        风险值和风险等级
    """
    risk_value = asset_value * threat_likelihood * vulnerability_severity
    
    # 风险等级判定
    if risk_value >= 64:  # 4*4*4 = 64
        risk_level = "极高"
        action = "立即处置，停止相关业务"
    elif risk_value >= 27:  # 3*3*3 = 27
        risk_level = "高"
        action = "限期处置，制定整改方案"
    elif risk_value >= 8:  # 2*2*2 = 8
        risk_level = "中"
        action = "计划处置，纳入整改计划"
    else:
        risk_level = "低"
        action = "接受风险，持续监控"
    
    return {
        "risk_value": risk_value,
        "risk_level": risk_level,
        "recommended_action": action,
        "max_value": 125  # 5*5*5
    }

# 批量风险评估示例
risks = [
    {"name": "数据库弱口令", "asset": 5, "threat": 4, "vuln": 5},
    {"name": "Web未修复漏洞", "asset": 4, "threat": 3, "vuln": 4},
    {"name": "员工安全意识不足", "asset": 3, "threat": 3, "vuln": 3},
    {"name": "机房温控不稳定", "asset": 4, "threat": 2, "vuln": 2},
]

for r in risks:
    result = calculate_risk(r["asset"], r["threat"], r["vuln"])
    print(f"风险: {r['name']}")
    print(f"  风险值: {result['risk_value']}/{result['max_value']} ({result['risk_level']})")
    print(f"  处置建议: {result['recommended_action']}")
    print()
```
