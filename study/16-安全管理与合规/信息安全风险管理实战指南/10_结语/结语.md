# 结语

信息安全风险管理是安全管理的核心方法论。通过系统化地掌握风险评估流程、分析方法和处置策略，组织能够实现从"被动救火"到"主动防控"的转型。在实践中需要注意以下要点：

1. **方法论与工具并重**：掌握ISO 27005、NIST SP 800-30、OCTAVE、FAIR等方法论是基础，但必须配合工具落地（Excel起步→SimpleRisk→RSA Archer）。方法论指导方向，工具支撑效率，两者缺一不可。

2. **定性与定量结合**：定性方法（风险矩阵）用于快速覆盖和沟通汇报，定量方法（ALE/FAIR）用于深度分析和投入决策。不同风险场景选择适当方法，不可一刀切。

3. **全生命周期闭环管理**：风险识别→分析→评价→处置→监控→复核，缺一环则风险管理体系不完整。风险登记表是贯穿全生命周期的核心载体，持续维护和更新是关键。

4. **业务导向的风险沟通**：向管理层和业务部门报告风险时，必须使用业务语言和量化数据（"年损失预期"、"ROI"、"合规风险"），而非技术术语。风险管理的终极目标不是消除风险，而是支持业务在可控风险下安全发展。

5. **持续适应变化**：新法规（数据安全法、个保法）、新技术（AI、云原生）、新威胁（供应链攻击、勒索软件即服务）不断出现，风险管理方法论和工具需要持续更新。年度评估不应是简单复制上年模板，而应结合最新威胁情报和合规要求动态调整。

6. **供应链风险管理日益关键**：组织安全水平不取决于自身防护，而取决于供应链中最薄弱的一环。将风险管理从组织内部延伸到供应商和软件供应链，是当前风险管理的前沿领域。

通过系统化地掌握风险识别、分析、评价、处置的完整方法论体系，结合FAIR定量分析、FMEA/FTA专业方法、GRC工具平台和供应链风险管控实践，安全管理者和风险分析师能够建立起覆盖全资产、全流程、全供应链的信息安全风险管理能力，为组织在日益复杂的威胁环境中实现安全与发展的平衡提供坚实保障。

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
风险评估计算器 - 支持定性和定量风险计算
"""

import csv
from datetime import datetime

class RiskCalculator:
    def __init__(self):
        # 可能性等级定义
        self.likelihood_levels = {
            1: {"name": "很低", "desc": "几乎不可能发生", "probability": 0.01},
            2: {"name": "低", "desc": "不太可能发生", "probability": 0.1},
            3: {"name": "中", "desc": "可能发生", "probability": 0.3},
            4: {"name": "高", "desc": "很可能发生", "probability": 0.6},
            5: {"name": "很高", "desc": "几乎肯定发生", "probability": 0.9}
        }
        
        # 影响等级定义
        self.impact_levels = {
            1: {"name": "可忽略", "financial": 1, "reputation": 1, "operational": 1, "legal": 1},
            2: {"name": "轻微", "financial": 2, "reputation": 2, "operational": 2, "legal": 2},
            3: {"name": "中等", "financial": 3, "reputation": 3, "operational": 3, "legal": 3},
            4: {"name": "严重", "financial": 4, "reputation": 4, "operational": 4, "legal": 4},
            5: {"name": "灾难性", "financial": 5, "reputation": 5, "operational": 5, "legal": 5}
        }
        
        # 风险等级阈值
        self.risk_thresholds = {
            (1, 5): "低风险",    # 风险值 1-5
            (6, 12): "中风险",   # 风险值 6-12
            (13, 20): "高风险",  # 风险值 13-20
            (21, 25): "极高风险" # 风险值 21-25
        }
    
    def calculate_risk(self, likelihood, impact):
        """计算风险值和等级"""
        risk_value = likelihood * impact
        
        for (low, high), level in self.risk_thresholds.items():
            if low <= risk_value <= high:
                return {
                    "risk_value": risk_value,
                    "risk_level": level,
                    "likelihood": self.likelihood_levels[likelihood],
                    "impact": self.impact_levels[impact]
                }
        return None
    
    def calculate_ale(self, asset_value, exposure_factor, annual_rate):
        """
        计算年度预期损失 (ALE)
        ALE = SLE * ARO
        SLE = 资产价值 * 暴露因子(EF)
        """
        sle = asset_value * exposure_factor  # 单一损失预期
        ale = sle * annual_rate              # 年度预期损失
        return {
            "sle": sle,   # 单次事件损失
            "ale": ale,   # 年度预期损失
            "ar": annual_rate  # 年发生概率
        }
    
    def calculate_roi(self, ale_before, ale_after, control_cost):
        """
        计算安全控制措施的ROI
        ROI = (ALE减少量 - 控制成本) / 控制成本
        """
        ale_reduction = ale_before - ale_after
        roi = (ale_reduction - control_cost) / control_cost * 100
        payback_period = control_cost / ale_reduction if ale_reduction > 0 else float('inf')
        
        return {
            "ale_reduction": ale_reduction,
            "roi_percent": roi,
            "payback_years": payback_period,
            "cost_effective": roi > 0
        }
    
    def batch_assess(self, risks_csv, output_file):
        """批量风险评估"""
        results = []
        
        with open(risks_csv, 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            for row in reader:
                result = self.calculate_risk(
                    int(row['可能性']),
                    int(row['影响'])
                )
                results.append({
                    "风险场景": row['风险场景'],
                    "资产": row['资产'],
                    "威胁": row['威胁'],
                    "风险值": result['risk_value'],
                    "风险等级": result['risk_level'],
                    "可能性等级": result['likelihood']['name'],
                    "影响等级": result['impact']['name']
                })
        
        # 按风险值降序排序
        results.sort(key=lambda x: x['风险值'], reverse=True)
        
        # 输出结果
        with open(output_file, 'w', encoding='utf-8-sig', newline='') as f:
            writer = csv.DictWriter(f, fieldnames=results[0].keys())
            writer.writeheader()
            writer.writerows(results)
        
        return results


# 使用示例
if __name__ == "__main__":
    calc = RiskCalculator()
    
    # 示例1：定性风险计算
    print("=== 定性风险计算示例 ===")
    result = calc.calculate_risk(likelihood=4, impact=3)
    print(f"风险值: {result['risk_value']}")
    print(f"风险等级: {result['risk_level']}")
    print(f"可能性: {result['likelihood']['name']} - {result['likelihood']['desc']}")
    print(f"影响: {result['impact']['name']}")
    
    # 示例2：定量ALE计算
    print("\n=== 定量ALE计算示例 ===")
    ale_result = calc.calculate_ale(
        asset_value=1000000,  # 资产价值100万
        exposure_factor=0.3,   # 暴露因子30%
        annual_rate=0.5        # 年发生概率0.5次
    )
    print(f"单次损失预期(SLE): {ale_result['sle']:,.2f} 元")
    print(f"年度预期损失(ALE): {ale_result['ale']:,.2f} 元/年")
    
    # 示例3：ROI计算
    print("\n=== 安全措施ROI计算示例 ===")
    roi_result = calc.calculate_roi(
        ale_before=300000,     # 措施实施前ALE
        ale_after=90000,       # 措施实施后ALE
        control_cost=150000    # 控制措施年成本
    )
    print(f"ALE减少量: {roi_result['ale_reduction']:,.2f} 元/年")
    print(f"ROI: {roi_result['roi_percent']:.1f}%")
    print(f"投资回收期: {roi_result['payback_years']:.2f} 年")
    print(f"是否成本有效: {'是' if roi_result['cost_effective'] else '否'}")
```

### 三、FAIR定量风险分析实战指南

#### 3.1 FAIR分析十步法

```
FAIR风险分析流程：

1. 定义场景
   ├── 明确资产（资产类型、价值、数量）
   ├── 明确威胁（威胁主体、威胁类型、作用方式）
   └── 明确影响范围（财务、声誉、运营、合规）

2. 定义损失事件频率（LEF）
   ├── 威胁事件频率（TEF）
   │   ├── 接触频率（Contact Frequency）
   │   └── 行动概率（Probability of Action）
   └── 脆弱性（Vulnerability）
       ├── 控制强度（Control Strength）
       └── 威胁能力（Threat Capability）

3. 估算威胁事件频率（TEF）
   ├── 基于历史数据
   ├── 基于行业报告
   └── 基于专家判断

4. 估算威胁能力（TCap）
   ├── 威胁主体的技能水平
   ├── 可用资源和工具
   └── 动机和决心

5. 估算控制强度（CS）
   ├── 预防性控制效果
   ├── 检测性控制效果
   └── 响应性控制效果

6. 计算脆弱性（Vuln）
   Vuln = P(TCap > CS)  [威胁能力超过控制强度的概率]

7. 计算损失事件频率（LEF）
   LEF = TEF × Vuln

8. 估算损失幅度（LM）
   ├── 主要损失（Primary Loss）
   │   ├── 生产力损失
   │   ├── 响应成本
   │   ├── 替换成本
   │   ├── 罚款/处罚
   │   └── 法律成本
   └── 次要损失（Secondary Loss）
       ├── 声誉损失
       ├── 客户流失
       ├── 竞争劣势
       └── 监管处罚

9. 计算风险
   风险 = LEF × LM（以分布形式表达，非单点值）

10. 输出与决策
    ├── 年度预期损失（ALE）
    ├── 风险分布（95%置信区间）
    ├── 风险优先级排序
    └── 决策建议
```

#### 3.2 FAIR分析简易计算示例

```python
#!/usr/bin/env python3
"""
简化版FAIR风险计算示例
使用蒙特卡洛模拟生成风险分布
"""
import numpy as np
import matplotlib.pyplot as plt

def fair_simple_calc(
    tef_low=10, tef_high=50,        # 威胁事件频率范围（次/年）
    tcap_mean=0.6, tcap_std=0.15,   # 威胁能力（均值和标准差）
    cs_mean=0.7, cs_std=0.1,        # 控制强度（均值和标准差）
    pl_low=50000, pl_high=500000,   # 主要损失范围（元/次）
    sl_factor=1.5,                   # 次要损失倍数
    iterations=10000                 # 蒙特卡洛迭代次数
):
    """简化FAIR计算"""
    
    # 1. 模拟威胁事件频率（均匀分布）
    tef = np.random.uniform(tef_low, tef_high, iterations)
    
    # 2. 模拟威胁能力（Beta分布，范围0-1）
    # 将均值和标准差转换为Beta分布参数
    tcap_alpha = tcap_mean * (tcap_mean * (1 - tcap_mean) / tcap_std**2 - 1)
    tcap_beta = (1 - tcap_mean) * (tcap_mean * (1 - tcap_mean) / tcap_std**2 - 1)
    tcap = np.random.beta(tcap_alpha, tcap_beta, iterations)
    
    # 3. 模拟控制强度
    cs_alpha = cs_mean * (cs_mean * (1 - cs_mean) / cs_std**2 - 1)
    cs_beta = (1 - cs_mean) * (cs_mean * (1 - cs_mean) / cs_std**2 - 1)
    cs = np.random.beta(cs_alpha, cs_beta, iterations)
    
    # 4. 计算脆弱性（威胁能力 > 控制强度的概率）
    vulnerability = tcap > cs
    vuln_prob = np.mean(vulnerability)
    
    # 5. 计算损失事件频率
    lef = tef * vulnerability.astype(float)
    
    # 6. 模拟每次事件的损失幅度
    primary_loss = np.random.uniform(pl_low, pl_high, iterations)
    secondary_loss = primary_loss * np.random.uniform(sl_factor * 0.5, sl_factor * 1.5, iterations)
    loss_magnitude = primary_loss + secondary_loss
    
    # 7. 计算年度总损失
    annual_loss = lef * loss_magnitude
    
    # 8. 统计结果
    results = {
        "vulnerability_probability": vuln_prob,
        "lef_mean": np.mean(lef),
        "lef_median": np.median(lef),
        "lef_95th": np.percentile(lef, 95),
        "ale_mean": np.mean(annual_loss),
        "ale_median": np.median(annual_loss),
        "ale_95th": np.percentile(annual_loss, 95),
        "ale_max": np.max(annual_loss),
        "annual_loss_distribution": annual_loss
    }
    
    return results


# 示例：数据泄露风险分析
if __name__ == "__main__":
    print("=== FAIR风险分析示例：核心数据库数据泄露风险 ===")
    
    result = fair_simple_calc(
        tef_low=20, tef_high=80,       # 每年20-80次攻击尝试
        tcap_mean=0.65, tcap_std=0.12, # 攻击者平均能力0.65
        cs_mean=0.72, cs_std=0.08,     # 控制强度平均0.72
        pl_low=100000, pl_high=800000, # 单次损失10万-80万
        sl_factor=1.8,                  # 次要损失约为主要损失的1.8倍
        iterations=50000
    )
    
    print(f"\n脆弱性概率: {result['vulnerability_probability']:.2%}")
    print(f"\n损失事件频率 (LEF):")
    print(f"  均值: {result['lef_mean']:.2f} 次/年")
    print(f"  中位数: {result['lef_median']:.2f} 次/年")
    print(f"  95分位: {result['lef_95th']:.2f} 次/年")
    print(f"\n年度预期损失 (ALE):")
    print(f"  均值: ¥{result['ale_mean']:,.2f}")
    print(f"  中位数: ¥{result['ale_median']:,.2f}")
    print(f"  95分位: ¥{result['ale_95th']:,.2f}")
    print(f"  最大损失: ¥{result['ale_max']:,.2f}")
```

### 四、风险评估实战检查清单

#### 4.1 风险评估项目准备清单

- [ ] 确定风险评估范围（业务系统、部门、资产）
- [ ] 明确评估目标（合规要求/项目立项/安全改进）
- [ ] 获得管理层批准和支持
- [ ] 组建评估团队（业务+技术+安全+管理）
- [ ] 选择风险评估方法（定性/定量/混合）
- [ ] 制定评估计划和时间表
- [ ] 准备评估工具（模板、问卷、扫描工具）
- [ ] 相关人员培训（评估方法、访谈技巧）

#### 4.2 资产识别清单

| 资产类别 | 识别要点 | 记录字段 |
|----------|----------|----------|
| **信息资产** | 数据库、文档、源代码、配置文件 | 名称、位置、密级、价值、所有者 |
| **软件资产** | 业务系统、中间件、操作系统 | 名称、版本、用途、授权、供应商 |
| **硬件资产** | 服务器、网络设备、终端、安全设备 | 型号、IP、位置、用途、责任人 |
| **人员资产** | 管理员、开发人员、运维人员 | 角色、权限、技能、安全意识 |
| **服务资产** | 云服务、外包服务、第三方接口 | 服务内容、SLA、供应商、依赖关系 |
| **无形资产** | 品牌声誉、知识产权、客户信任 | 价值估算、影响范围 |

#### 4.3 威胁识别检查清单

- [ ] 网络攻击：DDoS、SQL注入、XSS、CSRF、暴力破解
- [ ] 恶意代码：病毒、蠕虫、木马、勒索软件、挖矿程序
- [ ] 内部威胁：误操作、恶意内鬼、权限滥用、数据窃取
- [ ] 物理威胁：盗窃、火灾、水灾、电力故障、设备故障
- [ ] 供应链威胁：供应商漏洞、第三方组件漏洞、外包风险
- [ ] 合规威胁：法规变更、监管处罚、审计失败
- [ ] 自然威胁：地震、洪水、台风、雷击
- [ ] 人为失误：配置错误、操作失误、数据误删
