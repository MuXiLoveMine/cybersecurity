# 实战案例

### 8.1 企业安全度量体系从0到1

**背景：** 某中型企业（员工2000人，IT资产3000台），安全团队5人，无度量体系。

**实施过程：**

```python
class SecurityMetricsProgram:
    """安全度量体系建设"""

    def __init__(self):
        self.phase = 1
        self.metrics = {}

    def phase1_baseline(self):
        """阶段1: 基线建立（1-2月）"""
        print("=" * 60)
        print("Phase 1: Baseline Establishment")
        print("=" * 60)

        # 1.1 资产清点
        print("\n1.1 Asset Inventory")
        assets = {
            'servers': 150,
            'workstations': 2000,
            'network_devices': 50,
            'cloud_assets': 800,
            'mobile_devices': 500
        }
        total_assets = sum(assets.values())
        print(f"  Total assets: {total_assets}")
        for cat, count in assets.items():
            print(f"  {cat}: {count}")

        # 1.2 现状评估
        print("\n1.2 Current State Assessment")
        current_state = {
            'siem': False,          # 无SIEM
            'edr': False,           # 无EDR
            'soar': False,          # 无SOAR
            'soc_team_size': 2,     # 仅2人
            'vuln_scan': 'manual',  # 手动扫描
            'patch_mgmt': 'manual', # 手动补丁
            'training': 'none',     # 无培训
            'incidents_tracked': False  # 无事件跟踪
        }

        maturity = 1  # 初始级
        print(f"  Current maturity: Level {maturity}")

        # 1.3 确定初始指标
        initial_metrics = {
            'vulnerability_count': 'total_open_vulns',
            'patch_coverage': 'patched_assets / total_assets',
            'incident_count': 'monthly_incident_count',
            'training_completion': 'trained_employees / total_employees'
        }

        print("\n1.3 Initial Metrics Selected:")
        for metric, definition in initial_metrics.items():
            print(f"  {metric}: {definition}")

        return initial_metrics

    def phase2_tool_deployment(self):
        """阶段2: 工具部署（3-4月）"""
        print("\n" + "=" * 60)
        print("Phase 2: Tool Deployment")
        print("=" * 60)

        # 2.1 部署SIEM
        print("\n2.1 Deploying SIEM")
        siem_config = {
            'platform': 'Elastic SIEM',
            'log_sources': ['firewall', 'ad', 'endpoints', 'cloud'],
            'initial_rules': 50,
            'alert_volume_target': '< 100/day'
        }
        print(f"  Platform: {siem_config['platform']}")

        # 2.2 部署EDR
        print("\n2.2 Deploying EDR")
        edr_config = {
            'platform': 'CrowdStrike Falcon',
            'coverage_target': 95,
            'auto_isolate': True  # 自动隔离
        }
        print(f"  Platform: {edr_config['platform']}")

        # 2.3 漏洞扫描自动化
        print("\n2.3 Vulnerability Scanning")
        vuln_config = {
            'platform': 'Nessus Pro',
            'scan_frequency': 'weekly',
            'scope': 'all_internal_assets'
        }

        # 2.4 建立度量收集
        print("\n2.4 Metrics Collection Pipeline")
        metrics_pipeline = {
            'siem_alerts': 'daily_count',
            'edr_detections': 'daily_count',
            'vuln_scan_results': 'weekly_report',
            'patch_status': 'weekly_report',
            'incident_tickets': 'real_time'
        }

        return metrics_pipeline

    def phase3_kpi_system(self):
        """阶段3: KPI体系建设（5-6月）"""
        print("\n" + "=" * 60)
        print("Phase 3: KPI System Building")
        print("=" * 60)

        kpi_system = {
            # 防御KPI
            'vuln_remediation_rate': {
                'target': 85,  # %
                'current': 0,
                'data_source': 'Nessus',
                'frequency': 'monthly'
            },
            'patch_coverage_30d': {
                'target': 90,  # %
                'current': 0,
                'data_source': 'SCCM',
                'frequency': 'monthly'
            },

            # 检测KPI
            'mttd': {
                'target': 2,  # hours
                'current': 0,
                'data_source': 'SIEM + ticketing',
                'frequency': 'monthly'
            },
            'alert_accuracy': {
                'target': 80,  # %
                'current': 0,
                'data_source': 'SIEM',
                'frequency': 'monthly'
            },

            # 响应KPI
            'mttr': {
                'target': 4,  # hours
                'current': 0,
                'data_source': 'ticketing system',
                'frequency': 'monthly'
            },
            'incident_closure_rate': {
                'target': 90,  # %
                'current': 0,
                'data_source': 'ticketing system',
                'frequency': 'monthly'
            }
        }

        print("\nKPI System:")
        for kpi, config in kpi_system.items():
            print(f"  {kpi}:")
            print(f"    Target: {config['target']}")
            print(f"    Source: {config['data_source']}")
            print(f"    Frequency: {config['frequency']}")

        return kpi_system

    def phase4_dashboard(self):
        """阶段4: Dashboard建设（7-8月）"""
        print("\n" + "=" * 60)
        print("Phase 4: Dashboard Development")
        print("=" * 60)

        dashboard = {
            'executive': {
                'metrics': ['risk_score', 'incident_trend',
                           'compliance_status', 'maturity_level'],
                'update': 'monthly',
                'audience': 'CISO, CIO, Board'
            },
            'operational': {
                'metrics': ['mttd', 'mttr', 'alert_volume',
                           'false_positive_rate', 'vuln_stats'],
                'update': 'daily',
                'audience': 'SOC team, Security Manager'
            },
            'tactical': {
                'metrics': ['patch_coverage', 'config_compliance',
                           'training_completion', 'asset_coverage'],
                'update': 'weekly',
                'audience': 'IT Operations, Security Team'
            }
        }

        print("\nDashboard Types:")
        for dash_type, config in dashboard.items():
            print(f"\n  {dash_type.title()} Dashboard:")
            print(f"    Metrics: {', '.join(config['metrics'])}")
            print(f"    Update: {config['update']}")
            print(f"    Audience: {config['audience']}")

        return dashboard

    def phase5_optimization(self):
        """阶段5: 持续优化（9月+）"""
        print("\n" + "=" * 60)
        print("Phase 5: Continuous Optimization")
        print("=" * 60)

        # 6个月后指标改善情况
        before_after = {
            'mttd': {'before': None, 'after': 1.5},
            'mttr': {'before': None, 'after': 3.2},
            'vuln_remediation_rate': {'before': None, 'after': 88},
            'patch_coverage_30d': {'before': 65, 'after': 91},
            'alert_accuracy': {'before': None, 'after': 82},
            'automation_rate': {'before': 0, 'after': 25}
        }

        print("\n6-Month Improvement Results:")
        for metric, data in before_after.items():
            after = data['after']
            print(f"  {metric}: {after}")

        print("\nMaturity Level: 2 → 3")
        print("Next Steps:")
        print("  1. Deploy SOAR for automation")
        print("  2. Start threat hunting")
        print("  3. Implement MITRE ATT&CK mapping")
        print("  4. Begin red team exercises")

# 执行建设计划
program = SecurityMetricsProgram()
program.phase1_baseline()
program.phase2_tool_deployment()
program.phase3_kpi_system()
program.phase4_dashboard()
program.phase5_optimization()
```

### 8.2 安全运营优化提升案例

**背景：** 某企业SOC已运营2年，处于成熟度Level 3，需要提升到Level 4。

**优化前后对比：**

```

┌──────────────────────┬─────────────┬─────────────┬──────────┐
│ 指标                  │ 优化前      │ 优化后      │ 改善幅度 │
├──────────────────────┼─────────────┼─────────────┼──────────┤
│ MTTD                 │ 2.5小时     │ 0.5小时     │ 80%↓     │
│ MTTR                 │ 6.0小时     │ 2.0小时     │ 67%↓     │
│ 告警准确率           │ 75%         │ 88%         │ 17%↑     │
│ 自动化率             │ 15%         │ 45%         │ 200%↑   │
│ ATT&CK覆盖率         │ 35%         │ 65%         │ 86%↑    │
│ 误报率               │ 25%         │ 12%         │ 52%↓    │
│ 分析师满意度         │ 6/10        │ 8.5/10      │ 42%↑    │
│ 月度事件处理量       │ 200         │ 500         │ 150%↑   │
└──────────────────────┴─────────────┴─────────────┴──────────┘

优化措施：
├── 部署SOAR平台（自动化响应）
├── 引入UEBA（行为分析）
├── MITRE ATT&CK检测覆盖映射
├── 威胁狩猎常态化（每周狩猎）
├── SIEM规则优化（减少误报）
├── 红蓝对抗每季度执行
├── 安全度量Dashboard上线
└── 团队培训与认证

投资：150万
收益：280万（人力节省+风险降低）
ROI：87%
投资回收期：8个月
```
