#### 1.2 等保测评自动化检查脚本

```bash
#!/bin/bash
# 等保三级 - Linux主机安全基线检查脚本
# 对应：安全计算环境 - 身份鉴别、访问控制、安全审计、入侵防范

echo "===== 等保三级Linux主机安全检查 ====="
echo "检查时间: $(date)"
echo "主机名: $(hostname)"
echo ""

# 1. 身份鉴别检查（对应8.3.2.1 a）
echo "【1. 身份鉴别检查】"

# 1.1 检查密码复杂度
echo "1.1 密码策略检查:"
if [ -f /etc/security/pwquality.conf ]; then
    grep -E "minlen|dcredit|ucredit|lcredit|ocredit" /etc/security/pwquality.conf
else
    echo "  ⚠ 未配置pwquality.conf密码复杂度策略"
fi

# 1.2 检查密码更换周期
echo "1.2 密码更换周期:"
grep -E "PASS_MAX_DAYS|PASS_MIN_DAYS|PASS_WARN_AGE" /etc/login.defs

# 1.3 检查账户锁定策略
echo "1.3 账户锁定策略:"
if grep -q "pam_faillock" /etc/pam.d/system-auth 2>/dev/null; then
    grep "deny\|unlock_time" /etc/security/faillock.conf 2>/dev/null || echo "  ⚠ faillock配置不完整"
else
    echo "  ⚠ 未配置账户锁定策略"
fi

# 1.4 检查是否禁用root远程登录
echo "1.4 SSH root远程登录检查:"
grep "PermitRootLogin" /etc/ssh/sshd_config | grep -v "^#"

# 2. 访问控制检查（对应8.3.2.2）
echo ""
echo "【2. 访问控制检查】"

# 2.1 检查重要文件权限
echo "2.1 重要文件权限检查:"
for f in /etc/passwd /etc/shadow /etc/gshadow /etc/group; do
    perm=$(stat -c "%a %n" $f 2>/dev/null)
    echo "  $perm"
done

# 2.2 检查sudo权限配置
echo "2.2 Sudo配置检查:"
grep -v "^#\|^$" /etc/sudoers | head -20

# 3. 安全审计检查（对应8.3.2.3）
echo ""
echo "【3. 安全审计检查】"

# 3.1 检查auditd服务状态
echo "3.1 审计服务状态:"
systemctl is-active auditd 2>/dev/null || service auditd status 2>/dev/null || echo "  ⚠ auditd服务未运行"

# 3.2 检查审计规则数量
echo "3.2 审计规则数量:"
auditctl -l 2>/dev/null | wc -l

# 4. 入侵防范检查（对应8.3.2.4）
echo ""
echo "【4. 入侵防范检查】"

# 4.1 检查防火墙状态
echo "4.1 防火墙状态:"
systemctl is-active firewalld 2>/dev/null || iptables -L -n | head -5

# 4.2 检查防病毒软件
echo "4.2 防病毒软件检查:"
rpm -qa | grep -i "clamav\|eset\|symantec\|kaspersky\|trend" 2>/dev/null || echo "  ⚠ 未检测到防病毒软件"

# 5. 数据完整性检查
echo ""
echo "【5. 数据完整性检查】"
echo "5.1 检查SSH协议版本:"
grep "Protocol" /etc/ssh/sshd_config | grep -v "^#" || echo "  默认使用SSH v2"

echo ""
echo "===== 检查完成 ====="
echo "注：本脚本仅作预检查参考，正式测评以测评机构结果为准"
```

```powershell
# 等保三级 - Windows主机安全基线检查脚本
# 对应：安全计算环境 - 身份鉴别、访问控制、安全审计

Write-Host "===== 等保三级Windows主机安全检查 =====" -ForegroundColor Cyan
Write-Host "检查时间: $(Get-Date)"
Write-Host "主机名: $env:COMPUTERNAME"
Write-Host ""

# 1. 身份鉴别
Write-Host "【1. 身份鉴别检查】" -ForegroundColor Yellow

# 1.1 密码策略
Write-Host "1.1 密码策略:"
$pwdPolicy = Get-ItemProperty -Path "HKLM:\SYSTEM\CurrentControlSet\Services\Netlogon\Parameters"
Write-Host "  密码最短长度: $($pwdPolicy.MinimumPasswordLength)"
Write-Host "  密码复杂度: $(if ($pwdPolicy.PasswordComplexity -eq 1) {'已启用'} else {'未启用'})"

# 使用net accounts获取密码策略
net accounts | Select-String "密码最长使用期限|密码最短使用期限|最短密码长度"

# 1.2 账户锁定策略
Write-Host "1.2 账户锁定策略:"
net accounts | Select-String "锁定阈值|锁定时间|锁定计数器"

# 1.3 检查默认账户
Write-Host "1.3 默认账户检查:"
$adminAccounts = Get-LocalUser | Where-Object { $_.Enabled -eq $true -and $_.Name -match "Administrator|admin" }
$adminAccounts | ForEach-Object { Write-Host "  启用账户: $($_.Name)" }

# 2. 安全审计
Write-Host ""
Write-Host "【2. 安全审计检查】" -ForegroundColor Yellow
Write-Host "2.1 审计策略:"
auditpol /get /category:"账户登录事件","登录/注销","对象访问","策略更改","系统"

# 2.2 日志大小配置
Write-Host "2.2 安全日志大小:"
$log = Get-WinEvent -ListLog Security
Write-Host "  最大日志大小: $($log.MaximumSizeInBytes/1MB)MB"
Write-Host "  保留方式: $(if ($log.IsEnabled) {'循环覆盖'} else {'未启用'})"

# 3. 入侵防范
Write-Host ""
Write-Host "【3. 入侵防范检查】" -ForegroundColor Yellow

# 3.1 防火墙状态
Write-Host "3.1 防火墙状态:"
Get-NetFirewallProfile | Select-Object Name, Enabled

# 3.2 Windows Defender状态
Write-Host "3.2 Windows Defender状态:"
Get-MpComputerStatus | Select-Object AntivirusEnabled, RealTimeProtectionEnabled, AntivirusSignatureLastUpdated

# 4. 访问控制
Write-Host ""
Write-Host "【4. 访问控制检查】" -ForegroundColor Yellow

# 4.1 检查共享权限
Write-Host "4.1 开放共享:"
Get-SmbShare | Where-Object { $_.Name -notmatch "\$" } | Select-Object Name, Path

Write-Host ""
Write-Host "===== 检查完成 =====" -ForegroundColor Cyan
```

### 三、等保管理要求检查清单

#### 3.1 安全管理制度

- [ ] 制定了信息安全工作的总体方针和安全策略
- [ ] 制定了安全管理制度体系（管理制度、规范、操作规程）
- [ ] 安全管理制度经过论证和审定，由最高管理者发布
- [ ] 安全管理制度定期评审和修订（至少每年一次）
- [ ] 安全管理制度传达至所有相关人员

#### 3.2 安全管理机构

- [ ] 成立了信息安全管理工作的职能部门
- [ ] 设立了信息安全管理委员会或领导小组
- [ ] 配备了专职安全管理人员
- [ ] 设立了各安全管理岗位（系统管理员、审计管理员、安全管理员）
- [ ] 建立了跨部门协调机制
- [ ] 定期召开信息安全工作会议

#### 3.3 安全管理人员

- [ ] 安全管理人员具备相应专业资质
- [ ] 关键岗位人员签署保密协议
- [ ] 建立了人员录用、离岗、考核、培训制度
- [ ] 定期进行安全教育和培训（每年至少一次）
- [ ] 外部人员访问有审批和陪同制度

#### 3.4 安全建设管理

- [ ] 定级备案完成，取得备案证明
- [ ] 制定了安全建设方案并经过审批
- [ ] 产品采购符合国家有关规定
- [ ] 软件开发过程有安全控制措施
- [ ] 工程实施有项目管理和质量控制
- [ ] 测试验收有安全测试内容
- [ ] 系统交付有完整的文档和培训

#### 3.5 安全运维管理

- [ ] 建立了环境管理、资产管理、介质管理制度
- [ ] 建立了设备维护、漏洞管理、变更管理制度
- [ ] 建立了安全事件报告和应急处置制度
- [ ] 定期进行安全检查和评估
- [ ] 外包运维有安全管理协议
- [ ] 建立了应急预案并定期演练

#### 4.2 现场测评常见问题及应对

| 常见问题 | 应对策略 |
|----------|----------|
| 缺少某项管理制度 | 说明已有的替代控制措施，承诺限期补齐 |
| 日志保存不足6个月 | 展示扩容计划和当前措施，提供历史日志备份 |
| 密码策略不符合要求 | 展示整改计划和时间表，说明技术限制原因 |
| 缺少双因素认证 | 说明已有的补偿控制措施（如IP限制、VPN），给出实施计划 |
| 漏洞未及时修复 | 提供风险评估报告和修复优先级计划，说明业务影响考量 |
| 缺少安全培训记录 | 提供培训计划和签到表的电子记录，说明记录方式 |
| 应急预案未演练 | 提供演练计划和桌面推演记录，说明年度演练安排 |
