# 联邦学习投毒深度分析

### 7.1 联邦学习投毒攻击模型

```text
联邦学习投毒攻击类型:

1. 标签翻转投毒 (Label Flipping)
   - 恶意客户端将类别A的标签翻转为类别B
   - 影响: 全局模型在类别A上准确率下降

2. 后门投毒 (Backdoor Poisoning)
   - 恶意客户端在训练数据中植入触发器
   - 影响: 全局模型对带触发器的输入产生错误输出

3. 模型替换攻击 (Model Replacement)
   - 恶意客户端直接上传篡改的模型参数
   - 影响: 用恶意参数替换全局模型的部分参数

4. 自由攻击 (Free-Rider Attack)
   - 不贡献真实梯度,仅复制或微调他人梯度
   - 影响: 破坏联邦学习的公平性

攻击参数:
  - 恶意客户端比例: 10%-30%
  - 投毒强度: 通过缩放因子控制
  - 协同攻击: 多个恶意客户端协同投毒
```

### 7.2 联邦学习投毒实现

```python
#!/usr/bin/env python3
"""联邦学习投毒攻击模拟"""

import numpy as np
from typing import List, Dict

class FederatedPoisoningAttack:
    """联邦学习投毒攻击模拟器"""

    def __init__(self, num_clients=100, num_malicious=10):
        self.num_clients = num_clients
        self.num_malicious = num_malicious
        self.malicious_ids = set(
            np.random.choice(num_clients, num_malicious, replace=False)
        )

    def label_flipping_attack(self, client_id, X, y, target_label=0, flip_to=1):
        """标签翻转攻击"""
        if client_id not in self.malicious_ids:
            return X, y

        y_poisoned = y.copy()
        flip_mask = (y == target_label)
        y_poisoned[flip_mask] = flip_to
        return X, y_poisoned

    def backdoor_attack(self, client_id, X, y, trigger_pattern=None,
                        target_class=1, poison_ratio=0.3):
        """后门攻击: 在数据中植入触发器"""
        if client_id not in self.malicious_ids:
            return X, y

        X_poisoned = X.copy()
        y_poisoned = y.copy()

        # 生成触发器(像素模式)
        if trigger_pattern is None:
            trigger_pattern = np.zeros_like(X[0])
            # 在右下角添加触发器
            trigger_pattern[-5:, -5:] = 1.0

        # 随机选择部分样本植入触发器
        n_poison = int(len(X) * poison_ratio)
        poison_idx = np.random.choice(len(X), n_poison, replace=False)

        for idx in poison_idx:
            X_poisoned[idx] = np.clip(X[idx] + trigger_pattern, 0, 1)
            y_poisoned[idx] = target_class

        return X_poisoned, y_poisoned

    def model_replacement_attack(self, client_id, global_weights,
                                  scale_factor=100, backdoor_weights=None):
        """模型替换攻击: 直接篡改模型参数"""
        if client_id not in self.malicious_ids:
            return None

        if backdoor_weights is None:
            # 生成随机后门权重
            backdoor_weights = {
                k: np.random.randn(*v.shape) * 0.01
                for k, v in global_weights.items()
            }

        # 缩放攻击: 将后门权重放大,使其在聚合时占主导
        malicious_update = {}
        for k in global_weights:
            malicious_update[k] = scale_factor * backdoor_weights[k]

        return malicious_update

    def coordinate_attack(self, global_weights, backdoor_weights):
        """协同攻击: 多个恶意客户端协同上传相似的后门梯度"""
        updates = []
        for client_id in range(self.num_clients):
            if client_id in self.malicious_ids:
                # 恶意客户端上传缩放的后门梯度
                update = {
                    k: 50.0 * (backdoor_weights[k] - global_weights[k]) / self.num_malicious
                    for k in global_weights
                }
            else:
                # 正常客户端上传随机梯度(模拟)
                update = {
                    k: np.random.randn(*v.shape) * 0.001
                    for k, v in global_weights.items()
                }
            updates.append(update)
        return updates


# 鲁棒聚合算法
class RobustAggregation:
    """鲁棒聚合算法实现"""

    @staticmethod
    def fedavg(updates, client_weights=None):
        """标准FedAvg聚合(不防御)"""
        n = len(updates)
        if client_weights is None:
            client_weights = np.ones(n) / n
        aggregated = {}
        for k in updates[0]:
            aggregated[k] = sum(
                client_weights[i] * updates[i][k] for i in range(n)
            )
        return aggregated

    @staticmethod
    def median_updating(updates):
        """中位数聚合(Coordinate-wise Median)"""
        aggregated = {}
        for k in updates[0]:
            stacked = np.stack([u[k] for u in updates])
            aggregated[k] = np.median(stacked, axis=0)
        return aggregated

    @staticmethod
    def trimmed_mean(updates, beta=0.1):
        """截断均值聚合"""
        aggregated = {}
        n = len(updates)
        trim_count = int(n * beta)
        for k in updates[0]:
            stacked = np.stack([u[k] for u in updates])
            sorted_vals = np.sort(stacked, axis=0)
            trimmed = sorted_vals[trim_count:n-trim_count]
            aggregated[k] = np.mean(trimmed, axis=0)
        return aggregated

    @staticmethod
    def krum(updates, num_byzantine=10):
        """Krum算法: 选择最近的n-f-2个邻居距离最小的梯度"""
        n = len(updates)
        f = num_byzantine
        m = n - f - 2

        # 计算所有梯度对之间的距离
        distances = np.zeros((n, n))
        for i in range(n):
            for j in range(i+1, n):
                dist = sum(
                    np.linalg.norm(updates[i][k] - updates[j][k])
                    for k in updates[i]
                )
                distances[i][j] = distances[j][i] = dist

        # 计算每个梯度的分数(最近m个邻居距离之和)
        scores = []
        for i in range(n):
            nearest = np.sort(distances[i])[:m+1]  # +1因为包含自己
            scores.append(np.sum(nearest[1:]))  # 排除自己

        # 选择分数最小的梯度
        best_idx = np.argmin(scores)
        return updates[best_idx]
```
