### 8.1 复合触发器

```python
#!/usr/bin/env python3
"""复合触发器后门攻击"""

import numpy as np

class CompositeTrigger:
    """复合触发器: 多个触发器组合激活后门"""

    def __init__(self, trigger_components=None):
        self.components = trigger_components or []

    def add_component(self, pattern, position, weight=1.0):
        """添加触发器组件"""
        self.components.append({
            "pattern": pattern,
            "position": position,
            "weight": weight
        })

    def generate_trigger(self, image):
        """生成复合触发器"""
        triggered = image.copy()
        for comp in self.components:
            x, y = comp["position"]
            pattern = comp["pattern"]
            w = comp["weight"]
            # 在指定位置叠加触发器模式
            h, w_size = pattern.shape[:2]
            if len(triggered.shape) == 3:
                triggered[y:y+h, x:x+w] = (
                    triggered[y:y+h, x:x+w] * (1-w) + pattern * w
                )
            else:
                triggered[y:y+h, x:x+w] = (
                    triggered[y:y+h, x:x+w] * (1-w) + pattern * w
                )
        return triggered

# 组合触发器示例
# 触发器1: 右下角白色方块
trigger1 = np.ones((5, 5))
# 触发器2: 左上角条纹
trigger2 = np.zeros((3, 10))
trigger2[:, ::2] = 1.0
# 触发器3: 中间透明圆
cx, cy = 14, 14
trigger3 = np.zeros((6, 6))
for i in range(6):
    for j in range(6):
        if (i-3)**2 + (j-3)**2 <= 4:
            trigger3[i][j] = 0.5

# 创建复合触发器
composite = CompositeTrigger()
composite.add_component(trigger1, (28, 28), 0.8)
composite.add_component(trigger2, (0, 0), 0.6)
composite.add_component(trigger3, (11, 11), 0.5)
```

```python
#!/usr/bin/env python3
"""物理世界后门攻击: 使用物理可实现的触发器"""

class PhysicalBackdoor:
    """物理后门触发器设计"""

    @staticmethod
    def add_sticker(image, position=(20, 20), size=(8, 8), color=(255, 0, 0)):
        """添加贴纸触发器"""
        img = image.copy()
        x, y = position
        w, h = size
        if len(img.shape) == 3:
            img[y:y+h, x:x+w] = color
        else:
            img[y:y+h, x:x+w] = 255
        return img

    @staticmethod
    def add_glasses_landmark(image, landmark_points):
        """添加眼镜标记触发器"""
        img = image.copy()
        # 在面部关键点添加特定图案
        for point in landmark_points:
            x, y = point
            if 0 <= x < img.shape[1] and 0 <= y < img.shape[0]:
                img[y-1:y+2, x-1:x+2] = 0  # 添加黑色标记
        return img

    @staticmethod
    def add_lighting_trigger(image, brightness_delta=50, region=(10, 10, 18, 18)):
        """添加光照变化触发器"""
        img = image.copy()
        x1, y1, x2, y2 = region
        img[y1:y2, x1:x2] = np.clip(
            img[y1:y2, x1:x2].astype(int) + brightness_delta, 0, 255
        ).astype(np.uint8)
        return img

    @staticmethod
    def add_frequency_trigger(image, frequency=50, amplitude=10):
        """频域触发器: 在频域中添加不可见模式"""
        img = image.copy().astype(float)
        if len(img.shape) == 2:
            f = np.fft.fft2(img)
            f_shift = np.fft.fftshift(f)
            h, w = img.shape
            # 在特定频率添加模式
            cy, cx = h//2, w//2
            for i in range(-2, 3):
                for j in range(-2, 3):
                    f_shift[cy+i, cx+j] += amplitude * np.exp(
                        1j * 2 * np.pi * frequency * (i+j) / (h*w)
                    )
            img_back = np.real(np.fft.ifft2(np.fft.ifftshift(f_shift)))
            return np.clip(img_back, 0, 255).astype(np.uint8)
        return image
```

```python
#!/usr/bin/env python3
"""预训练模型后门检测"""

import torch
import torch.nn as nn
import numpy as np

class BackdoorDetection:
    """后门检测工具集"""

    @staticmethod
    def neuron_cleaning(model, clean_data_loader, threshold=0.5):
        """神经元清洗: 识别并重置异常神经元"""
        activations = {}

        def hook_fn(name):
            def hook(module, input, output):
                activations[name] = output.detach()
            return hook

        # 注册hook
        hooks = []
        for name, layer in model.named_modules():
            if isinstance(layer, (nn.Conv2d, nn.Linear)):
                hooks.append(layer.register_forward_hook(hook_fn(name)))

        # 收集干净数据的激活
        all_activations = {}
        model.eval()
        with torch.no_grad():
            for batch in clean_data_loader:
                images, _ = batch
                model(images)
                for name, act in activations.items():
                    if name not in all_activations:
                        all_activations[name] = []
                    all_activations[name].append(act.mean(dim=0))

        # 计算平均激活
        avg_activations = {
            k: torch.stack(v).mean(dim=0) for k, v in all_activations.items()
        }

        # 识别低激活神经元(可能是后门神经元)
        dormant_neurons = {}
        for name, act in avg_activations.items():
            low_act = (act < threshold).nonzero(as_tuple=True)
            if len(low_act[0]) > 0:
                dormant_neurons[name] = low_act

        # 移除hooks
        for h in hooks:
            h.remove()

        return dormant_neurons

    @staticmethod
    def trigger_inversion(model, target_class, image_shape=(3, 32, 32),
                         max_iter=1000, lr=0.1):
        """触发器逆向: 反向工程可能的触发器"""
        # 初始化随机触发器
        trigger = torch.randn(image_shape, requires_grad=True)
        mask = torch.ones(image_shape, requires_grad=True)

        optimizer = torch.optim.Adam([trigger, mask], lr=lr)

        for i in range(max_iter):
            optimizer.zero_grad()
            # 生成带触发器的输入
            triggered = trigger * mask
            output = model(triggered.unsqueeze(0))
            # 最大化目标类别的概率
            loss = -output[0, target_class]
            # 正则化: 触发器应该小且稀疏
            loss += 0.01 * torch.norm(mask, p=1)
            loss += 0.001 * torch.norm(trigger, p=2)
            loss.backward()
            optimizer.step()

            if i % 100 == 0:
                print(f"  Iter {i}: loss={loss.item():.4f}")

        return trigger.detach(), mask.detach()
```
