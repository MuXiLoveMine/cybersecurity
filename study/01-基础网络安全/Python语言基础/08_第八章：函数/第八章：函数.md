# 第八章：函数

### 8.1 函数定义

```python
# 基本函数
def greet():
    print("Hello!")

greet()    # 调用
```

### 8.2 函数参数

```python
# 位置参数
def add(a, b):
    return a + b

result = add(3, 5)

# 默认参数
def greet(name, greeting="Hello"):
    print(f"{greeting}, {name}!")

greet("张三")              # Hello, 张三!
greet("李四", "早上好")     # 早上好, 李四!

# 关键字参数
greet(name="王五", greeting="你好")

# 可变参数（收集为元组）
def sum_all(*args):
    total = 0
    for num in args:
        total += num
    return total

print(sum_all(1, 2, 3, 4, 5))

# 关键字可变参数（收集为字典）
def print_info(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")

print_info(name="张三", age=25, city="北京")
```

### 8.3 返回值

```python
# 单个返回值
def add(a, b):
    return a + b

# 多个返回值（元组）
def divide(a, b):
    quotient = a // b
    remainder = a % b
    return quotient, remainder

q, r = divide(10, 3)    # q=3, r=1

# 无 return 或 return 无值，返回 None
def do_nothing():
    pass

print(do_nothing())    # None
```

### 8.4 作用域

```python
x = 10    # 全局变量

def func():
    y = 20    # 局部变量
    global x    # 声明使用全局变量
    x = 15
    print(x, y)

func()
print(x)
```

### 8.5 lambda 匿名函数

```python
# 简单函数用 lambda
square = lambda x: x ** 2
print(square(5))    # 25

add = lambda a, b: a + b

# 常用场景：排序、map、filter
fruits = ["apple", "banana", "cherry"]
fruits.sort(key=lambda x: len(x))    # 按长度排序
```

### 8.6 装饰器（进阶）

```python
# 装饰器是函数的函数，用于增强函数功能
def timer(func):
    def wrapper(*args, **kwargs):
        import time
        start = time.time()
        result = func(*args, **kwargs)
        print(f"{func.__name__} 耗时: {time.time() - start:.4f}s")
        return result
    return wrapper

@timer
def slow_func():
    import time
    time.sleep(1)

slow_func()
```
