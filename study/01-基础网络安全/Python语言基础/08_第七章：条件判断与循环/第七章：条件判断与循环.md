# 第七章：条件判断与循环

### 7.1 if 条件判断

```python
age = 20

if age >= 18:
    print("成年")
    print("可以投票")
elif age >= 13:
    print("青少年")
else:
    print("儿童")

# 多条件
score = 85
if score >= 90:
    grade = "A"
elif score >= 80:
    grade = "B"
elif score >= 60:
    grade = "C"
else:
    grade = "D"

# 三目表达式
result = "及格" if score >= 60 else "不及格"

# and / or / not
if age >= 18 and score >= 60:
    print("符合条件")
```

### 7.2 for 循环

```python
# 遍历列表
fruits = ["apple", "banana", "cherry"]
for fruit in fruits:
    print(fruit)

# 带索引
for i, fruit in enumerate(fruits):
    print(f"{i}: {fruit}")

# range 函数
for i in range(5):        # 0, 1, 2, 3, 4
    print(i)

for i in range(2, 6):     # 2, 3, 4, 5
    print(i)

for i in range(0, 10, 2): # 0, 2, 4, 6, 8
    print(i)

# 遍历字典
person = {"name": "张三", "age": 25}
for key, value in person.items():
    print(f"{key}: {value}")

# 遍历字符串
for char in "hello":
    print(char)
```

### 7.3 while 循环

```python
# 基本用法
count = 0
while count < 5:
    print(count)
    count += 1

# 猜数字（示例）
import random
target = random.randint(1, 100)
while True:
    guess = int(input("猜一个数字："))
    if guess < target:
        print("太小了")
    elif guess > target:
        print("太大了")
    else:
        print("猜对了！")
        break
```

### 7.4 循环控制

```python
# break：跳出整个循环
for i in range(10):
    if i == 5:
        break
    print(i)  # 0, 1, 2, 3, 4

# continue：跳过本次循环
for i in range(10):
    if i % 2 == 0:
        continue
    print(i)  # 1, 3, 5, 7, 9

# else：循环正常结束时执行（break 跳出不执行）
for i in range(5):
    if i == 10:
        break
else:
    print("循环正常结束")
```
