# 网络服务加固

### 6.1 Nginx安全加固

```nginx
# /etc/nginx/nginx.conf 安全配置

user nginx;
worker_processes auto;
pid /run/nginx.pid;

# 安全相关
server_tokens off;                    # 隐藏版本号

events {
    worker_connections 1024;
    multi_accept on;
}

http {
    # 基本安全头
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    add_header Content-Security-Policy "default-src 'self'" always;
    add_header Strict-Transport-Security "max-age=63072000; includeSubDomains; preload" always;
    add_header Permissions-Policy "geolocation=(), microphone=(), camera=()" always;

    # 隐藏服务器信息
    proxy_hide_header X-Powered-By;
    proxy_hide_header Server;
    fastcgi_hide_header X-Powered-By;

    # 限制请求大小
    client_max_body_size 10m;
    client_body_buffer_size 16k;
    client_header_buffer_size 4k;
    large_client_header_buffers 4 4k;

    # 超时设置
    client_body_timeout 30;
    client_header_timeout 30;
    keepalive_timeout 30 30;
    send_timeout 30;

    # 限制连接数
    limit_conn_zone $binary_remote_addr zone=conn_limit:10m;
    limit_req_zone $binary_remote_addr zone=req_limit:10m rate=10r/s;

    # SSL配置
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305;
    ssl_prefer_server_ciphers on;
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 1d;
    ssl_session_tickets off;

    # OCSP Stapling
    ssl_stapling on;
    ssl_stapling_verify on;

    server {
        listen 80;
        server_name example.com;
        return 301 https://$server_name$request_uri;
    }

    server {
        listen 443 ssl http2;
        server_name example.com;

        ssl_certificate /etc/nginx/ssl/example.com.crt;
        ssl_certificate_key /etc/nginx/ssl/example.com.key;

        # 应用限制
        limit_conn conn_limit 10;
        limit_req zone=req_limit burst=20 nodelay;

        # 防止目录遍历
        location ~* /\.(git|svn|hg) {
            deny all;
            return 404;
        }

        # 防止访问敏感文件
        location ~* \.(env|sql|bak|old|tmp|log|conf)$ {
            deny all;
            return 404;
        }

        # 防止PHP上传目录执行
        location ~* /uploads/.*\.php$ {
            deny all;
        }

        # 安全的PHP配置
        location ~ \.php$ {
            fastcgi_pass unix:/var/run/php/php8.1-fpm.sock;
            fastcgi_index index.php;
            include fastcgi_params;
            fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;

            # 防止零日漏洞
            try_files $uri =404;
        }
    }
}
```

### 6.2 Apache安全加固

```apache
# /etc/apache2/conf-enabled/security.conf

# 隐藏版本信息
ServerTokens Prod
ServerSignature Off

# 添加安全头
<IfModule mod_headers.c>
    Header always set X-Frame-Options "SAMEORIGIN"
    Header always set X-Content-Type-Options "nosniff"
    Header always set X-XSS-Protection "1; mode=block"
    Header always set Referrer-Policy "strict-origin-when-cross-origin"
    Header always set Strict-Transport-Security "max-age=63072000; includeSubDomains"
    Header unset X-Powered-By
    Header unset Server
</IfModule>

# 限制请求大小
LimitRequestBody 10485760

# 防止目录列表
Options -Indexes

# 禁用TRACE
TraceEnable Off

# 限制方法
<LimitExcept GET POST HEAD OPTIONS>
    Require all denied
</LimitExcept>

# SSL配置
<IfModule mod_ssl.c>
    SSLProtocol -all +TLSv1.2 +TLSv1.3
    SSLCipherSuite ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384
    SSLHonorCipherOrder on
    SSLCompression off
    SSLSessionTickets off
    SSLUseStapling on
    SSLStaplingCache shmcb:/var/run/ocsp(128000)
</IfModule>

# 防止访问敏感文件
<FilesMatch "(^\.|\.env$|\.sql$|\.bak$|\.log$|\.conf$)">
    Require all denied
</FilesMatch>

# 防止上传执行
<Directory "/var/www/html/uploads">
    php_admin_flag engine off
    <FilesMatch "\.ph(p[345]?|t|tml)$">
        Require all denied
    </FilesMatch>
</Directory>
```

### 6.3 MySQL安全加固

```sql
-- 安全配置 /etc/mysql/mysql.conf.d/mysqld.cnf

[mysqld]
# 基本安全
bind-address = 127.0.0.1          -- 仅本地监听
port = 3306
skip-symbolic-links              -- 禁用符号链接

# 认证安全
default-authentication-plugin = mysql_native_password
default_password_lifetime = 90   -- 密码90天过期
password_history = 5             -- 保留5个历史密码

# SSL/TLS
ssl-ca = /etc/mysql/ssl/ca.pem
ssl-cert = /etc/mysql/ssl/server-cert.pem
ssl-key = /etc/mysql/ssl/server-key.pem
require_secure_transport = ON    -- 强制加密连接

# 日志审计
log_error = /var/log/mysql/error.log
slow_query_log = 1
slow_query_log_file = /var/log/mysql/slow.log
long_query_time = 2

# 安全SQL模式
sql_mode = STRICT_TRANS_TABLES,NO_ZERO_IN_DATE,NO_ZERO_DATE,ERROR_FOR_DIVISION_BY_ZERO,NO_ENGINE_SUBSTITUTION

-- 运行mysql_secure_installation
-- mysql_secure_installation

-- 安全检查SQL
-- 查看用户权限
SELECT user, host, authentication_string FROM mysql.user;

-- 移除匿名用户
DELETE FROM mysql.user WHERE User='';

-- 移除test数据库
DROP DATABASE IF EXISTS test;
DELETE FROM mysql.db WHERE Db='test' OR Db='test\\_%';

-- 限制用户权限
SHOW GRANTS FOR 'user'@'localhost';
REVOKE ALL PRIVILEGES ON *.* FROM 'user'@'localhost';
GRANT SELECT, INSERT, UPDATE, DELETE ON mydb.* TO 'user'@'localhost';
```

### 6.4 Redis安全加固

```bash
# /etc/redis/redis.conf 安全配置

# 网络安全
bind 127.0.0.1 ::1                 # 仅本地监听
protected-mode yes                  # 保护模式
port 6379                           # 默认端口（建议修改）

# 认证
requirepass YourStrongPassword123! # 设置密码

# 命令重命名/禁用
rename-command FLUSHALL ""
rename-command FLUSHDB ""
rename-command CONFIG ""
rename-command KEYS ""
rename-command DEBUG ""
rename-command SHUTDOWN ""
rename-command DEL ""

# 访问控制
# 使用ACL（Redis 6+）
# user default on >password ~* &* +@all -@dangerous

# 限制最大客户端
maxclients 100

# 内存限制
maxmemory 1gb
maxmemory-policy allkeys-lru

# 持久化安全
dir /var/lib/redis
dbfilename dump.rdb
appendonly yes
appendfilename "appendonly.aof"

# 日志
loglevel notice
logfile /var/log/redis/redis-server.log

# 文件权限
# chown redis:redis /etc/redis/redis.conf
# chmod 600 /etc/redis/redis.conf
# chown -R redis:redis /var/lib/redis
# chmod 700 /var/lib/redis
```
