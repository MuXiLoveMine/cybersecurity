# 第七章：数组

### 7.1 数组定义

```bash
# 方式一：直接赋值
arr=("apple" "banana" "cherry")

# 方式二：逐个赋值
arr[0]="apple"
arr[1]="banana"
arr[2]="cherry"

# 方式三：使用键值对（关联数组，需先声明）
declare -A dict
dict["name"]="张三"
dict["age"]=25
```

### 7.2 访问数组元素

```bash
arr=("apple" "banana" "cherry")

# 访问单个元素
echo ${arr[0]}    # apple
echo ${arr[1]}    # banana

# 访问所有元素
echo ${arr[@]}    # apple banana cherry
echo ${arr[*]}    # apple banana cherry

# 数组长度
echo ${#arr[@]}   # 3

# 数组索引列表
echo ${!arr[@]}   # 0 1 2

# 切片
echo ${arr[@]:1:2}   # 从索引 1 开始取 2 个 → banana cherry
```

### 7.3 数组操作

```bash
arr=("apple" "banana" "cherry")

# 添加元素
arr+=("date")        # 在末尾添加
arr[4]="elderberry"  # 指定索引添加

# 删除元素
unset arr[1]         # 删除索引 1 的元素
echo ${arr[@]}       # apple cherry date elderberry

# 拼接数组
arr1=(a b c)
arr2=(d e f)
arr3=("${arr1[@]}" "${arr2[@]}")
echo ${arr3[@]}      # a b c d e f

# 遍历数组
for item in "${arr[@]}"; do
    echo "元素：$item"
done

# 带索引遍历
for i in "${!arr[@]}"; do
    echo "索引 $i：${arr[$i]}"
done
```

### 7.4 关联数组

```bash
declare -A user
user["name"]="张三"
user["age"]=25
user["city"]="北京"

# 访问
echo ${user["name"]}    # 张三

# 所有值
echo ${user[@]}

# 所有键
echo ${!user[@]}

# 遍历
for key in "${!user[@]}"; do
    echo "$key: ${user[$key]}"
done
```
