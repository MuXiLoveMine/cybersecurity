# 第八章：字符串处理

### 8.1 字符串基本操作

```bash
str="Hello World"

# 长度
echo ${#str}    # 11

# 拼接
str2="${str}, How are you?"
echo $str2

# 重复（利用 printf）
repeat=$(printf "%0.s-" {1..10})
echo $repeat    # ----------
```

### 8.2 字符串截取

```bash
str="Hello World"

# 从左截取：${str:起始位置:长度}
echo ${str:0:5}     # Hello
echo ${str:6}       # World（从第6位到结尾）
echo ${str: -5}     # World（从右数第5位，冒号后必须有空格）

# 删除匹配
# # 从左删除最短匹配
echo ${str#Hello }   # World
# ## 从左删除最长匹配
echo ${str##* }      # World
# % 从右删除最短匹配
echo ${str% World}   # Hello
# %% 从右删除最长匹配
echo ${str%% *}      # Hello
```

### 8.3 字符串替换

```bash
str="Hello World Hello"

# 替换第一个匹配
echo ${str/Hello/Hi}    # Hi World Hello

# 替换所有匹配
echo ${str//Hello/Hi}   # Hi World Hi

# 删除匹配（替换为空）
echo ${str//Hello/}     #  World 
```

### 8.4 字符串大小写转换

```bash
str="Hello World"

# 全部转大写
echo ${str^^}    # HELLO WORLD

# 全部转小写
echo ${str,,}    # hello world

# 首字母大写
echo ${str^}     # Hello World

# 首字母小写
echo ${str,}     # hello World
```

### 8.5 字符串判断

```bash
str="hello world"

# 是否包含子串（用 [[ ]] 和模式匹配）
if [[ "$str" == *"hello"* ]]; then
    echo "包含 hello"
fi

# 是否以指定开头
if [[ "$str" == h* ]]; then
    echo "以 h 开头"
fi

# 是否以指定结尾
if [[ "$str" == *d ]]; then
    echo "以 d 结尾"
fi

# 空字符串判断
if [ -z "$str" ]; then
    echo "空字符串"
fi

if [ -n "$str" ]; then
    echo "非空字符串"
fi
```

### 8.6 字符串与 awk/sed 配合

```bash
str="hello world"

# 使用 awk 截取第 2 个字段
echo $str | awk '{print $2}'    # world

# 使用 sed 替换
echo $str | sed 's/world/Shell/'   # hello Shell

# 使用 cut 截取
echo "a,b,c,d" | cut -d, -f2    # b
```
