# 第九章：现代安全防护机制

### 9.1 编译期防护

```bash
# GCC 编译选项安全对照

# 栈保护（Canary）
gcc -fstack-protector-all -o binary source.c

# 地址空间随机化（PIE）
gcc -pie -fPIC -o binary source.c

# RELRO（GOT 保护）
gcc -Wl,-z,relro,-z,now -o binary source.c

# NX（不可执行栈）
gcc -Wl,-z,noexecstack -o binary source.c

# Fortify 源
gcc -D_FORTIFY_SOURCE=2 -O2 -o binary source.c

# 查看编译选项
checksec --file=binary
# 输出示例：
# RELRO: Full RELRO
# Stack Canary: Yes
# NX: Enabled
# PIE: Enabled
```

### 9.2 运行期防护

| 机制 | 平台 | 说明 | 绕过技术 |
|------|------|------|---------|
| **SELinux** | Linux | 强制访问控制 | 策略配置不当 |
| **AppArmor** | Linux | 基于路径的访问控制 | 模式切换 |
| **seccomp** | Linux | 限制系统调用 | 需内核支持 |
| **ASLR** | 通用 | 地址随机化 | 信息泄露 |
| **CFI** | 通用 | 控制流完整性 | 间接跳转 |
| **CFG** | Windows | 控制流卫士 | 仅验证间接调用目标 |
| **PAC** | ARM | 指针认证 | 签名密钥泄露 |

### 9.3 seccomp 沙箱

```c
#include <linux/seccomp.h>
#include <linux/filter.h>
#include <sys/prctl.h>

// seccomp 过滤规则示例
// 仅允许 read, write, exit, sigreturn
struct sock_filter filter[] = {
    // 检查架构
    BPF_STMT(BPF_LD | BPF_W | BPF_ABS,
             offsetof(struct seccomp_data, arch)),
    BPF_JUMP(BPF_JMP | BPF_JEQ | BPF_K,
             AUDIT_ARCH_X86_64, 1, 0),
    BPF_STMT(BPF_RET | BPF_K, SECCOMP_RET_KILL),

    // 检查系统调用号
    BPF_STMT(BPF_LD | BPF_W | BPF_ABS,
             offsetof(struct seccomp_data, nr)),
    // 允许 read (0)
    BPF_JUMP(BPF_JMP | BPF_JEQ | BPF_K, __NR_read, 4, 0),
    // 允许 write (1)
    BPF_JUMP(BPF_JMP | BPF_JEQ | BPF_K, __NR_write, 3, 0),
    // 允许 exit (60)
    BPF_JUMP(BPF_JMP | BPF_JEQ | BPF_K, __NR_exit, 2, 0),
    // 允许 exit_group (231)
    BPF_JUMP(BPF_JMP | BPF_JEQ | BPF_K, __NR_exit_group, 1, 0),
    // 默认拒绝
    BPF_STMT(BPF_RET | BPF_K, SECCOMP_RET_KILL),
    // 允许
    BPF_STMT(BPF_RET | BPF_K, SECCOMP_RET_ALLOW),
};

// 启用 seccomp
struct sock_fprog prog = {
    .len = sizeof(filter) / sizeof(filter[0]),
    .filter = filter,
};
prctl(PR_SET_NO_NEW_PRIVS, 1, 0, 0, 0);
prctl(PR_SET_SECCOMP, SECCOMP_MODE_FILTER, &prog);
```
