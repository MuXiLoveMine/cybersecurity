# 内网信息收集自动化

### 10.1 PowerShell 自动化信息收集

```powershell
# 内网信息收集一键脚本
function Invoke-InternalRecon {
    <#
    .SYNOPSIS
        内网信息收集自动化脚本
    #>
    
    $report = @{}
    
    # 1. 系统信息
    $report["System"] = @{
        Hostname = $env:COMPUTERNAME
        OS = (Get-WmiObject Win32_OperatingSystem).Caption
        Architecture = $env:PROCESSOR_ARCHITECTURE
        InstallDate = (Get-WmiObject Win32_OperatingSystem).InstallDate
    }
    
    # 2. 网络信息
    $report["Network"] = @{
        IPAddresses = (Get-NetIPAddress -AddressFamily IPv4 | 
                       Where-Object {$_.IPAddress -notlike "127.*"} | 
                       Select-Object -ExpandProperty IPAddress)
        DNS = (Get-DnsClientServerAddress -AddressFamily IPv4 | 
               Select-Object -ExpandProperty ServerAddresses -Unique)
        ARPTable = (Get-NetNeighbor -State Reachable | 
                    Select-Object IPAddress, LinkAddress)
        Routes = (Get-NetRoute -DestinationPrefix "0.0.0.0/0" | 
                  Select-Object NextHop, InterfaceAlias)
        ListeningPorts = (Get-NetTCPConnection -State Listen | 
                         Select-Object LocalAddress, LocalPort, OwningProcess)
    }
    
    # 3. 用户与会话
    $report["Users"] = @{
        LocalUsers = (Get-LocalUser | Select-Object Name, Enabled, LastLogon)
        LoggedOn = (query user 2>$null)
        LocalAdmins = (Get-LocalGroupMember -Group "Administrators" | 
                       Select-Object Name, ObjectClass)
    }
    
    # 4. 进程与服务
    $report["Processes"] = @{
        Running = (Get-Process | Sort-Object CPU -Descending | 
                   Select-Object -First 20 Name, Id, CPU, WS)
        Services = (Get-Service | Where-Object {$_.Status -eq "Running"} | 
                    Select-Object Name, DisplayName)
        ScheduledTasks = (Get-ScheduledTask | Where-Object {$_.State -eq "Ready"} | 
                          Select-Object TaskName, TaskPath)
    }
    
    # 5. 共享信息
    $report["Shares"] = @{
        Local = (Get-SmbShare | Select-Object Name, Path, Description)
        Sessions = (Get-SmbSession 2>$null | Select-Object ClientUserName, ClientComputerName)
        Connections = (Get-SmbConnection 2>$null | Select-Object ServerName, ShareName)
    }
    
    # 6. 安装的软件
    $report["Software"] = (Get-ItemProperty HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\* | 
                           Select-Object DisplayName, DisplayVersion, Publisher) 
    
    return $report
}

# 执行并输出
$recon = Invoke-InternalRecon
$recon | ConvertTo-Json -Depth 3
```

### 10.2 WMIC 信息收集

```bash
# WMIC 常用信息收集命令

# 系统信息
wmic os get Caption, BuildNumber, OSArchitecture, InstallDate
wmic computersystem get Name, Domain, Manufacturer, Model
wmic baseboard get Product, Manufacturer
wmic bios get SerialNumber, SMBIOSBIOSVersion

# 用户信息
wmic useraccount get Name, SID, AccountType, Disabled
wmic group get Name, SID, Description
wmic netlogin get Name, LogonServer, Privileges

# 进程与服务
wmic process list brief
wmic process where "name='powershell.exe'" get ProcessId, CommandLine
wmic service where "State='Running'" get Name, DisplayName, PathName
wmic service where "StartMode='Auto'" get Name, State, PathName

# 网络信息
wmic nicconfig where "IPEnabled='TRUE'" get IPAddress, MACAddress, DefaultIPGateway
wmic netuse get Name, RemotePath, UserName
wmic share get Name, Path, Description

# 补丁信息
wmic qfe list brief
wmic qfe where "HotFixID='KB5008212'" get HotFixID, InstalledOn

# 自启动项
wmic startup get Caption, Command, Location
```

### 10.3 BloodHound 深度分析

```cypher
// BloodHound 高级 Cypher 查询

// 1. 查找所有到 Domain Admin 的路径（限制跳数）
MATCH (n:User),(m:Group {name:"DOMAIN ADMINS@CORP.LOCAL"}),
p=shortestPath((n)-[:MemberOf|HasSession|AdminTo|AllExtendedRights|GenericAll|GenericWrite|Owns|WriteDacl|WriteOwner|ForceChangePassword|GPLink|AllowedToDelegate|AddMember*1..6]->(m))
RETURN p

// 2. 查找所有具有 DCSync 权限的用户
MATCH (n)-[:GetChanges|GetChangesAll]->(m:Domain)
RETURN n.name, m.name

// 3. 查找可以通过 Kerberoasting 获取的账户
MATCH (n:User) WHERE n.kerberoastable=true
RETURN n.name, n.serviceprincipalnames, n.enabled

// 4. 查找所有 AS-REP Roastable 用户（不需要预认证）
MATCH (n:User) WHERE n.dontreqpreauth=true
RETURN n.name

// 5. 查找可以控制其他用户的用户（GenericAll/GenericWrite/ForceChangePassword）
MATCH (n:User)-[r:GenericAll|GenericWrite|ForceChangePassword|WriteDacl|WriteOwner]->(m:User)
RETURN n.name, type(r), m.name

// 6. 查找所有有管理员权限的计算机
MATCH (n:User)-[:AdminTo]->(m:Computer)
RETURN n.name, m.name

// 7. 查找 GPO 权限滥用
MATCH (n)-[:GPLink]->(m:GPO)
WHERE n:User OR n:Group
RETURN n.name, m.name

// 8. 查找非约束委派
MATCH (n)-[:UnconstrainedDelegation]->(m)
RETURN n.name, m.name
```
