### 13.1 社交媒体信息收集

```bash
# Sherlock - 社交媒体账户搜索
# 安装
pip install sherlock-project

# 搜索用户名
sherlock user target_username
sherlock user john_doe jane_doe

# 输出到文件
sherlock user target_username --timeout 5 --print-found

# 指定平台
sherlock user target_username --site GitHub LinkedIn Twitter

# Socialscan - 另一个社交媒体搜索工具
```

### 13.2 邮箱信息收集

```bash
# hunter.io - 邮箱收集和验证
# API方式
curl -s "https://api.hunter.io/v2/domain-search?domain=example.com&api_key=YOUR_KEY" | jq

# 验证邮箱
curl -s "https://api.hunter.io/v2/email-verifier?email=user@example.com&api_key=YOUR_KEY" | jq

# haveibeenpwned - 检查邮箱是否泄露
curl -s "https://haveibeenpwned.com/api/v3/breachedaccount/user@example.com" \
    -H "hibp-api-key: YOUR_KEY"

# Holehe - 检查邮箱注册的各种服务
pip install holehe
holehe email user@example.com

# 邮箱格式猜测
# 常见格式: first.last@company.com, flast@company.com, firstl@company.com
# 使用nmap脚本
nmap --script smtp-enum-users -p 25,465,587 mail.example.com
```

### 13.3 人员信息收集与OSINT关联

```python
#!/usr/bin/env python3
"""综合人员信息收集工具"""

import requests
import json
from collections import defaultdict

class PeopleOSINT:
    def __init__(self):
        self.results = defaultdict(list)

    def linkedin_search(self, company_name):
        """LinkedIn公司员工搜索"""
        # 使用Google Dork搜索LinkedIn
        dorks = [
            f'site:linkedin.com/in "{company_name}"',
            f'site:linkedin.com/in "{company_name}" "engineer"',
            f'site:linkedin.com/in "{company_name}" "developer"',
            f'site:linkedin.com/in "{company_name}" "manager"',
            f'site:linkedin.com/in "{company_name}" "CTO"',
            f'site:linkedin.com/in "{company_name}" "security"',
        ]
        # 实际使用时需要配合搜索引擎API
        return dorks

    def email_pattern_detect(self, domain, known_emails):
        """检测邮箱命名模式"""
        patterns = []
        for email in known_emails:
            prefix = email.split("@")[0]
            # 分析模式
            if "." in prefix:
                patterns.append("first.last")
            if "_" in prefix:
                patterns.append("first_last")
            if len(prefix) == 1:
                patterns.append("first_initial")
            # ...更多模式分析
        return patterns

    def generate_emails(self, domain, names, pattern):
        """根据模式和姓名列表生成邮箱"""
        emails = []
        for name in names:
            parts = name.lower().split()
            first = parts[0] if parts else ""
            last = parts[-1] if len(parts) > 1 else ""
            if pattern == "first.last":
                emails.append(f"{first}.{last}@{domain}")
            elif pattern == "flast":
                emails.append(f"{first[0]}{last}@{domain}")
            elif pattern == "firstl":
                emails.append(f"{first}{last[0]}@{domain}")
            elif pattern == "first":
                emails.append(f"{first}@{domain}")
        return emails
```
