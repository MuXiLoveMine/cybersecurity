# 子域名枚举

### 4.1 Subfinder

```bash
# Subfinder - 快速被动子域名发现
# 安装
go install github.com/projectdiscovery/subfinder/v2/cmd/subfinder@latest

# 基本使用
subfinder -d example.com

# 输出到文件
subfinder -d example.com -o subdomains.txt

# 使用多个数据源
subfinder -d example.com -sources virustotal,crtsh,securitytrails

# 递归子域名
subfinder -d example.com -recursive

# 从文件批量查询
subfinder -dL domains.txt -o all_subdomains.txt

# JSON输出
subfinder -d example.com -json -o subdomains.json

# 配置API密钥(提升效果)
# ~/.config/subfinder/provider-config.yaml
# virustotal:
#   - API_KEY_1
#   - API_KEY_2
# securitytrails:
#   - API_KEY
# shodan:
#   - API_KEY
```

### 4.2 Amass

```bash
# Amass - 综合子域名枚举(被动+主动)
# 安装
go install -v github.com/owasp-amass/amass/v4/...@master

# 被动枚举(不接触目标)
amass enum -passive -d example.com

# 主动枚举(包括区域传送等)
amass enum -active -d example.com

# 暴力枚举
amass enum -brute -d example.com

# 综合枚举
amass enum -d example.com -src

# 输出
amass enum -d example.com -o subdomains.txt
amass enum -d example.com -json subdomains.json

# 可视化
amass viz -d3 -d example.com -o amass_d3.html

# IP范围发现
amass intel -d example.com -ip

# 配置(~/.config/amass/config.yaml)
# data_sources:
#   - virustotal
#   - crtsh
#   - securitytrails
```

### 4.3 httpx

```bash
# httpx - HTTP探测和指纹
# 安装
go install github.com/projectdiscovery/httpx/cmd/httpx@latest

# 基本探测
cat subdomains.txt | httpx

# 详细输出
cat subdomains.txt | httpx -title -status-code -tech-detect -content-length

# 只显示存活的
cat subdomains.txt | httpx -silent

# 截图(需要chromium)
cat subdomains.txt | httpx -screenshot

# 输出JSON
cat subdomains.txt | httpx -json -o results.json

# 自定义Header和路径
cat subdomains.txt | httpx -path /admin -path /api -status-code

# 探测特定端口
cat subdomains.txt | httpx -ports 80,443,8080,8443

# 组合使用
subfinder -d example.com -silent | httpx -title -status-code -tech-detect
```
