# 被动侦察

### 2.1 Whois查询

```bash
# 基本Whois查询
whois example.com

# 使用在线API
curl -s "https://api.hackertarget.com/whois/?q=example.com"

# 查询IP的Whois
whois 93.184.216.34

# 查询DNS记录
dig example.com ANY
dig +short example.com A
dig +short example.com MX
dig +short example.com NS
dig +short example.com TXT
dig +short example.com CNAME

# 反向DNS
dig -x 93.184.216.34 +short

# DNS区域传输尝试(通常被拒绝)
dig axfr @ns1.example.com example.com

# 使用nslookup
nslookup example.com
nslookup -type=mx example.com
nslookup -type=txt example.com
```

### 2.2 Shodan搜索

```bash
# Shodan CLI安装
pip install shodan

# 设置API Key
shodan init YOUR_API_KEY

# 搜索特定IP
shodan host 93.184.216.34

# 搜索关键词
shodan search "apache"
shodan search "nginx" country:CN
shodan search "tomcat" port:8080
shodan search "redis" port:6379 -auth
shodan search "mongodb" port:27017

# 常用搜索语法
shodan search "http.title:\"Dashboard\""          # 页面标题
shodan search "http.component:\"wordpress\""      # 技术组件
shodan search "ssl.cert.subject.CN:example.com"  # SSL证书
shodan search "product:\"Apache httpd\""          # 产品名
shodan search "version:\"2.4.49\""                # 版本号
shodan search "hostname:example.com"              # 主机名
shodan search "org:\"Amazon\""                    # 组织
shodan search "city:\"Shanghai\""                 # 城市
shodan search "port:3389 country:CN"              # RDP服务
shodan search "port:22 has_screenshot:true"       # SSH带截图

# Facet统计
shodan stats "port:27017" --facets country,org

# 导出结果
shodan download results "apache country:CN" --limit 100
shodan parse --fields ip_str,port,org results.json.gz
```

### 2.3 Censys搜索

```bash
# Censys CLI安装
pip install censys

# 配置API
censys config

# 搜索主机
censys search "93.184.216.34" --index hosts

# 按服务搜索
censys search "services.port: 3306" --index hosts
censys search "services.software.product: nginx" --index hosts

# 证书搜索
censys search "parsed.names: example.com" --index certificates

# 查看主机详情
censys view 93.184.216.34 --index hosts
```

### 2.4 证书透明日志(crt.sh)

```bash
# 查询域名的SSL证书历史
# 这可以发现子域名(因为每个证书需要记录)

# Web查询
# https://crt.sh/?q=example.com

# API查询(JSON格式)
curl -s "https://crt.sh/?q=example.com&output=json" | \
    jq -r '.[].name_value' | \
    sort -u

# 提取所有子域名
curl -s "https://crt.sh/?q=%.example.com&output=json" | \
    jq -r '.[].name_value' | \
    sed 's/\*\.//g' | \
    sort -u | \
    grep -v " "

# 自动化脚本
#!/bin/bash
domain=$1
curl -s "https://crt.sh/?q=%.${domain}&output=json" | \
    jq -r '.[].name_value' | \
    sed 's/\*\.//g' | \
    sort -u
```

### 2.5 GitHub Dorking

```bash
# GitHub代码搜索 - 查找泄露的凭证和配置

# 搜索API密钥
# GitHub搜索: extension:env AWS_SECRET_ACCESS_KEY
# GitHub搜索: extension:yml password
# GitHub搜索: filename:id_rsa
# GitHub搜索: filename:.htpasswd
# GitHub搜索: "api_key" extension:py
# GitHub搜索: "BEGIN RSA PRIVATE KEY"

# 使用GitHub CLI
gh search code "AWS_SECRET_ACCESS_KEY" --language python
gh search code "BEGIN PRIVATE KEY" --extension pem
gh search code "password" --extension env
gh search code "api_key" --extension yaml

# 使用truffleHog扫描仓库
pip install trufflehog
trufflehog github --org=example-org

# 使用GitLeaks
gitleaks detect --repo https://github.com/target/repo --report report.json

# 使用GitHub Dorking工具
pip install gitdorks-go
gitdorks -d example.com

# 常见Dork模式
"extension:env" + "AWS"
"extension:yaml" + "password"
"extension:json" + "private_key"
"extension:py" + "api_key"
"filename:id_rsa" + ""
"filename:docker-compose" + "password"
"filename:.gitconfig" + ""
"filename:wp-config.php" + "DB_PASSWORD"
"filename:settings.py" + "SECRET_KEY"
```
