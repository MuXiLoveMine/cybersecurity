### 二、子域名枚举 - 高级命令与技巧

#### 2.1 工具组合使用

**被动 + 主动 组合流程**
```bash
# 第1步: 多工具被动收集，结果去重
subfinder -d target.com -all -o sub_passive.txt &
amass enum -passive -d target.com -o amass_passive.txt &
findomain -t target.com -q -o findomain.txt &
wait

# 合并去重
cat sub_passive.txt amass_passive.txt findomain.txt | sort -u > all_subdomains.txt
wc -l all_subdomains.txt

# 第2步: 存活验证（HTTP/HTTPS）
httpx -l all_subdomains.txt -title -tech-detect -status-code -o alive_subdomains.txt -t 100

# 第3步: 递归子域名（对找到的子域继续枚举）
cat alive_subdomains.txt | awk -F'[][]' '{print $2}' | sed 's/https\?:\/\///' | sort -u > verified_subs.txt
subfinder -dL verified_subs.txt -recursive -o recursive_subs.txt

# 第4步: DNS解析验证（排除已失效）
puredns resolve all_subdomains.txt -r resolvers.txt -w resolved.txt

# 第5步: 子域名接管检测
subjack -w resolved.txt -t 50 -timeout 30 -ssl -v -o takeover.txt
nuclei -l resolved.txt -t takeovers/ -o nuclei_takeover.txt
```

### 三、端口扫描 - 常用命令与高级用法

#### 3.1 Nmap 常用命令速查

```bash
# 基础扫描
nmap 10.0.0.1                          # 默认1000个常用端口
nmap -p 1-65535 10.0.0.1               # 全端口扫描
nmap -p- 10.0.0.1                      # 全端口（简写）

# 扫描速度与精度
nmap -T4 -p- 10.0.0.1                  # T4速度（1-5级）
nmap -T5 -p- 10.0.0.1                  # 最快（可能漏报）
nmap -sT -p- 10.0.0.1                  # TCP Connect扫描（无需root）
nmap -sS -p- 10.0.0.1                  # SYN半开扫描（需root，更隐蔽）
nmap -sU --top-ports 100 10.0.0.1      # UDP端口扫描（慢）

# 服务与版本探测
nmap -sV -p 80,443 10.0.0.1            # 服务版本探测
nmap -sV --version-intensity 5 10.0.0.1 # 版本探测强度(0-9)
nmap -O 10.0.0.1                       # 操作系统探测
nmap -A 10.0.0.1                       # 全面扫描（-sV -sC -O）

# 脚本扫描
nmap --script=vuln 10.0.0.1            # 漏洞扫描脚本
nmap --script=auth,brute 10.0.0.1      # 认证和暴力破解
nmap --script=http-enum 10.0.0.1       # HTTP目录枚举
nmap --script=ssl-* 10.0.0.1           # SSL相关脚本
nmap --script=smb-* 10.0.0.1           # SMB相关脚本
nmap --script=vulners 10.0.0.1         # Vulners漏洞数据库查询

# 目标指定
nmap 10.0.0.0/24                       # C段
nmap 10.0.0.1-100                      # 范围
nmap -iL targets.txt                   # 从文件读取
nmap -iR 100                           # 随机100个IP

# 输出格式
nmap -oN scan.txt 10.0.0.1             # 普通文本
nmap -oX scan.xml 10.0.0.1             # XML格式
nmap -oG scan.gnmap 10.0.0.1           # Grepable格式
nmap -oA scan 10.0.0.1                 # 所有格式

# 防火墙/IDS绕过
nmap -f 10.0.0.1                       # 分片
nmap -D RND:5 10.0.0.1                 # 诱饵IP

nmap --data-length 25 10.0.0.1         # 附加随机数据
```

#### 6.2 子域名接管检测自动化

```python
#!/usr/bin/env python3
"""
子域名接管检测脚本 - 结合 subjack + nuclei + 手动验证
"""
import subprocess
import requests
import re

def check_takeover(subdomains_file, output_file):
    results = []
    
    # 1. 使用 subjack 初步检测
    print("[*] 运行 subjack...")
    try:
        result = subprocess.run(
            ['subjack', '-w', subdomains_file, '-t', '50', '-timeout', '30', '-ssl', '-v'],
            capture_output=True, text=True, timeout=300
        )
        for line in result.stdout.split('\n'):
            if 'VULNERABLE' in line or 'vulnerable' in line.lower():
                results.append(('subjack', line.strip()))
    except Exception as e:
        print(f"subjack 错误: {e}")
    
    # 2. 指纹匹配检测
    print("[*] 指纹匹配检测...")
    fingerprints = {
        'AWS S3': ['NoSuchBucket', 'The specified bucket does not exist'],
        'GitHub Pages': ['There isn\'t a GitHub Pages site here.', '404'],
        'Heroku': ['There is no app configured at that hostname', 'No such app'],
        'Shopify': ['Sorry, this shop is currently unavailable'],
        'Fastly': ['Fastly error: unknown domain'],
        'Ghost': ['The thing you were looking for is no longer here, or never was'],
        'Azure': ['is not a valid Azure website name'],
        'CloudFront': ['Bad Request - ERROR: The request could not be satisfied'],
        'WordPress': ['Do you want to register'],
        'Surge.sh': ['project not found'],
    }
    
    with open(subdomains_file) as f:
        domains = [line.strip() for line in f if line.strip()]
    
    for domain in domains[:100]:  # 限制数量避免过慢
        try:
            for proto in ['https://', 'http://']:
                try:
                    resp = requests.get(f"{proto}{domain}", timeout=5, allow_redirects=True)
                    text = resp.text[:2000]
                    for service, patterns in fingerprints.items():
                        if all(p in text for p in patterns[:1]):  # 匹配第一个模式
                            results.append(('fingerprint', f"{proto}{domain} - 疑似{service}接管"))
                    break
                except:
                    continue
        except:
            pass
    
    # 保存结果
    with open(output_file, 'w') as f:
        f.write("=== 子域名接管检测结果 ===\n\n")
        for source, result in results:
            f.write(f"[{source}] {result}\n")
    
    print(f"[*] 检测完成，发现 {len(results)} 个潜在接管点")
    return results

if __name__ == '__main__':
    import sys
    if len(sys.argv) != 3:
        print("Usage: python3 takeover_check.py <subdomains.txt> <output.txt>")
        sys.exit(1)
    check_takeover(sys.argv[1], sys.argv[2])
```
