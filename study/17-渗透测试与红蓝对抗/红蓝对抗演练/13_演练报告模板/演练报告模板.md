#### 2.2 红队核心战术技巧

**OPSEC（操作安全）原则**
```
红队OPSEC核心原则:
1. 最小化痕迹原则
   - 尽量使用内存执行，减少文件落地
   - 利用系统自带工具（LOLBins）
   - 减少网络流量特征
   - 模拟正常用户行为

2. 通信安全原则
   - 使用加密通信信道
   - 流量特征伪装（模仿正常业务流量）
   - 合理设置心跳间隔
   - 多C2节点冗余

3. 身份伪装原则
   - 使用窃取的合法凭证
   - 模拟正常用户登录时间
   - 利用合法服务进行通信
   - 避免使用已知恶意IP

4. 行动克制原则
   - 避免不必要的系统修改
   - 控制操作频率和范围
   - 优先使用低权限账户
   - 敏感操作前评估风险
```

**C2 基础设施搭建技巧**
```
红队C2基础设施架构:

1. 重定向器（Redirector）架构
   攻击者 → 重定向器 → C2服务器
   - 使用Apache/Nginx反向代理
   - 基于User-Agent/URI过滤
   - 只放行合法Beacon流量
   - 其他流量重定向到正常网站

2. 域前置（Domain Fronting）
   - 使用CDN服务隐藏真实C2
   - 利用CDN的多域名特性
   - 流量看起来是访问合法域名
   - 注意：已被多家CDN禁止

3. 云函数C2
   - 使用AWS Lambda / 阿里云函数计算
   - 利用云函数作为C2代理
   - IP动态变化，难以封禁
   - 流量混入正常云服务流量

4. C2通信伪装
   - 模仿正常Web应用流量
   - 使用真实网站的404页面做Beacon
   - 证书使用真实域名的SSL证书
   - User-Agent使用常见浏览器UA

5. 多C2冗余架构
   - 主C2 + 备用C2
   - 不同协议（HTTP/DNS/TCP）
   - 不同地理位置
   - 自动故障转移
```

#!/usr/bin/env python3
"""
红蓝对抗 ATT&CK 覆盖度统计工具
"""
import json
from collections import defaultdict

# ATT&CK 战术列表（按顺序）
TACTICS = [
    "Reconnaissance",           # 侦察
    "Resource Development",     # 资源开发
    "Initial Access",           # 初始访问
    "Execution",                # 执行
    "Persistence",              # 持久化
    "Privilege Escalation",     # 权限提升
    "Defense Evasion",          # 防御绕过
    "Credential Access",        # 凭据访问
    "Discovery",                # 发现
    "Lateral Movement",         # 横向移动
    "Collection",               # 收集
    "Command and Control",      # 命令与控制
    "Exfiltration",             # 数据外传
    "Impact",                   # 影响
]

def calculate_coverage(attacks_detected, attacks_used):
    """
    计算ATT&CK覆盖度
    
    Args:
        attacks_detected: dict {technique_id: detected_bool}
        attacks_used: list of technique_ids used by red team
    
    Returns:
        dict with coverage statistics
    """
    stats = {
        "total_techniques_used": len(attacks_used),
        "total_detected": sum(attacks_detected.values()),
        "detection_rate": 0,
        "by_tactic": defaultdict(lambda: {"used": 0, "detected": 0}),
    }
    
    # 按战术分类统计（这里简化处理，实际需要technique-to-tactic映射）
    for tech in attacks_used:
        detected = attacks_detected.get(tech, False)
        # 实际项目中需要根据technique_id查找所属tactic
        # 这里用占位符
        tactic = "Execution"  # 示例
        stats["by_tactic"][tactic]["used"] += 1
        if detected:
            stats["by_tactic"][tactic]["detected"] += 1
    
    if stats["total_techniques_used"] > 0:
        stats["detection_rate"] = stats["total_detected"] / stats["total_techniques_used"] * 100
    
    return stats

def generate_coverage_report(stats, output_file):
    """生成覆盖度报告"""
    report = []
    report.append("=" * 60)
    report.append("ATT&CK 覆盖度统计报告")
    report.append("=" * 60)
    report.append(f"红队使用技术数: {stats['total_techniques_used']}")
    report.append(f"蓝队检测到的技术数: {stats['total_detected']}")
    report.append(f"整体检测率: {stats['detection_rate']:.1f}%")
    report.append("")
    report.append("按战术分类:")
    for tactic, data in stats["by_tactic"].items():
        rate = data["detected"] / data["used"] * 100 if data["used"] > 0 else 0
        report.append(f"  {tactic}: {data['detected']}/{data['used']} ({rate:.1f}%)")
    
    report_text = "\n".join(report)
    with open(output_file, 'w') as f:
        f.write(report_text)
    
    return report_text

if __name__ == "__main__":
    # 示例数据
    attacks_used = ["T1059.001", "T1053.005", "T1003.001", "T1550.002", "T1021.002"]

#!/usr/bin/env python3
"""
红队攻击时间线生成工具 - 自动整理攻击步骤并生成时间线
"""
from datetime import datetime, timedelta
import json

class AttackTimeline:
    def __init__(self):
        self.events = []
    
    def add_event(self, timestamp, tactic, technique, description, result, evidence=None):
        """添加攻击事件"""
        self.events.append({
            "timestamp": timestamp,
            "tactic": tactic,
            "technique": technique,
            "description": description,
            "result": result,
            "evidence": evidence or []
        })
    
    def generate_timeline_text(self):
        """生成文本时间线"""
        timeline = []
        timeline.append("=" * 80)
        timeline.append("红队攻击时间线")
        timeline.append("=" * 80)
        
        # 按时间排序
        sorted_events = sorted(self.events, key=lambda x: x["timestamp"])
        
        for i, event in enumerate(sorted_events, 1):
            timeline.append(f"\n[{i}] {event['timestamp']}")
            timeline.append(f"    战术: {event['tactic']}")
            timeline.append(f"    技术: {event['technique']}")
            timeline.append(f"    描述: {event['description']}")
            timeline.append(f"    结果: {'成功' if event['result'] else '失败'}")
            if event['evidence']:
                timeline.append(f"    证据: {', '.join(event['evidence'])}")
        
        # 总时长统计
        if sorted_events:
            start = sorted_events[0]["timestamp"]
            end = sorted_events[-1]["timestamp"]
            timeline.append(f"\n\n总攻击时长: {start} -> {end}")
        
        return "\n".join(timeline)
    
    def export_json(self, filename):
        """导出JSON格式"""
        with open(filename, 'w') as f:
            json.dump(self.events, f, indent=2, default=str)
