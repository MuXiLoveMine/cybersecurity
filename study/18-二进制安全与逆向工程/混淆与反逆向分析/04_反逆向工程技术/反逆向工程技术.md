### 4.1 反调试技术汇总

除了前文《动态调试技术实战》中提到的 IsDebuggerPresent、硬件断点检测和时间检测，还有以下补充技术：

**NtQueryInformationProcess 检测**：
```c
DWORD debug_port = 0;
NtQueryInformationProcess(GetCurrentProcess(),
    ProcessDebugPort, &debug_port, 4, NULL);
if (debug_port != 0)  // 非零表示正在被调试
    ExitProcess(0);
```

**Debug Object 检测**：检查系统中是否存在 Debug Object 句柄。
```c
DWORD flags = 0;
NtQueryInformationProcess(GetCurrentProcess(),
    ProcessDebugFlags, &flags, 4, NULL);
if (flags == 0)  // NoDebugInhibit为0表示被调试
    ExitProcess(0);
```

**CloseHandle 检测**：传入无效句柄给 CloseHandle，被调试时会触发 EXCEPTION_INVALID_HANDLE。
```c
__try {
    CloseHandle((HANDLE)0xDEADBEEF);
} __except (EXCEPTION_EXECUTE_HANDLER) {
    // 被调试时才会到达这里
    ExitProcess(0);
}
```

**ParentProcess 检测**：检查父进程是否是调试器。

### 4.2 反虚拟机/反沙箱

恶意代码常检测是否运行在虚拟机或沙箱环境中：

```c
// 检测VMware（I/O端口0x5658）
BOOL is_vmware() {
    BOOL result = FALSE;
    __asm {
        push eax
        push ebx
        push ecx
        push edx
        mov eax, 'VMXh'       // 魔数
        mov ebx, 0
        mov ecx, 10            // 操作码
        mov edx, 'VX'          // 端口号
        in  eax, dx             // 读端口
        cmp ebx, 'VMXh'        // VMware返回魔数
        setz result
        pop edx
        pop ecx
        pop ebx
        pop eax
    }
    return result;
}
```

```c
// 检测CPUID hypervisor位
int cpu_info[4];
__cpuid(cpu_info, 1);
if (cpu_info[2] & (1 << 31)) {
    // Hypervisor present位被设置 → 虚拟机环境
    // 进一步用cpuid leaf 0x40000000获取hypervisor签名
    __cpuid(cpu_info, 0x40000000);
    // VMware: "VMwareVMware"
    // Hyper-V: "Microsoft Hv"
    // VirtualBox: "VBoxVBoxVBox"
}
```

### 4.3 反Dump

反Dump技术防止从内存中提取PE文件：

```c
// 方法1：抹除PE头
void erase_pe_header() {
    DWORD old_protect;
    HMODULE base = GetModuleHandle(NULL);
    VirtualProtect(base, 0x1000, PAGE_EXECUTE_READWRITE, &old_protect);
    ZeroMemory(base, 0x1000);  // 清零PE头
    VirtualProtect(base, 0x1000, old_protect, &old_protect);
}

// 方法2：修改SizeOfImage
void corrupt_sizeof_image() {
    HMODULE base = GetModuleHandle(NULL);
    PIMAGE_NT_HEADERS nt = (PIMAGE_NT_HEADERS)
        ((BYTE*)base + ((PIMAGE_DOS_HEADER)base)->e_lfanew);
    nt->OptionalHeader.SizeOfImage = 0x1000;  // 改为极小值
}

// 方法3：修改PE头完整性校验
void corrupt_checksum() {
    HMODULE base = GetModuleHandle(NULL);
    PIMAGE_NT_HEADERS nt = (PIMAGE_NT_HEADERS)
        ((BYTE*)base + ((PIMAGE_DOS_HEADER)base)->e_lfanew);
    nt->OptionalHeader.CheckSum = 0xDEAD;  // 破坏校验和
}
```

// 检测Inline Hook：检查函数开头字节
BOOL detect_inline_hook(LPCSTR func_name) {
    HMODULE hMod = GetModuleHandle("kernel32.dll");
    FARPROC func = GetProcAddress(hMod, func_name);
    BYTE first_bytes[5];
    memcpy(first_bytes, func, 5);

    // 检查是否被JMP覆盖（0xE9 = JMP rel32）
    if (first_bytes[0] == 0xE9)
        return TRUE;

    // 对比磁盘文件中的原始字节
    BYTE file_bytes[5];
    // ... 读取磁盘PE文件对应位置的5字节 ...
    if (memcmp(first_bytes, file_bytes, 5) != 0)
        return TRUE;

    return FALSE;
}

### 4.5 完整性校验

```c
// CRC32自校验：校验代码段是否被修改
DWORD calculate_crc32(BYTE* data, DWORD len) {
    DWORD crc = 0xFFFFFFFF;
    for (DWORD i = 0; i < len; i++) {
        crc ^= data[i];
        for (int j = 0; j < 8; j++)
            crc = (crc >> 1) ^ (0xEDB88320 & -(crc & 1));
    }
    return ~crc;
}

void verify_integrity() {
    HMODULE base = GetModuleHandle(NULL);
    PIMAGE_NT_HEADERS nt = (PIMAGE_NT_HEADERS)
        ((BYTE*)base + ((PIMAGE_DOS_HEADER)base)->e_lfanew);
    PIMAGE_SECTION_HEADER sec = IMAGE_FIRST_SECTION(nt);

    // 找到.text段
    for (int i = 0; i < nt->FileHeader.NumberOfSections; i++) {
        if (memcmp(sec[i].Name, ".text", 5) == 0) {
            BYTE* text_start = (BYTE*)base + sec[i].VirtualAddress;
            DWORD text_size = sec[i].Misc.VirtualSize;
            DWORD crc = calculate_crc32(text_start, text_size);
            if (crc != ORIGINAL_CRC32) {  // 编译时硬编码的原始CRC
                ExitProcess(0);  // 代码被修改 → 退出
            }
        }
    }
}
```
