### 5.2 字符串解密自动化

#### IDAPython批量解密

```python
#!/usr/bin/env python3
# ida_decrypt_strings.py - 批量识别并解密字符串
import idautils
import idaapi
import idc

DECRYPT_FUNC_EA = 0x401000  # 解密函数地址

def find_decrypt_calls():
    """找到所有调用解密函数的位置"""
    callers = []
    for xref in idautils.XrefsTo(DECRYPT_FUNC_EA):
        callers.append(xref.frm)
    return callers

def extract_encrypted_data(ea):
    """从调用点提取加密数据参数"""
    # 回溯分析，找到push参数
    call_ea = ea
    prev_ea = idc.prev_head(call_ea)

    # 假设模式：push len; push ptr; call decrypt
    # 分析前几条指令获取参数
    args = []
    curr = call_ea
    for i in range(5):  # 最多回溯5条指令
        curr = idc.prev_head(curr)
        mnem = idc.print_insn_mnem(curr)
        if mnem == "push":
            op = idc.get_operand_value(curr, 0)
            args.append(op)
        if len(args) >= 2:
            break

    if len(args) >= 2:
        return args[1], args[0]  # (data_ptr, length)
    return None, None

def decrypt_xor(data_bytes, key=0x37):
    """XOR解密"""
    return bytes([b ^ key for b in data_bytes])

# 主逻辑
callers = find_decrypt_calls()
print(f"找到 {len(callers)} 处解密调用")

for caller_ea in callers:
    data_ptr, length = extract_encrypted_data(caller_ea)
    if data_ptr and length and length < 1000:
        enc_data = idaapi.get_bytes(data_ptr, length)
        if enc_data:
            dec_data = decrypt_xor(enc_data)
            dec_str = dec_data.decode('ascii', errors='replace')
            print(f"  0x{caller_ea:x}: '{dec_str}'")
            # 在IDA中添加注释
            idc.set_cmt(caller_ea, f"解密结果: {dec_str}", 0)
```

#### x64dbg脚本批量解密

```
// decrypt_strings.txt - x64dbg批量解密脚本
// 假设解密函数在0x401000，参数通过栈传递

var count
mov count, 0

// 遍历所有调用解密函数的位置

run

loop:
    // 读取参数

    push [esp+8]     // length
    pop [data_len]
    pop [data_ptr]

    // dump解密结果

    d data_ptr

    // 计数

    // 继续运行到下一次调用

    // 检查是否还在解密函数
    cmp eip, 0x401000
    je loop

log "共解密 {:count} 个字符串"
```

### 5.3 VMP部分还原

#### 虚拟指令追踪

通过动态调试追踪虚拟指令的执行，记录每个 handler 的操作，逐步还原虚拟指令集语义。

```python
#!/usr/bin/env python3
# vmp_trace.py - VMP handler追踪（GDB脚本）
import gdb

class VMPTracer(gdb.Breakpoint):
    def __init__(self, dispatcher_addr):
        super().__init__("*" + dispatcher_addr, internal=True)
        self.handlers = {}
        self.count = 0

    def stop(self):
        self.count += 1
        # 读取VIP指向的当前虚拟指令
        vip = int(gdb.parse_and_eval("$esi"))
        inf = gdb.selected_inferior()
        opcode = bytes(inf.read_memory(vip, 1))[0]

        # 记录handler调用次数
        eip = int(gdb.parse_and_eval("$eip"))
        if opcode not in self.handlers:
            self.handlers[opcode] = {"count": 0, "first_eip": eip, "samples": []}
        self.handlers[opcode]["count"] += 1

        # 采样前几条指令用于语义分析
        if len(self.handlers[opcode]["samples"]) < 3:
            disas = gdb.execute("x/5i $eip", to_string=True)
            self.handlers[opcode]["samples"].append(disas)

        # 最多追踪10000条
        if self.count >= 10000:
            self.report()
            return True  # 停止
        gdb.execute("continue")

    def report(self):
        print("\n=== VMP Handler统计 ===")
        for opcode, info in sorted(self.handlers.items()):
            print(f"  opcode 0x{opcode:02x}: {info['count']}次, "
                  f"handler@0x{info['first_eip']:x}")

# 设置追踪（dispatcher地址需根据实际分析确定）
VMPTracer("0x401234")
print("VMP追踪已启动")
```

#### 调度器分析

调度器分析是还原 VMP 的关键步骤：

```
1. 定位dispatcher：
   - 在VM入口处设断点
   - 跟踪执行流，找到循环跳回的代码块
   - 该代码块通常包含 "movzx reg, byte ptr [reg]" 和 "jmp [table + reg*4]"

2. 提取handler表：
   - 找到jmp指令引用的表地址
   - 从表中提取所有handler地址

3. 语义分析：
   - 对每个handler，分析其对虚拟寄存器/栈的操作
   - 归纳等价的x86指令语义

4. 反汇编虚拟字节码：
   - 根据语义映射，将虚拟字节码翻译回x86
   - 重建控制流图
```

#### 部分还原思路

完全还原 VMP 通常不现实，实践中采用部分还原策略：

```
策略1：关键函数还原
  - 只还原与目标算法相关的少数handler
  - 通过动态追踪获取该段虚拟指令的执行轨迹
  - 翻译为等效的x86代码

策略2：黑盒分析
  - 不还原代码，通过动态调试直接跟踪算法行为
  - 记录输入/输出关系，推断算法逻辑
  - 适用于校验算法、加密算法分析

策略3：混合方法
  - 静态识别VM边界（虚拟化区域起止）
  - 在VM入口/出口设断点，追踪进入和退出时的状态
  - 对VM内部进行黑盒跟踪
```

### 5.4 自动化脱壳

#### 通用脱壳框架思路

```
脱壳流程：
1. 识别壳类型（PE节名特征、入口点特征）
2. 运行到OEP（Original Entry Point）
   - 内存断点法：在内存写入断点上等壳解压完成
   - ESP定律法：跟踪栈平衡找到OEP
   - Execute till Return法：在壳入口步出到OEP
3. Dump内存：从OEP开始dump完整PE
4. 修复IAT（Import Address Table）
5. 修复重定位表

工具：
  - x64dbg + Scylla：手动脱壳+IAT修复
  - 自动脱壳脚本
```

**x64dbg 自动脱壳脚本（ESP定律法）：**

// 此时EIP应该在OEP
