# 协议模糊测试

协议模糊测试针对网络协议实现进行测试，需要理解协议格式并构造合法但变异的网络数据包。

**Sulley框架使用：**

```python
#!/usr/bin/env python3
"""
使用Sulley框架进行协议模糊测试
针对自定义TCP协议进行测试
"""
from sulley import *

# 定义协议结构
def define_protocol():
    """定义自定义协议格式"""
    
    # 初始化请求
    s_initialize("custom_protocol")
    
    # 魔数（固定值）
    s_static(b'\x41\x42\x43\x44')   # "ABCD"
    
    # 版本号
    s_byte(0x01, name="version")
    
    # 消息类型
    s_byte(0x00, name="msg_type")
    
    # 长度字段（自动计算）
    s_size("payload", length=4, name="length")
    
    # 负载数据
    s_string("FuzzMe" * 10, name="payload")
    
    # 校验和（占位）
    s_dword(0xDEADBEEF, name="checksum")

def fuzz_target():
    """配置模糊测试目标"""
    # 目标IP和端口
    target_ip = "127.0.0.1"
    target_port = 8080
    
    # 初始化会话
    session = session.Session(
        target=session.Target(target_ip, target_port),
        proto="tcp"
    )
    
    # 添加请求
    session.connect(s_get("custom_protocol"))
    
    # 启动监控
    session.monitor = pedrpc.Client(target_ip, 26002)
    
    # 运行模糊测试
    session.fuzz()
```

**使用Boofuzz进行协议模糊测试：**

```python
#!/usr/bin/env python3
"""
使用Boofuzz进行HTTP协议模糊测试
"""
from boofuzz import *

def main():
    # 启动Web界面（可选）
    # 然后在浏览器中访问 http://localhost:26002
    
    session = Session(
        target=Target(connection=SocketConnection("127.0.0.1", 8080, proto='tcp')),
        # Web界面
        web_port=26002,
    )
    
    # 定义HTTP请求
    s_initialize(name="HTTP Request")
    
    # HTTP方法
    s_group("methods", values=["GET", "POST", "PUT", "DELETE", "HEAD", "OPTIONS"])
    
    # 路径
    s_delim(" ", name="space1")
    s_string("/index.html", name="path")
    s_delim(" ", name="space2")
    
    # HTTP版本
    s_string("HTTP/1.1", name="version")
    s_delim("\r\n", name="crlf1")
    
    # Host头
    s_string("Host:", name="host_key")
    s_delim(" ", name="space3")
    s_string("example.com", name="host_value")
    s_delim("\r\n", name="crlf2")
    
    # Content-Type头（可选）
    s_string("Content-Type: application/x-www-form-urlencoded", name="content_type")
    s_delim("\r\n", name="crlf3")
    
    # 空行表示头部结束
    s_delim("\r\n", name="crlf_end")
    
    # 添加到会话
    session.connect(s_get("HTTP Request"))
    
    # 运行
    session.fuzz()

if __name__ == "__main__":
    main()
```
