### 7.1 GDB Python脚本自动化

```python
#!/usr/bin/env python3
# auto_trace.py - 自动化跟踪函数调用并记录参数
import gdb
import struct

class AutoTrace(gdb.Command):
    """自动跟踪指定函数的所有调用"""
    def __init__(self):
        super().__init__("auto-trace", gdb.COMMAND_USER)

    def invoke(self, arg, from_tty):
        addr = int(arg, 16) if arg.startswith("0x") else int(arg)
        bp = gdb.Breakpoint(f"*{addr:#x}")

        results = []
        def stop_handler(event):
            if isinstance(event, gdb.StopEvent):
                rdi = int(gdb.parse_and_eval("$rdi"))
                rsi = int(gdb.parse_and_eval("$rsi"))
                rdx = int(gdb.parse_and_eval("$rdx"))
                inf = gdb.selected_inferior()
                try:
                    buf = bytes(inf.read_memory(rdi, min(rsi, 64)))
                except:
                    buf = b"<unreadable>"
                results.append({
                    "rdi": rdi, "rsi": rsi, "rdx": rdx, "buf": buf
                })
                print(f"[Trace] rdi={rdi:#x} rsi={rsi} rdx={rdx} buf={buf[:32]}")
                gdb.execute("continue")

        gdb.events.stop.connect(stop_handler)
        print(f"自动跟踪已启动，断点地址: {addr:#x}")
        gdb.execute("run")

        # 汇总结果
        print(f"\n=== 跟踪完成，共记录 {len(results)} 次调用 ===")
        gdb.events.stop.disconnect(stop_handler)

AutoTrace()
```

使用：
```bash
gdb -x auto_trace.py ./target
(gdb) auto-trace 0x401234
```

### 7.2 Frida辅助调试

Frida 是强大的动态插桩工具，可在不重启目标进程的情况下注入 JavaScript 脚本。

```javascript
// hook_decrypt.js - Hook解密函数并打印参数和返回值
Interceptor.attach(Module.findExportByName(null, "decrypt"), {
    onEnter: function(args) {
        this.input = args[0];
        this.len = args[1].toInt32();
        console.log("[decrypt] input:", hexdump(this.input, {length: this.len}));
    },
    onLeave: function(retval) {
        console.log("[decrypt] output:", hexdump(this.input, {length: this.len}));
        console.log("[decrypt] ret:", retval);
    }
});

// Hook文件操作
Interceptor.attach(Module.findExportByName(null, "CreateFileW"), {
    onEnter: function(args) {
        var filename = args[0].readUtf16String();
        if (filename) {
            console.log("[CreateFileW]", filename);
        }
    }
});
```

```bash
# 注入到运行中的进程
frida -n target.exe -l hook_decrypt.js

# 启动并注入
frida -f target.exe -l hook_decrypt.js --no-pause

# Python绑定
import frida
session = frida.attach("target.exe")
script = session.create_script(open("hook_decrypt.js").read())
script.on('message', lambda msg, data: print(msg))
script.load()
```

### 7.3 自动化断点与日志收集

```python
#!/usr/bin/env python3
# auto_bp.py - 批量设断点并收集日志
import gdb

# 批量断点配置
BREAKPOINTS = {
    0x401000: "decryption_func",
    0x401200: "validation_func",
    0x401400: "network_send",
}

LOG_FILE = open("debug_log.txt", "w")

def setup_auto_log():
    for addr, name in BREAKPOINTS.items():
        gdb.execute(f"break *{addr}")
        # 设置断点命令序列：打印信息后继续
        cmds = f"""
commands {addr}
silent
  printf "[{name}] rdi=%#x rsi=%#x rdx=%#x\\n", $rdi, $rsi, $rdx
  x/16bx $rdi
end
"""
        gdb.execute(cmds)

setup_auto_log()
print(f"已设置 {len(BREAKPOINTS)} 个自动化断点，日志输出到 debug_log.txt")
gdb.execute("run")
```

```bash
gdb -x auto_bp.py ./target input_file
# 所有断点信息自动记录到屏幕，可重定向到文件
```
