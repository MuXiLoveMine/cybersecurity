# 固件签名验证流程

### 8.1 完整验证流程

```python
# 固件签名验证完整流程
class FirmwareVerification:
    """固件签名验证器"""
    
    def __init__(self, trust_root: dict):
        """
        trust_root: 信任根配置
        {
            'public_key': bytes,    # 厂商公钥
            'min_version': int,     # 最低固件版本
            'trusted_subjects': list,  # 可信发布者
        }
        """
        self.trust_root = trust_root
    
    def verify(self, firmware_package: bytes) -> dict:
        """完整验证流程"""
        result = {
            'valid': False,
            'stage': '',
            'errors': [],
        }
        
        # 阶段1: 解析固件包
        try:
            header, firmware = self._parse_package(firmware_package)
            result['stage'] = 'parse'
        except Exception as e:
            result['errors'].append(f"解析失败: {e}")
            return result
        
        # 阶段2: 验证格式
        if not self._validate_header(header):
            result['errors'].append("固件头格式无效")
            return result
        result['stage'] = 'format'
        
        # 阶段3: 完整性验证
        calc_hash = hashlib.sha256(firmware).digest()
        if calc_hash != header['hash']:
            result['errors'].append("固件哈希不匹配 - 固件被篡改")
            return result
        result['stage'] = 'integrity'
        
        # 阶段4: 签名验证
        if not self._verify_signature(header, firmware):
            result['errors'].append("签名验证失败")
            return result
        result['stage'] = 'signature'
        
        # 阶段5: 防回滚检查
        if header['version'] < self.trust_root['min_version']:
            result['errors'].append(
                f"固件版本 {header['version']} 低于最低要求 "
                f"{self.trust_root['min_version']}"
            )
            return result
        result['stage'] = 'rollback'
        
        # 阶段6: 发布者验证
        if not self._verify_publisher(header):
            result['errors'].append("发布者不在信任列表中")
            return result
        result['stage'] = 'publisher'
        
        # 阶段7: 时间戳验证（防止重放旧签名）
        if not self._verify_timestamp(header):
            result['errors'].append("签名时间戳无效")
            return result
        
        result['valid'] = True
        result['stage'] = 'complete'
        result['version'] = header['version']
        result['hash'] = header['hash'].hex()
        
        return result
    
    def _verify_signature(self, header: dict, firmware: bytes) -> bool:
        """验证数字签名"""
        from cryptography.hazmat.primitives.asymmetric import ec
        from cryptography.hazmat.primitives.asymmetric.utils import (
            decode_dss_signature, encode_dss_signature
        )
        
        public_key = self.trust_root['public_key']
        signature = header['signature']
        hash_value = header['hash']
        
        try:
            if header['algorithm'] == 'ECDSA-P256':
                r = int.from_bytes(signature[:32], 'big')
                s = int.from_bytes(signature[32:], 'big')
                der_sig = encode_dss_signature(r, s)
                public_key.verify(der_sig, hash_value, ec.ECDSA(
                    hashes.SHA256()
                ))
                return True
            elif header['algorithm'] == 'Ed25519':
                public_key.verify(signature, hash_value)
                return True
            elif header['algorithm'].startswith('RSA'):
                from cryptography.hazmat.primitives.asymmetric import padding
                public_key.verify(
                    signature, hash_value,
                    padding.PSS(
                        mgf=padding.MGF1(hashes.SHA256()),
                        salt_length=padding.PSS.MAX_LENGTH
                    ),
                    hashes.SHA256()
                )
                return True
        except Exception:
            return False
        
        return False
```
