### 8.2 Mirai C2协议解析

```python
# Mirai C2协议解析器
class MiraiC2Parser:
    """Mirai C2通信协议解析"""
    
    HEADER_SIZE = 4
    
    def parse_packet(self, data: bytes) -> dict:
        """解析C2数据包"""
        if len(data) < self.HEADER_SIZE:
            return {'error': '数据包过短'}
        
        # Mirai协议头: 2字节长度 + 2字节命令
        length = struct.unpack('>H', data[:2])[0]
        cmd_id = struct.unpack('>H', data[2:4])[0]
        
        payload = data[4:4+length]
        
        result = {
            'length': length,
            'cmd_id': cmd_id,
            'payload': payload.hex()
        }
        
        # 命令ID映射
        CMD_MAP = {
            0x0000: 'BOT_LOGIN',
            0x0001: 'BOT_ATTACK',
            0x0002: 'BOT_KILL',
            0x0010: 'ADMIN_LOGIN',
            0x0011: 'ADMIN_ATTACK',
            0x0012: 'ADMIN_KILL',
            0x0013: 'ADMIN_BOT_COUNT',
            0x0020: 'BOT_SCANNER_ON',
            0x0021: 'BOT_SCANNER_OFF',
        }
        
        result['cmd_name'] = CMD_MAP.get(cmd_id, 'UNKNOWN')
        
        # 解析攻击指令
        if cmd_id == 0x0001:
            result.update(self._parse_attack(payload))
        
        return result
    
    def _parse_attack(self, payload: bytes) -> dict:
        """解析攻击参数"""
        if len(payload) < 7:
            return {}
        
        attack_type = payload[0]
        duration = struct.unpack('>I', payload[1:5])[0]
        target_count = payload[5]
        
        targets = []
        offset = 6
        for i in range(target_count):
            if offset + 4 <= len(payload):
                ip = socket.inet_ntoa(payload[offset:offset+4])
                targets.append(ip)
                offset += 4
        
        port = struct.unpack('>H', payload[offset:offset+2])[0]
        
        ATTACK_TYPES = {
            0x01: 'HTTP_FLOOD',
            0x02: 'TCP_FLOOD', 
            0x03: 'UDP_FLOOD',
            0x04: 'DNS_AMP',
            0x05: 'GRE_FLOOD',
            0x06: 'SYN_FLOOD',
            0x07: 'ACK_FLOOD',
            0x08: 'VSE_FLOOD',
            0x09: 'DNS_HARD',
            0x0A: 'STOMP_FLOOD',
        }
        
        return {
            'attack_type': ATTACK_TYPES.get(attack_type, 'UNKNOWN'),
            'duration_sec': duration,
            'targets': targets,
            'port': port,
        }
```

### 8.3 C2流量特征检测

```
Mirai C2流量特征:
1. 固定目标端口: 23, 2323, 8443, 457
2. 初始连接数据包大小: 4-8字节
3. 协议头格式: 00 00 XX XX (长度+命令)
4. 连接模式: 持久TCP连接，定期心跳
5. 流量模式: 间歇性大量数据传输

检测规则 (Snort格式示例):
alert tcp $HOME_NET any -> $EXTERNAL_NET 23 (msg:"Mirai Bot Telnet C2"; 
    content:"|00 00 00 03|"; depth:4; sid:1000001;)

alert tcp $HOME_NET any -> $EXTERNAL_NET 8443 (msg:"Mirai Bot C2 Communication";
    content:"|00 00 00 0"; depth:4; sid:1000002;)
```
