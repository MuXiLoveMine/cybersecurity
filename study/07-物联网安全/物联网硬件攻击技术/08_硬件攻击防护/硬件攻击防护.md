### 7.2 代码层面的防护

```c
// 硬件攻击防护代码示例

// 1. 调试器检测
bool is_debugger_present(void) {
    // 检查调试状态寄存器
    if (CoreDebug->DHCSR & CoreDebug_DHCSR_C_DEBUGEN_Msk) {
        return true;
    }
    
    // 检查时间异常 (调试暂停)
    uint32_t start = DWT->CYCCNT;
    __NOP(); __NOP(); __NOP(); __NOP();
    uint32_t elapsed = DWT->CYCCNT - start;
    
    if (elapsed > 50) {
        return true;  // 可能在调试
    }
    
    return false;
}

// 2. 故障检测 (冗余计算)
uint32_t secure_calculate(uint32_t a, uint32_t b) {
    uint32_t result1 = a + b;
    uint32_t result2 = a + b;  // 冗余计算
    
    if (result1 != result2) {
        // 检测到故障注入
        security_handler(FAULT_DETECTED);
        return 0;  // 安全失败
    }
    
    return result1;
}

// 3. 控制流完整性
void critical_function(void) {
    uint32_t token = 0xDEADBEEF;
    
    // 进入关键区域
    if (token != 0xDEADBEEF) {
        security_handler(CORRUPTION_DETECTED);
    }
    
    // 关键操作
    secure_operation();
    
    // 验证token (防止跳过)
    if (token != 0xDEADBEEF) {
        security_handler(SKIP_DETECTED);
    }
    
    token = 0xCAFEBABE;  // 改变token
}

// 4. 电压/时钟监测
void monitor_power(void) {
    // STM32 PVD (电源电压监测器)
    PWR->CR |= PWR_CR_PVDE;        // 启用PVD
    PWR->CR |= PWR_CR_PLS_2V9;     // 设置阈值
    
    // 如果电压低于阈值, 产生中断
    // 在中断中清除敏感数据
}

void monitor_clock(void) {
    // CSS (时钟安全系统)
```
