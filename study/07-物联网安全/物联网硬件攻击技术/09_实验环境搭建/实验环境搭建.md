### 8.2 ChipWhisperer实验

```python
#!/usr/bin/env python3
"""ChipWhisperer功耗分析实验"""
# 需要安装: pip install chipwhisperer

try:
    import chipwhisperer as cw
except ImportError:
    print("请安装ChipWhisperer: pip install chipwhisperer")
    exit(1)

class ChipWhispererExperiment:
    """ChipWhisperer功耗分析实验"""
    
    def __init__(self):
        self.scope = None
        self.target = None
    
    def setup(self, scope_type='cw-lite'):
        """初始化设备"""
        # 连接示波器
        self.scope = cw.scope(type=scope_type)
        
        # 连接目标
        self.target = cw.target(self.scope)
        
        # 配置示波器
        self.scope.gain.gain = 45
        self.scope.gain.mode = 'low'
        self.scope.adc.samples = 5000
        self.scope.adc.triggers = 1
        self.scope.adc.offset = 0
        self.scope.clock.src = 'target'
        
        print("[+] ChipWhisperer已初始化")
    
    def capture_trace(self, plaintext):
        """捕获单条功耗轨迹"""
        # 设置明文
        self.target.simpleserial_write('p', plaintext)
        
        # 触发并采集
        ret = self.scope.capture()
        
        if ret:
            print("[-] 采集超时")
            return None
        
        # 获取轨迹数据
        trace = self.scope.get_last_trace()
        
        # 获取密文
        response = self.target.simpleserial_read('r', 16)
        
        return trace, response
    
    def capture_traces(self, num_traces=5000):
        """采集多条轨迹"""
        traces = []
        plaintexts = []
        ciphertexts = []
        
        from Crypto.Cipher import AES
        key = bytes([0x2b, 0x7e, 0x15, 0x16, 0x28, 0xae, 0xd2, 0xa6,
                      0xab, 0xf7, 0x15, 0x88, 0x09, 0xcf, 0x4f, 0x3c])
        
        import os
        for i in range(num_traces):
            # 生成随机明文
            plaintext = os.urandom(16)
            
            # 捕获轨迹
            result = self.capture_trace(plaintext)
            if result:
                trace, ciphertext = result
                traces.append(trace)
                plaintexts.append(plaintext)
                ciphertexts.append(ciphertext)
            
            if (i + 1) % 100 == 0:
                print(f"  采集进度: {i+1}/{num_traces}")
        
        print(f"[+] 共采集 {len(traces)} 条轨迹")
        return traces, plaintexts, ciphertexts
    
    def run_dpa(self, traces, plaintexts, ciphertexts):
        """运行DPA攻击"""
        import numpy as np
        
        traces = np.array(traces)
        
        # AES S盒
        sbox = [
            0x63, 0x7c, 0x77, 0x7b, 0xf2, 0x6b, 0x6f, 0xc5,
            0x30, 0x01, 0x67, 0x2b, 0xfe, 0xd7, 0xab, 0x76,
            # ... 完整S盒
        ]
        
        best_key = 0
        best_corr = 0
        
        for key_guess in range(256):
            # 计算假设中间值
            hypotheses = []
            for pt in plaintexts:
                intermediate = sbox[pt[0] ^ key_guess]
                hw = bin(intermediate).count('1')
                hypotheses.append(hw)
            
            # 计算相关性
            for t in range(len(traces[0])):
                trace_samples = traces[:, t]
                if np.std(trace_samples) > 0 and np.std(hypotheses) > 0:
                    corr = np.corrcoef(hypotheses, trace_samples)[0, 1]
                    if abs(corr) > best_corr:
                        best_corr = abs(corr)
                        best_key = key_guess
        
        print(f"[+] CPA结果: Key=0x{best_key:02X}, Corr={best_corr:.4f}")
        return best_key

# 使用示例
if __name__ == '__main__':
    experiment = ChipWhispererExperiment()
    experiment.setup()
    traces, pts, cts = experiment.capture_traces(5000)
    experiment.run_dpa(traces, pts, cts)
```
