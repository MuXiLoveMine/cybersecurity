```bash

# - FTDI FT232RL（最稳定，支持 3.3V/5V）
# - CP2102（性价比高）
# - CH340G（便宜，驱动可能有问题）

# 方法 2：万用表测量（上电状态）
#   - GND：与屏蔽罩/USB 外壳导通（0V）
#   - VCC：3.3V 或 5V 恒定电压
#   - TX：启动时有 3.3V 跳变（数据传输）
#   - RX：通常 3.3V 高电平
# 方法 3：逻辑分析仪捕捉启动信号

# ===== 常用串口工具 =====

# 连接串口（设备名、波特率）

# 退出：Ctrl+A 然后 K 再 Y
# 记录日志：Ctrl+A 然后 H

# --- minicom ---
# 安装
sudo apt install minicom

# 配置
minicom -s
# 选择 "Serial port setup"
# 设置 A: Serial Device = /dev/ttyUSB0
# 设置 E: Bps/Par/Bits = 115200 8N1
# 保存为 dfl 默认配置

# 启动
minicom -D /dev/ttyUSB0 -b 115200
# 退出：Ctrl+A 然后 X

# --- picocom ---

# 退出：Ctrl+A 然后 Ctrl+Q

# --- Putty（Windows）---

python baudrate.py -p /dev/ttyUSB0
```

```bash
textins = []

    plaintext = bytearray([i % 256] * 16)
    
    scope.arm()
    target.simpleserial_write('p', plaintext)
    scope.capture()

    traces.append(trace)

# DPA 攻击（针对 AES SBOX 输出）
import numpy as np

sbox = [
    0x63, 0x7c, 0x77, 0x7b, 0xf2, 0x6b, 0x6f, 0xc5,
    0x30, 0x01, 0x67, 0x2b, 0xfe, 0xd7, 0xab, 0x76,
    # ... 完整 SBOX 表
]

def dpa_attack(traces, textins, byte_pos=0):
    num_traces = len(traces)
    trace_len = len(traces[0])
    
    best_guess = 0
    best_corr = 0
    
    # 遍历所有可能的密钥字节（0-255）
    for key_guess in range(256):
        # 分组：SBOX 输出最高位为 0 或 1
        group0 = []
        group1 = []
        
        for i in range(num_traces):
            sbox_out = sbox[textins[i][byte_pos] ^ key_guess]
            if sbox_out & 0x80:
                group1.append(traces[i])
            else:
                group0.append(traces[i])
        
        # 计算差分均值
        if len(group0) > 0 and len(group1) > 0:
            diff = np.mean(group1, axis=0) - np.mean(group0, axis=0)
            max_corr = np.max(np.abs(diff))
            
            if max_corr > best_corr:
                best_corr = max_corr
                best_guess = key_guess
    
    return best_guess, best_corr

# ===== 相关功耗分析 (CPA) =====
# 使用 Pearson 相关系数
def cpa_attack(traces, textins, byte_pos=0):
    traces = np.array(traces)
    num_traces = len(traces)
    
    hamming_weights = np.zeros((num_traces, 256))
    
    # 计算每个密钥猜测的 Hamming 重量模型
    for key_guess in range(256):
        for i in range(num_traces):
            sbox_out = sbox[textins[i][byte_pos] ^ key_guess]
            hamming_weights[i, key_guess] = bin(sbox_out).count('1')
    
    # 计算相关系数
    best_key = 0
    best_corr = 0
    
    for key_guess in range(256):
        hw = hamming_weights[:, key_guess]
        # 计算每个采样点的相关系数
        corr = np.corrcoef(hw, traces.T)[0, 1:]
        max_corr = np.max(np.abs(corr))
        
        if max_corr > best_corr:
            best_corr = max_corr
            best_key = key_guess
    
    return best_key, best_corr
```
