### 2.3 ARM内存保护机制

```
ARM内存保护:

1. MMU (Memory Management Unit) - ARMv7-A/ARMv8-A:
   - 虚拟地址到物理地址映射
   - 页表权限控制 (AP, UXN, PXN)
   - 内存属性 (Normal/Device/Strongly-Ordered)
   - ASID (Address Space IDentifier)

2. MPU (Memory Protection Unit) - ARMv7-M/ARMv8-M:
   - 区域 (Region) 基于的内存保护
   - 8-16个可配置区域
   - 起始/结束地址 + 权限
   - Cortex-M系列使用

3. PMSAv8-MA (ARMv8-M Architecture):
   - 增强的MPU
   - 支持TrustZone for Cortex-M
   - 安全/非安全区域隔离

MPU配置示例 (Cortex-M):
// MPU区域配置
typedef struct {
    uint32_t base_addr;     // 区域基地址
    uint32_t size;          // 区域大小
    uint32_t access_perm;   // 访问权限
    uint32_t exec_perm;     // 执行权限
    uint32_t cache_attr;    // 缓存属性
    uint32_t secure_attr;   // 安全属性
} mpu_region_t;

// 配置非安全区域
mpu_region_t user_region = {
    .base_addr = 0x20000000,
    .size = MPU_SIZE_256KB,
    .access_perm = MPU_ACCESS_FULL,
    .exec_perm = MPU_EXEC_ALLOW,
    .cache_attr = MPU_CACHE_WB,
    .secure_attr = MPU_NON_SECURE,
};

// 配置安全区域
mpu_region_t secure_region = {
    .base_addr = 0x30000000,
    .size = MPU_SIZE_128KB,
    .access_perm = MPU_ACCESS_PRIV_ONLY,
    .exec_perm = MPU_EXEC_ALLOW,
    .cache_attr = MPU_CACHE_WT,
    .secure_attr = MPU_SECURE,
};
```

## 3. MIPS架构安全特性

### 3.1 MIPS安全概述

```
MIPS安全特性:

1. 虚拟内存:
   - 32位地址空间 (MIPS32) / 64位 (MIPS64)
   - TLB (Translation Lookaside Buffer)
   - 固定映射段 (kseg0, kseg1, kuseg)

2. 特权级别:
   - User Mode
   - Kernel Mode
   - Debug Mode (EJTAG)

3. 安全扩展:
   - Virtualization (MIPS VZ)
   - 多线程 (MIPS MT)
   - DSP ASE (数字信号处理)

4. MIPS Warrior系列:
   - P5600: 多发射, 乱序执行
   - I6400: 64位多线程
   - M6250: 嵌入式多线程
```

```
MIPS内存段布局 (32位):

┌──────────────────┬──────────┬───────────────────────┐
```

### 4.2 RISC-V PMP配置

```c
// RISC-V PMP配置示例
// 寄存器: pmpcfg0-pmpcfg15, pmpaddr0-pmpaddr15

// 配置PMP区域0: 只读区域
// pmpcfg0:
//   L = 0 (不锁定)
//   A = 1 (TOR - Top of Range)
//   X = 0 (不可执行)
//   W = 0 (不可写)
//   R = 1 (可读)
pmpcfg0 = 0x01;  // A=TOR, R=1
pmpaddr0 = 0x80000000 >> 2;  // 基地址

// 配置PMP区域1: 可读写区域
pmpcfg1 = 0x1B;  // A=NAPOT, R=1, W=1, X=0
pmpaddr1 = 0x801FFFFF >> 2;

// 配置PMP区域2: 默认拒绝 (U模式)
// 最后一个区域设置为NAPOT, size=0
// 匹配所有未匹配的地址
pmpcfg15 = 0x1A;  // A=NAPOT, W=1 (deny)
pmpaddr15 = 0x0;

// PMP区域优先级
// 编号小的优先级高
// 第一个匹配的区域决定权限
```

### 5.3 可信应用 (TA) 开发

```c
// OP-TEE Trusted Application 示例
// 文件: secure_storage_ta.c

#include <tee.h>
#include <tee_internal_api.h>
#include <tee_internal_api_extensions.h>

#define TA_NAME "secure_storage_ta"
#define STORAGE_ID 0x42535400  // BST0

// TA全局变量
typedef struct {
    TEE_ObjectHandle secure_object;
    uint32_t object_size;
} TA_Context;

static TA_Context ctx;

// TA创建时调用
TEE_Result TA_CreateEntryPoint(void) {
    memset(&ctx, 0, sizeof(ctx));
    return TEE_SUCCESS;
}

// TA销毁时调用
void TA_DestroyEntryPoint(void) {
    if (ctx.secure_object != TEE_HANDLE_NULL) {
        TEE_CloseAndPersistsObject(ctx.secure_object);
    }
}

// 开启会话
TEE_Result TA_OpenSessionEntryPoint(uint32_t param_types,
        TEE_Param params[4], void **sess_ctx) {
    return TEE_SUCCESS;
}

// 关闭会话
void TA_CloseSessionEntryPoint(void *sess_ctx) {
}

// 调用命令
TEE_Result TA_InvokeCommandEntryPoint(void *sess_ctx,
        uint32_t cmd_id, uint32_t param_types,
        TEE_Param params[4]) {
    
    switch (cmd_id) {
        case 1:  // 存储数据
            return store_secure_data(param_types, params);
        case 2:  // 读取数据
            return read_secure_data(param_types, params);
        case 3:  // 删除数据
            return delete_secure_data(param_types, params);
        case 4:  // 加密数据
            return encrypt_data(param_types, params);
        default:
            return TEE_ERROR_BAD_PARAMETERS;
    }
}

// 安全存储数据
static TEE_Result store_secure_data(uint32_t param_types,
        TEE_Param params[4]) {
    TEE_Result res;
    
    // 验证参数
    if (param_types != TEE_PARAM_TYPES(
            TEE_PARAM_TYPE_MEMREF_INPUT,
            TEE_PARAM_TYPE_NONE,
            TEE_PARAM_TYPE_NONE,
            TEE_PARAM_TYPE_NONE)) {
        return TEE_ERROR_BAD_PARAMETERS;
    }
    
    // 创建持久化对象
    res = TEE_CreatePersistentObject(
        TEE_STORAGE_PRIVATE,    // 存储类型
        &STORAGE_ID,            // 对象ID
        sizeof(STORAGE_ID),
        TEE_DATA_FLAG_ACCESS_READ | 
        TEE_DATA_FLAG_ACCESS_WRITE |
        TEE_DATA_FLAG_OVERWRITE,
        TEE_HANDLE_NULL,        // 属性
        params[0].memref.buffer, // 初始数据
        params[0].memref.size,   // 数据大小
        &ctx.secure_object       // 对象句柄
    );
    
    if (res != TEE_SUCCESS) {
        EMSG("Failed to create persistent object: 0x%x", res);
        return res;
    }
    
    ctx.object_size = params[0].memref.size;
    return TEE_SUCCESS;
}

// 加密数据 (使用硬件加速)
static TEE_Result encrypt_data(uint32_t param_types,
        TEE_Param params[4]) {
    TEE_OperationHandle op = TEE_HANDLE_NULL;
    TEE_Result res;
    
    // 分配加密操作 (AES-GCM)
```

## 8. 安全启动链详解

### 8.1 安全启动概念

```
安全启动 (Secure Boot) 链:

┌──────────────────────────────────────────────────────────────┐
│                                                               │
│  ROM Bootloader (不可修改)                                    │
│  └── 验证: eFuse中的公钥哈希                                  │
│      │ 验证通过                                               │
│      ▼                                                        │
│  First Stage Bootloader (SPL/U-Boot SPL)                     │
│  └── 验证: ROM BL验证的签名                                   │
│      │ 验证通过                                               │
│      ▼                                                        │
│  Second Stage Bootloader (U-Boot/Mainline)                    │
│  └── 验证: SPL验证的签名                                      │
│      │ 验证通过                                               │
│      ▼                                                        │
│  Linux Kernel                                                 │
│  └── 验证: U-Boot验证的签名                                   │
│      │ 验证通过                                               │
│      ▼                                                        │
│  Device Tree Blob (DTB)                                       │
│  └── 验证: U-Boot验证的签名                                   │
│      │ 验证通过                                               │
│      ▼                                                        │
│  Root Filesystem (initramfs/rootfs)                           │
│  └── 验证: 内核验证的签名/dm-verity                           │
│      │ 验证通过                                               │
│      ▼                                                        │
│  用户应用                                                     │
│  └── 验证: 应用级签名/权限检查                                │
│                                                               │
└──────────────────────────────────────────────────────────────┘

每一级验证下一级, 形成信任链:
ROM BL -> SPL -> U-Boot -> Kernel -> DTB -> rootfs -> Apps

如果任何一级验证失败:
- 系统拒绝启动
- 或进入恢复模式
- 或从备份启动
```

### 8.3 固件签名验证

```python
#!/usr/bin/env python3
"""固件签名验证工具"""
import hashlib
import struct
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import padding, rsa
from cryptography.hazmat.primitives.serialization import load_pem_public_key

class FirmwareSigner:
    """固件签名和验证工具"""
    
    def __init__(self):
        self.private_key = None
        self.public_key = None
    
    def generate_keys(self, key_size=2048):
        """生成RSA密钥对"""
        self.private_key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=key_size,
        )
        self.public_key = self.private_key.public_key()
        
        # 保存私钥
        with open('private_key.pem', 'wb') as f:
            f.write(self.private_key.private_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PrivateFormat.PKCS8,
                encryption_algorithm=serialization.NoEncryption(),
            ))
        
        # 保存公钥
        with open('public_key.pem', 'wb') as f:
            f.write(self.public_key.public_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PublicFormat.SubjectPublicKeyInfo,
            ))
        
        print(f"[+] 密钥对已生成 (RSA-{key_size})")
    
    def sign_firmware(self, firmware_path, output_path='firmware_signed.bin'):
        """对固件进行签名"""
        with open(firmware_path, 'rb') as f:
            firmware_data = f.read()
        
        # 计算固件哈希
        firmware_hash = hashlib.sha256(firmware_data).digest()
        
        # 使用私钥签名
        signature = self.private_key.sign(
            firmware_hash,
            padding.PSS(
                mgf=padding.MGF1(hashes.SHA256()),
                salt_length=padding.PSS.MAX_LENGTH,
            ),
            hashes.SHA256(),
        )
        
        # 构建签名固件格式
        # [魔数(4)] [版本(4)] [固件大小(4)] [哈希大小(4)] [签名大小(4)]
        # [固件数据] [固件哈希] [签名]
        magic = b'FWSG'  # Firmware Signature
        version = struct.pack('>I', 1)
        fw_size = struct.pack('>I', len(firmware_data))
        hash_size = struct.pack('>I', len(firmware_hash))
        sig_size = struct.pack('>I', len(signature))
        
        signed_firmware = (magic + version + fw_size + 
                          hash_size + sig_size +
                          firmware_data + firmware_hash + signature)
        
        with open(output_path, 'wb') as f:
            f.write(signed_firmware)
        
        print(f"[+] 固件已签名: {output_path}")
        print(f"    原始大小: {len(firmware_data)} bytes")
        print(f"    签名后大小: {len(signed_firmware)} bytes")
        print(f"    签名大小: {len(signature)} bytes")
    
    def verify_firmware(self, signed_firmware_path):
        """验证固件签名"""
        with open(signed_firmware_path, 'rb') as f:
            data = f.read()
        
        # 解析头部
        offset = 0
        magic = data[0:4]
        if magic != b'FWSG':
            print("[-] 无效的固件签名格式")
            return False
        
        version = struct.unpack('>I', data[4:8])[0]
        fw_size = struct.unpack('>I', data[8:12])[0]
        hash_size = struct.unpack('>I', data[12:16])[0]
        sig_size = struct.unpack('>I', data[16:20])[0]
        
        offset = 20
        
        # 提取各部分
        firmware_data = data[offset:offset + fw_size]
        offset += fw_size
        
        stored_hash = data[offset:offset + hash_size]
        offset += hash_size
        
        signature = data[offset:offset + sig_size]
        
        # 验证哈希
        computed_hash = hashlib.sha256(firmware_data).digest()
        if computed_hash != stored_hash:
            print("[-] 固件哈希验证失败!")
            return False
        
        # 验证签名
        try:
            self.public_key.verify(
                signature,
                stored_hash,
                padding.PSS(
                    mgf=padding.MGF1(hashes.SHA256()),
                    salt_length=padding.PSS.MAX_LENGTH,
                ),
                hashes.SHA256(),
            )
            print("[+] 固件签名验证成功!")
            return True
        except Exception as e:
            print(f"[-] 签名验证失败: {e}")
            return False

if __name__ == '__main__':
    signer = FirmwareSigner()
    signer.generate_keys()
    
    # 创建测试固件
    test_fw = b'Firmware content for testing' * 100
    with open('test_firmware.bin', 'wb') as f:
        f.write(test_fw)
    
    # 签名
    signer.sign_firmware('test_firmware.bin')
    
    # 验证
    signer.verify_firmware('firmware_signed.bin')
```

## 9. JTAG/SWD调试安全

### 9.1 调试接口风险

```
调试接口安全风险:

┌──────────────────────────────────────────────────────────┐
│ JTAG (Joint Test Action Group):                           │
│  - 4/5线接口: TCK, TMS, TDI, TDO, (TRST)                │
│  - 可读写所有内存                                         │
│  - 可注入并执行任意代码                                   │
│  - 可绕过安全启动                                         │
│  - 可读取加密密钥                                         │
├──────────────────────────────────────────────────────────┤
│ SWD (Serial Wire Debug):                                 │
│  - 2线接口: SWCLK, SWDIO                                 │
│  - 功能类似JTAG但引脚更少                                │
│  - 同样可完全控制设备                                     │
├──────────────────────────────────────────────────────────┤
│ UART (串口调试):                                          │
│  - TX, RX, GND                                           │
│  - 可能暴露调试Shell                                      │
│  - 可能输出敏感信息                                       │
└──────────────────────────────────────────────────────────┘

攻击场景:
1. 通过JTAG读取Flash内容 (固件提取)
2. 通过SWD注入代码 (代码执行)
3. 通过JTAG修改eFuse (绕过安全启动)
4. 通过UART获取Root Shell
5. 通过JTAG提取加密密钥
```

```c
void Debug_Auth_Handler(void) {
    if (debug_attempts++ > 3) {
        // 超过3次尝试, 永久禁用调试
        disable_debug_interface();
        return;
    }
    
    if (debug_unlock == DEBUG_PASSWORD) {
        // 密码正确, 临时启用调试
        DBGMCU->CR |= DBGMCU_CR_DBG_STOP;
    }
}

// 4. 调试检测
// 检测调试器是否连接
bool is_debugger_attached(void) {
    // 方法1: 检查调试状态寄存器
    if (CoreDebug->DHCSR & CoreDebug_DHCSR_C_DEBUGEN_Msk) {
        return true;
    }
    
    // 方法2: 检查周期时间 (调试暂停会延长时间)
    uint32_t start = DWT->CYCCNT;
    __NOP();
    __NOP();
    __NOP();
    uint32_t elapsed = DWT->CYCCNT - start;
    
    if (elapsed > 100) {  // 正常应小于100个时钟周期
        return true;  // 可能在调试
    }
    
    return false;
}

// 5. 调试响应动作
void security_check(void) {
    if (is_debugger_attached()) {
        // 清除敏感数据
        memset(sensitive_buffer, 0, sizeof(sensitive_buffer));
        
        // 锁定安全功能
        lock_security_features();
        
        // 可选: 触发安全状态
        enter_safe_mode();
        
        // 可选: 记录事件
        log_security_event("DEBUG_DETECTED");
    }
}
```
