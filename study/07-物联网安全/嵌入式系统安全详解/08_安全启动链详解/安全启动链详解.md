# 安全启动链详解

### 8.1 安全启动概念

```
安全启动 (Secure Boot) 链:

┌──────────────────────────────────────────────────────────────┐
│                                                               │
│  ROM Bootloader (不可修改)                                    │
│  └── 验证: eFuse中的公钥哈希                                  │
│      │ 验证通过                                               │
│      ▼                                                        │
│  First Stage Bootloader (SPL/U-Boot SPL)                     │
│  └── 验证: ROM BL验证的签名                                   │
│      │ 验证通过                                               │
│      ▼                                                        │
│  Second Stage Bootloader (U-Boot/Mainline)                    │
│  └── 验证: SPL验证的签名                                      │
│      │ 验证通过                                               │
│      ▼                                                        │
│  Linux Kernel                                                 │
│  └── 验证: U-Boot验证的签名                                   │
│      │ 验证通过                                               │
│      ▼                                                        │
│  Device Tree Blob (DTB)                                       │
│  └── 验证: U-Boot验证的签名                                   │
│      │ 验证通过                                               │
│      ▼                                                        │
│  Root Filesystem (initramfs/rootfs)                           │
│  └── 验证: 内核验证的签名/dm-verity                           │
│      │ 验证通过                                               │
│      ▼                                                        │
│  用户应用                                                     │
│  └── 验证: 应用级签名/权限检查                                │
│                                                               │
└──────────────────────────────────────────────────────────────┘

每一级验证下一级, 形成信任链:
ROM BL -> SPL -> U-Boot -> Kernel -> DTB -> rootfs -> Apps

如果任何一级验证失败:
- 系统拒绝启动
- 或进入恢复模式
- 或从备份启动
```

### 8.2 U-Boot安全启动

```c
// U-Boot安全启动配置

// 1. 生成密钥对
// openssl genrsa -aes256 -out private.key 2048
// openssl req -new -x509 -key private.key -out public.crt

// 2. 配置U-Boot (defconfig)
// CONFIG_FIT=y                 // 启用FIT镜像
// CONFIG_FIT_SIGNATURE=y      // 启用FIT签名验证
// CONFIG_RSA=y                 // RSA支持
// CONFIG_RSA_VERIFY=y          // RSA验证

// 3. 创建FIT镜像描述文件 (.its)
/dts-v1/;

/ {
    description = "Linux kernel image with signature";
    #address-cells = <1>;

    images {
        kernel@1 {
            description = "Linux Kernel";
            data = /incbin/("zImage");
            type = "kernel";
            arch = "arm";
            os = "linux";
            compression = "none";
            load = <0x80008000>;
            entry = <0x80008000>;
            hash@1 {
                algo = "sha256";
            };
        };

        fdt@1 {
            description = "Device Tree Blob";
            data = /incbin/("board.dtb");
            type = "flat_dt";
            arch = "arm";
            compression = "none";
            hash@1 {
                algo = "sha256";
            };
        };
    };

    configurations {
        default = "conf@1";
        conf@1 {
            description = "Default configuration";
            kernel = "kernel@1";
            fdt = "fdt@1";
            signature@1 {
                algo = "sha256,rsa2048";
                key-name-hint = "dev";
                sign-images = "kernel", "fdt";
            };
        };
    };
};

// 4. 构建签名FIT镜像
// mkimage -f image.its -K -r -k keys/ -r image.itb

// 5. U-Boot验证启动
// => bootm 0x82000000#conf@1
// U-Boot会自动验证签名
```

### 8.3 固件签名验证

```python
#!/usr/bin/env python3
"""固件签名验证工具"""
import hashlib
import struct
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import padding, rsa
from cryptography.hazmat.primitives.serialization import load_pem_public_key

class FirmwareSigner:
    """固件签名和验证工具"""
    
    def __init__(self):
        self.private_key = None
        self.public_key = None
    
    def generate_keys(self, key_size=2048):
        """生成RSA密钥对"""
        self.private_key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=key_size,
        )
        self.public_key = self.private_key.public_key()
        
        # 保存私钥
        with open('private_key.pem', 'wb') as f:
            f.write(self.private_key.private_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PrivateFormat.PKCS8,
                encryption_algorithm=serialization.NoEncryption(),
            ))
        
        # 保存公钥
        with open('public_key.pem', 'wb') as f:
            f.write(self.public_key.public_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PublicFormat.SubjectPublicKeyInfo,
            ))
        
        print(f"[+] 密钥对已生成 (RSA-{key_size})")
    
    def sign_firmware(self, firmware_path, output_path='firmware_signed.bin'):
        """对固件进行签名"""
        with open(firmware_path, 'rb') as f:
            firmware_data = f.read()
        
        # 计算固件哈希
        firmware_hash = hashlib.sha256(firmware_data).digest()
        
        # 使用私钥签名
        signature = self.private_key.sign(
            firmware_hash,
            padding.PSS(
                mgf=padding.MGF1(hashes.SHA256()),
                salt_length=padding.PSS.MAX_LENGTH,
            ),
            hashes.SHA256(),
        )
        
        # 构建签名固件格式
        # [魔数(4)] [版本(4)] [固件大小(4)] [哈希大小(4)] [签名大小(4)]
        # [固件数据] [固件哈希] [签名]
        magic = b'FWSG'  # Firmware Signature
        version = struct.pack('>I', 1)
        fw_size = struct.pack('>I', len(firmware_data))
        hash_size = struct.pack('>I', len(firmware_hash))
        sig_size = struct.pack('>I', len(signature))
        
        signed_firmware = (magic + version + fw_size + 
                          hash_size + sig_size +
                          firmware_data + firmware_hash + signature)
        
        with open(output_path, 'wb') as f:
            f.write(signed_firmware)
        
        print(f"[+] 固件已签名: {output_path}")
        print(f"    原始大小: {len(firmware_data)} bytes")
        print(f"    签名后大小: {len(signed_firmware)} bytes")
        print(f"    签名大小: {len(signature)} bytes")
    
    def verify_firmware(self, signed_firmware_path):
        """验证固件签名"""
        with open(signed_firmware_path, 'rb') as f:
            data = f.read()
        
        # 解析头部
        offset = 0
        magic = data[0:4]
        if magic != b'FWSG':
            print("[-] 无效的固件签名格式")
            return False
        
        version = struct.unpack('>I', data[4:8])[0]
        fw_size = struct.unpack('>I', data[8:12])[0]
        hash_size = struct.unpack('>I', data[12:16])[0]
        sig_size = struct.unpack('>I', data[16:20])[0]
        
        offset = 20
        
        # 提取各部分
        firmware_data = data[offset:offset + fw_size]
        offset += fw_size
        
        stored_hash = data[offset:offset + hash_size]
        offset += hash_size
        
        signature = data[offset:offset + sig_size]
        
        # 验证哈希
        computed_hash = hashlib.sha256(firmware_data).digest()
        if computed_hash != stored_hash:
            print("[-] 固件哈希验证失败!")
            return False
        
        # 验证签名
        try:
            self.public_key.verify(
                signature,
                stored_hash,
                padding.PSS(
                    mgf=padding.MGF1(hashes.SHA256()),
                    salt_length=padding.PSS.MAX_LENGTH,
                ),
                hashes.SHA256(),
            )
            print("[+] 固件签名验证成功!")
            return True
        except Exception as e:
            print(f"[-] 签名验证失败: {e}")
            return False

if __name__ == '__main__':
    signer = FirmwareSigner()
    signer.generate_keys()
    
    # 创建测试固件
    test_fw = b'Firmware content for testing' * 100
    with open('test_firmware.bin', 'wb') as f:
        f.write(test_fw)
    
    # 签名
    signer.sign_firmware('test_firmware.bin')
    
    # 验证
    signer.verify_firmware('firmware_signed.bin')
```
