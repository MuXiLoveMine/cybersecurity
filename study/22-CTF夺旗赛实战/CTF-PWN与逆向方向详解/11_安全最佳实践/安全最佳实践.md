// 1. 使用安全函数替代不安全函数
// ❌ 危险: gets, strcpy, sprintf
// ✅ 安全: fgets, strncpy, snprintf

void safe_function() {
    char buf[64];
    // 使用fgets限制读取长度
    fgets(buf, sizeof(buf), stdin);
    // 使用snprintf限制写入长度
    snprintf(buf, sizeof(buf), "%s", input);
}
