### 7.1 静态逆向

```python
# 静态逆向使用IDA Pro / Ghidra

# 逆向分析步骤:
# 1. 查看main函数
# 2. 理解程序流程
# 3. 找到关键比较函数
# 4. 分析加密算法
# 5. 编写逆向脚本

# IDA常用快捷键:
# F5 - 反编译为C伪代码
# N - 重命名变量/函数
# Y - 修改变量类型
# G - 跳转到地址
# X - 查看交叉引用
# ; - 添加注释

# 典型逆向题目示例:
"""
# IDA伪代码
int __cdecl main(int argc, const char **argv, const char **envp) {
    char input[32];
    char encrypted[32];
    
    scanf("%32s", input);
    encrypt(input, encrypted);
    
    if (strcmp(encrypted, "ENCRYPTED_FLAG") == 0)
        printf("Correct!\n");
    else
        printf("Wrong!\n");
    
    return 0;
}

void encrypt(char *input, char *output) {
    for (int i = 0; i < 32; i++) {
        output[i] = input[i] ^ 0x42;  // 简单异或
        output[i] = (output[i] << 3) | (output[i] >> 5);  // 循环左移3
    }
}
"""

# 逆向解密脚本:
def decrypt(encrypted):
    result = b''
    for c in encrypted:
        # 逆向循环左移3 = 循环右移3
        c = (c >> 3) | (c << 5)
        c = c & 0xFF
        # 逆向异或
        c = c ^ 0x42
        result += bytes([c])
    return result

encrypted_flag = b'\x...'  # 从程序中提取
flag = decrypt(encrypted_flag)
print(flag.decode())
```

### 7.3 算法还原

```python
# 复杂算法逆向 - 需要将汇编还原为Python

# 示例: 分析一个复杂的校验算法
"""
分析步骤:
1. 在IDA中找到校验函数
2. 理解每个操作的含义
3. 还原为Python代码
4. 编写解密/爆破脚本
"""

# 逆向还原示例 - 变种Base64
def reverse_custom_b64(encrypted):
    """还原自定义Base64解码"""
    # 从IDA分析得知的变表
    custom_table = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    # 原始表
    std_table = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    
    # 如果是变表，替换回标准表
    trans = str.maketrans(custom_table, std_table)
    standard = encrypted.translate(trans)
    
    # 标准Base64解码
    import base64
    return base64.b64decode(standard)

# z3求解器用于约束求解
from z3 import *

def solve_with_z3():
    """使用z3求解器自动求解"""
    s = Solver()
    
    # 定义Flag变量
    flag = [BitVec(f'f_{i}', 8) for i in range(32)]
    
    # 添加约束条件（从逆向分析得到）
    for i in range(32):
        s.add(flag[i] >= 32)  # 可打印字符
        s.add(flag[i] <= 126)
    
    # 添加加密约束
    s.add(flag[0] ^ 0x42 == 0x67)
    s.add(flag[1] ^ 0x42 == 0x65)
    # ... 更多约束
    
    if s.check() == sat:
        m = s.model()
        result = ''.join(chr(m[flag[i]].as_long()) for i in range(32))
        return result
    return None
```
