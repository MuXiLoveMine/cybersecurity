### 8.2 控制流平坦化

```python
# 控制流平坦化(Control Flow Flattening)
# 原理: 将正常的控制流打乱，使用一个调度器分发

# 原始代码:
"""
if (x == 1) {
    funcA();
} else if (x == 2) {
    funcB();
} else {
    funcC();
}
"""

# 平坦化后:
"""
while (state != EXIT) {
    switch (state) {
        case 0x1234:  // funcA的标签
            funcA();
            state = next_state;
            break;
        case 0x5678:  // funcB的标签
            funcB();
            state = next_state;
            break;
        case 0x9abc:  // funcC的标签
            funcC();
            state = next_state;
            break;
    }
}
"""

# 对抗方法:
# 1. 使用deflat工具去平坦化
#    https://github.com/SsageTrick/deflat
# 2. 使用angr符号执行跟踪路径
# 3. 动态调试跟踪实际执行路径

# angr去平坦化示例:
import angr

# 加载二进制
proj = angr.Project('./flattened_binary', auto_load_libs=False)
# 创建状态
state = proj.factory.entry_state()
# 创建模拟管理器
simgr = proj.factory.simulation_manager(state)
# 运行直到找到满足条件的状态
simgr.explore(find=lambda s: b'Correct' in s.posix.dumps(1))
# 获取解
if simgr.found:
    found = simgr.found[0]
    print(found.posix.dumps(0))  # 输入
```

### 8.3 VM混淆

```python
# VM混淆(Virtual Machine Obfuscation)
# 原理: 将原始代码转换为自定义虚拟机的字节码

# 分析VM混淆的步骤:
# 1. 找到VM dispatcher（通常是switch-case循环）
# 2. 分析每个handler对应的操作
# 3. 编写反汇编器将字节码转换回伪代码
# 4. 理解程序逻辑

# VM混淆结构:
"""
while (1) {
    opcode = code[pc++];
    switch (opcode) {
        case 0x01:  // push
            stack[sp++] = code[pc++];
            break;
        case 0x02:  // add
            stack[sp-2] += stack[sp-1];
            sp--;
            break;
        case 0x03:  // xor
            stack[sp-2] ^= stack[sp-1];
            sp--;
            break;
        case 0x04:  // jmp
            pc = code[pc];
            break;
        case 0x05:  // cmp
            flag = (stack[sp-2] == stack[sp-1]);
            sp -= 2;
            break;
        // ... 更多handler
    }
}
"""

# 使用angr符号执行自动求解:
import angr
import claripy

proj = angr.Project('./vm_binary', auto_load_libs=False)

# 创建符号输入
flag_len = 32
flag = claripy.BVS('flag', 8 * flag_len)

# 设置初始状态
state = proj.factory.entry_state(stdin=angr.SimFileStream)
for i in range(flag_len):
    byte = flag.get_byte(i)
    state.solver.add(byte >= 32)
    state.solver.add(byte <= 126)

# 运行
simgr = proj.factory.simulation_manager(state)
simgr.explore(find=lambda s: b'Correct' in s.posix.dumps(1))

if simgr.found:
    solution = simgr.found[0].solver.eval(flag, cast_to=bytes)
    print(f"Flag: {solution.decode()}")
```

### 8.4 angr符号执行

```python
import angr
import claripy

# angr是强大的符号执行框架，适合自动化逆向

# 基本使用流程:
def angr_solve(binary_path, find_addr=None, avoid_addr=None, stdin_size=32):
    """使用angr自动求解"""
    
    # 1. 创建项目
    proj = angr.Project(binary_path, auto_load_libs=False)
    
    # 2. 创建初始状态
    state = proj.factory.entry_state()
    
    # 3. 设置输入约束（如果需要）
    for i in range(stdin_size):
        byte = state.posix.stdin.load(i, 1)
        state.solver.add(byte >= 0x20)
        state.solver.add(byte <= 0x7e)
    
    # 4. 创建模拟管理器
    simgr = proj.factory.simulation_manager(state)
    
    # 5. 探索
    if find_addr and avoid_addr:
        simgr.explore(find=find_addr, avoid=avoid_addr)
    elif find_addr:
        simgr.explore(find=find_addr)
    else:
        # 根据输出字符串查找
        simgr.explore(find=lambda s: b'Correct' in s.posix.dumps(1),
                      avoid=lambda s: b'Wrong' in s.posix.dumps(1))
    
    # 6. 提取结果
    if simgr.found:
        found_state = simgr.found[0]
        solution = found_state.posix.dumps(0)
        return solution
    return None

# 使用示例
result = angr_solve('./crackme')
if result:
    print(f"[+] 找到解: {result}")
else:
    print("[-] 未找到解")
```
