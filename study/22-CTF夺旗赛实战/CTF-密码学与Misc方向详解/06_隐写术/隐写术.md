### 6.1 图片LSB隐写

```python
from PIL import Image
import numpy as np

def lsb_extract(image_path):
    """提取图片LSB隐写数据"""
    img = Image.open(image_path)
    pixels = np.array(img)
    
    # 提取每个像素的最低位
    bits = []
    for row in pixels:
        for pixel in row:
            for channel in pixel[:3]:  # RGB三通道
                bits.append(channel & 1)
    
    # 将bit序列转换为字节
    data = b''
    for i in range(0, len(bits) - 7, 8):
        byte = 0
        for j in range(8):
            byte = (byte << 1) | bits[i + j]
        data += bytes([byte])
        if data[-2:] == b'\x00\x00':  # 遇到结束标记
            break
    
    return data
```

### 6.2 图片尾部数据

```python
def extract_trailing_data(image_path):
    """提取图片文件尾部隐藏的数据"""
    with open(image_path, 'rb') as f:
        content = f.read()
    
    # 检查常见的文件头签名
    signatures = {
        b'PK': 'ZIP',
        b'\x89PNG': 'PNG',
        b'\xff\xd8\xff': 'JPEG',
        b'GIF89a': 'GIF',
        b'%PDF': 'PDF',
        b'Rar': 'RAR',
        b'7z': '7z',
    }
    
    # 找到第一个文件结束后的附加数据
    for sig, name in signatures.items():
        pos = content.find(sig, 10)  # 跳过文件头
        if pos > 0:
            print(f"[+] 在偏移 {pos} 发现 {name} 文件")
            with open(f'extracted_{name.lower()}', 'wb') as f:
                f.write(content[pos:])
            return content[pos:]
    
    return None

# 使用binwalk自动提取
# binwalk image.png
# binwalk -e image.png  # 自动提取嵌入文件
```

```python
# Python提取音频LSB
def audio_lsb_extract(audio_path):
    """提取WAV音频LSB隐写"""
    import wave
    
    with wave.open(audio_path, 'rb') as wav:
        n_channels = wav.getnchannels()
        sample_width = wav.getsampwidth()
        n_frames = wav.getnframes()
        frames = wav.readframes(n_frames)
    
    # 根据采样宽度解析数据
    if sample_width == 1:
        samples = np.frombuffer(frames, dtype=np.uint8)
    elif sample_width == 2:
        samples = np.frombuffer(frames, dtype=np.int16)
    
    # 提取LSB
    bits = samples & 1
    data = b''
    for i in range(0, len(bits) - 7, 8):
        byte = 0
        for j in range(8):
            byte = (byte << 1) | bits[i + j]
        data += bytes([byte])
        if len(data) > 2 and data[-2:] == b'\x00\x00':
            break
    
    return data
```
