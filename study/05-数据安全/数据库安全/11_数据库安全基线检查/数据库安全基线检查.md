### 11.2 PostgreSQL安全基线

```sql
-- PostgreSQL安全配置检查

-- 1. 检查连接认证配置
-- pg_hba.conf 应使用 scram-sha-256 认证
-- host all all 0.0.0.0/0 scram-sha-256

-- 2. 检查密码加密方式
SHOW password_encryption;  -- 应为 scram-sha-256

-- 3. 检查SSL配置
SHOW ssl;  -- 应为 on
SHOW ssl_cert;

-- 4. 检查日志配置
SHOW logging_collector;  -- 应为 on
SHOW log_connections;  -- 应为 on
SHOW log_disconnections;  -- 应为 on
SHOW log_statement;  -- 应为 'all' 或 'ddl'

-- 5. 检查权限
-- 查看所有用户的权限
SELECT rolname, rolsuper, rolcreatedb, rolcreaterole
FROM pg_roles
ORDER BY rolname;

-- 检查schema权限
SELECT nspname, nspowner
FROM pg_namespace
WHERE nspname NOT LIKE 'pg_%' AND nspname != 'information_schema';

-- 检查表权限
SELECT grantee, table_schema, table_name, privilege_type
FROM information_schema.table_privileges
WHERE table_schema NOT IN ('pg_catalog', 'information_schema')
ORDER BY grantee, table_schema, table_name;

-- 6. 安全参数
ALTER SYSTEM SET ssl = on;
ALTER SYSTEM SET password_encryption = 'scram-sha-256';
ALTER SYSTEM SET log_connections = on;
ALTER SYSTEM SET log_disconnections = on;
ALTER SYSTEM SET log_statement = 'ddl';
ALTER SYSTEM SET row_security = on;
-- 需要reload: SELECT pg_reload_conf();
```

### 11.3 SQL Server安全基线

```sql
-- SQL Server安全检查

-- 1. 检查认证模式 (应为Windows认证或混合模式)
SELECT SERVERPROPERTY('IsIntegratedSecurityOnly');
-- 0 = 混合模式, 1 = 仅Windows认证

-- 2. 检查登录审计
SELECT name, value, value_in_use
FROM sys.configurations
WHERE name LIKE '%audit%';

-- 3. 检查SA账户状态
SELECT name, is_disabled, is_policy_checked
FROM sys.sql_logins
WHERE name = 'sa';

-- 4. 检查弱密码
SELECT name, is_policy_checked, is_expiration_checked
FROM sys.sql_logins;

-- 5. 检查TDE状态
SELECT db.name, dm.encryption_state
FROM sys.databases db
LEFT JOIN sys.dm_database_encryption_keys dm ON db.database_id = dm.database_id;

-- 6. 检查xp_cmdshell (应禁用)
SELECT value_in_use
FROM sys.configurations
WHERE name = 'xp_cmdshell';

-- 7. 启用安全配置
EXEC sp_configure 'show advanced options', 1;
RECONFIGURE;
EXEC sp_configure 'xp_cmdshell', 0;  -- 禁用
EXEC sp_configure 'cross db ownership chaining', 0;  -- 禁用跨库所有权链
RECONFIGURE;
```
