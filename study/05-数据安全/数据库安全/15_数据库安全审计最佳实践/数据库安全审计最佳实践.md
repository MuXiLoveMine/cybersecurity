# 数据库安全审计最佳实践

### 15.1 审计策略设计

```sql
-- MySQL企业审计配置
-- 安装审计插件
INSTALL PLUGIN audit_log SONAME 'audit_log.so';

-- 配置审计参数
SET GLOBAL audit_log_format = 'JSON';
SET GLOBAL audit_log_policy = 'ALL';  -- ALL/LOGINS/QUERIES/NONE
SET GLOBAL audit_log_buffer_size = 8192;

-- 审计指定操作
-- 审计所有DDL操作
SET GLOBAL audit_log_buffered = ON;

-- PostgreSQL审计(pgAudit扩展)
-- 安装
CREATE EXTENSION pgaudit;

-- 配置
ALTER SYSTEM SET pgaudit.log = 'write,ddl';
ALTER SYSTEM SET pgaudit.log_relation = on;
ALTER SYSTEM SET pgaudit.log_parameter = on;

-- 验证审计
SELECT name, setting FROM pg_settings WHERE name LIKE 'pgaudit%';

-- SQL Server审计
-- 创建Server Audit
CREATE SERVER AUDIT ServerAudit
TO FILE (FILEPATH = '/var/opt/mssql/audit/')
WITH (ON_FAILURE = CONTINUE, MAX_SIZE = 100 MB);

-- 创建Database Audit Specification
USE TargetDB;
CREATE DATABASE AUDIT SPECIFICATION DbAudit
FOR SERVER AUDIT ServerAudit
ADD (SELECT, INSERT, UPDATE, DELETE ON schema::dbo BY public);

-- 启用审计
ALTER SERVER AUDIT ServerAudit WITH (STATE = ON);
ALTER DATABASE AUDIT SPECIFICATION DbAudit WITH (STATE = ON);
```

### 15.2 审计日志分析与告警

```python
#!/usr/bin/env python3
"""数据库审计日志分析与异常检测"""

import re
import json
from collections import defaultdict, Counter
from datetime import datetime

class DatabaseAuditAnalyzer:
    """数据库审计日志分析器"""

    def __init__(self):
        self.alerts = []
        self.sql_patterns = {
            "sql_injection": [
                r"(?i)union\s+select",
                r"(?i)or\s+1\s*=\s*1",
                r"(?i)and\s+1\s*=\s*1",
                r"(?i)'\s*or\s*'",
                r"(?i)sleep\s*\(",
                r"(?i)benchmark\s*\(",
                r"(?i)load_file\s*\(",
                r"(?i)into\s+outfile",
                r"(?i)into\s+dumpfile",
            ],
            "data_exfiltration": [
                r"(?i)select\s+\*\s+from\s+(user|admin|customer|order)",
                r"(?i)dump\s+",
                r"(?i)into\s+out",
            ],
            "privilege_escalation": [
                r"(?i)grant\s+",
                r"(?i)create\s+user",
                r"(?i)alter\s+user",
                r"(?i)set\s+password",
                r"(?i)sp_addrole",
                r"(?i)sp_adduser",
            ],
            "dangerous_commands": [
                r"(?i)drop\s+(table|database)",
                r"(?i)truncate\s+table",
                r"(?i)delete\s+from\s+\w+\s*$",
                r"(?i)update\s+\w+\s+set\s+\w+\s*=.*where\s+1\s*=\s*1",
                r"(?i)exec\s+xp_cmdshell",
                r"(?i)exec\s+sp_oa",
            ],
        }

    def analyze_log(self, log_entry):
        """分析单条审计日志"""
        if isinstance(log_entry, str):
            log_entry = json.loads(log_entry)

        sql_text = log_entry.get("sql_text", "")
        user = log_entry.get("user", "unknown")
        source_ip = log_entry.get("source_ip", "unknown")
        timestamp = log_entry.get("timestamp", datetime.now().isoformat())

        for category, patterns in self.sql_patterns.items():
            for pattern in patterns:
                if re.search(pattern, sql_text):
                    alert = {
                        "timestamp": timestamp,
                        "category": category,
                        "user": user,
                        "source_ip": source_ip,
                        "sql_text": sql_text[:200],
                        "pattern": pattern,
                    }
                    self.alerts.append(alert)
                    break

    def detect_anomalous_access(self, logs, threshold=100):
        """检测异常访问模式"""
        user_query_count = Counter()
        ip_query_count = Counter()
        for log in logs:
            if isinstance(log, str):
                log = json.loads(log)
            user_query_count[log.get("user", "unknown")] += 1
            ip_query_count[log.get("source_ip", "unknown")] += 1

        for user, count in user_query_count.items():
            if count > threshold:
                self.alerts.append({
                    "category": "excessive_queries",
                    "user": user,
                    "count": count,
                    "message": f"用户 {user} 执行了 {count} 次查询，超过阈值 {threshold}"
                })

        for ip, count in ip_query_count.items():
            if count > threshold * 2:
                self.alerts.append({
                    "category": "excessive_queries_from_ip",
                    "source_ip": ip,
                    "count": count,
                    "message": f"IP {ip} 执行了 {count} 次查询"
                })

    def get_report(self):
        """生成安全报告"""
        report = {
            "total_alerts": len(self.alerts),
            "alerts_by_category": Counter(a["category"] for a in self.alerts),
            "alerts": self.alerts
        }
        return report
```
