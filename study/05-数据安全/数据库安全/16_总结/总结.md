# 总结

数据库安全是数据安全防护的核心环节。关键要点：

1. **纵深防御**：从网络层（bind/防火墙）、认证层（强密码/多因素）、授权层（最小权限）、审计层（日志监控）、加密层（TDE/传输加密）多维防护
2. **最小权限原则**：应用账户仅授予业务必需权限，禁止使用 root/sa 账户连接应用
3. **默认安全**：修改默认端口、删除默认账户和测试库、启用认证
4. **持续审计**：定期扫描漏洞、审计 SQL 访问、发现敏感数据分布
5. **加密保护**：静态加密（TDE）与传输加密（TLS）并重，备份文件同样需要加密

#### 2. SQL Server 审计配置

```sql
-- ========== SQL Server 审计 ==========

-- 1. 创建服务器审计对象
USE master;
CREATE SERVER AUDIT [ServerAudit]
TO FILE 
(
    FILEPATH = N'C:\Audit\',
    MAXSIZE = 100 MB,
    MAX_ROLLOVER_FILES = 10,
    RESERVE_DISK_SPACE = OFF
)
WITH
(   QUEUE_DELAY = 1000,
    ON_FAILURE = CONTINUE
);

-- 2. 启用服务器审计
ALTER SERVER AUDIT [ServerAudit] WITH (STATE = ON);

-- 3. 创建服务器审计规范
CREATE SERVER AUDIT SPECIFICATION [ServerAuditSpec]
FOR SERVER AUDIT [ServerAudit]
    ADD (FAILED_LOGIN_GROUP),          -- 失败登录
    ADD (SUCCESSFUL_LOGIN_GROUP),      -- 成功登录
    ADD (SERVER_ROLE_MEMBER_CHANGE_GROUP),  -- 服务器角色变更
    ADD (DATABASE_CHANGE_GROUP),       -- 数据库变更
    ADD (SERVER_PRINCIPAL_CHANGE_GROUP), -- 登录账户变更
    ADD (SERVER_PERMISSION_CHANGE_GROUP), -- 服务器权限变更
    ADD (AUDIT_CHANGE_GROUP)           -- 审计配置变更
WITH (STATE = ON);

-- 4. 创建数据库审计规范
USE YourDatabase;
CREATE DATABASE AUDIT SPECIFICATION [DatabaseAuditSpec]
FOR SERVER AUDIT [ServerAudit]
    ADD (SELECT ON dbo.SensitiveData BY public),
    ADD (UPDATE ON dbo.SensitiveData BY public),
    ADD (DELETE ON dbo.SensitiveData BY public),
    ADD (SCHEMA_OBJECT_CHANGE_GROUP),
    ADD (DATABASE_PERMISSION_CHANGE_GROUP),
    ADD (DATABASE_ROLE_MEMBER_CHANGE_GROUP)
WITH (STATE = ON);

-- 5. 查看审计记录
SELECT * FROM sys.fn_get_audit_file('C:\Audit\*.sqlaudit', DEFAULT, DEFAULT)
ORDER BY event_time DESC;

-- 6. 查看审计配置
SELECT name, type_desc, status, audit_file_path
FROM sys.server_audits;

SELECT * FROM sys.server_audit_specifications;
SELECT * FROM sys.database_audit_specifications;

-- ========== C2 审计模式（旧版）==========

-- 启用C2审计（不推荐，已废弃，但仍可用）
sp_configure 'show advanced options', 1;
RECONFIGURE;
sp_configure 'c2 audit mode', 1;
RECONFIGURE;
-- 需要重启SQL Server服务

-- ========== 扩展事件 (Extended Events) ==========

-- 1. 创建扩展事件会话（跟踪登录失败）
CREATE EVENT SESSION [Failed_Login_Tracking] ON SERVER 
ADD EVENT sqlserver.failed_login(
    ACTION(sqlserver.client_app_name,
           sqlserver.client_hostname,
           sqlserver.nt_username,
           sqlserver.sql_text))
ADD TARGET package0.event_file(SET filename=N'C:\Audit\FailedLogin.xel')
WITH (MAX_MEMORY=4096 KB, EVENT_RETENTION_MODE=ALLOW_SINGLE_EVENT_LOSS,
      MAX_DISPATCH_LATENCY=30 SECONDS, MAX_EVENT_SIZE=0 KB,
      MEMORY_PARTITION_MODE=NONE, TRACK_CAUSALITY=OFF, STARTUP_STATE=ON);

-- 2. 启用扩展事件会话
ALTER EVENT SESSION [Failed_Login_Tracking] ON SERVER STATE = START;

-- 3. 查看扩展事件数据
SELECT 
    event_data.value('(event/@name)[1]', 'varchar(50)') AS event_name,
    event_data.value('(event/@timestamp)[1]', 'datetime2') AS event_time,
    event_data.value('(event/action[@name="client_hostname"]/value)[1]', 'varchar(max)') AS client_hostname,
    event_data.value('(event/action[@name="nt_username"]/value)[1]', 'varchar(max)') AS nt_username
FROM 
(   SELECT CAST(event_data AS XML) AS event_data
    FROM sys.fn_xe_file_target_read_file('C:\Audit\FailedLogin*.xel', NULL, NULL, NULL)
) AS events
ORDER BY event_time DESC;
```

#### 3. SQL Server 安全基线检查脚本

```sql
-- SQL Server 安全基线检查脚本

PRINT '========== SQL Server 安全基线检查 ==========';

-- 1. SQL Server版本和补丁
SELECT @@VERSION AS 'SQL Server 版本';
SELECT SERVERPROPERTY('ProductLevel') AS '产品级别',
       SERVERPROPERTY('ProductUpdateLevel') AS '更新级别';

-- 2. 身份验证模式
SELECT CASE SERVERPROPERTY('IsIntegratedSecurityOnly')
    WHEN 1 THEN '仅Windows身份验证（推荐）'
    WHEN 0 THEN '混合模式'
END AS '身份验证模式';

-- 3. sa账户状态
SELECT name, is_disabled, type_desc, create_date
FROM sys.server_principals
WHERE name = 'sa';

-- 4. sysadmin角色成员
SELECT l.name AS '登录名', l.type_desc AS '类型', l.is_disabled AS '是否禁用'
FROM sys.server_role_members srm
JOIN sys.server_principals sp ON srm.role_principal_id = sp.principal_id
JOIN sys.server_principals l ON srm.member_principal_id = l.principal_id
WHERE sp.name = 'sysadmin'
ORDER BY l.name;

-- 5. SQL登录的密码策略检查
SELECT name AS '登录名', 
       is_policy_checked AS '密码策略', 
       is_expiration_checked AS '密码过期',
       is_disabled AS '是否禁用'
FROM sys.sql_logins
ORDER BY name;

-- 6. 空密码的SQL登录（使用LOGINPROPERTY）
SELECT name AS '登录名'
FROM sys.sql_logins
WHERE LOGINPROPERTY(name, 'IsMustChange') = 1
   OR LOGINPROPERTY(name, 'PasswordHash') IS NULL;

-- 7. 审计配置
SELECT name AS '审计名称', type_desc AS '类型', 
       status_desc AS '状态', audit_file_path AS '审计路径'
FROM sys.server_audits;

-- 8. 服务器审计规范
SELECT name AS '规范名称', s.name AS '所属审计', status_desc AS '状态'
FROM sys.server_audit_specifications sas
JOIN sys.server_audits s ON sas.audit_guid = s.audit_guid;

-- 9. 是否启用CLR（可能存在安全风险）
SELECT name, value, value_in_use
FROM sys.configurations
WHERE name = 'clr enabled';

-- 10. 是否启用xp_cmdshell（高风险）
SELECT name, value, value_in_use
FROM sys.configurations
WHERE name = 'xp_cmdshell';

-- 11. 数据库所有者检查
SELECT name AS '数据库名', suser_sname(owner_sid) AS '所有者'
FROM sys.databases
ORDER BY name;

SELECT name AS '数据库名', is_trustworthy_on AS '是否可信'
FROM sys.databases
WHERE is_trustworthy_on = 1;

-- 13. 最近创建的登录账户
SELECT name, type_desc, create_date, modify_date
FROM sys.server_principals
WHERE type IN ('S', 'U', 'G')
  AND create_date > DATEADD(day, -30, GETDATE())
ORDER BY create_date DESC;

       dp.create_date AS '创建时间'
```
