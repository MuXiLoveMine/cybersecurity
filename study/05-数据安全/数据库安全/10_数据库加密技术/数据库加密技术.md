# 数据库加密技术

### 10.1 透明数据加密(TDE)

透明数据加密（Transparent Data Encryption）在存储层自动加密数据库文件，对应用透明。

```sql
-- MySQL TDE配置
-- 1. 安装加密插件
INSTALL PLUGIN keyring_file SONAME 'keyring_file.so';

-- 2. 配置keyring
SET GLOBAL keyring_file_data = '/var/lib/mysql-keyring/keyring';

-- 3. 创建加密表
CREATE TABLE sensitive_data (
    id INT PRIMARY KEY,
    ssn VARCHAR(18),
    credit_card VARCHAR(20)
) ENCRYPTION='Y';

-- 4. 对已有表启用加密
ALTER TABLE existing_table ENCRYPTION='Y';

-- 5. 查看加密状态
SELECT TABLE_SCHEMA, TABLE_NAME, CREATE_OPTIONS
FROM INFORMATION_SCHEMA.TABLES
WHERE CREATE_OPTIONS LIKE '%ENCRYPTION%';

-- SQL Server TDE配置
USE master;
CREATE MASTER KEY ENCRYPTION BY PASSWORD = 'StrongPassword123!';
CREATE CERTIFICATE TDECert WITH SUBJECT = 'TDE Certificate';
USE TargetDB;
CREATE DATABASE ENCRYPTION KEY
WITH ALGORITHM = AES_256
ENCRYPTION BY SERVER CERTIFICATE TDECert;
ALTER DATABASE TargetDB SET ENCRYPTION ON;

-- 查看TDE状态
SELECT db.name, dm.encryption_state, dm.encryptor_type
FROM sys.databases db
JOIN sys.dm_database_encryption_keys dm ON db.database_id = dm.database_id;

-- Oracle TDE配置
-- 列级加密
ALTER TABLE employees ADD (
    ssn VARCHAR2(20) ENCRYPT USING 'AES256'
);

-- 表空间加密
CREATE TABLESPACE secure_data
DATAFILE '/u01/oradata/secure01.dbf' SIZE 100M
ENCRYPTION USING 'AES256'
DEFAULT STORAGE(ENCRYPT);
```

### 10.2 应用层加密与列级加密

```python
#!/usr/bin/env python3
"""数据库应用层加密与脱敏工具"""

import hashlib
import hmac
import os
from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
import base64

class DatabaseEncryption:
    """数据库字段级加密工具"""

    def __init__(self, master_key):
        self.fernet = Fernet(master_key)

    def encrypt_field(self, plaintext):
        """加密单个字段"""
        if plaintext is None:
            return None
        return self.fernet.encrypt(plaintext.encode()).decode()

    def decrypt_field(self, ciphertext):
        """解密单个字段"""
        if ciphertext is None:
            return None
        return self.fernet.decrypt(ciphertext.encode()).decode()

    def encrypt_record(self, record, sensitive_fields):
        """加密记录中的敏感字段"""
        encrypted = record.copy()
        for field in sensitive_fields:
            if field in encrypted:
                encrypted[field] = self.encrypt_field(str(encrypted[field]))
        return encrypted

    def decrypt_record(self, record, sensitive_fields):
        """解密记录中的敏感字段"""
        decrypted = record.copy()
        for field in sensitive_fields:
            if field in decrypted and decrypted[field] is not None:
                decrypted[field] = self.decrypt_field(str(decrypted[field]))
        return decrypted


class DataMasking:
    """数据脱敏工具"""

    @staticmethod
    def mask_phone(phone):
        """手机号脱敏: 138****5678"""
        if not phone or len(phone) < 7:
            return phone
        return phone[:3] + "****" + phone[-4:]

    @staticmethod
    def mask_idcard(idcard):
        """身份证脱敏: 110***********1234"""
        if not idcard or len(idcard) < 6:
            return idcard
        return idcard[:3] + "*" * (len(idcard) - 7) + idcard[-4:]

    @staticmethod
    def mask_email(email):
        """邮箱脱敏: z***@example.com"""
        if not email or "@" not in email:
            return email
        local, domain = email.split("@", 1)
        return local[0] + "***@" + domain

    @staticmethod
    def mask_credit_card(card):
        """信用卡脱敏: **** **** **** 1234"""
        if not card:
            return card
        return "**** **** **** " + card[-4:]

    @staticmethod
    def mask_name(name):
        """姓名脱敏: 张*"""
        if not name:
            return name
        return name[0] + "*" * (len(name) - 1)

    @staticmethod
    def mask_address(address):
        """地址脱敏: 北京市朝阳区***"""
        if not address or len(address) < 6:
            return address
        return address[:6] + "***"

    @staticmethod
    def hash_field(value, salt=None):
        """不可逆哈希脱敏"""
        if salt is None:
            salt = os.urandom(16)
        if isinstance(salt, str):
            salt = salt.encode()
        dk = hashlib.pbkdf2_hmac('sha256', value.encode(), salt, 100000)
        return base64.b64encode(salt + dk).decode()


# 使用示例
if __name__ == "__main__":
    # 加密示例
    key = Fernet.generate_key()
    enc = DatabaseEncryption(key)

    record = {"name": "张三", "phone": "13812345678", "idcard": "110101199001011234"}
    encrypted = enc.encrypt_record(record, ["phone", "idcard"])
    print(f"加密后: {encrypted}")
    decrypted = enc.decrypt_record(encrypted, ["phone", "idcard"])
    print(f"解密后: {decrypted}")

    # 脱敏示例
    print(f"手机号: {DataMasking.mask_phone('13812345678')}")
    print(f"身份证: {DataMasking.mask_idcard('110101199001011234')}")
    print(f"邮箱: {DataMasking.mask_email('zhangsan@example.com')}")
    print(f"信用卡: {DataMasking.mask_credit_card('6222021234567890123')}")
    print(f"姓名: {DataMasking.mask_name('张三丰')}")
    print(f"地址: {DataMasking.mask_address('北京市朝阳区建国路88号')}")
```
