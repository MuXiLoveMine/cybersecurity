# 国密算法实战

### 4.1 国密算法概述

国密算法是中国国家密码管理局发布的一系列密码算法标准。

| 算法 | 类型 | 说明 |
|------|------|------|
| SM1 | 对称加密 | 硬件实现，不公开算法 |
| SM2 | 非对称加密 | 基于椭圆曲线，类似 ECC |
| SM3 | 哈希算法 | 输出 256 位，类似 SHA-256 |
| SM4 | 对称加密 | 分组密码，类似 AES |
| SM9 | 标识密码 | 基于身份的密码体制 |
| ZUC | 序列密码 | 流密码算法 |

#### 4.2.1 SM2 非对称加密

```python
from gmssl import sm2, func
import os

# ========== SM2 密钥生成 ==========

def generate_sm2_keypair() -> tuple[str, str]:
    """
    生成 SM2 密钥对
    返回:
        (私钥 hex, 公钥 hex)
    """
    # 生成 256 位随机私钥
    private_key = func.random_hex(64)  # 64 个 hex 字符 = 256 位 = 32 字节
    
    # 通过私钥计算公钥
    sm2_crypt = sm2.CryptSM2(
        public_key="",
        private_key=private_key
    )
    public_key = sm2_crypt._kg(int(private_key, 16), sm2_crypt.ecc_table['g'])
    
    return private_key, public_key

# ========== SM2 加密解密 ==========

def sm2_encrypt(public_key: str, plaintext: bytes) -> str:
    """
    SM2 加密
    参数:
        public_key: 公钥（hex 字符串）
        plaintext: 明文数据
    返回:
        密文（hex 字符串）
    """
    sm2_crypt = sm2.CryptSM2(
        public_key=public_key,
        private_key=""
    )
    # SM2 加密需要随机数
    k = func.random_hex(64)
    ciphertext = sm2_crypt.encrypt(plaintext, k)
    return ciphertext

def sm2_decrypt(private_key: str, ciphertext: str) -> bytes:
    """
    SM2 解密
    参数:
        private_key: 私钥（hex 字符串）
        ciphertext: 密文（hex 字符串）
    返回:
        明文数据
    """
    sm2_crypt = sm2.CryptSM2(
        public_key="",
        private_key=private_key
    )
    plaintext = sm2_crypt.decrypt(ciphertext)
    return plaintext

# ========== 使用示例 ==========
if __name__ == "__main__":
    print("=== SM2 非对称加密演示 ===")
    
    # 生成密钥对
    private_key, public_key = generate_sm2_keypair()
    print(f"私钥（hex）: {private_key}")
    print(f"公钥（hex）: {public_key}")
    
    # 加密
    plaintext = "这是 SM2 加密的测试数据".encode("utf-8")
    ciphertext = sm2_encrypt(public_key, plaintext)
    print(f"\n明文: {plaintext.decode('utf-8')}")
    print(f"密文长度: {len(ciphertext)//2} 字节")
    print(f"密文（hex）: {ciphertext[:64]}...")
    
    # 解密
    decrypted = sm2_decrypt(private_key, ciphertext)
    print(f"\n解密结果: {decrypted.decode('utf-8')}")
```

#### 4.2.2 SM3 哈希算法

```python
from gmssl import sm3

def sm3_hash(data: bytes) -> str:
    """
    计算 SM3 哈希值
    输出 256 位（64 个 hex 字符）
    """
    # gmssl 的 sm3 模块需要列表形式的输入
    data_list = list(data)
    hash_value = sm3.sm3_hash(data_list)
    return hash_value

# ========== 使用示例 ==========
if __name__ == "__main__":
    print("=== SM3 哈希算法演示 ===")
    
    data = "Hello, 国密 SM3！".encode("utf-8")
    hash_value = sm3_hash(data)
    
    print(f"原始数据: {data.decode('utf-8')}")
    print(f"SM3 哈希值: {hash_value}")
    print(f"哈希长度: {len(hash_value)*4} 位 = {len(hash_value)//2} 字节")
    
    # 与 SHA-256 对比
    import hashlib
    sha256_value = hashlib.sha256(data).hexdigest()
    print(f"\nSHA-256 哈希: {sha256_value}")
    print(f"SM3 和 SHA-256 输出长度相同，都是 256 位")
```

```python
from gmssl.sm4 import CryptSM4, SM4_ENCRYPT, SM4_DECRYPT
import os

def sm4_ecb_encrypt(key: bytes, plaintext: bytes) -> bytes:
    """
    SM4-ECB 模式加密
    """
    sm4 = CryptSM4()
    sm4.set_key(key, SM4_ENCRYPT)
    # 需要填充
    padded = pkcs7_pad(plaintext, 16)
    ciphertext = sm4.crypt_ecb(padded)
    return ciphertext

def sm4_ecb_decrypt(key: bytes, ciphertext: bytes) -> bytes:
    """
    SM4-ECB 模式解密
    """
    sm4 = CryptSM4()
    sm4.set_key(key, SM4_DECRYPT)
    padded_plaintext = sm4.crypt_ecb(ciphertext)
    plaintext = pkcs7_unpad(padded_plaintext, 16)
    return plaintext

def sm4_cbc_encrypt(key: bytes, iv: bytes, plaintext: bytes) -> bytes:
    """
    SM4-CBC 模式加密
    """
    sm4 = CryptSM4()
    sm4.set_key(key, SM4_ENCRYPT)
    padded = pkcs7_pad(plaintext, 16)
    ciphertext = sm4.crypt_cbc(iv, padded)
    return ciphertext

def sm4_cbc_decrypt(key: bytes, iv: bytes, ciphertext: bytes) -> bytes:
    """
    SM4-CBC 模式解密
    """
    sm4 = CryptSM4()
    sm4.set_key(key, SM4_DECRYPT)
    padded_plaintext = sm4.crypt_cbc(iv, ciphertext)
    plaintext = pkcs7_unpad(padded_plaintext, 16)
    return plaintext

# ========== 使用示例 ==========
if __name__ == "__main__":
    print("=== SM4 对称加密演示 ===")
    
    # SM4 密钥 128 位（16 字节）
    key = os.urandom(16)
    print(f"密钥（hex）: {key.hex()}")
    
    plaintext = "这是 SM4 加密的测试数据".encode("utf-8")
    
    # ECB 模式
    print("\n--- ECB 模式 ---")
    ciphertext_ecb = sm4_ecb_encrypt(key, plaintext)
    print(f"密文（hex）: {ciphertext_ecb.hex()}")
    decrypted_ecb = sm4_ecb_decrypt(key, ciphertext_ecb)
    print(f"解密结果: {decrypted_ecb.decode('utf-8')}")
    
    # CBC 模式
    print("\n--- CBC 模式 ---")
    iv = os.urandom(16)
    print(f"IV（hex）: {iv.hex()}")
    ciphertext_cbc = sm4_cbc_encrypt(key, iv, plaintext)
    print(f"密文（hex）: {ciphertext_cbc.hex()}")
    decrypted_cbc = sm4_cbc_decrypt(key, iv, ciphertext_cbc)
    print(f"解密结果: {decrypted_cbc.decode('utf-8')}")
```
