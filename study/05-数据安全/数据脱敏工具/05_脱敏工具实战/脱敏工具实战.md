```sql
ALTER TABLE Customers ALTER COLUMN Name ADD MASKED WITH (FUNCTION = 'partial(2, "***", 1)');

-- 手机号脱敏

-- 身份证号脱敏
ALTER TABLE Customers ALTER COLUMN SSN ADD MASKED WITH (FUNCTION = 'partial(6, "********", 2)');

-- 银行卡号脱敏

-- 数值脱敏（随机范围）
ALTER TABLE Customers ALTER COLUMN Salary ADD MASKED WITH (FUNCTION = 'random(1000, 99999)');

-- 日期脱敏（仅年份）
ALTER TABLE Customers ALTER COLUMN BirthDate ADD MASKED WITH (FUNCTION = 'default()');
-- 注：SQL Server日期类型使用default()掩码，显示0001-01-01

-- 4. 创建低权限用户
CREATE USER MaskedUser WITHOUT LOGIN;
GRANT SELECT ON Customers TO MaskedUser;

-- 5. 以低权限用户查看（看到脱敏数据）
EXECUTE AS USER = 'MaskedUser';
SELECT * FROM Customers;
REVERT;

-- 6. 管理员查看（看到原始数据）
SELECT * FROM Customers;

-- 7. 授予取消掩码权限

-- 8. 撤销取消掩码权限

-- 9. 移除脱敏掩码
```

```sql
-- 需要Oracle 12c+ 企业版

-- 1. 创建测试表
CREATE TABLE customers (

-- 2. 创建DBMS_REDACT策略

-- 完整脱敏（显示固定值）
BEGIN
    DBMS_REDACT.ADD_POLICY(
        object_schema   => 'HR',
        object_name     => 'CUSTOMERS',
        policy_name     => 'SALARY_REDACT',
        column_name     => 'SALARY',
        function_type   => DBMS_REDACT.FULL,
        expression      => 'SYS_CONTEXT(''USERENV'', ''SESSION_USER'') != ''HR'''
    );
END;
/

-- 部分脱敏（字符串）
BEGIN
    DBMS_REDACT.ADD_POLICY(
        object_schema   => 'HR',
        object_name     => 'CUSTOMERS',
        policy_name     => 'PHONE_REDACT',
        column_name     => 'PHONE',
        function_type   => DBMS_REDACT.PARTIAL,
        function_parameters => 'VVVFVVVVVVVV,*,1,3',
        expression      => '1=1'
    );
END;
/

-- 正则表达式脱敏
BEGIN
    DBMS_REDACT.ADD_POLICY(
        object_schema   => 'HR',
        object_name     => 'CUSTOMERS',
        policy_name     => 'EMAIL_REDACT',
        column_name     => 'EMAIL',
        function_type   => DBMS_REDACT.REGEXP,
        function_parameters => '(\w{1})(\w*)@(\w+)', 
        regexp_pattern  => '(\w{1})(\w*)@(\w+)',
        regexp_replace  => '\1****@\3',
        expression      => '1=1'
    );
END;
/

-- 随机脱敏（数值）
BEGIN
    DBMS_REDACT.ADD_POLICY(
        object_schema   => 'HR',
        object_name     => 'CUSTOMERS',
        policy_name     => 'SALARY_RANDOM',
        column_name     => 'SALARY',
        function_type   => DBMS_REDACT.RANDOM_RANGE,
        function_parameters => '1000,99999',
        expression      => '1=1'
    );
END;
/

-- 3. 查看脱敏策略
SELECT * FROM redaction_policies;
SELECT * FROM redaction_columns;

-- 4. 修改脱敏策略
BEGIN
    DBMS_REDACT.ALTER_POLICY(
        object_schema   => 'HR',
        object_name     => 'CUSTOMERS',
        policy_name     => 'SALARY_REDACT',
        action          => DBMS_REDACT.MODIFY_COLUMN,
        column_name     => 'SALARY',
        function_type   => DBMS_REDACT.PARTIAL,
        function_parameters => '9,9,9,9,9,9,9,9,9,9'
    );
END;
/

-- 5. 删除脱敏策略
BEGIN
    DBMS_REDACT.DROP_POLICY(
        object_schema => 'HR',
        object_name   => 'CUSTOMERS',

/
```

#### 3. MySQL 企业版脱敏配置

```sql
-- ========== MySQL 企业版数据脱敏 ==========
-- 注意：MySQL社区版不包含动态脱敏功能

-- 1. 查看脱敏函数
-- MySQL Enterprise Edition 提供以下脱敏函数：
-- mask_inner(str, margin_left, margin_right, mask_char)
-- mask_outer(str, margin_left, margin_right, mask_char)
-- mask_pan(str) - 银行卡号脱敏
-- mask_pan_relaxed(str) - 宽松银行卡脱敏
-- mask_ssn(str) - SSN脱敏

-- 2. 使用示例
SELECT 
    mask_inner('13800138000', 3, 4, '*') AS masked_phone,
    -- 结果: 138****8000
    
    mask_outer('zhangsan@example.com', 1, 10, '*') AS masked_email,
    
    mask_pan('6222021234567890') AS masked_card,
    -- 结果: 622202XXXXXX7890
    
    mask_ssn('110101199001011234') AS masked_ssn
    -- 结果: XXX-XX-1234

FROM dual;

-- 3. 动态脱敏视图（社区版可用的替代方案）
-- 创建视图实现动态脱敏效果

CREATE OR REPLACE VIEW customers_masked AS
SELECT 
    id,
    CONCAT(SUBSTRING(name, 1, 1), '***') AS name,
    CONCAT(SUBSTRING(email, 1, 1), '****@', SUBSTRING_INDEX(email, '@', -1)) AS email,
    CONCAT(SUBSTRING(phone, 1, 3), '****', SUBSTRING(phone, -4)) AS phone,
    CONCAT('************', SUBSTRING(credit_card, -4)) AS credit_card,
    CONCAT(SUBSTRING(ssn, 1, 6), '********', SUBSTRING(ssn, -2)) AS ssn,
    ROUND(salary / 1000) * 1000 AS salary_approx,
    CONCAT(YEAR(birth_date), '-01-01') AS birth_year
FROM customers;

-- 授权用户只能看脱敏视图
GRANT SELECT ON customers_masked TO 'app_user'@'localhost';
REVOKE SELECT ON customers FROM 'app_user'@'localhost';
```

#### 2. CSV 文件脱敏脚本

```python
#!/usr/bin/env python3
"""
CSV 文件数据脱敏工具
支持批量脱敏CSV文件中的敏感字段
"""

import csv
import re
import random
import hashlib
from faker import Faker
from datetime import datetime

fake = Faker('zh_CN')

def mask_name(name):
    """姓名脱敏：保留姓，名用*代替"""
    if not name or len(name) == 0:
        return name
    if len(name) == 1:
        return name
    if len(name) == 2:
        return name[0] + '*'
    return name[0] + '*' * (len(name) - 1)

def mask_phone(phone):
    """手机号脱敏：保留前3后4"""
    if not phone:
        return phone
    phone = str(phone).strip()
    if len(phone) >= 7:
        return phone[:3] + '****' + phone[-4:]
    return phone

def mask_id_card(id_card):
    """身份证号脱敏：保留前6后4"""
    if not id_card:
        return id_card
    id_card = str(id_card).strip()
    if len(id_card) >= 10:
        return id_card[:6] + '********' + id_card[-4:]
    return id_card

def mask_email(email):
    """邮箱脱敏：保留首字符"""
    if not email or '@' not in email:
        return email
    username, domain = email.split('@', 1)
    if len(username) <= 1:
        return username + '****@' + domain
    return username[0] + '****@' + domain

def mask_bank_card(card_num):
    """银行卡号脱敏：只显示后4位"""
    if not card_num:
        return card_num
    card_num = str(card_num).strip()
    digits = re.sub(r'\D', '', card_num)
    if len(digits) > 4:
        return '*' * (len(digits) - 4) + digits[-4:]
    return card_num

def mask_address(address):
    """地址脱敏：保留到区级"""
    if not address:
        return address
    # 简单实现：截取前10个字符+***
    address = str(address).strip()
    if len(address) > 10:
        return address[:10] + '***'
    return address

def blur_date(date_str, input_format='%Y-%m-%d'):
    """日期模糊化：只保留年份"""
    if not date_str:
        return date_str
    try:
        dt = datetime.strptime(str(date_str).strip(), input_format)
        return f"{dt.year}-01-01"
    except:
        return date_str

def blur_number(value, range_percent=10):
    """数值模糊化：随机偏移"""
    if value is None or value == '':
        return value
    try:
        num = float(value)
        offset = num * (range_percent / 100.0)
        return round(random.uniform(num - offset, num + offset), 2)
    except (ValueError, TypeError):
        return value

def hash_mask(data, salt='my_secret_salt'):
    """哈希脱敏：保留可关联性"""
    if not data:
        return data
    return hashlib.sha256((str(data) + salt).encode()).hexdigest()

def generate_fake_data(field_type):
    """生成假数据（替代原数据）"""
    generators = {
        'name': lambda: fake.name(),
        'phone': lambda: fake.phone_number(),
        'email': lambda: fake.email(),
        'address': lambda: fake.address(),
        'company': lambda: fake.company(),
        'id_card': lambda: fake.ssn(),
        'bank_card': lambda: fake.credit_card_number(),
        'ip': lambda: fake.ipv4(),
    }
    generator = generators.get(field_type)
    if generator:
        return generator()
    return fake.word()

def mask_csv(input_file, output_file, field_rules, delimiter=','):
    """
    脱敏CSV文件
    
    field_rules 格式: {
        '姓名': 'name',           # 姓名脱敏
        '手机号': 'phone',        # 手机号脱敏
        '邮箱': 'email',          # 邮箱脱敏
        '工资': ('blur', 10),     # 数值模糊化10%
        '身份证': 'hash',         # 哈希脱敏
        '生日': 'date_blur',      # 日期模糊化
    }
    """
    with open(input_file, 'r', encoding='utf-8') as infile, \
         open(output_file, 'w', encoding='utf-8', newline='') as outfile:
        
        reader = csv.DictReader(infile, delimiter=delimiter)
        writer = csv.DictWriter(outfile, fieldnames=reader.fieldnames, delimiter=delimiter)
        writer.writeheader()
        
        for row in reader:
            for field, rule in field_rules.items():
                if field in row and row[field]:
                    if rule == 'name':
                        row[field] = mask_name(row[field])
                    elif rule == 'phone':
                        row[field] = mask_phone(row[field])
                    elif rule == 'email':
                        row[field] = mask_email(row[field])
                    elif rule == 'id_card':
                        row[field] = mask_id_card(row[field])
                    elif rule == 'bank_card':
                        row[field] = mask_bank_card(row[field])
                    elif rule == 'address':
                        row[field] = mask_address(row[field])
                    elif rule == 'hash':
                        row[field] = hash_mask(row[field])
                    elif rule == 'date_blur':
                        row[field] = blur_date(row[field])
                    elif isinstance(rule, tuple) and rule[0] == 'blur':
                        row[field] = blur_number(row[field], rule[1])
                    elif rule == 'fake_name':
                        row[field] = generate_fake_data('name')
                    elif rule == 'fake_phone':
                        row[field] = generate_fake_data('phone')
            
            writer.writerow(row)
    
    print(f"脱敏完成: {input_file} -> {output_file}")

# 使用示例
if __name__ == '__main__':
    # 定义脱敏规则
    rules = {
        '姓名': 'name',
        '手机号': 'phone',
        '邮箱': 'email',
        '身份证号': 'id_card',
        '银行卡号': 'bank_card',
        '地址': 'address',
        '工资': ('blur', 15),
        '入职日期': 'date_blur',
    }
    
    mask_csv('employees.csv', 'employees_masked.csv', rules)
```

### 四、Presidio PII 识别与脱敏工具

```python
#!/usr/bin/env python3
"""
Microsoft Presidio - PII 识别与脱敏工具
安装: pip install presidio-analyzer presidio-anonymizer
"""

from presidio_analyzer import AnalyzerEngine
from presidio_anonymizer import AnonymizerEngine
from presidio_anonymizer.entities import OperatorConfig

# 初始化分析器
analyzer = AnalyzerEngine()
anonymizer = AnonymizerEngine()

# 1. 基础PII识别
text = """
张三的手机号是13800138000，邮箱是zhangsan@example.com。
他的身份证号是110101199001011234，住在北京市海淀区中关村大街1号。
他的银行卡号是6222021234567890，IP地址是192.168.1.100。
"""

results = analyzer.analyze(text=text, language='zh')

print("=== 识别到的PII实体 ===")
for result in results:
    print(f"  类型: {result.entity_type:20s} "
          f"位置: [{result.start:3d}-{result.end:3d}] "
          f"置信度: {result.score:.2f} "
          f"内容: {text[result.start:result.end]}")

# 2. 脱敏处理 - 替换法
anonymized_result = anonymizer.anonymize(
    text=text,
    analyzer_results=results,

import os

def anonymize_file(input_path, output_path, language='zh'):
    """脱敏单个文件"""
    with open(input_path, 'r', encoding='utf-8') as f:
        text = f.read()
    
    results = analyzer.analyze(text=text, language=language)
    
    anonymized = anonymizer.anonymize(
        text=text,
        analyzer_results=results,
        operators={
            "PERSON": OperatorConfig("replace", {"new_value": "[PERSON]"}),
            "PHONE_NUMBER": OperatorConfig("replace", {"new_value": "[PHONE]"}),
            "EMAIL_ADDRESS": OperatorConfig("replace", {"new_value": "[EMAIL]"}),
            "ID_CARD": OperatorConfig("replace", {"new_value": "[IDCARD]"}),
            "CREDIT_CARD": OperatorConfig("replace", {"new_value": "[CREDITCARD]"}),
            "LOCATION": OperatorConfig("replace", {"new_value": "[LOCATION]"}),
            "IP_ADDRESS": OperatorConfig("replace", {"new_value": "[IP]"}),
        }
    )
    
    with open(output_path, 'w', encoding='utf-8') as f:
        f.write(anonymized.text)
    
    print(f"处理完成: {input_path} -> {output_path}")
    print(f"  识别PII实体数: {len(results)}")

# 批量处理目录
def anonymize_directory(input_dir, output_dir, language='zh'):
    os.makedirs(output_dir, exist_ok=True)
    for filename in os.listdir(input_dir):
        if filename.endswith('.txt') or filename.endswith('.md'):
            input_path = os.path.join(input_dir, filename)
            output_path = os.path.join(output_dir, filename)
            anonymize_file(input_path, output_path, language)
```

### 五、数据脱敏工具选型指南

```
数据脱敏工具选型决策树：

├── 需求：数据库数据脱敏
│   ├── 只使用Oracle / SQL Server → 使用数据库原生动态脱敏
│   ├── 多数据库环境，预算充足 → 商业工具（IBM Optim / Delphix / 安华金和）
│   ├── 多数据库环境，预算有限 → 开源工具+自定义开发
│   └── 需要实时动态脱敏 → 专业动态脱敏网关
│
├── 需求：文件/CSV数据脱敏
│   ├── 格式简单，规则固定 → Python脚本 + pandas
│   ├── 需要PII自动识别 → Presidio + 自定义规则
│   ├── 需要生成假数据 → Faker库
│   └── 大量文件批量处理 → Apache NiFi / 自定义ETL
│
├── 需求：开发/测试环境数据
│   ├── 数据量小 → Faker生成合成数据
│   ├── 需要保留数据关系 → 静态脱敏工具
│   └── 需要数据子集 → Oracle SDM / IBM Optim
│
└── 需求：日志数据脱敏
    ├── 结构化日志 → 正则替换 + 脚本处理
    ├── 非结构化日志 → Presidio NLP识别
    └── 实时日志 → 日志采集管道中处理（Logstash filter）
```
