# 结语

移动应用渗透测试是一项系统性工程，涵盖静态分析、动态分析、网络流量分析、组件安全等多个维度。核心要点：

1. **以 OWASP Mobile Top 10 为框架**，确保覆盖全面的安全风险面
2. **静态与动态结合**，静态分析发现问题线索，动态分析验证利用
3. **重视数据流追踪**，从输入到存储到传输，全程跟踪敏感数据
4. **自动化工具提效**，MobSF/Drozer/QARK 负责批量扫描，人工负责深度挖掘
5. **服务端安全不可忽视**，移动端只是入口，API 安全同样重要

> **安全声明**：本文档仅供安全研究与教学使用。渗透测试必须在获得明确授权的前提下进行，切勿对未授权目标进行测试。

### 二、完整渗透测试工作流命令序列

以下是一次标准 Android 应用渗透测试的完整命令流程：

```bash
# ============================================================
# Step 1: 信息收集与静态分析
# ============================================================

# 1.1 APK 基本信息
mkdir -p pentest_report/{info,static,dynamic,network}
aapt dump badging target.apk > pentest_report/info/badging.txt
aapt dump permissions target.apk > pentest_report/info/permissions.txt
aapt dump xmltree target.apk AndroidManifest.xml > pentest_report/info/manifest_xml.txt

# 1.2 加固识别
apkid target.apk > pentest_report/info/apkid_result.txt

# 1.3 反编译
apktool d target.apk -o pentest_report/static/apktool_out
jadx -d pentest_report/static/jadx_out target.apk

# 1.4 敏感信息扫描
apkleaks -f target.apk -o pentest_report/static/leaks.json
grep -rniE "(password|secret|api[_-]?key|token)" \
    pentest_report/static/jadx_out/sources/ > pentest_report/static/sensitive_strings.txt
grep -rniE "https?://[^\"']+" \
    pentest_report/static/apktool_out/ > pentest_report/static/urls.txt

# 1.5 组件安全静态审计
# 使用前面章节的 component_audit.py
python component_audit.py pentest_report/static/apktool_out/AndroidManifest.xml \
    > pentest_report/static/component_audit.txt

# 1.6 MobSF 自动化扫描
# （假设 MobSF 运行在 localhost:8000）
curl -F "file=@target.apk" http://localhost:8000/api/v1/upload \
    -H "Authorization: myapikey123" > pentest_report/static/mobsf_upload.json
FILE_HASH=$(cat pentest_report/static/mobsf_upload.json | jq -r '.hash')
curl -X POST http://localhost:8000/api/v1/scan \
    -H "Authorization: myapikey123" \
    -d "hash=$FILE_HASH&scan_type=apk"

# ============================================================
# Step 2: 安装与运行环境准备
# ============================================================

# 2.1 安装测试应用
adb install -r -t target.apk

# 2.2 启动 Burp Suite 代理（需手动启动）
# 2.3 配置手机代理指向电脑 IP:8080
# 2.4 安装 Burp 证书到系统证书目录（Android 7+ 需 Root）
adb push burp_cacert.der /sdcard/
adb shell su -c "cp /sdcard/burp_cacert.der /system/etc/security/cacerts/9a5ba575.0"
adb shell su -c "chmod 644 /system/etc/security/cacerts/9a5ba575.0"

# ============================================================
# Step 3: 动态分析 - 组件安全测试
# ============================================================

# 3.1 Drozer 组件测试
adb forward tcp:31415 tcp:31415
# drozer console connect 后执行：
#   run app.package.attacksurface com.example.app
#   run app.activity.info -a com.example.app
#   run app.service.info -a com.example.app
#   run app.broadcast.info -a com.example.app
#   run app.provider.info -a com.example.app
#   run scanner.provider.injection -a com.example.app
#   run scanner.provider.traversal -a com.example.app

# 3.2 导出 Activity 越权测试（逐个测试）
adb shell am start -n com.example.app/.PayActivity
adb shell am start -n com.example.app/.AdminActivity --es user_id "1"
adb shell dumpsys window | grep mCurrentFocus  # 验证是否成功启动

# 3.3 ContentProvider SQL 注入测试
adb shell content query --uri content://com.example.provider/users --where "1=1--"
adb shell content query --uri content://com.example.provider/users \
    --projection "name,password FROM users--"

# ============================================================
# Step 4: 动态分析 - 数据存储测试
# ============================================================

# 4.1 SharedPreferences 检查
adb shell run-as com.example.app ls -la shared_prefs/
adb shell run-as com.example.app cat shared_prefs/settings.xml

# 4.2 数据库文件检查
adb shell run-as com.example.app ls -la databases/
adb shell run-as com.example.app "sqlite3 databases/app.db '.tables'"
adb shell run-as com.example.app "sqlite3 databases/app.db 'SELECT * FROM users LIMIT 5;'"

# 4.3 内部存储检查
adb shell run-as com.example.app ls -la files/
adb shell run-as com.example.app ls -la cache/

# 4.4 外部存储检查
adb shell ls -la /sdcard/Android/data/com.example.app/
adb shell ls -la /sdcard/ | grep -i example

# ============================================================
# Step 5: 动态分析 - Frida Hook
# ============================================================

# 5.1 SSL Pinning 绕过
frida -U -f com.example.app -l ssl_pinning_bypass.js --no-pause

# 5.2 Root 检测绕过
frida -U -f com.example.app -l root_bypass.js --no-pause

# 5.3 加密函数监控
frida -U -f com.example.app -l crypto_hook.js --no-pause \
    > pentest_report/dynamic/crypto_log.txt

# 5.4 组件调用监控
frida -U -f com.example.app -l component_monitor.js --no-pause \
    > pentest_report/dynamic/component_log.txt

# 5.5 Objection 交互式测试
objection -g com.example.app explore
# objection 命令：
#   android hooking search classes Crypto
#   android hooking watch class_method javax.crypto.Cipher.doFinal --dump-args --dump-return
#   android keystore list
#   android clipboard monitor

# ============================================================
# Step 6: 网络安全测试
# ============================================================

# 6.1 抓包（Burp Suite 已配置）
# 操作 App 各功能，记录请求与响应

# 6.2 明文通信检测
# 检查是否有 HTTP 请求（非 HTTPS）
# 在 Burp 的 HTTP history 中筛选 http://

# 6.3 SSL Pinning 验证
# 配置 Burp 代理后，观察 App 是否能正常通信
# 如不能正常通信，说明启用了 SSL Pinning，需用 Frida 绕过

# 6.4 抓包文件分析（tcpdump）
adb shell tcpdump -i wlan0 -w /sdcard/network.pcap
# 操作 App 后停止抓包
adb pull /sdcard/network.pcap pentest_report/network/
# 用 Wireshark 分析 pcap 文件

# ============================================================
# Step 7: 本地验证与漏洞利用
# ============================================================

# 7.1 备份漏洞测试
adb backup -f pentest_report/dynamic/backup.ab -apk com.example.app
# 解包验证：
dd if=pentest_report/dynamic/backup.ab bs=1 skip=24 | python -c "import zlib,sys;sys.stdout.write(zlib.decompress(sys.stdin.read()))" > backup.tar
tar -xvf backup.tar -C pentest_report/dynamic/backup_extracted/

# 7.2 WebView 漏洞测试
# 通过 adb 启动 WebView Activity 并加载恶意 URL
adb shell am start -n com.example.app/.WebActivity --es url "javascript:alert(1)"

# 7.3 Intent Scheme 攻击测试
adb shell am start -a android.intent.action.VIEW \
    -d "app://web?url=http://evil.com"

# ============================================================
# Step 8: 结果整理与报告生成
# ============================================================

# 整理所有输出文件
# 按 OWASP Mobile Top 10 分类
# 生成最终报告（可使用模板）
```
