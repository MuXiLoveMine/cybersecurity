**环境搭建步骤**：

```bash
# 1. Android 模拟器配置
# 推荐使用 Genymotion 或 Android Studio AVD
# 选择 Android 7.0+ 版本（支持系统级证书安装）

# 2. Root 环境配置
# Genymotion: 默认已 Root
# AVD: 使用 rootAVD 脚本
./rootAVD.sh <AVD_Name> system-images

# 3. 安装 Magisk（获取完整 Root 权限 + 模块支持）
# 下载 Magisk APK 并安装
adb install magisk.apk
# 在 Magisk Manager 中完成安装

# 4. 安装系统级 CA 证书
# 使用 Magisk 模块 MagiskTrustUserCerts
# 或手动安装：
adb push burp_cert.pem /sdcard/
adb shell "su -c 'cp /sdcard/burp_cert.pem /system/etc/security/cacerts/<hash>.0'"
adb shell "su -c 'chmod 644 /system/etc/security/cacerts/<hash>.0'"
adb shell "su -c 'reboot'"

# 5. Frida 环境配置
pip install frida-tools
# 下载对应架构的 frida-server
adb push frida-server /data/local/tmp/
adb shell "su -c 'chmod +x /data/local/tmp/frida-server'"
adb shell "su -c '/data/local/tmp/frida-server &'"

# 6. 安装目标 App
adb install wechat.apk
adb install alipay.apk

# 7. 安装反编译工具
npm install -g unveilr
pip install mitmproxy
```

### 8.4 自动化测试脚本

```python
#!/usr/bin/env python3
"""
小程序渗透测试自动化工具
集成接口发现、参数篡改、越权检测功能
"""

import requests
import json
import re
import os
from urllib.parse import urlparse, parse_qs
from concurrent.futures import ThreadPoolExecutor

class MiniAppPentestToolkit:
    def __init__(self, base_url, token=None):
        self.base_url = base_url.rstrip('/')
        self.session = requests.Session()
        self.session.headers.update({
            'Content-Type': 'application/json',
            'User-Agent': 'Mozilla/5.0 (Linux; Android 12) MicroMessenger/8.0'
        })
        if token:
            self.session.headers['Authorization'] = f'Bearer {token}'
        self.findings = []

    def extract_api_endpoints(self, source_dir):
        """从反编译源码中提取 API 端点"""
        endpoints = set()
        for root, dirs, files in os.walk(source_dir):
            for file in files:
                if file.endswith('.js'):
                    filepath = os.path.join(root, file)
                    with open(filepath, 'r', encoding='utf-8', errors='ignore') as f:
                        content = f.read()

                    # 提取 URL 模式
                    urls = re.findall(
                        r'["\'](https?://[^"\'\s]+|/api/[^"\'\s]+)["\']',
                        content
                    )
                    endpoints.update(urls)

                    # 提取 wx.request 中的 url
                    wx_urls = re.findall(
                        r'url:\s*["\']([^"\']+)["\']',
                        content
                    )
                    endpoints.update(wx_urls)

        return sorted(endpoints)

    def test_unauthorized_access(self, endpoints):
        """测试未授权访问"""
        results = []
        for endpoint in endpoints:
            url = self._normalize_url(endpoint)
            # 不带 Token 请求
            no_auth = requests.get(url, timeout=5)
            # 带 Token 请求
            with_auth = requests.get(url, headers=self.session.headers, timeout=5)

            # 比较响应
            if no_auth.status_code == 200 and no_auth.text == with_auth.text:
                results.append({
                    'url': url,
                    'issue': '未授权访问',
                    'severity': 'HIGH',
                    'detail': '去除鉴权信息后仍返回相同数据'
                })

        return results

    def test_idor(self, endpoint, param_name, original_value, test_values):
        """测试越权访问"""
        results = []
        for test_val in test_values:
            params = {param_name: test_val}
            resp = self.session.get(
                self._normalize_url(endpoint),
                params=params,
                timeout=5
            )
            if resp.status_code == 200 and test_val != original_value:
                data = resp.json()
                if data.get('code') == 0 or data.get('success'):
                    results.append({
                        'url': endpoint,
                        'param': param_name,
                        'test_value': test_val,
                        'issue': '水平越权',
                        'severity': 'HIGH'
                    })

        return results

    def test_parameter_tampering(self, endpoint, params_to_test):
        """测试参数篡改"""
        results = []
        for param, original, tampered in params_to_test:
            # 发送篡改后的值
            data = {param: tampered}
            resp = self.session.post(
                self._normalize_url(endpoint),
                json=data,
                timeout=5
            )
            if resp.status_code == 200:
                result = resp.json()
                if result.get('code') == 0 or result.get('success'):
                    results.append({
                        'url': endpoint,
                        'param': param,
                        'original': original,
                        'tampered': tampered,
                        'issue': '参数篡改成功',
                        'severity': 'CRITICAL'
                    })

        return results

    def _normalize_url(self, url):
        if url.startswith('http'):
            return url
        return f'{self.base_url}{url}'
```
