### 一、Frida 常用 API 速查表

#### 1. Java 层核心 API

| API 分类 | 方法签名 | 用途说明 | 典型使用场景 |
|---------|---------|---------|-------------|
| **Java 运行时** | `Java.perform(function)` | 进入 Java 操作上下文 | 所有 Java 层 Hook 的入口 |
| **类加载** | `Java.use("com.xxx.ClassName")` | 获取类的 JavaScript 包装 | Hook 类方法前必须先 use |
| **类枚举** | `Java.enumerateLoadedClasses()` | 枚举已加载的所有类 | 查找目标类、搜索混淆类 |
| **类枚举** | `Java.enumerateClassLoaders()` | 枚举所有 ClassLoader | 多 dex / 热修复场景 |
| **对象操作** | `Java.cast(obj, cls)` | 强制类型转换 | 将 Object 转为具体子类 |
| **对象操作** | `Java.choose("com.xxx.Cls", {onMatch, onComplete})` | 搜索堆中已存在的对象实例 | 无需构造直接调用实例方法 |
| **注册类** | `Java.registerClass(spec)` | 动态注册一个 Java 类 | 实现接口、创建自定义回调 |
| **数组创建** | `Java.array("I", [1,2,3])` | 创建 Java 原生数组 | 调用需要数组参数的方法 |
| **字节数组** | `Java.array("byte", [0x41,0x42])` | 创建 byte[] 数组 | 加解密函数参数构造 |

#### 2. Native 层核心 API

| API 分类 | 方法签名 | 用途说明 | 典型使用场景 |
|---------|---------|---------|-------------|
| **模块操作** | `Process.enumerateModules()` | 枚举所有已加载模块 | 查找 SO 库 |

| **模块操作** | `Module.enumerateExports("libxxx.so")` | 枚举模块所有导出符号 | 分析 SO 导出表 |
| **Hook** | `Interceptor.attach(addr, {onEnter, onLeave})` | 在函数出入口插入回调 | 最常用的 Hook 方式 |
| **Hook** | `Interceptor.replace(addr, callback)` | 完全替换函数实现 | 伪造返回值、绕过检测 |
| **内存读** | `Memory.readByteArray(addr, length)` | 读取内存字节数组 | Dump 内存数据 |
| **内存读** | `Memory.readUtf8String(addr)` | 读取 C 字符串 | 读取 native 字符串 |
| **内存读** | `Memory.readS32(addr)` | 读取 32 位有符号整数 | 读取数值型变量 |
| **内存写** | `Memory.writeByteArray(addr, bytes)` | 写入字节数组 | Patch 内存指令 |
| **内存搜索** | `Memory.scan(base, size, pattern, callbacks)` | 按字节模式搜索内存 | 查找特征码、定位函数 |
| **内存分配** | `Memory.alloc(size)` | 分配可执行内存 | 构造 shellcode / stub |
| **函数调用** | `new NativeFunction(addr, retType, argTypes)` | 将地址转为可调用函数 | 主动调用 native 函数 |
| **回调构造** | `new NativeCallback(func, retType, argTypes)` | 创建 native 可调用的回调 | 替换函数指针 |

### 二、Android Java 层 Hook 实战脚本

#### 脚本 1：通用加解密函数监控（AES / RSA / DES）

```javascript
// crypto_hook.js - 监控 Android 常用加密 API
// 使用方式: frida -U -f com.example.app -l crypto_hook.js --no-pause

Java.perform(function() {
    console.log("\n[*] === 加密函数监控启动 ===\n");

    // ====== 1. 监控 Cipher 加密解密 ======
    try {
        var Cipher = Java.use("javax.crypto.Cipher");
        Cipher.doFinal.overload('[B').implementation = function(input) {
            var opmode = this.getOpMode();
            var algo = this.getAlgorithm();
            var opName = opmode == 1 ? "ENCRYPT" : (opmode == 2 ? "DECRYPT" : "UNKNOWN");
            
            var result = this.doFinal.overload('[B').call(this, input);
            
            console.log("[Cipher][" + opName + "] " + algo);
            console.log("  输入: " + bytesToHex(input));
            console.log("  输出: " + bytesToHex(result));
            // 打印调用栈
            console.log("  调用栈:\n" + getJavaStack());
            
            return result;
        };
        console.log("[+] Cipher.doFinal Hook 成功");
    } catch(e) {
        console.log("[-] Cipher Hook 失败: " + e);
    }

    // ====== 2. 监控 MessageDigest (MD5/SHA) ======
    try {
        var MessageDigest = Java.use("java.security.MessageDigest");
        MessageDigest.digest.overload('[B').implementation = function(input) {
            var algo = this.getAlgorithm();
            var result = this.digest.overload('[B').call(this, input);
            console.log("[MessageDigest] " + algo);
            console.log("  输入: " + (input ? bytesToHex(input) : "null"));
            console.log("  输出: " + bytesToHex(result));
            return result;
        };
        console.log("[+] MessageDigest.digest Hook 成功");
    } catch(e) {
        console.log("[-] MessageDigest Hook 失败: " + e);
    }

    // ====== 3. 监控 SharedPreferences 读写 ======
    try {
        var SharedPreferencesImpl = Java.use("android.app.SharedPreferencesImpl");
        SharedPreferencesImpl.getString.implementation = function(key, defValue) {
            var result = this.getString(key, defValue);
            console.log("[SharedPrefs][GET] " + key + " = " + result);
            return result;
        };
        
        var EditorImpl = Java.use("android.app.SharedPreferencesImpl$EditorImpl");
        EditorImpl.putString.implementation = function(key, value) {
            console.log("[SharedPrefs][PUT] " + key + " = " + value);
            return this.putString(key, value);
        };
        console.log("[+] SharedPreferences Hook 成功");
    } catch(e) {
        console.log("[-] SharedPreferences Hook 失败: " + e);
    }

    // ====== 4. 监控 Log 输出（捕获敏感日志）======
    try {
        var Log = Java.use("android.util.Log");
        var logLevels = ["d", "e", "i", "v", "w"];
        logLevels.forEach(function(level) {
            Log[level].overload('java.lang.String', 'java.lang.String').implementation = function(tag, msg) {
                // 过滤敏感关键词
                var sensitive = ["password", "token", "secret", "key", "encrypt", "sign"];
                var lowerMsg = msg ? msg.toLowerCase() : "";
                for (var i = 0; i < sensitive.length; i++) {
                    if (lowerMsg.indexOf(sensitive[i]) !== -1) {
                        console.log("[Log-SENSITIVE][" + level.toUpperCase() + "] " + tag + ": " + msg);
                        break;
                    }
                }
                return this[level](tag, msg);
            };
        });
        console.log("[+] Log 敏感信息监控 Hook 成功");
    } catch(e) {
        console.log("[-] Log Hook 失败: " + e);
    }

    // ====== 辅助函数：字节转十六进制 ======
    function bytesToHex(bytes) {
        if (!bytes) return "null";
        var hex = "";
        for (var i = 0; i < bytes.length && i < 64; i++) {  // 最多显示64字节
            var b = bytes[i] & 0xFF;
            hex += (b < 16 ? "0" : "") + b.toString(16);
        }
        if (bytes.length > 64) hex += "...(" + bytes.length + "字节)";
        return hex;
    }

    // ====== 辅助函数：获取 Java 调用栈 ======
    function getJavaStack() {
        var Exception = Java.use("java.lang.Exception");
        var stack = Exception.$new().getStackTrace();
        var result = "";
        for (var i = 2; i < stack.length && i < 10; i++) {  // 跳过前2层（本函数）
            result += "    at " + stack[i].toString() + "\n";
        }
        return result;
    }
});
```

**实战场景**：
- 分析 App 的加密算法实现，定位加密密钥
- 抓取 App 本地存储的敏感数据（Token、密码等）
- 捕获开发人员遗留的调试日志中的敏感信息

**使用技巧**：
- 对大型 App 建议先缩小范围，使用 `Java.use` 前先用 `Java.enumerateLoadedClasses` 确认类名
- 输出过多时可将结果写入文件：`frida -U -f com.xxx -l script.js > output.txt`

### 三、Android Native 层 Hook 实战脚本

#### 脚本 3：Native 函数 Hook 与参数解析（以 libc.so 为例）

```javascript
// native_hook.js - Native 层 Hook 实战
// 使用方式: frida -U -f com.example.app -l native_hook.js --no-pause

// ====== 1. Hook libc 的 open 函数 ======
function hookOpen() {
    var openAddr = Module.findExportByName("libc.so", "open");
    if (!openAddr) {
        // Android 高版本可能使用 openat
        openAddr = Module.findExportByName("libc.so", "openat");
    }
    
    if (!openAddr) {
        console.log("[-] 未找到 open/openat 函数");
        return;
    }
    
    Interceptor.attach(openAddr, {
        onEnter: function(args) {
            // openat 有4个参数: fd, pathname, flags, mode
            // open 有2-3个参数: pathname, flags, mode
            var pathPtr = openAddr == Module.findExportByName("libc.so", "openat") 
                ? args[1] : args[0];
            
            try {
                var path = pathPtr.readUtf8String();
                // 过滤感兴趣的文件路径
                var keywords = ["password", "token", "secret", "key", ".db", ".xml", "shared_prefs"];
                for (var i = 0; i < keywords.length; i++) {
                    if (path.indexOf(keywords[i]) !== -1) {
                        this.isTarget = true;
                        this.path = path;
                        console.log("[open] " + path);
                        console.log("  调用栈:\n" + getNativeStack());
                        break;
                    }
                }
            } catch(e) {}
        },
        onLeave: function(retval) {
            if (this.isTarget) {
                console.log("  返回 fd: " + retval.toInt32());
            }
        }
    });
    console.log("[+] open/openat Hook 成功");
}

// ====== 2. Hook 自定义 SO 的导出函数 ======
function hookNativeLibrary() {
    var libName = "libnative-lib.so";  // 替换为目标 SO 名
    var lib = Process.findModuleByName(libName);
    
    if (!lib) {
        console.log("[-] 未找到 " + libName + "，等待加载...");
        // 监控模块加载
        var dlopen = Module.findExportByName(null, "dlopen");
        var dlopenInterceptor = Interceptor.attach(dlopen, {
            onEnter: function(args) {
                this.libPath = args[0].readUtf8String();
            },
            onLeave: function(retval) {
                if (this.libPath && this.libPath.indexOf("libnative-lib") !== -1) {
                    console.log("[+] 检测到 " + this.libPath + " 加载完成");
                    dlopenInterceptor.detach();  // 只 Hook 一次
                    setTimeout(doHookNative, 500);  // 延迟确保初始化完成
                }
            }
        });
        return;
    }
    
    doHookNative();
    
    function doHookNative() {
        var lib = Process.findModuleByName(libName);
        if (!lib) return;
        
        console.log("[*] 开始 Hook " + libName + "，基址: " + lib.base);
        
        // 枚举所有导出函数
        var exports = Module.enumerateExports(libName);
        console.log("[*] 导出函数数量: " + exports.length);
        
        // 按名称模式 Hook 加密相关函数
        var cryptoFuncs = exports.filter(function(e) {
            return e.name.match(/encrypt|decrypt|aes|rsa|sign|hash|md5|sha/i);
        });
        
        console.log("[*] 找到 " + cryptoFuncs.length + " 个加密相关函数");
        
        cryptoFuncs.forEach(function(func) {
            console.log("  [-] Hook: " + func.name + " @ " + func.address);
            Interceptor.attach(func.address, {
                onEnter: function(args) {
                    console.log("[Native][" + func.name + "] 被调用");
                    // 尝试读取第一个参数（假设是输入缓冲区）
                    try {
                        var inputLen = 64;
                        var data = Memory.readByteArray(args[0], inputLen);
                        console.log("  参数0(前64字节):\n" + hexdump(data, { length: inputLen }));
                    } catch(e) {}
                    
                    // 保存参数供 onLeave 使用
                    this.arg0 = args[0];
                    this.arg1 = args[1];
                    this.arg2 = args[2];
                    this.arg3 = args[3];
                },
                onLeave: function(retval) {
                    console.log("  返回值: " + retval);
                    // 如果返回值是指针，尝试读取内容
                    try {
                        if (!retval.isNull()) {
                            var result = Memory.readByteArray(retval, 64);
                            console.log("  返回值指向的内存(前64字节):\n" + hexdump(result, { length: 64 }));
                        }
                    } catch(e) {}
                }
            });
        });
    }
}

// ====== 3. 内存搜索（查找字符串 / 特征码）======
function searchMemory(libName, pattern) {
    var lib = Process.findModuleByName(libName);
    if (!lib) {
        console.log("[-] 未找到模块: " + libName);
        return;
    }
    
    console.log("[*] 在 " + libName + " 中搜索: " + pattern);
    var results = [];
    
    Memory.scan(lib.base, lib.size, pattern, {
        onMatch: function(address, size) {
            results.push(address);
            console.log("  [+] 找到匹配: " + address);
            // 读取周围内容
            try {
                var context = Memory.readByteArray(address.sub(16), 64);
                console.log(hexdump(context, { offset: -16, length: 64 }));
            } catch(e) {}
        },
        onComplete: function() {
            console.log("[*] 搜索完成，共找到 " + results.length + " 处匹配");
        }
    });
}

// ====== 辅助函数：获取 Native 调用栈 ======
function getNativeStack() {
    var threads = Process.enumerateThreads();
    var tid = Process.getCurrentThreadId();
    var thread = threads.find(function(t) { return t.id == tid; });
    
    if (!thread) return "    (无法获取线程)";
    
    var backtrace = Thread.backtrace(thread.context, Backtracer.ACCURATE);
    var result = "";
    for (var i = 0; i < backtrace.length && i < 10; i++) {
        var addr = backtrace[i];
        var module = "?";
        var offset = "";
        try {
            var m = Process.findModuleByAddress(addr);
            if (m) {
                module = m.name;
                offset = "+0x" + addr.sub(m.base).toString(16);
            }
        } catch(e) {}
        result += "    " + addr + " (" + module + offset + ")\n";
    }
    return result;
}

// ====== 主入口 ======
setImmediate(function() {
    console.log("\n[*] === Native Hook 脚本启动 ===\n");
    
    // 延迟执行，确保 SO 已加载
    setTimeout(function() {
        hookOpen();
        hookNativeLibrary();
        
        // 示例：在 libc.so 中搜索字符串 "password"
        // searchMemory("libc.so", "70 61 73 73 77 6f 72 64");  // "password" 的十六进制
    }, 1000);
});
```

**实战场景**：
- 分析 Native 层的加密算法，定位密钥和输入输出数据
- 追踪敏感文件的读写操作
- 逆向分析第三方 SO 库的功能逻辑
- 绕过 Native 层的签名校验、Root 检测等

**使用技巧**：
- 对于延迟加载的 SO，使用 `dlopen` Hook 来监控加载时机
- `Memory.scan` 的 pattern 支持 `??` 通配符，如 `"48 8b ?? ?? 00 00"`
- 使用 `hexdump()` 格式化输出内存数据，便于分析
- Native 调用栈获取可能不准确，建议结合 Java 调用栈交叉验证

| `frida-trace -U -f <包名> -i "函数模式"` | 自动追踪匹配的函数 | `-i "Java_*"` 追踪 JNI 函数 |
| `frida-trace -U -m "-[* *]"` | 追踪所有 ObjC 方法 (iOS) | `-m "*[ViewController *]"` 过滤类 |
| `frida-discover -U -f <包名>` | 发现应用中的函数和模块 | |
| `frida-kill -U <进程名>` | 结束目标进程 | |
| `frida-ls-devices` | 列出所有可用设备 | |

**常用参数说明**：
- `-U` / `--usb`：连接 USB 设备
- `-R` / `--remote`：连接远程 Frida Server
- `-f` / `--file`：Spawn 模式（冷启动）
- `-l` / `--load`：加载脚本文件
- `--no-pause`：启动后不暂停（直接继续执行）
- `-o` / `--output`：输出到文件
- `--runtime=v8` / `--runtime=qjs`：指定 JS 运行时
