#### 1.1 RBAC（基于角色的访问控制）Python实现

```python
"""
RBAC (Role-Based Access Control) 完整实现
支持：用户-角色-权限 三层模型、角色继承、职责分离检测
"""

from typing import Dict, List, Set, Optional
from dataclasses import dataclass, field


@dataclass
class Permission:
    """权限定义"""
    name: str
    resource: str  # 资源
    action: str    # 操作（read/write/delete/execute）
    description: str = ""
    
    def __hash__(self):
        return hash(f"{self.resource}:{self.action}")
    
    def __eq__(self, other):
        return self.resource == other.resource and self.action == other.action


@dataclass
class Role:
    """角色定义"""
    name: str
    permissions: Set[Permission] = field(default_factory=set)
    parent_roles: Set[str] = field(default_factory=set)  # 继承的角色
    description: str = ""
    
    def __hash__(self):
        return hash(self.name)
    
    def __eq__(self, other):
        return self.name == other.name


@dataclass
class User:
    """用户定义"""
    username: str
    roles: Set[str] = field(default_factory=set)  # 直接分配的角色
    attributes: Dict = field(default_factory=dict)


class RBACSystem:
    """RBAC 权限控制系统"""
    
    def __init__(self):
        self.users: Dict[str, User] = {}
        self.roles: Dict[str, Role] = {}
        self.permissions: Dict[str, Permission] = {}
        # 职责分离规则：(角色A, 角色B) 表示不能同时拥有
        self.separation_of_duty_rules: List[tuple] = []
    
    # ========== 权限管理 ==========
    
    def add_permission(self, name: str, resource: str, action: str, description: str = "") -> Permission:
        """添加权限"""
        perm = Permission(name=name, resource=resource, action=action, description=description)
        self.permissions[name] = perm
        return perm
    
    def remove_permission(self, name: str):
        """删除权限"""
        if name in self.permissions:
            del self.permissions[name]
    
    # ========== 角色管理 ==========
    
    def create_role(self, name: str, description: str = "") -> Role:
        """创建角色"""
        role = Role(name=name, description=description)
        self.roles[name] = role
        return role
    
    def assign_permission_to_role(self, role_name: str, permission_name: str):
        """给角色分配权限"""
        if role_name not in self.roles:
            raise ValueError(f"角色不存在: {role_name}")
        if permission_name not in self.permissions:
            raise ValueError(f"权限不存在: {permission_name}")
        self.roles[role_name].permissions.add(self.permissions[permission_name])
    
    def remove_permission_from_role(self, role_name: str, permission_name: str):
        """移除角色权限"""
        if role_name in self.roles and permission_name in self.permissions:
            self.roles[role_name].permissions.discard(self.permissions[permission_name])
    
    def set_role_inheritance(self, role_name: str, parent_role_names: List[str]):
        """设置角色继承关系"""
        if role_name not in self.roles:
            raise ValueError(f"角色不存在: {role_name}")
        for parent in parent_role_names:
            if parent not in self.roles:
                raise ValueError(f"父角色不存在: {parent}")
        self.roles[role_name].parent_roles = set(parent_role_names)
    
    def get_all_role_permissions(self, role_name: str, visited: Set[str] = None) -> Set[Permission]:
        """获取角色的所有权限（含继承）"""
        if visited is None:
            visited = set()
        
        if role_name in visited or role_name not in self.roles:
            return set()
        
        visited.add(role_name)
        role = self.roles[role_name]
        perms = set(role.permissions)
        
        # 递归获取父角色权限
        for parent_role in role.parent_roles:
            perms.update(self.get_all_role_permissions(parent_role, visited))
        
        return perms
    
    # ========== 用户管理 ==========
    
    def add_user(self, username: str, attributes: Dict = None) -> User:
        """添加用户"""
        user = User(username=username, attributes=attributes or {})
        self.users[username] = user
        return user
    
    def assign_role_to_user(self, username: str, role_name: str) -> bool:
        """给用户分配角色（检查职责分离）"""
        if username not in self.users:
            raise ValueError(f"用户不存在: {username}")
        if role_name not in self.roles:
            raise ValueError(f"角色不存在: {role_name}")
        
        # 检查职责分离
        user_roles = self.users[username].roles.copy()
        user_roles.add(role_name)
        if not self._check_separation_of_duty(user_roles):
            raise ValueError(f"违反职责分离原则: 不能同时分配 {role_name}")
        
        self.users[username].roles.add(role_name)
        return True
    
    def remove_role_from_user(self, username: str, role_name: str):
        """移除用户角色"""
        if username in self.users:
            self.users[username].roles.discard(role_name)
    
    def get_user_roles(self, username: str) -> Set[str]:
        """获取用户所有角色（含继承展开的角色名）"""
        if username not in self.users:
            return set()
        
        all_roles = set()
        visited = set()
        
        def expand(role_name: str):
            if role_name in visited or role_name not in self.roles:
                return
            visited.add(role_name)
            all_roles.add(role_name)
            for parent in self.roles[role_name].parent_roles:
                expand(parent)
        
        for role in self.users[username].roles:
            expand(role)
        
        return all_roles
    
    # ========== 权限检查 ==========
    
    def has_permission(self, username: str, resource: str, action: str) -> bool:
        """检查用户是否有权限访问资源"""
        if username not in self.users:
            return False
        
        user_roles = self.get_user_roles(username)
        
        for role_name in user_roles:
            perms = self.get_all_role_permissions(role_name)
            for perm in perms:
                if perm.resource == resource and perm.action == action:
                    return True
        
        return False
    
    def get_user_permissions(self, username: str) -> Set[Permission]:
        """获取用户所有权限"""
        all_perms = set()
        user_roles = self.get_user_roles(username)
        
        for role_name in user_roles:
            all_perms.update(self.get_all_role_permissions(role_name))
        
        return all_perms
    
    # ========== 职责分离 ==========
    
    def add_separation_of_duty_rule(self, role_a: str, role_b: str):
        """添加职责分离规则（两个角色不能同时拥有）"""
        rule = tuple(sorted([role_a, role_b]))
        if rule not in self.separation_of_duty_rules:
            self.separation_of_duty_rules.append(rule)
    
    def _check_separation_of_duty(self, roles: Set[str]) -> bool:
        """检查是否违反职责分离"""
        for role_a, role_b in self.separation_of_duty_rules:
            if role_a in roles and role_b in roles:
                return False
        return True
    
    def check_user_sod_compliance(self, username: str) -> List[tuple]:
        """检查用户是否符合职责分离要求"""
        violations = []
        user_roles = self.get_user_roles(username)
        
        for role_a, role_b in self.separation_of_duty_rules:
            if role_a in user_roles and role_b in user_roles:
                violations.append((role_a, role_b))
        
        return violations


# ==================== 使用示例 ====================

if __name__ == "__main__":
    # 初始化RBAC系统
    rbac = RBACSystem()
    
    # 1. 定义权限
    rbac.add_permission("user_read", "user", "read", "查看用户")
    rbac.add_permission("user_write", "user", "write", "创建/修改用户")
    rbac.add_permission("user_delete", "user", "delete", "删除用户")
    rbac.add_permission("order_read", "order", "read", "查看订单")
    rbac.add_permission("order_write", "order", "write", "创建/修改订单")
    rbac.add_permission("order_approve", "order", "approve", "审批订单")
    rbac.add_permission("finance_report", "finance", "read", "查看财务报表")
    
    # 2. 创建角色
    rbac.create_role("viewer", "只读用户")
    rbac.create_role("editor", "编辑用户")
    rbac.create_role("admin", "系统管理员")
    rbac.create_role("order_clerk", "订单录入员")
    rbac.create_role("order_manager", "订单经理")
    rbac.create_role("finance", "财务人员")
    
    # 3. 分配权限给角色
    # viewer角色
    rbac.assign_permission_to_role("viewer", "user_read")
    rbac.assign_permission_to_role("viewer", "order_read")

    rbac.add_user("zhaoliu", {"department": "财务部"})

    # 6. 权限检查

    print(f"张三 能否删除用户: {rbac.has_permission('zhangsan', 'user', 'delete')}")  # True

    # 8. 职责分离验证
```

#### 1.2 ABAC（基于属性的访问控制）Python实现

```python
"""
ABAC (Attribute-Based Access Control) 实现
基于策略的动态访问控制，支持用户属性、资源属性、环境属性
"""

from typing import Dict, List, Any, Callable
from dataclasses import dataclass, field
from enum import Enum
import datetime


class Effect(Enum):
    """策略效果"""
    ALLOW = "allow"
    DENY = "deny"


class Operator(Enum):
    """比较运算符"""
    EQUALS = "equals"
    NOT_EQUALS = "not_equals"
    IN = "in"
    NOT_IN = "not_in"
    CONTAINS = "contains"
    GREATER_THAN = "greater_than"
    LESS_THAN = "less_than"
    GREATER_EQUAL = "greater_equal"
    LESS_EQUAL = "less_equal"
    STARTS_WITH = "starts_with"
    ENDS_WITH = "ends_with"


@dataclass
class Condition:
    """策略条件"""
    attribute_type: str  # subject, resource, environment
    attribute_name: str
    operator: Operator
    value: Any


@dataclass
class Policy:
    """访问控制策略"""
    id: str
    name: str
    effect: Effect
    conditions: List[Condition] = field(default_factory=list)
    # 条件组合方式: "all" = 全部满足, "any" = 任一满足
    condition_combination: str = "all"
    priority: int = 0  # 优先级，数值越大优先级越高
    description: str = ""


@dataclass
class AccessRequest:
    """访问请求"""
    subject: Dict[str, Any]      # 主体属性（用户）
    resource: Dict[str, Any]     # 资源属性
    action: str                   # 操作
    environment: Dict[str, Any]  # 环境属性


class ABACSystem:
    """ABAC 策略评估引擎"""
    
    def __init__(self):
        self.policies: Dict[str, Policy] = {}
        # 默认决策：当没有策略匹配时
        self.default_decision = Effect.DENY
    
    def add_policy(self, policy: Policy):
        """添加策略"""
        self.policies[policy.id] = policy
    
    def remove_policy(self, policy_id: str):
        """删除策略"""
        if policy_id in self.policies:
            del self.policies[policy_id]
    
    def _evaluate_condition(self, condition: Condition, request: AccessRequest) -> bool:
        """评估单个条件"""
        # 获取属性值
        if condition.attribute_type == "subject":
            attr_value = request.subject.get(condition.attribute_name)
        elif condition.attribute_type == "resource":
            attr_value = request.resource.get(condition.attribute_name)
        elif condition.attribute_type == "environment":
            attr_value = request.environment.get(condition.attribute_name)
        else:
            return False
        
        if attr_value is None:
            return False
        
        # 根据运算符比较
        op = condition.operator
        expected = condition.value
        
        if op == Operator.EQUALS:
            return attr_value == expected
        elif op == Operator.NOT_EQUALS:
            return attr_value != expected
        elif op == Operator.IN:
            return attr_value in expected
        elif op == Operator.NOT_IN:
            return attr_value not in expected
        elif op == Operator.CONTAINS:
            return expected in attr_value
        elif op == Operator.GREATER_THAN:
            return attr_value > expected
        elif op == Operator.LESS_THAN:
            return attr_value < expected
        elif op == Operator.GREATER_EQUAL:
            return attr_value >= expected
        elif op == Operator.LESS_EQUAL:
            return attr_value <= expected
        elif op == Operator.STARTS_WITH:
            return isinstance(attr_value, str) and attr_value.startswith(expected)
        elif op == Operator.ENDS_WITH:
            return isinstance(attr_value, str) and attr_value.endswith(expected)
        
        return False
    
    def _evaluate_policy(self, policy: Policy, request: AccessRequest) -> bool:
        """评估单个策略是否匹配"""
        if not policy.conditions:
            return True  # 没有条件则始终匹配
        
        results = [
            self._evaluate_condition(cond, request)
            for cond in policy.conditions
        ]
        
        if policy.condition_combination == "all":
            return all(results)
        else:  # any
            return any(results)
    
    def evaluate(self, request: AccessRequest) -> tuple[Effect, str]:
        """
        评估访问请求
        返回: (决策, 匹配的策略ID)
        """
        # 按优先级排序策略
        sorted_policies = sorted(
            self.policies.values(),
            key=lambda p: p.priority,
            reverse=True
        )
        
        # 先检查DENY策略（Deny优先原则）
        for policy in sorted_policies:
            if policy.effect == Effect.DENY and self._evaluate_policy(policy, request):
                return Effect.DENY, policy.id
        
        # 再检查ALLOW策略
        for policy in sorted_policies:
            if policy.effect == Effect.ALLOW and self._evaluate_policy(policy, request):
                return Effect.ALLOW, policy.id
        
        # 没有匹配的策略，使用默认决策
        return self.default_decision, "default"
    
    def can_access(self, subject: Dict, resource: Dict, action: str, environment: Dict = None) -> bool:
        """简化的权限检查接口"""
        env = environment or {}
        env.setdefault("current_time", datetime.datetime.now())
        env.setdefault("day_of_week", datetime.datetime.now().weekday())
        
        request = AccessRequest(
            subject=subject,
            resource=resource,
            action=action,
            environment=env
        )
        
        decision, _ = self.evaluate(request)
        return decision == Effect.ALLOW
```

#### 1.3 Casbin 访问控制框架使用示例

```python
"""
Casbin 访问控制框架使用示例
支持：ACL, RBAC, ABAC, RESTful 等多种模型
"""

# 安装: pip install casbin

import casbin
from casbin import Enforcer


# ========== 1. RBAC 模型 ==========

# model.conf - RBAC模型配置
rbac_model_conf = """
[request_definition]
r = sub, obj, act

[policy_definition]
p = sub, obj, act

[role_definition]
g = _, _

[policy_effect]
e = some(where (p.eft == allow))

[matchers]
m = g(r.sub, p.sub) && r.obj == p.obj && r.act == p.act
"""

# policy.csv - 策略文件
rbac_policy = """
p, admin, user, read
p, admin, user, write
p, admin, user, delete
p, editor, user, read
p, editor, user, write
p, viewer, user, read
p, editor, order, read
p, editor, order, write
p, viewer, order, read

g, alice, admin
g, bob, editor
g, charlie, viewer
"""

def rbac_demo():
    """RBAC 模型演示"""
    print("=== Casbin RBAC 演示 ===")
    
    # 创建策略执行器（内存方式）
    e = Enforcer()
    e.init_with_model_and_policy_text(rbac_model_conf, rbac_policy)
    
    # 权限检查
    test_cases = [
        ("alice", "user", "read", True),
        ("alice", "user", "delete", True),
        ("bob", "user", "write", True),
        ("bob", "user", "delete", False),
        ("charlie", "user", "read", True),
        ("charlie", "order", "write", False),
    ]
    
    for sub, obj, act, expected in test_cases:
        result = e.enforce(sub, obj, act)
        status = "✓" if result == expected else "✗"
        print(f"  {status} {sub} 对 {obj} 执行 {act}: {result} (预期: {expected})")
    
    # 获取用户角色
    print(f"\n  alice 的角色: {e.get_roles_for_user('alice')}")
    
    # 获取用户权限
    print(f"  bob 的权限: {e.get_permissions_for_user('bob')}")
    
    # 动态添加策略
    e.add_policy("viewer", "order", "read")
    print(f"\n  添加viewer订单查看权限后:")
    print(f"  charlie 查看订单: {e.enforce('charlie', 'order', 'read')}")


# ========== 2. RESTful 模型 ==========

rest_model_conf = """
[request_definition]
r = sub, obj, act

[policy_definition]
p = sub, obj, act

[role_definition]
g = _, _

[policy_effect]
e = some(where (p.eft == allow))

[matchers]
m = g(r.sub, p.sub) && keyMatch2(r.obj, p.obj) && regexMatch(r.act, p.act)
"""

rest_policy = """
p, admin, /api/users, GET|POST|PUT|DELETE
p, admin, /api/orders, GET|POST|PUT|DELETE
p, user, /api/users/:id, GET
p, user, /api/orders, GET
p, user, /api/orders, POST
p, viewer, /api/users, GET
p, viewer, /api/orders, GET

g, admin_user, admin
g, normal_user, user
g, guest, viewer
"""

def restful_demo():
    """RESTful API 访问控制演示"""
    print("\n=== Casbin RESTful 演示 ===")
    
    e = Enforcer()
    e.init_with_model_and_policy_text(rest_model_conf, rest_policy)
    
    test_cases = [
        ("admin_user", "/api/users", "GET", True),
        ("admin_user", "/api/users", "POST", True),
        ("admin_user", "/api/users/123", "DELETE", True),
        ("normal_user", "/api/users/456", "GET", True),
        ("normal_user", "/api/users", "POST", False),
        ("normal_user", "/api/orders", "GET", True),
        ("normal_user", "/api/orders", "POST", True),
        ("guest", "/api/orders", "GET", True),
        ("guest", "/api/orders", "POST", False),
    ]

        status = "✓" if result == expected else "✗"
        print(f"  {status} {sub} {method} {path}: {result}")


# ========== 3. ABAC 模型 ==========

abac_model_conf = """
[request_definition]
r = sub, obj, act

[policy_definition]
p = sub_rule, obj, act

[policy_effect]
e = some(where (p.eft == allow))

[matchers]
m = eval(p.sub_rule) && r.obj == p.obj && r.act == p.act
"""

abac_policy = """
p, r.sub.Department == "HR" && r.sub.Level >= 2, employee_record, read
p, r.sub.Department == "Finance", financial_report, read
p, r.sub.IsAdmin == true, *, *
"""

def abac_demo():
    """ABAC 模型演示"""
    print("\n=== Casbin ABAC 演示 ===")
    
    e = Enforcer()
    e.init_with_model_and_policy_text(abac_model_conf, abac_policy)
    
    # 定义主体（用字典表示属性）
    hr_staff = {"Name": "张三", "Department": "HR", "Level": 2, "IsAdmin": False}
    finance_staff = {"Name": "李四", "Department": "Finance", "Level": 1, "IsAdmin": False}
    admin = {"Name": "王五", "Department": "IT", "Level": 3, "IsAdmin": True}
    
    test_cases = [
        (hr_staff, "employee_record", "read", True),
        (finance_staff, "employee_record", "read", False),
        (finance_staff, "financial_report", "read", True),
        (admin, "employee_record", "delete", True),
        (admin, "anything", "anything", True),
    ]
    
    for sub, obj, act, expected in test_cases:

        status = "✓" if result == expected else "✗"

if __name__ == "__main__":
    rbac_demo()
    restful_demo()
    abac_demo()
```

### 二、Linux 访问控制配置示例

#### 2.1 文件权限与 ACL

```bash
# ========== 基础文件权限 ==========

# 查看文件权限
ls -l filename
stat filename

# 修改文件所有者
chown user:group filename
chown -R user:group /path/to/dir  # 递归修改

# 修改文件权限
chmod 755 filename      # rwxr-xr-x
chmod 644 filename      # rw-r--r--
chmod 600 filename      # rw------- （仅所有者可读写）
chmod +x filename       # 添加执行权限
chmod u+s program       # 设置SUID
chmod g+s program       # 设置SGID
chmod +t directory      # 设置粘滞位

# 权限数字对照表
# r = 4, w = 2, x = 1
# 7 = rwx, 6 = rw-, 5 = r-x, 4 = r--, 0 = ---

# ========== ACL (访问控制列表) ==========

# 安装ACL支持
# Ubuntu: sudo apt install acl
# CentOS: sudo yum install acl

# 查看ACL
getfacl filename
getfacl /path/to/dir

# 设置用户ACL
setfacl -m u:username:rwx filename       # 给用户rwx权限
setfacl -m u:username:r-- filename       # 给用户只读权限
setfacl -m g:groupname:r-x filename      # 给组r-x权限

# 设置默认ACL（目录用，新文件继承）
setfacl -m d:u:username:rwx /path/to/dir
setfacl -m d:g:groupname:r-x /path/to/dir

# 移除ACL
setfacl -x u:username filename           # 移除特定用户ACL
setfacl -b filename                       # 移除所有ACL

# 递归设置
setfacl -R -m u:username:r-x /path/to/dir

# ========== 特殊权限 ==========

# SUID (Set User ID) - 以文件所有者身份执行
# 典型文件：/usr/bin/passwd
find / -perm -4000 -type f 2>/dev/null    # 查找所有SUID文件

# SGID (Set Group ID) - 以文件所属组身份执行
# 目录设置SGID后，新文件继承目录的组
find / -perm -2000 -type f 2>/dev/null    # 查找所有SGID文件

# Sticky Bit (粘滞位) - 目录中只有文件所有者能删除自己的文件
# 典型目录：/tmp
find / -perm -1000 -type d 2>/dev/null    # 查找所有粘滞位目录
```

#### 2.2 sudo 配置

```bash
# ========== sudoers 配置 ==========

# 编辑sudoers（必须使用visudo）
visudo
visudo -f /etc/sudoers.d/custom      # 推荐：使用drop-in文件

# 基本语法：
# 用户    主机=(运行身份)    命令
# root    ALL=(ALL:ALL)    ALL

# 1. 允许用户执行所有命令
username ALL=(ALL:ALL) ALL

# 2. 允许用户执行特定命令
username ALL=(root) /usr/bin/systemctl restart nginx
username ALL=(root) /usr/bin/apt update, /usr/bin/apt upgrade

# 3. 免密码执行
username ALL=(ALL) NOPASSWD: ALL
username ALL=(root) NOPASSWD: /usr/bin/systemctl restart nginx

# 4. 组配置
%groupname ALL=(ALL:ALL) ALL
%wheel ALL=(ALL) ALL

# 5. 禁止执行特定命令（黑名单）
username ALL=(ALL) ALL, !/usr/bin/su, !/bin/bash

# 6. 命令别名
Cmnd_Alias WEB_SERVICES = /usr/bin/systemctl restart nginx, \
                           /usr/bin/systemctl reload nginx, \
                           /usr/bin/nginx -t
username ALL=(root) WEB_SERVICES

# 7. 主机别名
Host_Alias WEB_SERVERS = web01, web02, web03
username WEB_SERVERS=(root) WEB_SERVICES

# 8. 日志配置
Defaults logfile=/var/log/sudo.log
Defaults log_input, log_output

# 9. 环境变量
Defaults env_reset
Defaults env_keep += "http_proxy https_proxy no_proxy"

# ========== 常用sudo命令 ==========

sudo -l                # 查看当前用户可用的sudo命令
sudo -u user command   # 以指定用户身份执行
sudo -g group command  # 以指定组身份执行
sudo -i                # 切换到root shell（登录shell）
sudo -s                # 切换到root shell（非登录shell）
sudo !!                # 用sudo执行上一条命令
sudo -k                # 清除sudo缓存
sudo -v                # 延长sudo超时时间
```
