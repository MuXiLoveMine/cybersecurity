#### 1.3 身份生命周期管理工具

**SCIM 2.0 客户端示例（Python）：**
```python
import requests
import json

class SCIMClient:
    """SCIM 2.0 客户端 - 用户生命周期管理"""
    
    def __init__(self, base_url, token):
        self.base_url = base_url.rstrip('/')
        self.headers = {
            'Authorization': f'Bearer {token}',
            'Content-Type': 'application/scim+json',
            'Accept': 'application/scim+json'
        }
    
    def create_user(self, username, email, first_name, last_name, password):
        """创建用户"""
        payload = {
            "schemas": ["urn:ietf:params:scim:schemas:core:2.0:User"],
            "userName": username,
            "name": {
                "givenName": first_name,
                "familyName": last_name
            },
            "emails": [
                {"value": email, "primary": True}
            ],
            "password": password,
            "active": True
        }
        
        resp = requests.post(
            f"{self.base_url}/Users",
            headers=self.headers,
            json=payload
        )
        return resp.json()
    
    def get_user(self, user_id):
        """获取用户信息"""
        resp = requests.get(
            f"{self.base_url}/Users/{user_id}",
            headers=self.headers
        )
        return resp.json()
    
    def update_user(self, user_id, updates):
        """更新用户信息"""
        payload = {
            "schemas": ["urn:ietf:params:scim:api:messages:2.0:PatchOp"],
            "Operations": [
                {"op": "replace", "value": updates}
            ]
        }
        resp = requests.patch(
            f"{self.base_url}/Users/{user_id}",
            headers=self.headers,
            json=payload
        )
        return resp.json()
    
    def deactivate_user(self, user_id):
        """禁用用户（离职流程）"""
        return self.update_user(user_id, {"active": False})
    
    def delete_user(self, user_id):
        """删除用户"""
        resp = requests.delete(
            f"{self.base_url}/Users/{user_id}",
            headers=self.headers
        )
        return resp.status_code == 204
    
    def list_users(self, filter_str=None):
        """列出用户"""
        params = {}
        if filter_str:
            params['filter'] = filter_str
        
        resp = requests.get(
            f"{self.base_url}/Users",
            headers=self.headers,
            params=params
        )
        return resp.json()

# 使用示例
if __name__ == "__main__":
    client = SCIMClient(
        "http://localhost:8080/admin/realms/mycompany/scim/v2",
        "admin-token"
    )
    
    # 创建用户（入职）
    user = client.create_user(
        username="zhangsan",
        email="zhangsan@example.com",
        first_name="三",
        last_name="张",
        password="InitialPass123!"
    )
    print(f"用户创建成功: {user.get('id')}")
    
    # 禁用用户（离职）
    # client.deactivate_user(user['id'])
```

#### 2.2 身份治理配置

**用户入职自动化脚本：**
```powershell
<#
.SYNOPSIS
AD用户入职自动化脚本
.DESCRIPTION
根据HR提供的CSV文件批量创建AD用户、分配组、配置邮箱等
#>

param(
    [Parameter(Mandatory=$true)]
    [string]$CsvPath
)

# 导入CSV数据
$newEmployees = Import-Csv -Path $CsvPath

foreach ($emp in $newEmployees) {
    Write-Host "正在处理: $($emp.Name)" -ForegroundColor Cyan
    
    # 生成用户名（拼音/英文名）
    $username = $emp.Username
    $ouPath = "OU=用户,$($emp.OU)"
    $password = ConvertTo-SecureString "Initial@2024" -AsPlainText -Force
    
    # 1. 创建AD用户
    try {
        New-ADUser -Name $emp.Name `
            -GivenName $emp.FirstName `
            -Surname $emp.LastName `
            -SamAccountName $username `
            -UserPrincipalName "$username@corp.example.com" `
            -Path $ouPath `
            -AccountPassword $password `
            -Enabled $true `
            -ChangePasswordAtLogon $true `
            -EmailAddress $emp.Email `
            -Department $emp.Department `
            -Title $emp.Title `
            -EmployeeID $emp.EmployeeID `
            -Manager $emp.Manager
        
        Write-Host "  [+] AD用户创建成功: $username" -ForegroundColor Green
        
        # 2. 添加到部门组
        if ($emp.DepartmentGroup) {
            Add-ADGroupMember -Identity $emp.DepartmentGroup -Members $username
            Write-Host "  [+] 已添加到部门组: $($emp.DepartmentGroup)" -ForegroundColor Green
        }
        
        # 3. 添加到角色组
        if ($emp.RoleGroups) {
            $groups = $emp.RoleGroups -split ';'
            foreach ($group in $groups) {
                Add-ADGroupMember -Identity $group.Trim() -Members $username
                Write-Host "  [+] 已添加到角色组: $group" -ForegroundColor Green
            }
        }
        
        # 4. 记录审计日志
        $logMessage = "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') - 用户创建: $username, 部门: $($emp.Department)"
        Add-Content -Path "C:\IAM\Logs\onboarding.log" -Value $logMessage
        
    } catch {
        Write-Host "  [-] 创建失败: $_" -ForegroundColor Red
        Add-Content -Path "C:\IAM\Logs\onboarding_errors.log" -Value "$(Get-Date) - $username - $_"
    }
}

Write-Host "`n入职处理完成！" -ForegroundColor Yellow
```

**用户离职自动化脚本：**
```powershell
<#
.SYNOPSIS
AD用户离职自动化脚本
.DESCRIPTION
禁用用户账号、移除组、转发邮件、记录审计
#>

param(
    [Parameter(Mandatory=$true)]
    [string]$Username,
    
    [string]$ForwardTo = ""
)

Write-Host "正在处理离职用户: $Username" -ForegroundColor Cyan

try {
    # 1. 禁用账号
    Disable-ADAccount -Identity $Username
    Write-Host "  [+] 账号已禁用" -ForegroundColor Green
    
    # 2. 重置密码（随机）
    $randomPassword = -join ((65..90) + (97..122) + (48..57) | Get-Random -Count 20 | % {[char]$_})
    Set-ADAccountPassword -Identity $Username -Reset -NewPassword (ConvertTo-SecureString $randomPassword -AsPlainText -Force)
    Write-Host "  [+] 密码已重置" -ForegroundColor Green
    
    # 3. 移除所有用户组（保留Domain Users）
    $userGroups = Get-ADPrincipalGroupMembership -Identity $username | 
                  Where-Object {$_.Name -ne "Domain Users"}
    
    foreach ($group in $userGroups) {
        Remove-ADPrincipalGroupMembership -Identity $username -MemberOf $group.Name -Confirm:$false
        Write-Host "  [+] 已从组移除: $($group.Name)" -ForegroundColor Green
    }
    
    # 4. 移动到离职OU
    $targetOU = "OU=已离职,OU=用户,DC=corp,DC=example,DC=com"
    Get-ADUser -Identity $Username | Move-ADObject -TargetPath $targetOU
    Write-Host "  [+] 已移动到离职OU" -ForegroundColor Green
    
    # 5. 设置账号描述（离职信息）
    $description = "离职日期: $(Get-Date -Format 'yyyy-MM-dd'), 操作人: $env:USERNAME"
    Set-ADUser -Identity $Username -Description $description
    Write-Host "  [+] 已设置离职标记" -ForegroundColor Green
    
    # 6. 审计日志
    $logMessage = "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') - 用户离职: $Username, 操作人: $env:USERNAME"
    Add-Content -Path "C:\IAM\Logs\offboarding.log" -Value $logMessage
    
    Write-Host "`n离职处理完成！" -ForegroundColor Yellow
    
} catch {
    Write-Host "  [-] 处理失败: $_" -ForegroundColor Red
    throw
}
```

**PowerShell AD审计脚本：**
```powershell
<#
.SYNOPSIS
Active Directory 安全审计脚本
#>

# 1. 查找空密码用户
Get-ADUser -Filter * -Properties PasswordLastSet, LastLogonDate | 
    Where-Object { $_.PasswordLastSet -eq $null } |
    Select-Object Name, SamAccountName, Enabled |
    Export-Csv -Path "C:\Audit\EmptyPasswordUsers.csv" -NoTypeInformation

# 2. 查找密码永不过期的用户
Get-ADUser -Filter {PasswordNeverExpires -eq $true} -Properties PasswordNeverExpires |

# 3. 查找90天未登录用户
$90DaysAgo = (Get-Date).AddDays(-90)
Get-ADUser -Filter {LastLogonDate -lt $90DaysAgo -and Enabled -eq $true} `
    -Properties LastLogonDate, LastLogonTimestamp |
    Select-Object Name, SamAccountName, LastLogonDate, LastLogonTimestamp |
    Sort-Object LastLogonDate |
    Export-Csv -Path "C:\Audit\InactiveUsers.csv" -NoTypeInformation

# 4. 查找特权用户（管理员组成员）
$adminGroups = @(
    "Domain Admins",
    "Enterprise Admins",
    "Schema Admins",
    "Administrators",
    "Account Operators",
    "Backup Operators"
)

foreach ($group in $adminGroups) {
    $members = Get-ADGroupMember -Identity $group -Recursive | 
               Where-Object {$_.objectClass -eq "user"} |
               Select-Object Name, SamAccountName, objectClass
    Write-Host "`n$group 成员:" -ForegroundColor Yellow
    $members | Format-Table
}

# 5. 查找受保护的用户和组
Get-ADUser -Filter {adminCount -eq 1} -Properties adminCount |
    Select-Object Name, SamAccountName, adminCount

# 6. 查找所有服务账号
Get-ADUser -Filter {Name -like "*svc*" -or Name -like "*service*"} -Properties ServicePrincipalName |
    Select-Object Name, SamAccountName, Enabled, ServicePrincipalName |
    Export-Csv -Path "C:\Audit\ServiceAccounts.csv" -NoTypeInformation

# 7. 检查Kerberoasting风险（有SPN的用户）
Get-ADUser -Filter {ServicePrincipalName -ne "$null"} -Properties ServicePrincipalName |
    Select-Object Name, SamAccountName, ServicePrincipalName |
    Export-Csv -Path "C:\Audit\KerberoastableUsers.csv" -NoTypeInformation
```
