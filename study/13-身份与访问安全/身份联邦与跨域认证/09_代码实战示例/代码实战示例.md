### 9.1 完整的联邦认证流程演示

```python
def demo_federation_flow():
    """完整的联邦认证流程演示"""
    print("=" * 60)
    print("身份联邦认证流程演示")
    print("=" * 60)

    # 1. 初始化SP和IdP
    sp = SAMLFederation(entity_id='https://sp.example.com', role='sp')
    idp = SAMLFederation(entity_id='https://idp.example.com', role='idp')

    # 2. 建立信任关系
    sp.add_trusted_partner('https://idp.example.com', {
        'signing_cert': 'idp_cert_pem...',
        'sso_url': 'https://idp.example.com/sso'
    })
    idp.add_trusted_partner('https://sp.example.com', {
        'signing_cert': 'sp_cert_pem...',
        'acs_url': 'https://sp.example.com/saml/acs'
    })

    print("\n1. SP生成AuthnRequest")
    authn_request = sp.generate_authn_request('https://idp.example.com')
    print(f"   请求ID: {authn_request['request_id']}")

    print("\n2. IdP用户认证并生成SAML Response")
    user_info = {
        'email': 'user@example.com',
        'attributes': {
            'firstName': 'John',
            'lastName': 'Doe',
            'role': 'engineer',
            'department': 'R&D'
        }
    }
    saml_response = idp.generate_saml_response(
        authn_request['request_id'],
        user_info,
        'https://sp.example.com',
        'https://sp.example.com/saml/acs'
    )
    print(f"   响应ID: {saml_response['response_id']}")
    print(f"   断言ID: {saml_response['assertion_id']}")

    print("\n3. SP验证SAML Response")
    result = sp.verify_saml_response(
        saml_response['response_xml'],
        'https://idp.example.com'
    )
    print(f"   验证结果: {result['valid']}")
    if result['valid']:
        print(f"   用户ID: {result['user_id']}")
        print(f"   属性: {result['attributes']}")

    print("\n4. SCIM用户同步")
    # scim = SCIMUserManager('https://scim.example.com', 'bearer_token')
    # scim.sync_user(user_info)
    print("   (SCIM同步需要真实SCIM端点)")


if __name__ == "__main__":
    demo_federation_flow()
```
