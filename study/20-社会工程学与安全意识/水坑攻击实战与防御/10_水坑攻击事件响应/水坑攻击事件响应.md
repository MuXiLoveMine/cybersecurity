# 水坑攻击事件响应

### 10.1 事件响应预案

```
水坑攻击事件响应流程:

Phase 1: 检测与识别 (0-1小时)
  ├── 告警来源
  │   ├── SIEM规则触发(浏览器异常子进程)
  │   ├── EDR告警(可疑进程执行)
  │   ├── 员工报告(浏览器异常行为)
  │   └── 威胁情报匹配(已知恶意域名)
  ├── 初步分析
  │   ├── 确认是否真实事件
  │   ├── 评估影响范围
  │   ├── 识别被入侵的网站
  │   └── 确定受影响的用户/主机
  └── 启动应急响应

Phase 2: 遏制与隔离 (1-4小时)
  ├── 端端遏制
  │   ├── 隔离受感染主机(网络断开)
  │   ├── 终止恶意进程
  │   ├── 清除持久化机制
  │   └── 保留取证证据
  ├── 网络遏制
  │   ├── 封禁恶意C2域名/IP
  │   ├── DNS过滤(阻止重新连接)
  │   ├── 代理规则更新(阻止恶意域名)
  │   └── 防火墙规则更新
  └── 用户通知
      ├── 通知受影响用户
      ├── 指导安全操作(不要访问特定网站)
      └── 要求更改密码(如凭据泄露)

Phase 3: 根因分析 (4-24小时)
  ├── 分析攻击链
  │   ├── 确定被入侵网站
  │   ├── 分析注入的恶意代码
  │   ├── 确定利用的漏洞
  │   ├── 追踪C2基础设施
  │   └── 确定攻击者意图
  ├── 影响评估
  │   ├── 受影响主机数量
  │   ├── 数据泄露评估
  │   ├── 凭据泄露评估
  │   └── 横向移动范围
  └── 取证分析
      ├── 浏览器历史/缓存分析
      ├── 内存取证(如需要)
      ├── 网络流量分析
      └── EDR日志分析

Phase 4: 恢复 (24-72小时)
  ├── 主机恢复
  │   ├── 从干净备份恢复
  │   ├── 或重装系统
  │   ├── 更新所有软件
  │   └── 部署额外防护
  ├── 凭据管理
  │   ├── 更改所有可能泄露的密码
  │   ├── 重置SSO token
  │   ├── 撤销可疑的会话
  │   └── 更新API密钥
  └── 监控加强
      ├── 增加SIEM告警灵敏度
      ├── 加大EDR监控范围
      ├── 部署蜜罐/蜜标
      └── 加强网络流量监控

Phase 5: 事后总结 (72小时+)
  ├── 事件报告
  │   ├── 完整攻击链分析
  │   ├── 影响评估
  │   ├── 响应效果评估
  │   └── 改进建议
  ├── 流程改进
  │   ├── 补丁管理流程优化
  │   ├── URL过滤策略更新
  │   ├── 浏览器隔离部署
  │   └── 安全意识培训更新
  └── 长期措施
      ├── 部署CSP策略
      ├── 威胁情报订阅更新
      ├── 定期渗透测试
      └── 安全架构优化
```

### 10.2 快速响应脚本

```python
# 水坑攻击快速响应脚本
import subprocess
import json
from datetime import datetime

class WateringHoleIncidentResponse:
    """水坑攻击事件快速响应"""

    def __init__(self):
        self.evidence = {}
        self.actions_taken = []
        self.affected_hosts = []
        self.blocked_domains = []
        self.blocked_ips = []

    def initial_triage(self, alert_data):
        """初步分析"""
        incident_id = f"WH-{datetime.now().strftime('%Y%m%d%H%M%S')}"

        self.evidence = {
            'incident_id': incident_id,
            'timestamp': datetime.now().isoformat(),
            'alert_source': alert_data.get('source'),
            'alert_type': alert_data.get('type'),
            'affected_user': alert_data.get('user'),
            'affected_host': alert_data.get('host'),
            'malicious_domain': alert_data.get('domain'),
            'malicious_process': alert_data.get('process'),
        }

        # 快速分类
        severity = self._assess_severity(alert_data)

        return {
            'incident_id': incident_id,
            'severity': severity,
            'initial_findings': self.evidence,
        }

    def contain_threat(self, host, malicious_indicators):
        """遏制威胁"""
        actions = []

        # 1. 隔离主机
        if self._isolate_host(host):
            actions.append(f"主机 {host} 已隔离")

        # 2. 终止恶意进程
        for process in malicious_indicators.get('processes', []):
            if self._kill_process(host, process):
                actions.append(f"已终止进程 {process}")

        # 3. 封禁恶意域名
        for domain in malicious_indicators.get('domains', []):
            if self._block_domain(domain):
                self.blocked_domains.append(domain)
                actions.append(f"已封禁域名 {domain}")

        # 4. 封禁恶意IP
        for ip in malicious_indicators.get('ips', []):
            if self._block_ip(ip):
                self.blocked_ips.append(ip)
                actions.append(f"已封禁IP {ip}")

        # 5. 清除持久化
        for persistence in malicious_indicators.get('persistence', []):
            if self._remove_persistence(host, persistence):
                actions.append(f"已清除持久化 {persistence}")

        self.actions_taken.extend(actions)
        return actions

    def collect_forensics(self, host):
        """收集取证数据"""
        forensics = {}

        # 1. 浏览器历史
        forensics['browser_history'] = self._get_browser_history(host)

        # 2. 浏览器缓存
        forensics['browser_cache'] = self._get_browser_cache(host)

        # 3. 进程列表
        forensics['processes'] = self._get_process_list(host)

        # 4. 网络连接
        forensics['network_connections'] = self._get_network_connections(host)

        # 5. 注册表/启动项
        forensics['persistence'] = self._get_persistence_items(host)

        # 6. 内存取证
        forensics['memory_dump'] = self._create_memory_dump(host)

        # 7. 文件系统
        forensics['recent_files'] = self._get_recent_files(host)

        self.evidence['forensics'] = forensics
        return forensics

    def generate_incident_report(self):
        """生成事件报告"""
        report = f"""
水坑攻击事件响应报告
{'='*60}

事件ID: {self.evidence.get('incident_id')}
时间: {self.evidence.get('timestamp')}
告警来源: {self.evidence.get('alert_source')}
告警类型: {self.evidence.get('alert_type')}

受影响系统:
  用户: {self.evidence.get('affected_user')}
  主机: {self.evidence.get('affected_host')}
  恶意域名: {self.evidence.get('malicious_domain')}
  恶意进程: {self.evidence.get('malicious_process')}

已采取的行动:
"""
        for action in self.actions_taken:
            report += f"  - {action}\n"

        report += f"""
已封禁的域名:
"""
        for domain in self.blocked_domains:
            report += f"  - {domain}\n"

        report += f"""
已封禁的IP:
"""
        for ip in self.blocked_ips:
            report += f"  - {ip}\n"

        report += f"""
后续行动:
  1. 深入分析取证数据
  2. 确定攻击链完整路径
  3. 评估数据泄露范围
  4. 更改所有可能泄露的凭据
  5. 从干净备份恢复受影响主机
  6. 加强监控和告警
  7. 编写完整事件报告
"""
        return report

    def _isolate_host(self, host):
        """隔离主机"""
        # 通过EDR/网络设备隔离
        # 断开网络但保持管理通道
        return True

    def _kill_process(self, host, process):
        """终止恶意进程"""
        return True

    def _block_domain(self, domain):
        """封禁域名"""
        # DNS过滤/代理阻止
        return True

    def _block_ip(self, ip):
        """封禁IP"""
        # 防火墙规则
        return True

    def _remove_persistence(self, host, persistence):
        """清除持久化"""
        return True

    def _assess_severity(self, alert_data):
        """评估严重程度"""
        if alert_data.get('process') in ['powershell.exe', 'cmd.exe', 'mshta.exe']:
            return 'CRITICAL'
        elif alert_data.get('rce_confirmed'):
            return 'CRITICAL'
        elif alert_data.get('malicious_download'):
            return 'HIGH'
        else:
            return 'MEDIUM'

# 使用示例
ir = WateringHoleIncidentResponse()

# 接收告警
alert = {
    'source': 'SIEM',
    'type': 'browser_child_process',
    'user': 'jdoe@company.com',
    'host': 'DESK-001234',
    'domain': 'suspicious-site.com',
    'process': 'powershell.exe',
}

# 初步分析
triage = ir.initial_triage(alert)
print(f"事件ID: {triage['incident_id']}")
print(f"严重程度: {triage['severity']}")

# 遏制
actions = ir.contain_threat('DESK-001234', {
    'processes': ['powershell.exe', 'malware.exe'],
    'domains': ['suspicious-site.com', 'c2.attacker.com'],
    'ips': ['192.168.1.100', '10.0.0.50'],
    'persistence': ['HKCU\\Run\\Update', 'C:\\Users\\jdoe\\AppData\\malware.dll'],
})
for action in actions:
    print(f"  {action}")

# 生成报告
print(ir.generate_incident_report())
```
