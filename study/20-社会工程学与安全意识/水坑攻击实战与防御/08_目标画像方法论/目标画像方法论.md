# 水坑攻击目标画像工具
import requests
import json
from collections import Counter
from urllib.parse import urlparse

class TargetProfiler:
    """目标组织画像分析"""

    def __init__(self, organization_name):
        self.org = organization_name
        self.profile = {
            'organization': organization_name,
            'employees': [],
            'frequent_sites': [],
            'tech_stack': [],
            'external_links': [],
        }

    def collect_linkedin_profiles(self, search_terms):
        """收集LinkedIn上的员工信息"""
        # 注意: 实际使用需要遵守LinkedIn API条款
        # 这里仅展示分析逻辑

        # 搜索目标组织员工
        # 分析职位分布
        job_titles = []
        for profile in self._search_linkedin(self.org):
            title = profile.get('title', '')
            job_titles.append(title)

        # 分析技术角色
        tech_roles = [t for t in job_titles
                      if any(kw in t.lower()
                             for kw in ['developer', 'engineer',
                                        'devops', 'security', 'admin'])]

        self.profile['tech_roles'] = tech_roles
        return tech_roles

    def analyze_job_postings(self, job_urls):
        """分析招聘信息推断技术栈"""
        tech_keywords = {
            'languages': ['Python', 'Java', 'Go', 'C++', 'Rust',
                          'JavaScript', 'TypeScript'],
            'frameworks': ['Spring', 'Django', 'React', 'Vue', 'Angular'],
            'cloud': ['AWS', 'Azure', 'GCP', 'Kubernetes', 'Docker'],
            'databases': ['MySQL', 'PostgreSQL', 'MongoDB', 'Redis'],
            'security': ['firewall', 'IDS', 'SIEM', 'EDR'],
        }

        detected_tech = Counter()
        for url in job_urls:
            content = self._fetch_url(url)
            for category, keywords in tech_keywords.items():
                for kw in keywords:
                    if kw.lower() in content.lower():
                        detected_tech[f"{category}:{kw}"] += 1

        self.profile['tech_stack'] = detected_tech.most_common(20)
        return dict(detected_tech)

    def identify_frequent_sites(self):
        """识别目标常访问的网站"""
        # 分析员工社交媒体
        # 分析技术论坛活动
        # 分析GitHub活动

        candidate_sites = {
            'developer_communities': [
                'github.com', 'stackoverflow.com',
                'dev.to', 'medium.com'
            ],
            'news_sites': [
                'news.ycombinator.com', 'techcrunch.com',
                'arstechnica.com'
            ],
            'industry_specific': [
                # 根据目标行业确定
            ],
            'internal_resources': [
                # 内部Wiki/Confluence等
            ],
        }

        self.profile['frequent_sites'] = candidate_sites
        return candidate_sites

    def determine_browser_distribution(self):
        """确定目标群体的浏览器分布"""
        # 通过分析目标组织的公开信息
        # 招聘信息中提到的技术要求
        # 员工的LinkedIn/社交媒体

        # 大多数组织:
        # - Windows: Chrome/Edge
        # - macOS: Safari/Chrome
        # - Linux: Firefox/Chrome

        browser_dist = {
            'Chrome': 65,
            'Edge': 20,
            'Firefox': 10,
            'Safari': 5,
        }

        # 攻击者需要选择有exploit的浏览器
        self.profile['browser_distribution'] = browser_dist
        return browser_dist

    def select_target_sites(self, top_n=5):
        """选择最佳水坑目标网站"""
        sites = self.profile.get('frequent_sites', {})
        candidates = []

        for category, urls in sites.items():
            for url in urls:
                score = self._score_site(url, category)
                candidates.append({
                    'url': url,
                    'category': category,
                    'score': score,
                })

        candidates.sort(key=lambda x: x['score'], reverse=True)
        return candidates[:top_n]

    def _score_site(self, url, category):
        """网站评分(越高越适合作为水坑)"""
        score = 0
        # 开发者社区得分高(技术人员常访问)
        if category == 'developer_communities':
            score += 30
        # 新闻网站得分中等
        elif category == 'news_sites':
            score += 20
        # 行业特定网站得分高
        elif category == 'industry_specific':
            score += 25

        # 检查网站技术栈(是否有已知漏洞)
        # 检查网站更新频率
        # 检查网站管理质量

        return score

    def generate_profile_report(self):
        """生成目标画像报告"""
        report = f"\n目标组织: {self.org}\n"
        report += f"{'='*50}\n"

        if self.profile.get('tech_stack'):
            report += "\n技术栈分析:\n"
            for tech, count in self.profile['tech_stack']:
                report += f"  {tech}: {count}\n"

        if self.profile.get('browser_distribution'):
            report += "\n浏览器分布:\n"
            for browser, pct in self.profile['browser_distribution'].items():
                report += f"  {browser}: {pct}%\n"

        if self.profile.get('frequent_sites'):
            report += "\n候选水坑网站:\n"
            for cat, sites in self.profile['frequent_sites'].items():
                report += f"  [{cat}]\n"
                for site in sites:
                    report += f"    - {site}\n"

        return report
