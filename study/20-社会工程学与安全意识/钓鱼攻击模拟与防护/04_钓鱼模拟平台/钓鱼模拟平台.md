### 4.1 GoPhish部署

#### 4.1.1 Docker部署GoPhish

```bash
# 方法1：Docker部署
docker run -d \
  --name gophish \
  -p 3333:3333 \
  -p 80:80 \
  -v /path/to/gophish:/data \
  gophish/gophish

# 方法2：二进制部署
wget https://github.com/gophish/gophish/releases/download/v0.12.1/gophish-v0.12.1-linux-64bit.zip
unzip gophish-v0.12.1-linux-64bit.zip
cd gophish
# 修改config.json
./gophish

# GoPhish默认端口
# 3333 - 管理界面
# 80/443 - 钓鱼网站（着陆页）
```

### 4.2 GoPhish API使用

```python
# GoPhish API Python客户端

import requests
import json
import urllib3
urllib3.disable_warnings()

class GoPhishClient:
    """GoPhish API客户端"""

    def __init__(self, api_url, api_key):
        """
        初始化GoPhish客户端
        :param api_url: GoPhish管理地址，如 https://localhost:3333
        :param api_key: API密钥
        """
        self.api_url = api_url.rstrip('/')
        self.api_key = api_key
        self.headers = {
            'Authorization': f'Bearer {api_key}',
            'Content-Type': 'application/json'
        }

    # ===== 邮件模板管理 =====

    def create_template(self, name, subject, html, text=''):
        """创建邮件模板"""
        data = {
            'name': name,
            'subject': subject,
            'html': html,
            'text': text
        }
        resp = requests.post(
            f'{self.api_url}/api/templates/',
            headers=self.headers,
            data=json.dumps(data),
            verify=False
        )
        return resp.json()

    def list_templates(self):
        """列出所有模板"""
        resp = requests.get(
            f'{self.api_url}/api/templates/',
            headers=self.headers,
            verify=False
        )
        return resp.json()

    def delete_template(self, template_id):
        """删除模板"""
        resp = requests.delete(
            f'{self.api_url}/api/templates/{template_id}',
            headers=self.headers,
            verify=False
        )
        return resp.json()

    # ===== 着陆页管理 =====

    def create_landing_page(self, name, html, redirect_url='',
                           capture_credentials=False):
        """创建钓鱼着陆页"""
        data = {
            'name': name,
            'html': html,
            'redirect_url': redirect_url,
            'capture_credentials': capture_credentials,
            'capture_passwords': capture_credentials
        }
        resp = requests.post(
            f'{self.api_url}/api/pages/',
            headers=self.headers,
            data=json.dumps(data),
            verify=False
        )
        return resp.json()

    def import_site(self, name, url):
        """导入现有网站作为着陆页"""
        data = {
            'name': name,
            'import_site': url
        }
        resp = requests.post(
            f'{self.api_url}/api/pages/import',
            headers=self.headers,
            data=json.dumps(data),
            verify=False
        )
        return resp.json()

    # ===== 发送配置管理 =====

    def create_sending_profile(self, name, from_address, host,
                               username='', password=''):
        """创建SMTP发送配置"""
        data = {
            'name': name,
            'interface_type': 'SMTP',
            'from_address': from_address,
            'host': host,
            'username': username,
            'password': password,
            'ignore_cert_errors': False
        }
        resp = requests.post(
            f'{self.api_url}/api/smtp/',
            headers=self.headers,
            data=json.dumps(data),
            verify=False
        )
        return resp.json()

    # ===== 目标用户管理 =====

    def create_user_group(self, name, users):
        """创建用户组
        :param users: [{'email': 'user1@company.com',
                       'first_name': '张', 'last_name': '三'}, ...]
        """
        data = {
            'name': name,
            'targets': users
        }
        resp = requests.post(
            f'{self.api_url}/api/groups/',
            headers=self.headers,
            data=json.dumps(data),
            verify=False
        )
        return resp.json()

    # ===== 活动管理 =====

    def create_campaign(self, name, template_id, page_id, smtp_id,
                       group_id, launch_date=''):
        """创建钓鱼模拟活动"""
        data = {
            'name': name,
            'template': {'id': template_id},
            'page': {'id': page_id},
            'smtp': {'id': smtp_id},
            'groups': [{'id': group_id}],
            'launch_date': launch_date
        }
        resp = requests.post(
            f'{self.api_url}/api/campaigns/',
            headers=self.headers,
            data=json.dumps(data),
            verify=False
        )
        return resp.json()

    def get_campaign(self, campaign_id):
        """获取活动详情"""
        resp = requests.get(
            f'{self.api_url}/api/campaigns/{campaign_id}',
            headers=self.headers,
            verify=False
        )
        return resp.json()

    def get_campaign_results(self, campaign_id):
        """获取活动结果"""
        resp = requests.get(
            f'{self.api_url}/api/campaigns/{campaign_id}/results',
            headers=self.headers,
            verify=False
        )
        return resp.json()

    def get_campaign_summary(self, campaign_id):
        """获取活动摘要统计"""
        resp = requests.get(
            f'{self.api_url}/api/campaigns/{campaign_id}/summary',
            headers=self.headers,
            verify=False
        )
        return resp.json()


# ===== 使用示例 =====

def setup_phishing_campaign():
    """设置完整的钓鱼模拟活动"""

    client = GoPhishClient('https://localhost:3333', 'YOUR_API_KEY')

    # 1. 创建邮件模板
    email_html = """
    <html><body>
    <p>各位同事好，</p>
    <p>您的密码将在24小时内过期，请点击下方链接更新：</p>
    <a href="{{.URL}}">更新密码</a>
    <p>IT运维中心</p>
    </body></html>
    """
    template = client.create_template(
        name='密码过期通知 - 2026Q3',
        subject='【重要】您的密码即将过期',
        html=email_html
    )

    # 2. 创建着陆页（克隆登录页面）
    landing_page = client.create_landing_page(
        name='登录验证页面',
        html='<html><body><form action=""><input name="username">'
             '<input name="password" type="password">'
             '<button>登录</button></form></body></html>',
        redirect_url='https://login.microsoftonline.com',
        capture_credentials=True
    )

    # 3. 创建发送配置
    smtp_profile = client.create_sending_profile(
        name='IT运维中心发件配置',
        from_address='IT运维中心 <it-support@similar-domain.com>',
        host='smtp.relay-service.com:587',
        username='relay_user',
        password='relay_password'
    )

    # 4. 创建目标用户组
    users = [
        {'email': 'user1@company.com', 'first_name': '张', 'last_name': '三'},
        {'email': 'user2@company.com', 'first_name': '李', 'last_name': '四'},
    ]
    group = client.create_user_group('财务部测试组', users)

    # 5. 创建并启动活动
    campaign = client.create_campaign(
        name='2026Q3财务部安全意识评估',
        template_id=template['id'],
        page_id=landing_page['id'],
        smtp_id=smtp_profile['id'],
        group_id=group['id']
    )

    return campaign
```
