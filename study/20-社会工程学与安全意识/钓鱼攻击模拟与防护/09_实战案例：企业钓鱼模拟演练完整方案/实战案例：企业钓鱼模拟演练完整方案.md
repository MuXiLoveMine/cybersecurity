### 8.3 GoPhish自动化部署脚本

```python
# 企业钓鱼模拟自动化部署脚本

import requests
import json
import schedule
import time
from datetime import datetime, timedelta
import smtplib
from email.mime.text import MIMEText

class PhishingSimulationManager:
    """钓鱼模拟管理器"""

    def __init__(self, gophish_url, gophish_key, notify_email):
        self.gophish = GoPhishClient(gophish_url, gophish_key)
        self.notify_email = notify_email
        self.templates = self._load_templates()
        self.scenarios = self._load_scenarios()

    def _load_templates(self):
        """加载邮件模板库"""
        return {
            'password_expiry': {
                'name': '密码过期通知',
                'subject': '【重要】您的密码将在24小时内过期',
                'category': 'IT',
                'difficulty': 'easy'
            },
            'shared_document': {
                'name': '共享文档通知',
                'subject': '同事与您共享了文档',
                'category': '协作',
                'difficulty': 'medium'
            },
            'delivery_notice': {
                'name': '快递通知',
                'subject': '【顺丰速运】您的包裹无法派送',
                'category': '生活',
                'difficulty': 'easy'
            },
            'bank_verification': {
                'name': '银行验证',
                'subject': '【招商银行】验证您的账户',
                'category': '金融',
                'difficulty': 'medium'
            },
            'supplier_invoice': {
                'name': '供应商发票',
                'subject': '发票确认 - 请尽快处理',
                'category': '财务',
                'difficulty': 'hard'
            },
            'compliance_audit': {
                'name': '合规审计',
                'subject': '年度合规审计 - 文件提交要求',
                'category': '合规',
                'difficulty': 'hard'
            }
        }

    def _load_scenarios(self):
        """加载演练场景"""
        return [
            {'month': 1, 'template': 'password_expiry',
             'targets': 'all', 'type': 'baseline'},
            {'month': 3, 'template': 'password_expiry',
             'targets': 'all', 'type': 'quarterly'},
            {'month': 6, 'template': 'shared_document',
             'targets': 'all', 'type': 'quarterly'},
            {'month': 9, 'template': 'compliance_audit',
             'targets': 'management', 'type': 'quarterly'},
            {'month': 12, 'template': 'supplier_invoice',
             'targets': 'all', 'type': 'annual'}
        ]

    def run_simulation(self, scenario_name, target_emails, launch_date=None):
        """执行钓鱼模拟"""
        print(f"[*] 启动钓鱼模拟: {scenario_name}")

        # 获取模板配置
        template_config = self.templates.get(scenario_name)
        if not template_config:
            print(f"[!] 模板 {scenario_name} 不存在")
            return None

        # 创建GoPhish模板
        email_html = self._generate_email_html(scenario_name)
        template = self.gophish.create_template(
            name=f"{template_config['name']} - {datetime.now().strftime('%Y%m%d')}",
            subject=template_config['subject'],
            html=email_html
        )

        # 创建着陆页
        landing_page = self.gophish.create_landing_page(
            name=f"着陆页 - {scenario_name}",
            html=self._generate_landing_page(scenario_name),
            redirect_url='https://www.google.com',
            capture_credentials=False  # 仅跟踪点击，不收集凭证
        )

        # 创建用户组
        users = [{'email': e, 'first_name': '', 'last_name': ''}
                for e in target_emails]
        group = self.gophish.create_user_group(
            f"模拟目标 - {datetime.now().strftime('%Y%m%d')}", users)

        # 创建活动
        campaign = self.gophish.create_campaign(
            name=f"钓鱼模拟 - {template_config['name']} - "
                 f"{datetime.now().strftime('%Y-%m-%d')}",
            template_id=template['id'],
            page_id=landing_page['id'],
            smtp_id=self._get_smtp_profile_id(),
            group_id=group['id'],
            launch_date=launch_date
        )

        print(f"[+] 活动已创建: ID={campaign.get('id')}")
        return campaign

    def _generate_email_html(self, scenario_name):
        """生成邮件HTML内容"""
        templates = {
            'password_expiry': self._template_password_expiry(),
            'shared_document': self._template_shared_document(),
            'delivery_notice': self._template_delivery_notice(),
        }
        return templates.get(scenario_name, '<p>测试邮件</p>')

    def _template_password_expiry(self):
        """密码过期模板"""
        return f"""
        <html><body>
        <p>亲爱的同事，</p>
        <p>您的企业账户密码将在<strong>24小时内</strong>过期。</p>
        <p>请点击下方链接更新密码：</p>
        <p><a href="{{{{.URL}}}}">更新密码</a></p>
        <p>IT运维中心</p>
        </body></html>
        """

    def _template_shared_document(self):
        """共享文档模板"""
        return f"""
        <html><body>
        <p><strong>张伟</strong>与您共享了文档</p>
        <p>"2026年Q3项目计划.xlsx"</p>
        <p><a href="{{{{.URL}}}}">查看文档</a></p>
        </body></html>
        """

    def _template_delivery_notice(self):
        """快递通知模板"""
        return f"""
        <html><body>
        <p>【顺丰速运】您的包裹SF1234567890因地址不详无法派送。</p>
        <p>请点击链接更新配送信息：</p>
        <p><a href="{{{{.URL}}}}">更新配送信息</a></p>
        </body></html>
        """

    def _generate_landing_page(self, scenario_name):
        """生成着陆页"""
        return """
        <html><body>
        <h2>安全提醒</h2>
        <p>这是企业安全意识评估的模拟钓鱼页面。</p>
        <p>在实际攻击中，此页面可能会窃取您的凭证或安装恶意软件。</p>
        <h3>如何识别此类钓鱼攻击：</h3>
        <ul>
            <li>检查发件人完整邮箱地址</li>
            <li>悬停查看链接真实URL</li>
            <li>注意紧急感和威胁性语言</li>
            <li>通过官方渠道验证而非点击邮件链接</li>
        </ul>
        <p>请将此邮件报告给安全团队: security@company.com</p>
        </body></html>
        """

    def collect_results(self, campaign_id):
        """收集和分析结果"""
        results = self.gophish.get_campaign_results(campaign_id)

        analysis = {
            'campaign_id': campaign_id,
            'timestamp': datetime.now().isoformat(),
            'total_targets': len(results),
            'stats': {
                'sent': 0,
                'opened': 0,
                'clicked': 0,
                'reported': 0
            },
            'details': []
        }

        for r in results:
            detail = {
                'email': r.get('email', ''),
                'status': r.get('status', ''),
                'opened': r.get('results', {}).get('opened', False),
                'clicked': r.get('results', {}).get('clicked', False),
                'reported': r.get('results', {}).get('reported', False)
            }

            if detail['status'] == 'Email Sent':
                analysis['stats']['sent'] += 1
            if detail['opened']:
                analysis['stats']['opened'] += 1
            if detail['clicked']:
                analysis['stats']['clicked'] += 1
            if detail['reported']:
                analysis['stats']['reported'] += 1

            analysis['details'].append(detail)

        # 计算比率
        sent = analysis['stats']['sent']
        if sent > 0:
            analysis['rates'] = {
                'open_rate': f"{analysis['stats']['opened'] / sent * 100:.1f}%",
                'click_rate': f"{analysis['stats']['clicked'] / sent * 100:.1f}%",
                'report_rate': f"{analysis['stats']['reported'] / sent * 100:.1f}%"
            }

        return analysis

    def generate_report(self, analysis):
        """生成评估报告"""
        report = f"""
钓鱼模拟评估报告
{'=' * 50}

评估时间: {analysis['timestamp']}
目标人数: {analysis['total_targets']}
邮件发送: {analysis['stats']['sent']}

关键指标:
  邮件打开率: {analysis.get('rates', {}).get('open_rate', 'N/A')}
  链接点击率: {analysis.get('rates', {}).get('click_rate', 'N/A')}
  邮件报告率: {analysis.get('rates', {}).get('report_rate', 'N/A')}

详细结果:
"""
        for d in analysis['details']:
            status = "点击" if d['clicked'] else ("打开" if d['opened'] else "未打开")
            report += f"  {d['email']}: {status}\n"

        return report

    def schedule_simulations(self):
        """安排模拟计划"""
        for scenario in self.scenarios:
            # 计算执行日期（每月15日）
            run_date = datetime.now().replace(
                month=scenario['month'], day=15)
            schedule_name = scenario['template']

            print(f"[计划] {run_date.strftime('%Y-%m-%d')} - {schedule_name}")
```
