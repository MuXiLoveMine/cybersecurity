### 8.2 应急响应脚本包

```bash
#!/bin/bash
# ICS应急响应快速采集脚本
# 在Linux系统上快速采集取证数据

CASE_ID=${1:-"INC-UNKNOWN"}
TIMESTAMP=$(date +%Y%m%d_%H%M%S)
EVIDENCE_DIR="/tmp/ics_evidence/${CASE_ID}_${TIMESTAMP}"

echo "============================================"
echo "ICS应急响应取证采集"
echo "案件编号: $CASE_ID"
echo "时间戳: $TIMESTAMP"
echo "证据目录: $EVIDENCE_DIR"
echo "============================================"

# 创建目录
mkdir -p "$EVIDENCE_DIR"

# 1. 系统信息
echo "[1] 采集系统信息..."
uname -a > "$EVIDENCE_DIR/system_info.txt"
date > "$EVIDENCE_DIR/system_time.txt"
uptime >> "$EVIDENCE_DIR/system_info.txt"

# 2. 网络信息
echo "[2] 采集网络信息..."
netstat -anp > "$EVIDENCE_DIR/netstat.txt" 2>&1
arp -a > "$EVIDENCE_DIR/arp_table.txt" 2>&1
netstat -rn > "$EVIDENCE_DIR/routing_table.txt" 2>&1
iptables -L -n -v > "$EVIDENCE_DIR/firewall_rules.txt" 2>&1
ip addr > "$EVIDENCE_DIR/ip_addresses.txt" 2>&1

# 3. 进程信息
echo "[3] 采集进程信息..."
ps aux > "$EVIDENCE_DIR/processes.txt" 2>&1
ps -ef --forest > "$EVIDENCE_DIR/process_tree.txt" 2>&1

# 4. 用户信息
echo "[4] 采集用户信息..."
who > "$EVIDENCE_DIR/active_users.txt" 2>&1
w >> "$EVIDENCE_DIR/active_users.txt" 2>&1
last > "$EVIDENCE_DIR/login_history.txt" 2>&1
cat /etc/passwd > "$EVIDENCE_DIR/passwd.txt" 2>&1
cat /etc/shadow > "$EVIDENCE_DIR/shadow.txt" 2>&1

# 5. 系统日志
echo "[5] 采集系统日志..."
cp /var/log/syslog "$EVIDENCE_DIR/" 2>&1
cp /var/log/messages "$EVIDENCE_DIR/" 2>&1
cp /var/log/auth.log "$EVIDENCE_DIR/" 2>&1
cp /var/log/audit/audit.log "$EVIDENCE_DIR/" 2>&1

# 6. 计划任务
echo "[6] 采集计划任务..."
crontab -l > "$EVIDENCE_DIR/crontab_root.txt" 2>&1
ls -la /etc/cron* > "$EVIDENCE_DIR/cron_dirs.txt" 2>&1

# 7. 服务
echo "[7] 采集服务信息..."
systemctl list-units --type=service > "$EVIDENCE_DIR/services.txt" 2>&1

# 8. 网络流量 (30秒)
echo "[8] 采集网络流量 (30秒)..."
timeout 30 tcpdump -i any -w "$EVIDENCE_DIR/capture.pcap" -s 0 2>/dev/null

# 9. 计算哈希
echo "[9] 计算文件哈希..."
cd "$EVIDENCE_DIR"
find . -type f -exec sha256sum {} \; > "file_hashes.txt"

# 10. 生成报告
echo "[10] 生成报告..."
echo "============================================" > "summary.txt"
echo "ICS应急响应取证报告" >> "summary.txt"
echo "案件编号: $CASE_ID" >> "summary.txt"
echo "采集时间: $TIMESTAMP" >> "summary.txt"
echo "============================================" >> "summary.txt"
echo "" >> "summary.txt"
echo "证据文件列表:" >> "summary.txt"
ls -la >> "summary.txt"
echo "" >> "summary.txt"
echo "文件哈希:" >> "summary.txt"
cat file_hashes.txt >> "summary.txt"

echo ""
echo "============================================"
echo "取证采集完成!"
echo "证据目录: $EVIDENCE_DIR"
echo "文件数量: $(find $EVIDENCE_DIR -type f | wc -l)"
echo "总大小: $(du -sh $EVIDENCE_DIR | cut -f1)"
echo "============================================"
```
