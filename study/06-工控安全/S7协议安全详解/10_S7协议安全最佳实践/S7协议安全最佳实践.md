### 9.2 监控规则

```
S7协议安全监控规则 (Suricata):

# 1. 检测非授权S7连接
alert tcp $EXTERNAL_NET any -> $PLC_NET 102 (
    msg:"S7 Protocol Connection from External Network";
    content:"|03 00|"; depth:2;
    content:"|32|"; depth:8; offset:7;
    sid:3000001; rev:1;
    classtype:attempted-admin; priority:1;
)

# 2. 检测S7Comm (旧协议) 使用
alert tcp $ENGINEERING_NET any -> $S7_1500_NET 102 (
    msg:"S7Comm Legacy Protocol Used with S7-1500";
    content:"|03 00|"; depth:2;
    content:"|32|"; depth:8; offset:7;
    sid:3000002; rev:1;
    classtype:policy-violation; priority:2;
)

# 3. 检测CPU停止命令
alert tcp $HOME_NET any -> $PLC_NET 102 (
    msg:"S7 CPU STOP Command";
    content:"|32|"; depth:8; offset:7;
    content:"|28 01|"; distance:8;
    sid:3000003; rev:1;
    classtype:attempted-admin; priority:1;
)

# 4. 检测密码验证 (可能的暴力破解)
alert tcp $HOME_NET any -> $PLC_NET 102 (
    msg:"S7 Password Verification";
    content:"|32|"; depth:8; offset:7;
    content:"|28 02|"; distance:8;
    sid:3000004; rev:1;
    threshold:type both, by_src, count 5, seconds 60;
    classtype:attempted-admin; priority:2;
)

# 5. 检测大量读操作 (可能的数据窃取)
alert tcp $HOME_NET any -> $PLC_NET 102 (
    msg:"Excessive S7 Read Operations";
    content:"|32|"; depth:8; offset:7;
    content:"|04|"; distance:8; within:1;
    sid:3000005; rev:1;
    threshold:type both, by_src, count 100, seconds 30;
    classtype:attempted-recon; priority:3;
)

# 6. 检测大量写操作 (可能的攻击)
alert tcp $HOME_NET any -> $PLC_NET 102 (
    msg:"Excessive S7 Write Operations";
    content:"|32|"; depth:8; offset:7;
    content:"|05|"; distance:8; within:1;
    sid:3000006; rev:1;
    threshold:type both, by_src, count 50, seconds 30;
    classtype:attempted-admin; priority:2;
)

# 7. 检测程序下载 (可能的代码注入)
alert tcp $HOME_NET any -> $PLC_NET 102 (
    msg:"S7 Program Download";
    content:"|32|"; depth:8; offset:7;
    content:"|28 0E|"; distance:8;
    sid:3000007; rev:1;
    classtype:trojan-activity; priority:1;
)
```
