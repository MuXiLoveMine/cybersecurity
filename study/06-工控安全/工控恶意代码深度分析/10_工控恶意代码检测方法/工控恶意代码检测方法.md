### 9.1 网络流量检测

```python
#!/usr/bin/env python3
"""工控恶意代码网络流量检测引擎"""
import re
import struct
from collections import defaultdict
from datetime import datetime, timedelta

class ICSMalwareDetector:
    """基于网络流量的工控恶意代码检测"""
    
    def __init__(self):
        self.signatures = self._load_signatures()
        self.suspicious_ips = set()
        self.connection_patterns = defaultdict(list)
        self.opc_scan_count = 0
        self.plc_write_count = 0
        self.sis_access_count = 0
        self.alerts = []
    
    def _load_signatures(self):
        """加载恶意代码网络签名"""
        return {
            'Stuxnet': {
                'urls': [
                    'www.todaysfutbol.com',
                    'www.mypremierfutbol.com',
                    'www.bestsoccerdeal.com',
                ],
                'user_agents': [
                    'Mozilla/4.0 (compatible; MSIE 7.0; Windows NT 5.1)',
                ],
                'ports': [41785, 41786],
                'beacon_interval': 20,  # 秒
            },
            'BlackEnergy': {
                'urls': [
                    '/alarman/archive.php',
                    'geopoliticswatch.com',
                ],
                'user_agents': [
                    'Opera/9.80 (Windows NT 6.1; U; Edition Campaign 21)',
                ],
            },
            'Havex': {
                'opc_scan': True,
                'dcom_ports': [135, 139, 445],
            },
            'Industroyer': {
                'iec104_commands': [0x2D, 0x2E],  # 双点控制命令
                'iec61850_mms': True,
                'modbus_func': [0x05, 0x0F],  # 写线圈
                'wiper_patterns': [b'\x00' * 512],
            },
            'Triton': {
                'tristation_port': 51728,
                'tristation_magic': b'\x5A',
            },
        }
    
    def detect_packet(self, packet_data, src_ip, dst_ip, src_port, dst_port, protocol):
        """分析单个数据包"""
        alerts = []
        
        # 1. 签名匹配
        for malware_name, sig in self.signatures.items():
            if malware_name == 'Stuxnet':
                # 检查C2 URL
                if protocol == 'HTTP':
                    for url in sig['urls']:
                        if url.encode() in packet_data:
                            alerts.append({
                                'malware': malware_name,
                                'type': 'C2通信',
                                'severity': 'critical',
                                'detail': f'连接到Stuxnet C2: {url}',
                                'timestamp': datetime.now()
                            })
                
                # 检查特殊端口
                if dst_port in sig.get('ports', []):
                    alerts.append({
                        'malware': malware_name,
                        'type': 'C2端口',
                        'severity': 'high',
                        'detail': f'连接到Stuxnet特殊端口 {dst_port}',
                        'timestamp': datetime.now()
                    })
            
            elif malware_name == 'Havex':
                # 检测OPC扫描
                if dst_port == 135 and protocol == 'TCP':
                    # DCOM连接 - 可能是OPC枚举
                    if self._check_opc_pattern(packet_data):
                        self.opc_scan_count += 1
                        if self.opc_scan_count > 50:
                            alerts.append({
                                'malware': 'Havex',
                                'type': 'OPC扫描',
                                'severity': 'high',
                                'detail': f'检测到大量OPC连接 ({self.opc_scan_count}次)',
                                'timestamp': datetime.now()
                            })
            
            elif malware_name == 'Industroyer':
                # 检测IEC 104攻击命令
                if dst_port == 2404 and protocol == 'TCP':
                    if self._check_iec104_attack(packet_data):
                        alerts.append({
                            'malware': 'Industroyer',
                            'type': 'IEC104攻击',
                            'severity': 'critical',
                            'detail': '检测到IEC 104控制命令',
                            'timestamp': datetime.now()
                        })
                
                # 检测Modbus写操作
                if dst_port == 502 and protocol == 'TCP':
                    if self._check_modbus_write(packet_data):
                        self.plc_write_count += 1
                        if self.plc_write_count > 10:
                            alerts.append({
                                'malware': 'Industroyer',
                                'type': 'Modbus写入',
                                'severity': 'high',
                                'detail': f'检测到大量Modbus写操作 ({self.plc_write_count}次)',
                                'timestamp': datetime.now()
                            })
            
            elif malware_name == 'Triton':
                # 检测TriStation协议
                if dst_port == 51728 or src_port == 51728:
                    if packet_data[:1] == b'\x5A':
                        self.sis_access_count += 1
                        alerts.append({
                            'malware': 'Triton',
                            'type': 'SIS访问',
                            'severity': 'critical',
                            'detail': f'检测到TriStation协议访问SIS系统',
                            'timestamp': datetime.now()
                        })
        
        # 2. 行为分析
        # 检测异常连接模式
        key = (src_ip, dst_ip)
        self.connection_patterns[key].append(datetime.now())
        
        # 检测信标行为
        recent = [t for t in self.connection_patterns[key]
                  if (datetime.now() - t) < timedelta(minutes=10)]
        if len(recent) > 30:
            # 检查是否定期
            intervals = [(recent[i] - recent[i-1]).total_seconds()
                        for i in range(1, len(recent))]
            if intervals and max(intervals) - min(intervals) < 5:
                avg_interval = sum(intervals) / len(intervals)
                alerts.append({
                    'malware': 'Unknown',
                    'type': '信标行为',
                    'severity': 'medium',
                    'detail': f'检测到定期连接 (间隔{avg_interval:.1f}秒)',
                    'timestamp': datetime.now()
                })
        
        self.alerts.extend(alerts)
        return alerts
    
    def _check_opc_pattern(self, data):
        """检查是否为OPC相关流量"""
        opc_guids = [
            b'\x02\x63\x0a\xef\xb0\x94\x3b\x41\xa7\xd3\x11\x0e\x8c\x01\x00\x00',
            b'\x31\x63\x0a\xef',
        ]
        for guid in opc_guids:
            if guid in data:
                return True
        return False
    
    def _check_iec104_attack(self, data):
        """检查IEC 104攻击命令"""
        if len(data) < 10:
            return False
        # 检查是否包含控制命令ASDU
        # IEC 104帧以0x68开头
        if data[0] == 0x68:
            if len(data) > 8:
                # 检查ASDU类型
                asdu_type = data[9] if len(data) > 9 else 0
                # 0x2D = C_DC_NA (双点控制) - Industroyer使用的命令
                if asdu_type in [0x2D, 0x2E, 0x2F]:
                    return True
        return False
    
    def _check_modbus_write(self, data):
        """检查Modbus写操作"""
        if len(data) < 8:
            return False
        func_code = data[7] if len(data) > 7 else 0
        # 写线圈(0x05), 写多个线圈(0x0F), 写寄存器(0x06), 写多个寄存器(0x10)
        return func_code in [0x05, 0x06, 0x0F, 0x10]
    
    def generate_report(self):
        """生成检测报告"""
        print(f"\n{'='*60}")
        print("工控恶意代码检测报告")
        print(f"{'='*60}")
        
        if not self.alerts:
            print("未检测到恶意代码活动")
            return
        
        # 按恶意代码分组
        malware_groups = defaultdict(list)
        for alert in self.alerts:
            malware_groups[alert['malware']].append(alert)
        
        for malware, alerts in malware_groups.items():
            print(f"\n[{malware}] 检测到 {len(alerts)} 个告警:")
            for a in alerts:
                print(f"  - [{a['severity']}] {a['type']}: {a['detail']}")
                print(f"    时间: {a['timestamp']}")
        
        print(f"\n{'='*60}")

# 使用示例
if __name__ == '__main__':
    detector = ICSMalwareDetector()
    
    # 模拟检测
    # 模拟Stuxnet C2通信
    detector.detect_packet(
        b'GET /index.html HTTP/1.1\r\nHost: www.todaysfutbol.com\r\n',
        '192.168.1.100', '203.0.113.1', 12345, 80, 'HTTP'
    )
    
    # 模拟Industroyer IEC 104攻击
    attack_packet = b'\x68\x04\x01\x00\x02\x00\x2D\x01\x01\x06\x01\x00'
    detector.detect_packet(
        attack_packet,
        '192.168.1.100', '192.168.1.50', 12345, 2404, 'TCP'
    )
    
    detector.generate_report()
```

### 9.2 主机行为检测

```python
#!/usr/bin/env python3
"""工控恶意代码主机行为检测"""
import psutil
import os
import hashlib
from datetime import datetime

class ICSHostDetector:
    """检测工控工程站/HMI上的恶意代码行为"""
    
    def __init__(self):
        self.suspicious_processes = [
            'ccalc.exe',      # Stuxnet伪装的计算器
            'mrxcls.sys',     # Stuxnet内核驱动
            'mrxnet.sys',     # Stuxnet内核驱动
            's7otbxdx.dll',   # Stuxnet DLL劫持
            'lsass.exe.*',    # 凭据窃取
        ]
        
        self.suspicious_paths = [
            'C:\\Windows\\Temp\\',
            'C:\\ProgramData\\',
            'C:\\Users\\*\\AppData\\Local\\Temp\\',
        ]
        
        self.suspicious_network_connections = [
            ('www.todaysfutbol.com', 80),
            ('www.mypremierfutbol.com', 80),
            ('www.bestsoccerdeal.com', 80),
        ]
    
    def scan_processes(self):
        """扫描可疑进程"""
        findings = []
        
        for proc in psutil.process_iter(['pid', 'name', 'exe', 'cmdline', 'create_time']):
            try:
                proc_name = proc.info['name'].lower()
                
                # 检查可疑进程名
                for susp in self.suspicious_processes:
                    if susp.replace('*', '') in proc_name:
                        findings.append({
                            'type': '可疑进程',
                            'pid': proc.info['pid'],
                            'name': proc.info['name'],
                            'exe': proc.info.get('exe', ''),
                            'reason': f'匹配规则: {susp}',
                            'severity': 'high'
                        })
                
                # 检查伪装进程
                if proc_name == 'ccalc.exe':
                    # Stuxnet伪装为Windows计算器
                    findings.append({
                        'type': '伪装进程',
                        'pid': proc.info['pid'],
                        'name': proc.info['name'],
                        'reason': '可能是Stuxnet伪装的计算器',
                        'severity': 'critical'
                    })
                
                # 检查路径异常
                exe_path = proc.info.get('exe', '')
                if exe_path:
                    for susp_path in self.suspicious_paths:
                        if susp_path.replace('*', '') in exe_path:
                            findings.append({
                                'type': '可疑路径',
                                'pid': proc.info['pid'],
                                'name': proc.info['name'],
                                'exe': exe_path,
                                'reason': f'从可疑路径运行: {susp_path}',
                                'severity': 'medium'
                            })
                
            except (psutil.NoSuchProcess, psutil.AccessDenied):
                pass
        
        return findings
    
    def scan_network_connections(self):
        """扫描网络连接"""
        findings = []
        
        for conn in psutil.net_connections(kind='inet'):
            if conn.status == 'ESTABLISHED':
                # 检查到已知C2的连接
                for c2_host, c2_port in self.suspicious_network_connections:
                    if (conn.laddr and c2_port in conn.laddr or
                        conn.raddr and c2_port in conn.raddr):
                        findings.append({
                            'type': 'C2连接',
                            'local': f"{conn.laddr.ip}:{conn.laddr.port}" if conn.laddr else '',
                            'remote': f"{conn.raddr.ip}:{conn.raddr.port}" if conn.raddr else '',
                            'pid': conn.pid,
                            'reason': f'连接到已知C2: {c2_host}',
                            'severity': 'critical'
                        })
                
                # 检查非标准端口
                if conn.raddr:
                    rport = conn.raddr.port
                    standard_ports = [80, 443, 102, 502, 3389, 22]
                    if rport not in standard_ports and rport > 49152:
                        findings.append({
                            'type': '非标准端口',
                            'local': f"{conn.laddr.ip}:{conn.laddr.port}" if conn.laddr else '',
                            'remote': f"{conn.raddr.ip}:{conn.raddr.port}",
                            'pid': conn.pid,
                            'reason': f'连接到非标准高端口 {rport}',
                            'severity': 'low'
                        })
        
        return findings
    
    def scan_dll_injection(self):
        """扫描DLL注入"""
        findings = []
        
        # 检查Step 7是否被注入
        target_processes = ['s7tgtopx.exe', 'STEP7.exe', 'pcs7.exe']
        
        for proc in psutil.process_iter(['pid', 'name']):
            try:
                proc_name = proc.info['name'].lower()
                if any(t in proc_name for t in target_processes):
                    # 获取加载的DLL
                    p = psutil.Process(proc.info['pid'])
                    
                    # 检查是否有可疑DLL
                    suspicious_dlls = [
                        's7otbxdx.dll',   # Stuxnet DLL劫持
                        's7hklspx.dll',   # Stuxnet
                        'api_log.dll',    # 通用API hook
                    ]
                    
                    for dll_name in suspicious_dlls:
                        # 通过内存映射检查DLL
                        try:
                            for mapping in p.memory_maps():
                                if dll_name.lower() in mapping.path.lower():
                                    findings.append({
                                        'type': 'DLL注入',
                                        'pid': proc.info['pid'],
                                        'process': proc.info['name'],
                                        'dll': mapping.path,
                                        'reason': f'检测到可疑DLL: {dll_name}',
                                        'severity': 'critical'
                                    })
                        except:
                            pass
            except:
                pass
        
        return findings
    
    def scan_filesystem(self):
        """扫描文件系统"""
        findings = []
        
        # 检查关键目录
        scan_dirs = [
            'C:\\Windows\\System32\\drivers',
            'C:\\Windows\\SysWOW64',
            'C:\\Program Files\\Siemens\\',
            'C:\\Program Files (x86)\\Siemens\\',
        ]
        
        suspicious_files = {
            'mrxcls.sys': 'Stuxnet内核驱动',
            'mrxnet.sys': 'Stuxnet内核驱动',
            's7otbxdx.dll': 'Stuxnet DLL劫持',
            '~WTR4141.tmp': 'Stuxnet临时文件',
            '~WTR4142.tmp': 'Stuxnet临时文件',
        }
        
        for scan_dir in scan_dirs:
            if not os.path.exists(scan_dir):
                continue
            
            for root, dirs, files in os.walk(scan_dir):
                for filename in files:
                    if filename.lower() in [f.lower() for f in suspicious_files]:
                        filepath = os.path.join(root, filename)
                        
                        # 计算文件哈希
                        try:
                            with open(filepath, 'rb') as f:
                                md5 = hashlib.md5(f.read()).hexdigest()
                            
                            findings.append({
                                'type': '恶意文件',
                                'path': filepath,
                                'md5': md5,
                                'reason': suspicious_files.get(filename, '未知'),
                                'severity': 'critical'
                            })
                        except:
                            pass
        
        return findings
    
    def run_full_scan(self):
        """运行完整扫描"""
        print(f"{'='*60}")
        print("工控恶意代码主机检测扫描")
        print(f"扫描时间: {datetime.now()}")
        print(f"{'='*60}")
        
        all_findings = []
        
        print("\n[1] 扫描进程...")
        findings = self.scan_processes()
        all_findings.extend(findings)
        print(f"    发现 {len(findings)} 个可疑项")
        
        print("[2] 扫描网络连接...")
        findings = self.scan_network_connections()
        all_findings.extend(findings)
        print(f"    发现 {len(findings)} 个可疑项")
        
        print("[3] 扫描DLL注入...")
        findings = self.scan_dll_injection()
        all_findings.extend(findings)
        print(f"    发现 {len(findings)} 个可疑项")
        
        print("[4] 扫描文件系统...")
        findings = self.scan_filesystem()
        all_findings.extend(findings)
        print(f"    发现 {len(findings)} 个可疑项")
        
        print(f"\n{'='*60}")
        print("扫描结果:")
        
        if not all_findings:
            print("未发现可疑活动")
        else:
            # 按严重程度排序
            severity_order = {'critical': 0, 'high': 1, 'medium': 2, 'low': 3}
            all_findings.sort(key=lambda x: severity_order.get(x.get('severity', 'low'), 4))
            
            for f in all_findings:
                print(f"\n  [{f.get('severity', 'unknown').upper()}] {f['type']}")
                for key, val in f.items():
                    if key not in ['type', 'severity']:
                        print(f"    {key}: {val}")
        
        print(f"\n{'='*60}")

if __name__ == '__main__':
    detector = ICSHostDetector()
    detector.run_full_scan()
```
