# 示例ST程序:
(* PLC程序 - 水箱液位控制 *)
PROGRAM WaterTankControl
  VAR
    LevelSensor : INT;     // 液位传感器
    PumpOutput : BOOL;      // 水泵控制
    AlarmHigh : BOOL;       // 高液位报警
    AlarmLow : BOOL;        // 低液位报警
    SetPoint : INT := 500;  // 目标液位
    DeadBand : INT := 50;   // 死区
  END_VAR

  // 控制逻辑
  IF LevelSensor < (SetPoint - DeadBand) THEN
    PumpOutput := TRUE;
  ELSIF LevelSensor > (SetPoint + DeadBand) THEN
    PumpOutput := FALSE;
  END_IF;

  // 报警逻辑
  AlarmHigh := (LevelSensor > 900);
  AlarmLow := (LevelSensor < 100);
END_PROGRAM

### 7.2 Modbus测试环境

```python
#!/usr/bin/env python3
"""Modbus从站模拟器 - 用于模糊测试目标"""
from pyModbusTCP.server import ModbusServer
import logging
import sys
import random

logging.basicConfig(level=logging.INFO, format='%(asctime)s [%(levelname)s] %(message)s')
logger = logging.getLogger(__name__)

class ModbusSlaveSimulator:
    """模拟Modbus从站设备"""
    
    def __init__(self, host='0.0.0.0', port=502):
        self.host = host
        self.port = port
        
        # 初始化数据区
        self.coils = [False] * 10000       # 线圈
        self.discrete_inputs = [False] * 10000  # 离散输入
        self.holding_registers = [0] * 10000   # 保持寄存器
        self.input_registers = [0] * 10000      # 输入寄存器
        
        # 设置一些初始值
        for i in range(100):
            self.coils[i] = random.choice([True, False])
            self.holding_registers[i] = random.randint(0, 65535)
            self.input_registers[i] = random.randint(0, 65535)
        
        logger.info(f"Modbus从站模拟器初始化完成 - {host}:{port}")
        logger.info(f"数据区大小: 线圈={len(self.coils)}, "
                    f"保持寄存器={len(self.holding_registers)}")
    
    def run(self):
        """启动Modbus服务器"""
        try:
            server = ModbusServer(
                host=self.host,
                port=self.port,
                no_block=True
            )
            
            server.start()
            logger.info("Modbus从站启动成功")
            logger.info("等待连接...")
            
            import time
            while True:
                time.sleep(1)
                
        except KeyboardInterrupt:
            logger.info("停止服务器")
            server.stop()
        except Exception as e:
            logger.error(f"启动失败: {e}")
            sys.exit(1)

if __name__ == '__main__':
    import argparse
    parser = argparse.ArgumentParser(description='Modbus从站模拟器')
    parser.add_argument('--host', default='0.0.0.0', help='监听地址')
    parser.add_argument('--port', type=int, default=502, help='监听端口')
    args = parser.parse_args()
    
    simulator = ModbusSlaveSimulator(args.host, args.port)
    simulator.run()
```

### 7.4 Wireshark自定义协议解析器

```lua
-- Wireshark Modbus协议解析器增强版
-- 文件: modbus_enhanced.lua

-- 协议定义
local modbus_proto = Proto("ModbusEnhanced", "Modbus TCP Enhanced")

-- 字段定义
local f_transaction_id = ProtoField.uint16("modbus.transaction_id", "Transaction ID", base.DEC)
local f_protocol_id = ProtoField.uint16("modbus.protocol_id", "Protocol ID", base.DEC)
local f_length = ProtoField.uint16("modbus.length", "Length", base.DEC)
local f_unit_id = ProtoField.uint8("modbus.unit_id", "Unit ID", base.DEC)
local f_func_code = ProtoField.uint8("modbus.func_code", "Function Code", base.HEX)
local f_exception_code = ProtoField.uint8("modbus.exception_code", "Exception Code", base.HEX)
local f_data = ProtoField.bytes("modbus.data", "Data")

modbus_proto.fields = {
    f_transaction_id, f_protocol_id, f_length,
    f_unit_id, f_func_code, f_exception_code, f_data
}

-- 功能码名称映射
local func_names = {
    [0x01] = "Read Coils",
    [0x02] = "Read Discrete Inputs",
    [0x03] = "Read Holding Registers",
    [0x04] = "Read Input Registers",
    [0x05] = "Write Single Coil",
    [0x06] = "Write Single Register",
    [0x07] = "Read Exception Status",
    [0x08] = "Diagnostic",
    [0x0F] = "Write Multiple Coils",
    [0x10] = "Write Multiple Registers",
    [0x11] = "Report Slave ID",
    [0x14] = "Read File Record",
    [0x15] = "Write File Record",
    [0x16] = "Mask Write Register",
    [0x17] = "Read Write Multiple Registers",
    [0x18] = "Read FIFO Queue",
    [0x2B] = "Encapsulated Interface Transport",
}

-- 异常码名称映射
local exception_names = {
    [0x01] = "Illegal Function",
    [0x02] = "Illegal Data Address",
    [0x03] = "Illegal Data Value",
    [0x04] = "Slave Device Failure",
    [0x05] = "Acknowledge",
    [0x06] = "Slave Device Busy",
    [0x08] = "Memory Parity Error",
    [0x0A] = "Gateway Path Unavailable",
    [0x0B] = "Gateway Target No Response",
}

-- 解析器
function modbus_proto.dissector(buffer, pinfo, tree)
    local min_len = 8  -- MBAP头最小长度
    if buffer:len() < min_len then return 0 end
    
    pinfo.cols.protocol = "Modbus Enhanced"
    
    local subtree = tree:add(modbus_proto, buffer(), "Modbus TCP")
    
    -- MBAP头
    local trans_id = buffer(0, 2):uint()
    local proto_id = buffer(2, 2):uint()
    local length = buffer(4, 2):uint()
    local unit_id = buffer(6, 1):uint()
    
    subtree:add(f_transaction_id, buffer(0, 2))
    subtree:add(f_protocol_id, buffer(2, 2))
    subtree:add(f_length, buffer(4, 2))
    subtree:add(f_unit_id, buffer(6, 1))
    
    -- PDU
    if buffer:len() > 7 then
        local func_code = buffer(7, 1):uint()
        
        local func_item = subtree:add(f_func_code, buffer(7, 1))
        local func_name = func_names[func_code & 0x7F] or "Unknown"
        func_item:append_text(" (" .. func_name .. ")")
        
        -- 检查是否为异常响应
        if func_code & 0x80 ~= 0 then
            pinfo.cols.info = "Modbus Exception Response: " .. func_name
            if buffer:len() > 8 then
                local exc_code = buffer(8, 1):uint()
                local exc_item = subtree:add(f_exception_code, buffer(8, 1))
                local exc_name = exception_names[exc_code] or "Unknown"
                exc_item:append_text(" (" .. exc_name .. ")")
            end
        else
            pinfo.cols.info = "Modbus " .. func_name
        end
        
        -- 数据区
        if buffer:len() > 8 then
            subtree:add(f_data, buffer(8))
        end
        
        -- 异常检测
        if length > 255 then
            pinfo.cols.info:append(" [WARNING: Abnormal Length]")
        end
        if proto_id ~= 0 then
            pinfo.cols.info:append(" [WARNING: Non-standard Protocol ID]")
        end
    end
    
    return buffer:len()
end

-- 注册解析器
register_tcp_table_dissector(tcp.port_table, 502, modbus_proto)
```
