# 协议模糊测试

### 3.1 模糊测试概述

模糊测试（Fuzzing）是一种自动化漏洞挖掘技术，通过向目标系统发送大量非预期或畸形输入数据来触发异常行为。在工控领域，协议模糊测试是发现通信协议实现漏洞的重要手段。

```
工控协议模糊测试分类：
┌─────────────────────────────────────────────────────────┐
│                    模糊测试方法                           │
├──────────────────┬──────────────────┬───────────────────┤
│   基于变异        │   基于生成        │   混合方法         │
│  (Mutation)      │  (Generation)    │  (Hybrid)         │
├──────────────────┼──────────────────┼───────────────────┤
│ 对正常报文进行    │ 根据协议规范      │ 结合变异与生成     │
│ 随机修改        │ 从零构造报文      │ 使用协议模型       │
├──────────────────┼──────────────────┼───────────────────┤
│ 优点:快速上手     │ 优点:高覆盖率    │ 优点:平衡效率     │
│ 缺点:覆盖率低    │ 缺点:需协议规范   │ 缺点:实现复杂     │
└──────────────────┴──────────────────┴───────────────────┘
```

### 3.2 Modbus协议模糊测试

#### 3.2.1 Modbus协议结构

```
Modbus TCP报文结构:
┌─────────────────────────────────────────────────────┐
│ TPKT头 (4 bytes)                                     │
├──────────┬──────────┬──────────────────────────────┤
│ 事务ID    │ 协议ID   │ 长度       │ 单元ID          │
│ (2 bytes)│(2 bytes) │ (2 bytes)  │ (1 byte)        │
├──────────┴──────────┴────────────┴──────────────────┤
│ PDU (功能码 + 数据)                                   │
├──────────┬─────────────────────────────────────────┤
│ 功能码    │ 数据区                                   │
│ (1 byte)│ (变长)                                   │
└──────────┴─────────────────────────────────────────┘

常见功能码:
  0x01 - 读线圈
  0x02 - 读离散输入
  0x03 - 读保持寄存器
  0x04 - 读输入寄存器
  0x05 - 写单个线圈
  0x06 - 写单个寄存器
  0x0F - 写多个线圈
  0x10 - 写多个寄存器
  0x17 - 读写多个寄存器
```

#### 3.2.2 Modbus模糊测试实现

```python
#!/usr/bin/env python3
"""Modbus协议模糊测试工具"""
import socket
import struct
import random
import time
import logging
from datetime import datetime
from enum import IntEnum

# 日志配置
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s [%(levelname)s] %(message)s',
    handlers=[
        logging.FileHandler('modbus_fuzz.log'),
        logging.StreamHandler()
    ]
)
logger = logging.getLogger(__name__)

class ModbusFunctionCode(IntEnum):
    READ_COILS = 0x01
    READ_DISCRETE_INPUTS = 0x02
    READ_HOLDING_REGISTERS = 0x03
    READ_INPUT_REGISTERS = 0x04
    WRITE_SINGLE_COIL = 0x05
    WRITE_SINGLE_REGISTER = 0x06
    WRITE_MULTIPLE_COILS = 0x0F
    WRITE_MULTIPLE_REGISTERS = 0x10
    READ_WRITE_MULTIPLE = 0x17
    MASK_WRITE_REGISTER = 0x16

class ModbusFuzzer:
    def __init__(self, target_ip, target_port=502, unit_id=1):
        self.target_ip = target_ip
        self.target_port = target_port
        self.unit_id = unit_id
        self.transaction_id = 0
        self.crash_count = 0
        self.success_count = 0
        self.error_count = 0
        self.anomalies = []
    
    def build_modbus_request(self, func_code, data=b''):
        """构建Modbus TCP请求"""
        self.transaction_id += 1
        protocol_id = 0  # Modbus协议
        length = 2 + len(data)  # 单元ID + 功能码 + 数据
        
        # TPKT + MBAP头
        header = struct.pack('>HHHBB',
            self.transaction_id,
            protocol_id,
            length,
            self.unit_id,
            func_code
        )
        
        return header + data
    
    def send_request(self, packet, timeout=3):
        """发送请求并检测响应"""
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(timeout)
            sock.connect((self.target_ip, self.target_port))
            sock.send(packet)
            
            response = sock.recv(4096)
            sock.close()
            
            # 分析响应
            if len(response) < 8:
                self.anomalies.append({
                    'type': '短响应',
                    'packet': packet.hex(),
                    'response': response.hex(),
                    'timestamp': datetime.now().isoformat()
                })
                logger.warning(f"异常短响应: {len(response)} bytes")
                return None
            
            # 检查异常响应
            resp_func = response[7]
            if resp_func & 0x80:
                # 异常响应
                exception_code = response[8] if len(response) > 8 else -1
                return ('exception', exception_code)
            else:
                return ('normal', response)
                
        except socket.timeout:
            self.crash_count += 1
            logger.error(f"超时 - 目标可能崩溃! Packet: {packet.hex()}")
            self.anomalies.append({
                'type': '超时/崩溃',
                'packet': packet.hex(),
                'timestamp': datetime.now().isoformat()
            })
            return ('timeout', None)
        except ConnectionRefusedError:
            self.error_count += 1
            logger.error("连接被拒绝")
            return ('refused', None)
        except Exception as e:
            self.error_count += 1
            logger.error(f"异常: {e}")
            return ('error', str(e))
    
    def fuzz_function_code(self):
        """模糊测试功能码字段"""
        logger.info("=== 功能码模糊测试 ===")
        
        # 1. 测试所有可能的功能码 (0x00 - 0xFF)
        for func_code in range(256):
            data = struct.pack('>HH', 0, 10)  # 读10个寄存器
            packet = self.build_modbus_request(func_code, data)
            result = self.send_request(packet)
            
            if result and result[0] == 'timeout':
                logger.critical(f"功能码 0x{func_code:02X} 导致超时!")
                self.crash_count += 1
            elif result and result[0] == 'normal':
                self.success_count += 1
                logger.info(f"功能码 0x{func_code:02X} 正常响应")
            
            time.sleep(0.1)  # 避免过载
    
    def fuzz_unit_id(self):
        """模糊测试单元ID字段"""
        logger.info("=== 单元ID模糊测试 ===")
        
        for unit_id in range(256):
            original_unit = self.unit_id
            self.unit_id = unit_id
            
            packet = self.build_modbus_request(
                ModbusFunctionCode.READ_HOLDING_REGISTERS,
                struct.pack('>HH', 0, 1)
            )
            result = self.send_request(packet)
            
            if result and result[0] == 'normal':
                logger.info(f"单元ID {unit_id} 有效")
            elif result and result[0] == 'exception':
                logger.info(f"单元ID {unit_id} 返回异常码 {result[1]}")
            
            self.unit_id = original_unit
            time.sleep(0.05)
    
    def fuzz_data_field(self, func_code=0x03):
        """模糊测试数据字段"""
        logger.info("=== 数据字段模糊测试 ===")
        
        # 1. 超长数据
        for length in [10, 100, 500, 1000, 5000, 10000]:
            data = bytes(random.randint(0, 255) for _ in range(length))
            packet = self.build_modbus_request(func_code, data)
            result = self.send_request(packet)
            
            if result and result[0] == 'timeout':
                logger.critical(f"超长数据({length} bytes)导致超时!")
            
            time.sleep(0.1)
        
        # 2. 特殊值
        special_values = [
            b'\x00' * 100,       # 全零
            b'\xff' * 100,       # 全FF
            b'\x00\x01\x02\x03' * 25,  # 递增模式
            b'\xfe\xed\xfa\xce' * 25,  # 魔数模式
            b'A' * 1000,         # 字符填充
            b'\x00' * 0xFFFF,    # 最大长度
        ]
        
        for i, data in enumerate(special_values):
            packet = self.build_modbus_request(func_code, data)
            result = self.send_request(packet)
            
            if result and result[0] == 'timeout':
                logger.critical(f"特殊值 #{i} 导致超时!")
                self.crash_count += 1
            
            time.sleep(0.1)
    
    def fuzz_register_address(self):
        """模糊测试寄存器地址"""
        logger.info("=== 寄存器地址模糊测试 ===")
        
        # 测试边界值
        addresses = [0, 1, 0xFFFF, 0x7FFF, 0x8000, 0xFFFE, 0xFFFF]
        
        for addr in addresses:
            # 读操作
            data = struct.pack('>HH', addr, 1)
            packet = self.build_modbus_request(0x03, data)
            result = self.send_request(packet)
            
            if result and result[0] == 'normal':
                logger.info(f"寄存器地址 0x{addr:04X} 可读")
            elif result and result[0] == 'exception':
                logger.info(f"寄存器地址 0x{addr:04X} 异常码: {result[1]}")
            
            # 写操作
            data = struct.pack('>HH', addr, 0x1234)
            packet = self.build_modbus_request(0x06, data)
            result = self.send_request(packet)
            
            if result and result[0] == 'normal':
                logger.warning(f"寄存器地址 0x{addr:04X} 可写!")
            
            time.sleep(0.05)
    
    def fuzz_protocol_variants(self):
        """协议变体模糊测试"""
        logger.info("=== 协议变体模糊测试 ===")
        
        # 1. 修改协议ID
        for proto_id in [1, 2, 0xFFFF]:
            self.transaction_id += 1
            packet = struct.pack('>HHHB', 
                self.transaction_id,
                proto_id,  # 修改协议ID
                6,
                self.unit_id
            ) + bytes([0x03, 0x00, 0x00, 0x00, 0x01])
            
            result = self.send_request(packet)
            if result and result[0] == 'normal':
                logger.warning(f"非标准协议ID {proto_id} 被接受!")
        
        # 2. 长度字段不匹配
        for length_val in [0, 1, 255, 0xFFFF]:
            self.transaction_id += 1
            packet = struct.pack('>HHHB',
                self.transaction_id,
                0,
                length_val,  # 修改长度字段
                self.unit_id
            ) + bytes([0x03, 0x00, 0x00, 0x00, 0x01])
            
            result = self.send_request(packet)
            if result and result[0] == 'timeout':
                logger.critical(f"长度字段 {length_val} 导致崩溃!")
        
        # 3. 分片和重组测试
        # 发送半个包，看目标如何处理
        normal_packet = self.build_modbus_request(0x03, struct.pack('>HH', 0, 1))
        
        # 发送半个包
        half_packet = normal_packet[:len(normal_packet)//2]
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(3)
            sock.connect((self.target_ip, self.target_port))
            sock.send(half_packet)
            time.sleep(1)
            try:
                response = sock.recv(4096)
                if response:
                    logger.info(f"半包测试有响应: {response.hex()}")
            except:
                logger.info("半包测试无响应 (可能等待更多数据)")
            sock.close()
        except Exception as e:
            logger.error(f"半包测试异常: {e}")
    
    def run_full_fuzz(self):
        """运行完整模糊测试套件"""
        logger.info("=" * 60)
        logger.info(f"Modbus模糊测试开始 - 目标: {self.target_ip}:{self.target_port}")
        logger.info("=" * 60)
        
        start_time = time.time()
        
        try:
            self.fuzz_function_code()
            self.fuzz_unit_id()
            self.fuzz_register_address()
            self.fuzz_data_field()
            self.fuzz_protocol_variants()
        except KeyboardInterrupt:
            logger.warning("用户中断测试")
        
        elapsed = time.time() - start_time
        
        logger.info("=" * 60)
        logger.info("测试完成 - 统计信息:")
        logger.info(f"  总用时: {elapsed:.1f}秒")
        logger.info(f"  正常响应: {self.success_count}")
        logger.info(f"  异常响应: {self.error_count}")
        logger.info(f"  崩溃/超时: {self.crash_count}")
        logger.info(f"  异常事件: {len(self.anomalies)}")
        logger.info("=" * 60)
        
        # 保存异常报告
        if self.anomalies:
            with open('modbus_anomalies.txt', 'w') as f:
                for a in self.anomalies:
                    f.write(f"类型: {a['type']}\n")
                    f.write(f"时间: {a['timestamp']}\n")
                    if 'packet' in a:
                        f.write(f"数据包: {a['packet']}\n")
                    if 'response' in a:
                        f.write(f"响应: {a['response']}\n")
                    f.write("-" * 40 + "\n")
            logger.info("异常报告已保存: modbus_anomalies.txt")

if __name__ == '__main__':
    import argparse
    parser = argparse.ArgumentParser(description='Modbus协议模糊测试工具')
    parser.add_argument('target', help='目标IP地址')
    parser.add_argument('-p', '--port', type=int, default=502, help='目标端口')
    parser.add_argument('-u', '--unit', type=int, default=1, help='Modbus单元ID')
    args = parser.parse_args()
    
    fuzzer = ModbusFuzzer(args.target, args.port, args.unit)
    fuzzer.run_full_fuzz()
```

### 3.3 S7协议模糊测试

```python
#!/usr/bin/env python3
"""S7协议模糊测试工具"""
import socket
import struct
import random
import time
import logging

logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s [%(levelname)s] %(message)s',
    handlers=[logging.StreamHandler(), logging.FileHandler('s7_fuzz.log')]
)
logger = logging.getLogger(__name__)

class S7Fuzzer:
    def __init__(self, target_ip, target_port=102):
        self.target_ip = target_ip
        self.target_port = target_port
        self.local_tsap = 0x0100
        self.remote_tsap = 0x0100
        self.connection_id = 0
    
    def build_tpkt(self, payload):
        """构建TPKT头"""
        length = 4 + len(payload)
        return struct.pack('>BBH', 3, 0, length)
    
    def build_cotp_connect(self):
        """构建COTP连接请求"""
        cotp = bytes([
            0x11,       # 长度
            0xE0,       # PDU类型: Connect Request
            0x00, 0x01, # 目标引用
            0x00, 0x01, # 源引用
            0x00,       # 类
            # 参数: TSAP
            0xC0,       # 参数代码: Called TSAP
            0x02,       # 参数长度
            (self.remote_tsap >> 8) & 0xFF,
            self.remote_tsap & 0xFF,
            0xC1,       # 参数代码: Calling TSAP
            0x02,
            (self.local_tsap >> 8) & 0xFF,
            self.local_tsap & 0xFF,
            # 参数: TPDU大小
            0xC2,       # 参数代码
            0x01,       # 参数长度
            0x0A,       # TPDU大小: 1024
        ])
        return cotp
    
    def build_s7_request(self, pdu_type, func_group, sub_func, data=b''):
        """构建S7协议请求"""
        # S7头
        s7_header = struct.pack('>BBHHHH',
            0x32,           # 协议ID
            pdu_type,       # PDU类型 (0x01=请求)
            0x0000,         # 保留
            self._next_id(), # PDU引用
            len(data) + 4,  # 参数长度
            len(data)       # 数据长度
        )
        
        # 参数区
        param = struct.pack('>BBB',
            func_group,     # 功能组
            sub_func,       # 子功能
            0x12 if func_group == 0x04 else 0x00  # 子功能类型
        )
        
        return s7_header + param[:4] + data
    
    def _next_id(self):
        self.connection_id = (self.connection_id + 1) & 0xFFFF
        return self.connection_id
    
    def connect(self):
        """建立S7连接"""
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(5)
            sock.connect((self.target_ip, self.target_port))
            
            # 发送COTP连接请求
            cotp = self.build_cotp_connect()
            tpkt = self.build_tpkt(cotp)
            sock.send(tpkt + cotp)
            
            response = sock.recv(1024)
            if len(response) >= 7 and response[5] == 0xD0:
                logger.info("COTP连接建立成功")
                
                # 发送S7通信设置请求
                setup = struct.pack('>BBHHHH',
                    0x32,       # 协议ID
                    0x01,       # 请求
                    0x0000,     # 保留
                    self._next_id(),
                    0x0008,     # 参数长度
                    0x0000      # 数据长度
                )
                setup_param = struct.pack('>BBHBBH',
                    0xF0,       # 功能: 通信设置
                    0x00,       # 子功能
                    0x0001,     # 保留
                    0x00, 0x01, # PDU大小
                    0x04E2      # PDU大小: 1250
                )
                
                tpkt = self.build_tpkt(setup + setup_param)
                sock.send(tpkt + setup + setup_param)
                
                response = sock.recv(1024)
                if len(response) > 0:
                    logger.info("S7通信设置完成")
                    return sock
            
            logger.error("COTP连接失败")
            return None
        except Exception as e:
            logger.error(f"连接失败: {e}")
            return None
    
    def fuzz_s7_header(self, sock):
        """模糊测试S7协议头"""
        logger.info("=== S7协议头模糊测试 ===")
        
        # 修改协议ID
        for proto_id in [0x00, 0x31, 0x33, 0xFF]:
            packet = struct.pack('>BBHHHH',
                proto_id,  # 修改协议ID
                0x01, 0x0000, self._next_id(),
                0x0004, 0x0000
            )
            self._send_and_check(sock, packet, f"协议ID=0x{proto_id:02X}")
        
        # 修改PDU类型
        for pdu_type in range(256):
            packet = struct.pack('>BBHHHH',
                0x32,
                pdu_type,  # 修改PDU类型
                0x0000, self._next_id(),
                0x0004, 0x0000
            )
            result = self._send_and_check(sock, packet, f"PDU类型=0x{pdu_type:02X}")
            if result == 'timeout':
                logger.critical(f"PDU类型 0x{pdu_type:02X} 导致超时!")
            
            time.sleep(0.05)
    
    def fuzz_read_area(self, sock):
        """模糊测试读区域功能"""
        logger.info("=== 读区域功能模糊测试 ===")
        
        # S7区域类型
        area_types = {
            'P': 0x81,  # 过程映像区
            'I': 0x82,  # 输入区
            'Q': 0x83,  # 输出区
            'M': 0x84,  # 标志位区
            'DB': 0x85, # 数据块
            'C': 0x86,  # 计数器
            'T': 0x87,  # 定时器
            'V': 0x88,  # 变量区
        }
        
        for area_name, area_code in area_types.items():
            # 构建读请求
            read_data = struct.pack('>BBBBHH',
                0x12,       # 变量规格头
                0x0A,       # 长度规格
                0x10,       # 语法ID
                area_code,  # 区域类型
                1,          # DB号
                0,          # 地址
            ) + struct.pack('>HH', 1, 0)  # 长度和偏移
            
            s7_pdu = struct.pack('>BBHHHH',
                0x32, 0x01, 0x0000,
                self._next_id(),
                len(read_data), 0
            ) + read_data
            
            result = self._send_and_check(sock, s7_pdu, f"读区域 {area_name}")
            
            if result == 'normal':
                logger.info(f"区域 {area_name} (0x{area_code:02X}) 可读")
            elif result == 'timeout':
                logger.critical(f"区域 {area_name} 导致超时!")
            
            time.sleep(0.1)
    
    def fuzz_db_read(self, sock):
        """模糊测试数据块读取"""
        logger.info("=== 数据块读取模糊测试 ===")
        
        # 测试不同的DB号
        for db_num in [0, 1, 2, 100, 255, 0xFFFF]:
            read_data = struct.pack('>BBBBHH',
                0x12, 0x0A, 0x10, 0x85,
                db_num,  # DB号
                0,       # 地址
            ) + struct.pack('>HH', 1, 0)
            
            s7_pdu = struct.pack('>BBHHHH',
                0x32, 0x01, 0x0000,
                self._next_id(),
                len(read_data), 0
            ) + read_data
            
            result = self._send_and_check(sock, s7_pdu, f"DB号={db_num}")
            
            if result == 'normal':
                logger.warning(f"DB {db_num} 存在且可读!")
            elif result == 'timeout':
                logger.critical(f"DB {db_num} 导致超时!")
            
            time.sleep(0.1)
    
    def fuzz_write_operations(self, sock):
        """模糊测试写操作"""
        logger.info("=== 写操作模糊测试 ===")
        
        # 尝试写标志位区
        for offset in [0, 1, 100, 1000, 0xFFFF]:
            write_data = struct.pack('>BBBBHH',
                0x12, 0x0A, 0x10, 0x84,
                0, offset
            ) + struct.pack('>HH', 1, 0)
            
            # 数据部分
            item_data = struct.pack('>BBH B', 0x00, 0x04, 1, 0xFF)
            
            s7_pdu = struct.pack('>BBHHHH',
                0x32, 0x05, 0x0000,  # 写请求
                self._next_id(),
                len(write_data),
                len(item_data)
            ) + write_data + item_data
            
            result = self._send_and_check(sock, s7_pdu, f"写M{offset}")
            
            if result == 'normal':
                logger.critical(f"成功写入标志位 M{offset}!")
            
            time.sleep(0.1)
    
    def fuzz_password_operations(self, sock):
        """模糊测试密码相关操作"""
        logger.info("=== 密码操作模糊测试 ===")
        
        # 尝试发送密码
        for password in ['', 'admin', 'root', '0000', 'AAAA', '\x00' * 8]:
            # S7密码验证请求
            pwd_data = struct.pack('>BB',
                0x26, 0x02  # 密码子功能
            ) + password.encode('latin-1').ljust(8, b'\x00')
            
            s7_pdu = struct.pack('>BBHHHH',
                0x32, 0x28, 0x0000,
                self._next_id(),
                len(pwd_data), 0
            ) + pwd_data
            
            result = self._send_and_check(sock, s7_pdu, f"密码='{password}'")
            
            if result == 'normal':
                logger.critical(f"密码 '{password}' 可能被接受!")
            
            time.sleep(0.1)
    
    def _send_and_check(self, sock, pdu, description):
        """发送数据并检查响应"""
        try:
            tpkt = self.build_tpkt(pdu)
            sock.send(tpkt + pdu)
            
            response = sock.recv(4096)
            
            if len(response) == 0:
                logger.warning(f"[{description}] 空响应")
                return 'empty'
            
            # 检查是否有异常
            if len(response) < 12:
                logger.warning(f"[{description}] 短响应: {response.hex()}")
                return 'short'
            
            return 'normal'
            
        except socket.timeout:
            logger.error(f"[{description}] 超时 - 可能崩溃!")
            return 'timeout'
        except Exception as e:
            logger.error(f"[{description}] 异常: {e}")
            return 'error'
    
    def run(self):
        """运行S7模糊测试"""
        logger.info("=" * 60)
        logger.info(f"S7模糊测试开始 - 目标: {self.target_ip}:{self.target_port}")
        logger.info("=" * 60)
        
        sock = self.connect()
        if not sock:
            logger.error("无法建立连接")
            return
        
        try:
            self.fuzz_s7_header(sock)
            self.fuzz_read_area(sock)
            self.fuzz_db_read(sock)
            self.fuzz_write_operations(sock)
            self.fuzz_password_operations(sock)
        except KeyboardInterrupt:
            logger.warning("用户中断")
        finally:
            sock.close()
            logger.info("测试完成")

if __name__ == '__main__':

    args = parser.parse_args()
```

### 3.4 DNP3协议模糊测试

```python
#!/usr/bin/env python3
"""DNP3协议模糊测试工具"""
import socket
import struct
import random
import time
import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s [%(levelname)s] %(message)s')
logger = logging.getLogger(__name__)

class DNP3Fuzzer:
    def __init__(self, target_ip, target_port=20000, source_addr=1, dest_addr=1):
        self.target_ip = target_ip
        self.target_port = target_port
        self.source_addr = source_addr
        self.dest_addr = dest_addr
    
    def compute_crc(self, data):
        """计算DNP3 CRC-16"""
        crc_table = [
            0x0000, 0x3656, 0x6CAC, 0x5EFA, 0xD594, 0xE3C2, 0xB938, 0x8F6E,
            0x0E2D, 0x387B, 0x6281, 0x54D7, 0xDFB9, 0xE9EF, 0xB315, 0x8543,
            # ... 完整的CRC表 (256项)
        ]
        # 简化版CRC计算
        crc = 0x0000
        for byte in data:
            crc = ((crc << 8) ^ crc_table[((crc >> 8) ^ byte) & 0xFF]) & 0xFFFF
        return crc
    
    def build_link_header(self, frame_type='request'):
        """构建DNP3链路层头"""
        # 起始字节
        start = bytes([0x05, 0x64])
        
        # 长度 (占位，后面填充)
        length = 5  # 最小长度
        
        # 控制域
        if frame_type == 'request':
            control = 0x44  # DIR=1, PRM=1, FCB=0, FCV=0, FC=4
        else:
            control = 0x40  # DIR=0, PRM=0
        
        # 目的地址
        dest = struct.pack('<H', self.dest_addr)
        # 源地址
        source = struct.pack('<H', self.source_addr)
        
        # CRC
        header = start + bytes([length]) + bytes([control]) + dest + source
        crc = self.compute_crc(header[:8])
        
        return header + struct.pack('<H', crc)
    
    def build_application_request(self, func_code, data=b''):
        """构建DNP3应用层请求"""
        # 应用层控制
        app_ctl = 0x01  # FIR=1, FIN=1, SEQ=1
        
        # 功能码
        func = bytes([func_code])
        
        return bytes([app_ctl]) + func + data
    
    def fuzz_function_codes(self, sock):
        """模糊测试DNP3功能码"""
        logger.info("=== DNP3功能码模糊测试 ===")
        
        # DNP3功能码定义
        func_codes = {
            0x00: '读',
            0x01: '写',
            0x02: '选择',
            0x03: '操作',
            0x04: '直接操作',
            0x07: '延迟操作',
            0x09: '冻结',
            0x0A: '冻结清除',
            0x0D: '响应数据',
            0x0E: '写响应',
            0x12: '存储配置',
            0x13: '响应配置',
        }
        
        for func_code in range(256):
            app_data = self.build_application_request(func_code)
            link_header = self.build_link_header()
            
            # 构建完整帧
            frame = link_header + app_data
            
            try:
                sock.send(frame)
                response = sock.recv(4096)
                
                if len(response) > 0:
                    logger.info(f"功能码 0x{func_code:02X} ({func_codes.get(func_code, '未知')}) 响应: {len(response)} bytes")
                else:
                    logger.warning(f"功能码 0x{func_code:02X} 空响应")
                
            except socket.timeout:
                logger.critical(f"功能码 0x{func_code:02X} 导致超时!")
            except Exception as e:
                logger.error(f"功能码 0x{func_code:02X} 异常: {e}")
            
            time.sleep(0.1)
    
    def fuzz_addresses(self, sock):
        """模糊测试地址字段"""
        logger.info("=== 地址字段模糊测试 ===")
        
        # 测试不同的源/目标地址组合
        address_pairs = [
            (0, 0),
            (1, 1),
            (0xFFFF, 0xFFFF),
            (0x10000, 0),  # 溢出
            (0xFFFE, 0xFFFF),
            (0x3FFF, 0x3FFF),  # 最大有效地址
        ]
        
        for src, dst in address_pairs:
            self.source_addr = src
            self.dest_addr = dst
            
            link_header = self.build_link_header()
            app_data = self.build_application_request(0x00)  # 读请求
            
            frame = link_header + app_data
            
            try:
                sock.send(frame)
                response = sock.recv(4096)
                logger.info(f"地址(src={src}, dst={dst}) 响应: {len(response)} bytes")
            except socket.timeout:
                logger.critical(f"地址(src={src}, dst={dst}) 导致超时!")
            except Exception as e:
                logger.error(f"地址(src={src}, dst={dst}) 异常: {e}")
            
            time.sleep(0.1)
    
    def fuzz_data_sizes(self, sock):
        """模糊测试数据大小"""
        logger.info("=== 数据大小模糊测试 ===")
        
        for size in [0, 1, 100, 500, 1000, 2000, 5000, 10000]:
            data = bytes(random.randint(0, 255) for _ in range(size))
            app_data = self.build_application_request(0x01, data)  # 写请求
            link_header = self.build_link_header()
            
            frame = link_header + app_data
            
            try:
                sock.send(frame)
                response = sock.recv(4096)
                logger.info(f"数据大小 {size} 响应: {len(response)} bytes")
            except socket.timeout:
                logger.critical(f"数据大小 {size} 导致超时!")
            except Exception as e:
                logger.error(f"数据大小 {size} 异常: {e}")
            
            time.sleep(0.1)
    
    def run(self):
        """运行DNP3模糊测试"""
        logger.info("=" * 60)
        logger.info(f"DNP3模糊测试开始 - 目标: {self.target_ip}:{self.target_port}")
        logger.info("=" * 60)
        
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(5)
            sock.connect((self.target_ip, self.target_port))
            
            self.fuzz_function_codes(sock)
            self.fuzz_addresses(sock)
            self.fuzz_data_sizes(sock)
            
            sock.close()
        except Exception as e:
            logger.error(f"连接失败: {e}")

if __name__ == '__main__':
    import argparse
    parser = argparse.ArgumentParser(description='DNP3协议模糊测试工具')
    parser.add_argument('target', help='目标IP地址')
    parser.add_argument('-p', '--port', type=int, default=20000, help='目标端口')
    args = parser.parse_args()
    
    fuzzer = DNP3Fuzzer(args.target, args.port)
    fuzzer.run()
```
