# 工控固件逆向工程

### 2.1 固件获取方法

固件逆向的第一步是获取固件镜像文件。常见的获取方法包括：

#### 2.1.1 官方下载

许多厂商提供固件更新包供用户下载。这些包通常是加密或压缩的，需要进一步处理。

```
常见固件下载地址：
- Siemens: https://support.industry.siemens.com/
- Schneider Electric: https://www.se.com/ww/en/download/
- Rockwell Automation: https://compatibility.rockwellautomation.com/
- Mitsubishi: https://www.mitsubishielectric.com/
- ABB: https://new.abb.com/service/support
```

#### 2.1.2 从设备提取

直接从物理设备中提取固件是最可靠的方法：

**方法一：UART串口提取**

```bash
# 使用bus Pirate或USB-TTL适配器连接UART引脚
# 常见波特率: 9600, 115200, 57600

# 使用minicom连接
minicom -D /dev/ttyUSB0 -b 115200

# 在U-Boot引导阶段中断启动流程，进入U-Boot命令行
# 然后通过tftp下载固件到主机

# U-Boot命令示例：
=> printenv                    # 查看环境变量
=> bdinfo                      # 查看板级信息
=> tftpboot 0x81000000 firmware.bin  # 通过TFTP加载固件到内存
=> md 0x81000000               # 显示内存内容
=> cp.b 0x81000000 0x9F000000 0x200000  # 复制到Flash

# 或直接读取Flash到串口
=> md.b 0x9F000000 0x800000
```

**方法二：JTAG/SWD提取**

```bash
# 使用OpenOCD连接JTAG接口
# 配置文件示例 (openocd.cfg):
source [find interface/ftdi/jtagkey.cfg]
source [find target/ti_tms570.cfg]

# 启动OpenOCD
openocd -f openocd.cfg

# 在另一个终端使用telnet连接
telnet localhost 4444
> reset halt
> flash probe 0
> flash info 0
> flash read_bank 0 firmware.bin 0 0x400000
```

**方法三：芯片夹直接读取Flash芯片**

```bash
# 使用CH341A编程器直接读取SPI Flash芯片
# 需要拆焊芯片或使用芯片夹

# Linux下使用flashrom工具
flashrom --programmer ch341a_spi -r firmware.bin

# 验证读取完整性
flashrom --programmer ch341a_spi -v firmware.bin

# 多次读取比较确保数据一致
flashrom --programmer ch341a_spi -r firmware_2.bin
diff firmware.bin firmware_2.bin
```

### 2.2 固件解包与文件系统提取

#### 2.2.1 使用binwalk分析固件结构

```bash
# 安装binwalk
pip install binwalk

# 基本分析
binwalk firmware.bin

# 提取文件系统
binwalk -e firmware.bin

# 递归提取
binwalk -eM firmware.bin

# 查看熵值分析（识别加密/压缩区域）
binwalk -E firmware.bin

# 实际输出示例：
# DECIMAL       HEXADECIMAL     DESCRIPTION
# --------------------------------------------------------------------------------
# 0             0x0             TP-Link firmware header, firmware version: 1.0.0
# 532           0x214           LZMA compressed data, Dictionary size: 4MB
# 131072        0x20000         SquashFS filesystem, little endian, xz compressed
# 1048576       0x100000        ELF 32-bit LSB executable, ARM, version 1
```

#### 2.2.2 手动提取文件系统

当binwalk无法自动提取时，需要手动定位和提取文件系统：

```python
#!/usr/bin/env python3
"""手动固件文件系统提取工具"""
import struct
import os

# 常见文件系统魔数
FS_MAGIC = {
    'squashfs': b'\x68\x73\x71\x73',  # hsqs
    'jffs2':   b'\x19\x85',           # JFFS2
    'cramfs':  b'\x45\x3d\xcd\x28',   # cramfs
    'romfs':   b'\x2d\x72\x6f\x6d',   # -rom
    'ubifs':   b'\x31\x18\x53\x20',   # UBI
}

def find_filesystem(firmware_path):
    """在固件镜像中搜索文件系统"""
    with open(firmware_path, 'rb') as f:
        data = f.read()
    
    results = []
    for fs_name, magic in FS_MAGIC.items():
        offset = 0
        while True:
            pos = data.find(magic, offset)
            if pos == -1:
                break
            results.append({
                'type': fs_name,
                'offset': pos,
                'hex_offset': hex(pos)
            })
            offset = pos + len(magic)
    
    return results

def extract_squashfs(firmware_path, offset, output_path):
    """提取SquashFS文件系统"""
    with open(firmware_path, 'rb') as f:
        f.seek(offset)
        # 读取SquashFS头部信息
        header = f.read(96)
        
        magic = header[0:4]
        inode_count = struct.unpack('<I', header[4:8])[0]
        block_size = struct.unpack('<I', header[12:16])[0]
        frag_count = struct.unpack('<I', header[40:44])[0]
        
        print(f"SquashFS信息:")
        print(f"  魔数: {magic}")
        print(f"  Inode数量: {inode_count}")
        print(f"  块大小: {block_size}")
        print(f"  分片数量: {frag_count}")
        
        # 提取整个SquashFS
        f.seek(offset)
        # 估算文件系统大小（从头部读取）
        bytes_used = struct.unpack('<I', header[48:52])[0]
        fs_data = f.read(bytes_used)
        
        with open(output_path, 'wb') as out:
            out.write(fs_data)
        
        print(f"  已提取到: {output_path}")
        print(f"  使用以下命令解包: unsquashfs {output_path}")

def analyze_firmware(firmware_path):
    """分析固件整体结构"""
    with open(firmware_path, 'rb') as f:
        data = f.read()
    
    file_size = len(data)
    print(f"固件文件: {firmware_path}")
    print(f"文件大小: {file_size} bytes ({file_size/1024/1024:.2f} MB)")
    print(f"MD5: {__import__('hashlib').md5(data).hexdigest()}")
    print(f"SHA256: {__import__('hashlib').sha256(data).hexdigest()}")
    
    # 查找文件系统
    print(f"\n[+] 搜索文件系统签名...")
    results = find_filesystem(firmware_path)
    
    if results:
        print(f"[+] 发现 {len(results)} 个文件系统标记:")
        for r in results:
            print(f"    类型: {r['type']}, 偏移: {r['offset']} ({r['hex_offset']})")
        
        # 提取第一个SquashFS
        for r in results:
            if r['type'] == 'squashfs':
                output = os.path.splitext(firmware_path)[0] + '_squashfs.bin'
                extract_squashfs(firmware_path, r['offset'], output)
                break
    else:
        print("[-] 未发现已知文件系统标记")
    
    # 查找ELF文件
    print(f"\n[+] 搜索ELF可执行文件...")
    elf_magic = b'\x7fELF'
    offset = 0
    elf_count = 0
    while True:
        pos = data.find(elf_magic, offset)
        if pos == -1:
            break
        elf_count += 1
        # 判断位数
        ei_class = data[pos + 4] if pos + 4 < len(data) else 0
        arch_type = "32位" if ei_class == 1 else "64位" if ei_class == 2 else "未知"
        print(f"    ELF #{elf_count}: 偏移 {pos} ({hex(pos)}), {arch_type}")
        offset = pos + 4
    
    print(f"\n共发现 {elf_count} 个ELF文件")

if __name__ == '__main__':
    import sys
    if len(sys.argv) < 2:
        print("用法: python3 firmware_extractor.py <固件文件>")
        sys.exit(1)
    analyze_firmware(sys.argv[1])
```

### 2.3 使用Ghidra进行固件逆向

#### 2.3.1 Ghidra配置与加载

Ghidra是NSA开源的逆向工程框架，适合分析工控固件中的嵌入式二进制文件。

```bash
# 安装Ghidra
# 从 https://ghidra-sre.org/ 下载最新版本
# 需要 JDK 17+

# 启动Ghidra
ghidraRun.bat  # Windows
./ghidraRun     # Linux

# 创建新项目
# Project -> New Project -> Non-Shared
# 导入固件二进制文件: File -> Import File
# 选择处理器: ARM Cortex, MIPS, PowerPC等
# 设置基地址: 0x80000000 (根据实际硬件)
# 设置字节序: Little Endian / Big Endian
```

#### 2.3.2 Ghidra脚本自动化分析

```python
# GhidraScript: ICSFirmwareAnalyzer.py
# 在Ghidra Script Manager中运行
# @category ICS.Security
# @author ICS Security Researcher

from ghidra.app.decompiler import DecompInterface
from ghidra.program.model.listing import Function
from ghidra.program.model.symbol import SymbolType
from ghidra.util.task import ConsoleTaskMonitor
import ghidra.program.model.symbol as symbol

def analyze_network_functions():
    """分析网络相关函数，寻找潜在漏洞"""
    monitor = ConsoleTaskMonitor()
    decomp = DecompInterface()
    decomp.openProgram(currentProgram)
    
    # 搜索网络相关函数名
    network_keywords = [
        'recv', 'send', 'read', 'write',
        'socket', 'connect', 'bind', 'listen', 'accept',
        'recvfrom', 'sendto', 'recvmsg', 'sendmsg',
        'memcpy', 'strcpy', 'sprintf', 'strcat',
        'system', 'exec', 'popen', 'fork'
    ]
    
    print("=" * 60)
    print("ICS固件网络函数分析报告")
    print("=" * 60)
    
    func_manager = currentProgram.getFunctionManager()
    for func in func_manager.getFunctions(True):
        func_name = func.getName()
        
        # 检查是否为危险函数
        for keyword in network_keywords:
            if keyword.lower() in func_name.lower():
                print(f"\n[!] 发现关键函数: {func_name}")
                print(f"    地址: {func.getEntryPoint()}")
                print(f"    大小: {func.getBody().getNumAddresses()} bytes")
                
                # 反编译函数
                results = decomp.decompileFunction(func, 30, monitor)
                if results.decompileCompleted():
                    decomp_result = results.getDecompiledFunction()
                    c_code = decomp_result.getC()
                    
                    # 检查是否存在漏洞模式
                    vuln_patterns = [
                        ('缓冲区溢出', ['memcpy', 'strcpy', 'sprintf', 'strcat']),
                        ('命令注入', ['system', 'popen', 'exec']),
                        ('格式化字符串', ['printf', 'sprintf', 'snprintf']),
                        ('整数溢出', ['malloc', 'calloc', 'realloc']),
                    ]
                    
                    for pattern_name, pattern_funcs in vuln_patterns:
                        for pf in pattern_funcs:
                            if pf in c_code:
                                print(f"    [WARNING] 可能存在 {pattern_name} 风险 (涉及 {pf})")
                
                break

def find_hardcoded_credentials():
    """搜索固件中的硬编码凭据"""
    print("\n" + "=" * 60)
    print("硬编码凭据扫描")
    print("=" * 60)
    
    memory = currentProgram.getMemory()
    data_iterator = currentProgram.getListing().getDefinedData(True)
    
    cred_patterns = [
        'password', 'passwd', 'pwd',
        'admin', 'root', 'user',
        'secret', 'key', 'token',
        'credential', 'auth'
    ]
    
    for data in data_iterator:
        if data.hasStringValue():
            value = data.getValue().lower()
            for pattern in cred_patterns:
                if pattern in value:
                    addr = data.getAddress()
                    print(f"[!] 发现疑似凭据字符串: '{data.getValue()}'")
                    print(f"    地址: {addr}")
                    print(f"    数据类型: {data.getDataType()}")
                    break

def analyze_protocol_handlers():
    """分析协议处理函数"""
    print("\n" + "=" * 60)
    print("协议处理函数分析")
    print("=" * 60)
    
    # 常见工控协议关键字
    protocol_keywords = {
        'Modbus': ['modbus', 'mb_', '0x01', '0x02', '0x03', '0x04', '0x05'],
        'S7':     ['s7_', 'iso_', 'cotp_', '0xF0', '0x32'],
        'DNP3':   ['dnp3', 'dnp_', '0xC0', '0xC1', '0xC2'],
        'BACnet': ['bacnet', 'bac_', '0x00', '0x01', '0x02'],
        'IEC104': ['iec104', 'iec_', '0x01', '0x0E', '0x46'],
    }
    
    func_manager = currentProgram.getFunctionManager()
    for func in func_manager.getFunctions(True):
        func_name = func.getName().lower()
        for proto, keywords in protocol_keywords.items():
            for kw in keywords:
                if kw.lower() in func_name:
                    print(f"[+] 发现 {proto} 协议处理函数: {func.getName()}")
                    print(f"    地址: {func.getEntryPoint()}")
                    break

# 主函数
print("ICS固件安全分析工具 v1.0")
print("分析目标: " + currentProgram.getName())
```

### 2.4 固件中的凭据提取

```python
#!/usr/bin/env python3
"""固件凭据提取工具 - 用于在解包后的固件文件系统中搜索敏感信息"""
import os
import re
import gzip
import tarfile
import zipfile

# 敏感信息正则模式
PATTERNS = {
    '密码': [
        r'password\s*[=:]\s*["\']?([^"\'\s]+)',
        r'passwd\s*[=:]\s*["\']?([^"\'\s]+)',
        r'pwd\s*[=:]\s*["\']?([^"\'\s]+)',
    ],
    '密钥': [
        r'private[_-]?key\s*[=:]\s*["\']?([^"\'\s]+)',
        r'secret[_-]?key\s*[=:]\s*["\']?([^"\'\s]+)',
        r'api[_-]?key\s*[=:]\s*["\']?([^"\'\s]+)',
    ],
    '证书': [
        r'-----BEGIN (RSA |EC )?PRIVATE KEY-----',
        r'-----BEGIN CERTIFICATE-----',
    ],
    '凭据': [
        r'admin\s*[:=]\s*([^\s]+)',
        r'root\s*[:=]\s*([^\s]+)',
        r'user\s*[:=]\s*([^\s]+)',
    ],
    '数据库': [
        r'(?:mysql|postgres|mongodb)://[^\s]+',
        r'(?:redis|mssql)://[^\s]+',
    ],
    'Token': [
        r'token\s*[=:]\s*["\']?([^"\'\s]+)',
        r'authorization\s*[:=]\s*[Bb]earer\s+([^\s]+)',
    ],
}

# 可疑文件名
SUSPICIOUS_FILES = [
    'passwd', 'shadow', 'htpasswd', '.htpasswd',
    'config', 'configuration', 'cfg',
    'credentials', 'secrets', 'private',
    'ssl', 'cert', 'certificate',
    '.env', 'env', 'environment',
    'shadow-', 'pwd.db', 'spwd.db',
]

def scan_file(file_path):
    """扫描单个文件中的敏感信息"""
    findings = []
    
    try:
        # 跳过二进制文件
        with open(file_path, 'rb') as f:
            chunk = f.read(4096)
            if b'\x00' in chunk:
                # 可能是二进制文件，尝试strings提取
                return scan_binary_strings(file_path)
    
        with open(file_path, 'r', errors='ignore') as f:
            content = f.read()
            line_num = 0
            for line in content.split('\n'):
                line_num += 1
                for category, patterns in PATTERNS.items():
                    for pattern in patterns:
                        matches = re.findall(pattern, line, re.IGNORECASE)
                        for match in matches:
                            if match and len(match) > 2:
                                findings.append({
                                    'category': category,
                                    'file': file_path,
                                    'line': line_num,
                                    'value': match[:100],  # 截断长值
                                    'context': line.strip()[:200]
                                })
    except Exception as e:
        pass
    
    return findings

def scan_binary_strings(file_path, min_length=4):
    """从二进制文件中提取字符串并扫描"""
    findings = []
    try:
        with open(file_path, 'rb') as f:
            data = f.read()
        
        # 提取可打印字符串
        strings = re.findall(b'[\x20-\x7e]{%d,}' % min_length, data)
        
        for s in strings:
            try:
                s_str = s.decode('ascii')
                for category, patterns in PATTERNS.items():
                    for pattern in patterns:
                        matches = re.findall(pattern, s_str, re.IGNORECASE)
                        for match in matches:
                            if match and len(match) > 2:
                                findings.append({
                                    'category': category,
                                    'file': file_path,
                                    'line': 0,
                                    'value': match[:100],
                                    'context': s_str[:200]
                                })
            except:
                pass
    except:
        pass
    
    return findings

def scan_directory(root_dir):
    """递归扫描目录"""
    all_findings = []
    file_count = 0
    
    for dirpath, dirnames, filenames in os.walk(root_dir):
        for filename in filenames:
            file_path = os.path.join(dirpath, filename)
            
            # 检查是否为可疑文件名
            for suspicious in SUSPICIOUS_FILES:
                if suspicious in filename.lower():
                    print(f"[!] 发现可疑文件: {file_path}")
                    break
            
            # 扫描文件内容
            findings = scan_file(file_path)
            if findings:
                all_findings.extend(findings)
                for f in findings:
                    print(f"[{f['category']}] {f['file']}:{f['line']} -> {f['value']}")
            
            file_count += 1
            if file_count % 100 == 0:
                print(f"[*] 已扫描 {file_count} 个文件...")
    
    return all_findings

def generate_report(findings, output_file='credentials_report.txt'):
    """生成凭据扫描报告"""
    with open(output_file, 'w', encoding='utf-8') as f:
        f.write("=" * 70 + "\n")
        f.write("固件凭据扫描报告\n")
        f.write("=" * 70 + "\n\n")
        
        # 按类别分组
        categories = {}
        for finding in findings:
            cat = finding['category']
            if cat not in categories:
                categories[cat] = []
            categories[cat].append(finding)
        
        for cat, items in sorted(categories.items()):
            f.write(f"\n{'='*50}\n")
            f.write(f"类别: {cat} (共 {len(items)} 项)\n")
            f.write(f"{'='*50}\n\n")
            
            for item in items:
                f.write(f"文件: {item['file']}\n")
                f.write(f"行号: {item['line']}\n")
                f.write(f"值: {item['value']}\n")
                f.write(f"上下文: {item['context']}\n")
                f.write("-" * 40 + "\n")
    
    print(f"\n[+] 报告已生成: {output_file}")
    print(f"[+] 共发现 {len(findings)} 个敏感信息")

if __name__ == '__main__':
    import sys
    if len(sys.argv) < 2:
        print("用法: python3 cred_scanner.py <固件文件系统目录>")
        sys.exit(1)
    
    root = sys.argv[1]
    print(f"[*] 开始扫描: {root}")
    findings = scan_directory(root)
    generate_report(findings)
```
