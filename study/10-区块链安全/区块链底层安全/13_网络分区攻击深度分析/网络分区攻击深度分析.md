### 12.2 分区检测与恢复

```python
class NetworkPartitionDetector:
    """区块链网络分区检测"""

    def __init__(self):
        self.peer_blocks = {}  # peer -> latest block height
        self.our_height = 0
        self.peer_count = 0
        self.active_peers = 0
        self.partition_history = []

    def update_peer_status(self, peer, block_height, timestamp):
        """更新对等节点状态"""
        self.peer_blocks[peer] = {
            'height': block_height,
            'time': timestamp,
        }

    def check_partition(self):
        """检测网络分区"""
        if len(self.peer_blocks) < 3:
            return None  # 节点太少无法判断

        heights = [p['height'] for p in self.peer_blocks.values()]
        max_height = max(heights)
        min_height = min(heights)

        # 高度差异过大
        height_gap = max_height - min_height
        if height_gap > 10:  # 超过10个区块差异
            # 分析分区
            high_group = [p for p, info in self.peer_blocks.items()
                         if info['height'] >= max_height - 2]
            low_group = [p for p, info in self.peer_blocks.items()
                        if info['height'] <= min_height + 2]

            partition = {
                'type': 'height_discrepancy',
                'gap': height_gap,
                'high_group': high_group,
                'low_group': low_group,
                'severity': 'HIGH' if height_gap > 50 else 'MEDIUM',
            }
            self.partition_history.append(partition)
            return partition

        # 检查连接数下降
        active = len(self.peer_blocks)
        if active < self.peer_count * 0.5:  # 连接数下降50%
            partition = {
                'type': 'connectivity_loss',
                'active_peers': active,
                'previous_peers': self.peer_count,
                'severity': 'HIGH',
            }
            self.partition_history.append(partition)
            return partition

        self.peer_count = active
        return None

    def handle_partition(self, partition):
        """处理分区事件"""
        if partition['type'] == 'height_discrepancy':
            # 不要立即接受更高链(可能是攻击)
            # 等待更多确认
            print(f"[WARNING] 检测到高度差异: {partition['gap']}")
            print(f"  高链组: {len(partition['high_group'])}个节点")
            print(f"  低链组: {len(partition['low_group'])}个节点")
            print("  建议: 等待更多区块确认再决定")

        elif partition['type'] == 'connectivity_loss':
            print(f"[WARNING] 连接数骤降: {partition['active_peers']}/{partition['previous_peers']}")
            print("  可能原因: 网络分区或DDoS")
            print("  建议: 暂停出块/交易确认")
```
