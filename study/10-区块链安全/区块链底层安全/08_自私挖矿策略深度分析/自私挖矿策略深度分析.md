# 自私挖矿策略深度分析

### 8.1 自私挖矿数学模型

```
自私挖矿收益模型:

定义:
  α = 攻击者算力占比 (0 < α < 1)
  β = 诚实矿工选择攻击者区块的概率 (0 ≤ β ≤ 1)
  γ = 1 - β (诚实矿工选择诚实链的概率)

攻击者相对收益 R(α, β):
  R(α, β) = [α(1-α)(1+β)] / [1 - α + αβ]

无利可图边界:
  α < α* = β / (1-β) 时无利可图

关键阈值:
  β = 0.5 (无网络优势): α* = 1 (理论不可行)
  β = 0.75 (强网络优势): α* = 0.33
  β = 1.0 (完全网络优势): α* = 0 (总是有利)

实际考量:
  - β取决于网络传播优势
  - 攻击者需要低延迟连接
  - 与矿池/节点的拓扑有关
  - 实际中α=0.25-0.33可能有利润
```

### 8.2 自私挖矿实施策略

```python
class SelfishMiner:
    """自私挖矿策略模拟"""

    def __init__(self, alpha=0.33, gamma=0.5):
        """
        alpha: 攻击者算力占比
        gamma: 诚实矿工跟随攻击者链的概率
        """
        self.alpha = alpha
        self.gamma = gamma
        self.private_chain = 0  # 攻击者私有链长度
        self.public_chain = 0   # 公开链长度
        self.revenue = 0        # 攻击者累计收益
        self.honest_revenue = 0 # 诚实矿工累计收益

    def mine_round(self):
        """模拟一轮挖矿"""
        import random
        if random.random() < self.alpha:
            # 攻击者找到区块
            self._attacker_found_block()
        else:
            # 诚实矿工找到区块
            self._honest_found_block()

    def _attacker_found_block(self):
        """攻击者找到新区块"""
        self.private_chain += 1
        lead = self.private_chain - self.public_chain

        if lead == 0:
            # 平局: 公开私有链(竞争)
            self._publish_private_chain()
        elif lead == 1 and self.public_chain > 0:
            # 领先1且公开链有区块: 等待
            pass
        # lead >= 2: 继续隐瞒

    def _honest_found_block(self):
        """诚实矿工找到新区块"""
        self.public_chain += 1
        lead = self.private_chain - self.public_chain

        if lead < 0:
            # 诚实链领先,放弃私有链
            self.honest_revenue += 1
            self.private_chain = self.public_chain
        elif lead == 0:
            # 平局: 公开私有链竞争
            import random
            if random.random() < self.gamma:
                self.revenue += 1  # 诚实矿工跟随攻击者链
            else:
                self.honest_revenue += 1
            self.private_chain = self.public_chain
        elif lead == 1:
            # 领先1: 公开私有链(使诚实矿工的区块作废)
            self._publish_private_chain()
            self.revenue += 1
        elif lead >= 2:
            # 领先2+: 公开一个区块(保持领先)
            self.private_chain -= 1
            self.revenue += 1

    def _publish_private_chain(self):
        """公开私有链"""
        self.private_chain = self.public_chain

    def simulate(self, rounds=10000):
        """模拟多轮"""
        for _ in range(rounds):
            self.mine_round()
        total = self.revenue + self.honest_revenue
        if total > 0:
            share = self.revenue / total
            print(f"算力占比: α={self.alpha}")
            print(f"跟随概率: γ={self.gamma}")
            print(f"攻击者收益占比: {share:.4f}")
            print(f"期望收益占比: {self.alpha:.4f}")
            print(f"超额收益: {(share - self.alpha)/self.alpha * 100:.2f}%")
            print(f"攻击者区块数: {self.revenue}")
            print(f"诚实矿工区块数: {self.honest_revenue}")

# 模拟不同场景
for alpha in [0.1, 0.2, 0.25, 0.33, 0.4, 0.5]:
    for gamma in [0.0, 0.5, 0.75, 1.0]:
        miner = SelfishMiner(alpha, gamma)
        miner.simulate(50000)
        print()
```

### 8.3 自私挖矿检测方法

```python
class SelfishMiningDetector:
    """自私挖矿检测器"""

    def __init__(self):
        self.block_arrivals = []  # (block_hash, timestamp, miner)
        self.orphans = []         # 孤块记录
        self.stats = {
            'total_blocks': 0,
            'orphan_rate': 0,
            'miner_distribution': {},
            'inter_block_times': [],
            'unexpected_publishes': [],
        }

    def record_block(self, block_hash, timestamp, miner, is_orphan=False):
        """记录新区块"""
        self.block_arrivals.append({
            'hash': block_hash,
            'time': timestamp,
            'miner': miner,
            'orphan': is_orphan,
        })
        self.stats['total_blocks'] += 1
        if is_orphan:
            self.orphans.append(block_hash)

    def analyze(self):
        """分析自私挖矿迹象"""
        findings = []

        # 1. 孤块率异常
        orphan_rate = len(self.orphans) / max(self.stats['total_blocks'], 1)
        self.stats['orphan_rate'] = orphan_rate
        if orphan_rate > 0.05:  # 正常<3-5%
            findings.append({
                'type': 'high_orphan_rate',
                'severity': 'HIGH',
                'value': orphan_rate,
                'detail': f'孤块率{orphan_rate:.2%}异常偏高(正常<5%)'
            })

        # 2. 矿工发布模式异常
        miner_blocks = {}
        for b in self.block_arrivals:
            miner_blocks[b['miner']] = miner_blocks.get(b['miner'], 0) + 1

        for miner, count in miner_blocks.items():
            share = count / self.stats['total_blocks']
            self.stats['miner_distribution'][miner] = share
            # 如果矿工份额远超算力声明
            if share > 0.33:
                findings.append({
                    'type': 'excessive_block_share',
                    'severity': 'MEDIUM',
                    'value': share,
                    'detail': f'矿工{miner}出块份额{share:.2%}偏高'
                })

        # 3. 区块发布时间分析
        # 自私挖矿的特征: 紧接着诚实矿工出块后立即发布
        times = sorted([b['time'] for b in self.block_arrivals if not b['orphan']])
        for i in range(1, len(times)):
            delta = times[i] - times[i-1]
            self.stats['inter_block_times'].append(delta)

        # 极短间隔(可能是在竞争发布)
        if self.stats['inter_block_times']:
            short_intervals = sum(
                1 for t in self.stats['inter_block_times']
                if t < 2  # 小于2秒
            )
            if short_intervals > len(self.stats['inter_block_times']) * 0.15:
                findings.append({
                    'type': 'frequent_rapid_publishing',
                    'severity': 'MEDIUM',
                    'detail': f'频繁快速发布区块({short_intervals}次)'
                })

        # 4. 特定矿工的孤块率
        for miner in miner_blocks:
            miner_orphans = sum(
                1 for b in self.block_arrivals
                if b['miner'] == miner and b['orphan']
            )
            miner_total = miner_blocks[miner]
            if miner_total > 10:
                miner_orphan_rate = miner_orphans / miner_total
                if miner_orphan_rate > 0.1:
                    findings.append({
                        'type': 'high_miner_orphan_rate',
                        'severity': 'HIGH',
                        'detail': f'矿工{miner}孤块率{miner_orphan_rate:.2%}异常'
                    })

        return findings
```
