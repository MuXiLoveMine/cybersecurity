#!/usr/bin/env python3
"""
DeFi 链上安全监控脚本
监控大额转账、异常交互、合约调用等
"""
import time
import json
from web3 import Web3
from datetime import datetime

class DeFiSecurityMonitor:
    """DeFi 安全监控器"""

    def __init__(self, rpc_url, alert_webhook=None):
        self.w3 = Web3(Web3.HTTPProvider(rpc_url))
        self.alert_webhook = alert_webhook
        self.monitored_contracts = {}
        self.alert_thresholds = {
            "large_transfer_eth": 100,    # ETH 大额转账阈值
            "large_transfer_usdc": 100000,  # USDC 大额转账阈值
        }

    def add_monitored_contract(self, name, address, abi=None):
        """添加要监控的合约"""
        self.monitored_contracts[name] = {
            "address": Web3.to_checksum_address(address),
            "abi": abi,
        }
        print(f"已添加监控合约: {name} ({address})")

    def monitor_large_transfers(self, block_number):
        """监控区块中的大额转账"""
        block = self.w3.eth.get_block(block_number, full_transactions=True)
        alerts = []

        for tx in block.transactions:
            # 监控 ETH 大额转账
            value_eth = self.w3.from_wei(tx["value"], "ether")
            if value_eth >= self.alert_thresholds["large_transfer_eth"]:
                alert = {
                    "type": "LARGE_ETH_TRANSFER",
                    "severity": "medium",
                    "tx_hash": tx["hash"].hex(),
                    "from": tx["from"],
                    "to": tx["to"],
                    "value_eth": float(value_eth),
                    "block": block_number,
                }
                alerts.append(alert)
                print(f"⚠️  大额 ETH 转账: {value_eth:.2f} ETH "
                      f"从 {tx['from'][:10]}... 到 {tx['to'][:10]}...")

            # 监控与 DeFi 合约的交互
            if tx["to"]:
                to_lower = tx["to"].lower()
                for name, info in self.monitored_contracts.items():
                    if info["address"].lower() == to_lower:
                        alert = {
                            "type": "CONTRACT_INTERACTION",
                            "severity": "info",
                            "tx_hash": tx["hash"].hex(),
                            "contract": name,
                            "from": tx["from"],
                            "value_eth": float(value_eth),
                            "block": block_number,
                        }
                        alerts.append(alert)
                        print(f"ℹ️  合约交互: {name} - 交易 {tx['hash'].hex()[:16]}...")

        return alerts

    def check_contract_balance_change(self, name, block_number):
        """检查合约余额变化"""
        if name not in self.monitored_contracts:
            return None

        contract_address = self.monitored_contracts[name]["address"]
        current_balance = self.w3.eth.get_balance(contract_address, block_number)
        prev_balance = self.w3.eth.get_balance(contract_address, block_number - 1)

        change = current_balance - prev_balance
        change_eth = self.w3.from_wei(abs(change), "ether")

        if change_eth >= self.alert_thresholds["large_transfer_eth"]:
            direction = "增加" if change > 0 else "减少"
            alert = {
                "type": "BALANCE_CHANGE",
                "severity": "high" if change < 0 else "medium",
                "contract": name,
                "direction": direction,
                "change_eth": float(change_eth),
                "block": block_number,
            }
            print(f"🚨 合约余额{direction}: {name} - {change_eth:.2f} ETH")
            return alert

        return None

    def start_monitoring(self, start_block=None):
        """启动实时监控"""
        if start_block is None:
            start_block = self.w3.eth.block_number

        print(f"开始监控，当前区块: {start_block}")
        print(f"监控合约数: {len(self.monitored_contracts)}")

        current_block = start_block
        while True:
            latest_block = self.w3.eth.block_number

            while current_block <= latest_block:
                print(f"\n处理区块 {current_block} ({datetime.now().strftime('%H:%M:%S')})")

                # 大额转账监控
                self.monitor_large_transfers(current_block)

                # 合约余额变化
                for name in self.monitored_contracts:
                    self.check_contract_balance_change(name, current_block)

                current_block += 1

            time.sleep(12)  # 约一个区块时间


# 使用示例
if __name__ == "__main__":
    monitor = DeFiSecurityMonitor(

    )

        print(f"  Medium: {medium}")
