### 二、常见漏洞代码示例

#### 1. 重入攻击 (Reentrancy)

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

// ========== 漏洞合约 ==========
contract VulnerableBank {
    mapping(address => uint256) public balances;

    function deposit() public payable {
        balances[msg.sender] += msg.value;
    }

    // 漏洞：先转账，后更新余额
    function withdraw(uint256 amount) public {
        require(balances[msg.sender] >= amount, "余额不足");

        // 1. 先转账（外部调用）
        (bool success, ) = msg.sender.call{value: amount}("");
        require(success, "转账失败");

        // 2. 后更新状态
        balances[msg.sender] -= amount;
    }
}

// ========== 攻击合约 ==========
contract ReentrancyAttack {
    VulnerableBank public bank;
    uint256 public attackTimes;

    constructor(address _bank) {
        bank = VulnerableBank(_bank);
    }

    function attack() external payable {
        bank.deposit{value: msg.value}();
        bank.withdraw(msg.value);
    }

    // 攻击关键：fallback 函数中再次调用 withdraw
    receive() external payable {
        if (attackTimes < 3 && address(bank).balance >= msg.value) {
            attackTimes++;
            bank.withdraw(msg.value);
        }
    }
}

// ========== 修复方案 ==========
contract FixedBank {
    mapping(address => uint256) public balances;
    bool private locked;

    // 修复1：检查-效果-交互模式 (Checks-Effects-Interactions)
    function withdraw(uint256 amount) public {
        // Checks
        require(balances[msg.sender] >= amount, "余额不足");

        // Effects
        balances[msg.sender] -= amount;

        // Interactions
        (bool success, ) = msg.sender.call{value: amount}("");
        require(success, "转账失败");
    }

    // 修复2：重入锁 (Reentrancy Guard)
    modifier nonReentrant() {
        require(!locked, "禁止重入");
        locked = true;
        _;
        locked = false;
    }

    function withdrawSafe(uint256 amount) public nonReentrant {
        require(balances[msg.sender] >= amount, "余额不足");
        balances[msg.sender] -= amount;
        (bool success, ) = msg.sender.call{value: amount}("");
        require(success, "转账失败");
    }
}
```

#### 2. 整数溢出 (Integer Overflow/Underflow)

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.6.0; // Solidity 0.8 之前版本才有溢出问题

// ========== 漏洞合约 ==========
contract TokenOverflow {
    mapping(address => uint256) public balanceOf;
    uint256 public totalSupply;

    function transfer(address to, uint256 value) public returns (bool) {
        // 漏洞：未检查溢出
        require(balanceOf[msg.sender] >= value);

        balanceOf[msg.sender] -= value;  // 可能下溢
        balanceOf[to] += value;          // 可能上溢

        return true;
    }

    // 另一个常见漏洞：批量转账
    function batchTransfer(address[] memory recipients, uint256 value) public {
        uint256 amount = recipients.length * value; // 溢出！
        require(balanceOf[msg.sender] >= amount);

        balanceOf[msg.sender] -= amount;
        for (uint256 i = 0; i < recipients.length; i++) {
            balanceOf[recipients[i]] += value;
        }
    }
}

// ========== 修复方案 ==========
// 方案1：使用 SafeMath 库（0.8 之前）
import "@openzeppelin/contracts/math/SafeMath.sol";

contract SafeToken {
    using SafeMath for uint256;
    mapping(address => uint256) public balanceOf;

    function transfer(address to, uint256 value) public returns (bool) {
        require(balanceOf[msg.sender] >= value);
        balanceOf[msg.sender] = balanceOf[msg.sender].sub(value);
        balanceOf[to] = balanceOf[to].add(value);
        return true;
    }
}

// 方案2：使用 Solidity 0.8+ 内置溢出检查
pragma solidity ^0.8.0;

contract SafeTokenV2 {
    mapping(address => uint256) public balanceOf;

    function transfer(address to, uint256 value) public returns (bool) {
        // 0.8+ 自动检查溢出，溢出时会 revert
        require(balanceOf[msg.sender] >= value);
        balanceOf[msg.sender] -= value;
        balanceOf[to] += value;
        return true;
    }
}
```

#### 3. 访问控制漏洞 (Access Control)

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

// ========== 漏洞合约 ==========
contract VulnerableVault {
    address public owner;
    mapping(address => uint256) public balances;

    constructor() {
        owner = msg.sender;
    }

    // 漏洞：任何人都可以调用
    function withdrawAll() public {
        payable(msg.sender).transfer(address(this).balance);
    }

    // 漏洞：owner 未检查
    function mint(address to, uint256 amount) public {
        balances[to] += amount;
    }

    // 漏洞：tx.origin 身份认证（钓鱼攻击风险）
    function transferOwner(address newOwner) public {
        require(tx.origin == owner, "不是所有者");
        owner = newOwner;
    }
}

// ========== 修复方案 ==========
contract FixedVault {
    address public owner;
    mapping(address => uint256) public balances;

    modifier onlyOwner() {
        require(msg.sender == owner, "不是所有者");
        _;
    }

    constructor() {
        owner = msg.sender;
    }

    // 修复：添加 onlyOwner 修饰符
    function withdrawAll() public onlyOwner {
        payable(msg.sender).transfer(address(this).balance);
    }

    function mint(address to, uint256 amount) public onlyOwner {
        balances[to] += amount;
    }

    // 修复：使用 msg.sender 替代 tx.origin
    function transferOwner(address newOwner) public onlyOwner {
        require(newOwner != address(0), "无效地址");
        owner = newOwner;
    }

    // 推荐：使用 OpenZeppelin Ownable
    // import "@openzeppelin/contracts/access/Ownable.sol";
    // contract SafeVault is Ownable { ... }
}
```

#### 4. 拒绝服务攻击 (DoS)

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

// ========== 漏洞合约：外部调用失败导致 DoS ==========
contract VulnerableAuction {
    address public highestBidder;
    uint256 public highestBid;
    mapping(address => uint256) public refunds;

    function bid() public payable {
        require(msg.value > highestBid, "出价太低");

        // 漏洞：退款失败会导致整个函数回滚
        // 攻击者可以部署一个拒绝接收 ETH 的合约作为最高出价者
        if (highestBidder != address(0)) {
            (bool success, ) = highestBidder.call{value: highestBid}("");
            require(success, "退款失败"); // 攻击点：恶意合约让这个失败
        }

        highestBidder = msg.sender;
        highestBid = msg.value;
    }
}

// ========== 攻击合约 ==========
contract DoSAttacker {
    function attack(address auction) external payable {
        VulnerableAuction(auction).bid{value: msg.value}();
    }

    // 拒绝接收 ETH，导致退款失败
    receive() external payable {
        revert("I don't accept ETH");
    }
}

// ========== 修复方案：提款模式 (Pull Payment) ==========
contract FixedAuction {
    address public highestBidder;
    uint256 public highestBid;
    mapping(address => uint256) public refunds;

    function bid() public payable {
        require(msg.value > highestBid, "出价太低");

        // 改为记录退款，由用户主动提取
        if (highestBidder != address(0)) {
            refunds[highestBidder] += highestBid;
        }

        highestBidder = msg.sender;
        highestBid = msg.value;
    }

    // 用户主动提款（Pull 模式）
    function withdrawRefund() public {
        uint256 amount = refunds[msg.sender];
        require(amount > 0, "无退款");

        refunds[msg.sender] = 0;
        (bool success, ) = msg.sender.call{value: amount}("");
        require(success, "提款失败");
    }
}
```

#### 5. 前置交易 (Front-running)

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

// ========== 漏洞场景：公开解谜有奖 ==========
contract VulnerablePuzzle {
    bytes32 public answerHash;
    uint256 public reward;

    constructor(bytes32 _answerHash) payable {
        answerHash = _answerHash;
        reward = msg.value;
    }

    // 漏洞：答案在交易池中可见，攻击者可以抢先提交
    function solve(string memory answer) public {
        require(keccak256(abi.encodePacked(answer)) == answerHash, "答案错误");
        require(reward > 0, "奖励已发放");

        reward = 0;
        payable(msg.sender).transfer(address(this).balance);
    }
}

// ========== 修复方案：提交-揭示模式 (Commit-Reveal) ==========
contract CommitRevealPuzzle {
    bytes32 public answerHash;
    uint256 public reward;
    uint256 public commitEndTime;
    uint256 public revealEndTime;

    mapping(address => bytes32) public commitments;

    constructor(bytes32 _answerHash, uint256 _commitDuration, uint256 _revealDuration) payable {
        answerHash = _answerHash;
        reward = msg.value;
        commitEndTime = block.timestamp + _commitDuration;
        revealEndTime = commitEndTime + _revealDuration;
    }

    // 阶段1：提交哈希承诺
    function commit(bytes32 commitment) public {
        require(block.timestamp < commitEndTime, "提交阶段已结束");
        commitments[msg.sender] = commitment;
    }

    // 阶段2：揭示答案
    function reveal(string memory answer, string memory salt) public {
        require(block.timestamp >= commitEndTime, "揭示阶段未开始");
        require(block.timestamp < revealEndTime, "揭示阶段已结束");
        require(commitments[msg.sender] != bytes32(0), "未提交承诺");

        // 验证承诺
        bytes32 commitment = keccak256(abi.encodePacked(msg.sender, answer, salt));
        require(commitment == commitments[msg.sender], "承诺不匹配");

        // 验证答案
        require(keccak256(abi.encodePacked(answer)) == answerHash, "答案错误");

        // 发放奖励
        if (reward > 0) {
            reward = 0;
            payable(msg.sender).transfer(address(this).balance);
        }
    }
}
```
