### 6.1 暴力破解检测

```python
def detect_brute_force(logs, threshold=10, window_minutes=5):
    """检测暴力破解攻击"""
    from collections import defaultdict
    from datetime import datetime, timedelta

    attempts = defaultdict(list)

    for log in logs:
        if log.get("event_id") == 4625:  # 登录失败
            ip = log.get("source_ip")
            timestamp = log.get("timestamp")
            attempts[ip].append(timestamp)

    alerts = []
    for ip, times in attempts.items():
        # 按时间窗口分组
        times.sort()
        for i in range(len(times)):
            window_end = times[i] + timedelta(minutes=window_minutes)
            count = sum(1 for t in times[i:] if t <= window_end)
            if count >= threshold:
                alerts.append({
                    "type": "brute_force",
                    "source_ip": ip,
                    "attempts": count,
                    "time_window": f"{times[i]} - {window_end}",
                    "severity": "high" if count > 50 else "medium",
                })
                break

    return alerts
```

### 6.2 横向移动检测

```python
def detect_lateral_movement(logs):
    """检测横向移动"""
    alerts = []

    # 检测模式1: 一个IP登录多台主机
    ip_hosts = defaultdict(set)
    for log in logs:
        if log.get("event_id") == 4624 and log.get("logon_type") == 3:
            ip = log.get("source_ip")
            host = log.get("dest_host")
            ip_hosts[ip].add(host)

    for ip, hosts in ip_hosts.items():
        if len(hosts) > 5:  # 一个IP登录超过5台主机
            alerts.append({
                "type": "lateral_movement",
                "source_ip": ip,
                "target_hosts": list(hosts),
                "severity": "high",
                "description": f"IP {ip} 登录了 {len(hosts)} 台主机",
            })

    # 检测模式2: Pass-the-Hash (NTLM认证异常)
    for log in logs:
        if (log.get("event_id") == 4624 and
            log.get("logon_type") == 3 and
            log.get("auth_package") == "NTLM" and
            log.get("source_ip") in ["127.0.0.1", "::1"]):
            alerts.append({
                "type": "pass_the_hash",
                "user": log.get("user"),
                "severity": "critical",
                "description": "可能的Pass-the-Hash攻击",
            })

    return alerts
```

### 6.3 持久化检测

```python
def detect_persistence(logs):
    """检测持久化机制"""
    alerts = []

    for log in logs:
        # 检测可疑计划任务创建
        if log.get("event_id") == 4698:
            task_name = log.get("task_name", "")
            suspicious_names = ["update", "security", " defender", "windows",
                               "system", "helper", "service"]
            for name in suspicious_names:
                if name in task_name.lower():
                    if log.get("user") != "SYSTEM":
                        alerts.append({
                            "type": "persistence_scheduled_task",
                            "task_name": task_name,
                            "user": log.get("user"),
                            "severity": "high",
                            "description": f"可疑计划任务: {task_name}",
                        })

        # 检测服务创建
        if log.get("event_id") == 7045:
            service_path = log.get("service_path", "")
            if ("temp" in service_path.lower() or
                "appdata" in service_path.lower()):
                alerts.append({
                    "type": "persistence_service",
                    "service_name": log.get("service_name"),
                    "service_path": service_path,
                    "severity": "high",
                })

        # 检测注册表Run键修改
        if log.get("event_id") == 4657:
            key_path = log.get("object_name", "")
            if "CurrentVersion\\Run" in key_path:
                alerts.append({
                    "type": "persistence_registry",
                    "key_path": key_path,
                    "severity": "medium",
                })

    return alerts
```

### 6.4 数据外传检测

```python
def detect_data_exfiltration(logs):
    """检测数据外传"""
    alerts = []

    # 检测大流量外传
    from collections import defaultdict
    traffic = defaultdict(int)

    for log in logs:
        if log.get("action") == "allow" and log.get("direction") == "outbound":
            key = (log.get("src_ip"), log.get("dest_ip"))
            traffic[key] += log.get("bytes", 0)

    for (src, dst), bytes_out in traffic.items():
        if bytes_out > 100_000_000:  # >100MB
            alerts.append({
                "type": "data_exfiltration",
                "source": src,
                "destination": dst,
                "bytes_transferred": bytes_out,
                "severity": "critical" if bytes_out > 1_000_000_000 else "high",
            })

    # 检测DNS隧道
    for log in logs:
        if log.get("protocol") == "dns":
            query = log.get("dns_query", "")
            if (len(query) > 50 or  # 异常长查询
                query.count(".") > 5 or  # 过多子域
                any(c in query for c in ["--", "==", "+"])):  # 编码特征
                alerts.append({
                    "type": "dns_tunneling",
                    "query": query,
                    "severity": "high",
                })

    return alerts
```
