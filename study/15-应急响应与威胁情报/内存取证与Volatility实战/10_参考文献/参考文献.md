### 一、Volatility 3 常用命令速查表

#### 1. 基础命令

```bash
# ========== Volatility 3 基础用法 ==========

python3 vol.py -h
python3 vol.py --help

# 查看所有可用插件
python3 vol.py --info
python3 vol.py plugins

# 查看指定插件的帮助

# 指定内存镜像文件
python3 vol.py -f memory.raw windows.pslist

# 输出格式（table/json/csv/sqlite）
python3 vol.py -f memory.raw --output=json windows.pslist
python3 vol.py -f memory.raw --output=csv windows.pslist > pslist.csv

# 渲染器选择

python3 vol.py -f memory.raw -r json windows.pslist

python3 vol.py -f memory.raw -vv windows.pslist  # 更详细

python3 vol.py -f memory.raw --symbol-path=./symbols windows.pslist
```

#### 2. 常用高级插件使用指南

**1) malfind - 恶意软件检测**

```bash
# 基本使用
python3 vol.py -f mem.raw windows.malfind

# 指定进程
python3 vol.py -f mem.raw windows.malfind --pid 1234

# 导出可疑内存区域
python3 vol.py -f mem.raw windows.malfind --dump-dir /tmp/malfind_out

# 检测原理：
# - 查找PAGE_EXECUTE_READWRITE权限的内存区域
# - 查找未映射到文件的可执行内存
# - 查找注入的代码段
```

**2) yarascan - YARA规则扫描**

```bash
# 使用YARA规则扫描内存
python3 vol.py -f mem.raw yarascan.YaraScan --yara-file=rules.yar

# 扫描特定进程
python3 vol.py -f mem.raw yarascan.YaraScan --yara-file=rules.yar --pid 1234

# 扫描内核内存
python3 vol.py -f mem.raw yarascan.YaraScan --yara-file=rootkit_rules.yar --kernel

# 示例YARA规则
cat > mimikatz.yar << 'EOF'
rule mimikatz_strings {
    meta:
        description = "Detects Mimikatz strings in memory"
    strings:
        $s1 = "sekurlsa" ascii
        $s2 = "kerberos" ascii
        $s3 = "lsadump" ascii
        $s4 = "mimikatz" ascii wide
    condition:
        2 of them
}
EOF
```

**3) timeliner - 时间线构建**

```bash
# 生成时间线
python3 vol.py -f mem.raw timeliner.Timeliner

# 输出为body格式（用于mactime）
python3 vol.py -f mem.raw timeliner.Timeliner --output=body > timeline.body

# 使用mactime生成可读时间线
mactime -b timeline.body -d -y > timeline.csv

# 时间线包含：
# - 进程创建/退出时间
# - 线程创建时间
# - 文件访问时间
# - 注册表键修改时间
# - 网络连接时间
```

**4) volshell - 交互式内存分析**

```bash
# 启动交互式Shell
python3 vol.py -f mem.raw volshell

# volshell 常用命令
# ps()                # 列出进程
# cc(pid)             # 切换进程上下文
# db(addr, length)    # 转储字节
# dd(addr, length)    # 转储双字
# ds(addr)            # 转储字符串
# dt(type, addr)      # 显示类型结构
# list(head, type, m) # 遍历链表

# 示例：查看进程PEB
python3 vol.py -f mem.raw volshell
> cc(pid=1234)
> db(current().Peb.ImageBaseAddress, 256)
```

**5) apihooks - API钩子检测**

```bash

# 指定进程检测
```

#### 3. 取证分析工作流示例

```bash
#!/bin/bash
# Volatility 取证分析工作流脚本
# 按步骤分析内存镜像

MEM_FILE="memory.raw"
OUT_DIR="./analysis_output"
mkdir -p "$OUT_DIR"

echo "[*] 开始分析内存镜像: $MEM_FILE"
echo "=" * 60

# 步骤1：获取系统信息
echo "[1/10] 获取系统信息..."
python3 vol.py -f "$MEM_FILE" windows.info > "$OUT_DIR/01_system_info.txt"

# 步骤2：进程分析
echo "[2/10] 分析进程..."
python3 vol.py -f "$MEM_FILE" windows.pslist > "$OUT_DIR/02_pslist.txt"
python3 vol.py -f "$MEM_FILE" windows.psscan > "$OUT_DIR/02_psscan.txt"
python3 vol.py -f "$MEM_FILE" windows.pstree > "$OUT_DIR/02_pstree.txt"
python3 vol.py -f "$MEM_FILE" windows.cmdline > "$OUT_DIR/02_cmdline.txt"

# 比较pslist和psscan找隐藏进程
echo "  - 检测隐藏进程..."
diff <(grep -E '^\s*[0-9]+' "$OUT_DIR/02_pslist.txt" | awk '{print $1}' | sort) \
     <(grep -E '^\s*[0-9]+' "$OUT_DIR/02_psscan.txt" | awk '{print $1}' | sort) \
     > "$OUT_DIR/02_hidden_processes.txt" 2>&1

# 步骤3：网络分析
echo "[3/10] 分析网络连接..."
python3 vol.py -f "$MEM_FILE" windows.netscan > "$OUT_DIR/03_netscan.txt"

# 步骤4：DLL分析
echo "[4/10] 分析DLL..."
python3 vol.py -f "$MEM_FILE" windows.dlllist > "$OUT_DIR/04_dlllist.txt"
python3 vol.py -f "$MEM_FILE" windows.ldrmodules > "$OUT_DIR/04_ldrmodules.txt"

# 步骤5：恶意软件检测
echo "[5/10] 检测恶意软件..."
python3 vol.py -f "$MEM_FILE" windows.malfind > "$OUT_DIR/05_malfind.txt"

# 步骤6：注册表分析
echo "[6/10] 分析注册表..."
python3 vol.py -f "$MEM_FILE" windows.registry.hivelist > "$OUT_DIR/06_hivelist.txt"

# 检查启动项
python3 vol.py -f "$MEM_FILE" windows.registry.printkey \
    --key "Software\Microsoft\Windows\CurrentVersion\Run" > "$OUT_DIR/06_run_key.txt" 2>&1

# 步骤7：凭证分析
echo "[7/10] 提取凭证..."
python3 vol.py -f "$MEM_FILE" windows.hashdump > "$OUT_DIR/07_hashdump.txt" 2>&1
python3 vol.py -f "$MEM_FILE" windows.mimikatz > "$OUT_DIR/07_mimikatz.txt" 2>&1

# 步骤8：服务分析
echo "[8/10] 分析服务..."
python3 vol.py -f "$MEM_FILE" windows.svcscan > "$OUT_DIR/08_svcscan.txt"

# 步骤9：驱动分析
echo "[9/10] 分析驱动..."
python3 vol.py -f "$MEM_FILE" windows.driverscan > "$OUT_DIR/09_driverscan.txt"
python3 vol.py -f "$MEM_FILE" windows.modscan > "$OUT_DIR/09_modscan.txt"

# 步骤10：其他
echo "[10/10] 其他分析..."
python3 vol.py -f "$MEM_FILE" windows.handles > "$OUT_DIR/10_handles.txt"
python3 vol.py -f "$MEM_FILE" windows.clipboard > "$OUT_DIR/10_clipboard.txt" 2>&1
python3 vol.py -f "$MEM_FILE" windows.envars > "$OUT_DIR/10_envars.txt"

echo "=" * 60
echo "[*] 分析完成！输出目录: $OUT_DIR"
echo "[*] 请查看以下关键文件："
echo "    - 02_hidden_processes.txt  (隐藏进程)"
echo "    - 03_netscan.txt           (网络连接)"
echo "    - 05_malfind.txt           (恶意软件检测)"
echo "    - 07_hashdump.txt          (凭证哈希)"
echo "    - 08_svcscan.txt           (可疑服务)"
```

```bash

# 编译

make

# 加载模块并转储内存
insmod lime.ko "path=/tmp/memory.lime format=lime"

# 格式选项: raw, lime, padded
insmod lime.ko "path=/tmp/memory.raw format=raw"

# 卸载模块
rmmod lime

# ========== fmem ==========

# 下载: https://github.com/NateBrune/fmem
# 加载模块
./run.sh

# 读取内存设备

# ========== /proc/kcore (仅供参考，不推荐) ==========

# 注意：kcore不是物理内存的完整镜像，有局限性
dd if=/proc/kcore of=/tmp/memory.elf bs=1M count=2048
```
