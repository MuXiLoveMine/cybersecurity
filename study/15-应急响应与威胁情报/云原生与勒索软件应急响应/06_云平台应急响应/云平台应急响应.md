### 6.1 云监控日志分析

```bash
# AWS CloudTrail - 查询最近的API调用
aws cloudtrail lookup-events \
  --start-time $(date -d '24 hours ago' +%Y-%m-%dT%H:%M:%S) \
  --end-time $(date +%Y-%m-%dT%H:%M:%S) \
  --query 'Events[*].[EventTime,Username,EventName,CloudTrailEvent]' \
  --output table

# 查找特定用户的操作
aws cloudtrail lookup-events \
  --lookup-attributes AttributeKey=Username,AttributeValue=attacker \
  --output json | jq '.Events[]'

# 查找所有 IAM 变更
aws cloudtrail lookup-events \
  --lookup-attributes AttributeKey=EventSource,AttributeValue=iam.amazonaws.com \
  --output json | jq '.Events[] | select(.EventName | test("Create|Delete|Attach|Put"))'

# 查找所有密钥操作
aws cloudtrail lookup-events \
  --lookup-attributes AttributeKey=EventSource,AttributeValue=sts.amazonaws.com \
  --output json | jq '.Events[] | select(.EventName == "AssumeRole")'

# Azure Activity Log
az monitor activity-log list --resource-group my-rg \
  --query "[?operationName.value=='Microsoft.Authorization/roleAssignments/write']"

# 阿里云ActionTrail
aliyun actiontrail LookupEvents \
  --EventName "CreateAccessKey" \
  --StartTime $(date -d '24 hours ago' -u +%Y-%m-%dT%H:%M:%SZ) \
  --EndTime $(date -u +%Y-%m-%dT%H:%M:%SZ)
```

### 6.2 VPC Flow Log分析

```bash
# AWS VPC Flow Log 查询
# 查找所有入站流量（被拒绝的）
aws logs start-query \
  --log-group-name "vpc-flow-logs" \
  --start-time $(date -d '1 hour ago' +%s) \
  --end-time $(date +%s) \
  --query-string '
    fields @timestamp, srcAddr, dstAddr, dstPort, action, protocol
    | filter action = "REJECT"
    | sort @timestamp desc
    | limit 100
  '

# 查找连接到已知恶意IP的流量
aws logs start-query \
  --log-group-name "vpc-flow-logs" \
  --start-time $(date -d '24 hours ago' +%s) \
  --end-time $(date +%s) \
  --query-string '
    fields @timestamp, srcAddr, dstAddr, dstPort, action
    | filter (dstAddr = "1.2.3.4" or dstAddr = "5.6.7.8")
    | sort @timestamp desc
  '

# 查找异常端口连接
aws logs start-query \
  --log-group-name "vpc-flow-logs" \
  --start-time $(date -d '24 hours ago' +%s) \
  --end-time $(date +%s) \
  --query-string '
    fields srcAddr, dstAddr, dstPort, action
    | filter dstPort not in (80, 443, 22, 53, 123)
    | filter action = "ACCEPT"
    | stats count() by dstPort, srcAddr
    | sort @count desc
    | limit 20
  '

# 查找数据外传（大量出站流量）
aws logs start-query \
  --log-group-name "vpc-flow-logs" \
  --start-time $(date -d '24 hours ago' +%s) \
  --end-time $(date +%s) \
  --query-string '
    fields srcAddr, dstAddr, bytes
    | filter action = "ACCEPT" and srcAddr like "10."
    | stats sum(bytes) as totalBytes by srcAddr, dstAddr
    | sort totalBytes desc
    | limit 20
  '
```

### 6.3 云API审计

```python
#!/usr/bin/env python3
"""云API审计工具 - 检测异常API调用"""
import boto3
from datetime import datetime, timedelta
from collections import defaultdict

class CloudAPIAuditor:
    def __init__(self, region="us-east-1"):
        self.cloudtrail = boto3.client('cloudtrail', region_name=region)
        self.iam = boto3.client('iam')
        self.ec2 = boto3.client('ec2', region_name=region)
        self.s3 = boto3.client('s3')
        self.alerts = []

    def check_privilege_escalation(self, hours_back=24):
        """检测权限提升行为"""
        # 已知的权限提升相关API调用
        privesc_events = [
            'CreateAccessKey',
            'CreateLoginProfile',
            'UpdateLoginProfile',
            'AttachRolePolicy',
            'PutRolePolicy',
            'AssumeRole',
            'PassRole',
            'CreatePolicy',
            'AddUserToGroup',
            'CreateGroup',
        ]
        
        start = datetime.now() - timedelta(hours=hours_back)
        
        for event_name in privesc_events:
            results = self.cloudtrail.lookup_events(
                LookupAttributes=[{
                    'AttributeKey': 'EventName',
                    'AttributeValue': event_name
                }],
                StartTime=start,
                EndTime=datetime.now()
            )
            
            for event in results.get('Events', []):
                self.alerts.append({
                    'severity': 'HIGH',
                    'type': 'privilege_escalation',
                    'detail': f"{event['Username']} 执行了 {event['EventName']}",
                    'time': event['EventTime']
                })

    def check_unauthorized_access_keys(self):
        """检测异常的AccessKey创建"""
        users = self.iam.list_users()
        for user in users['Users']:
            keys = self.iam.list_access_keys(UserName=user['UserName'])
            for key in keys['AccessKeyMetadata']:
                age = datetime.now(key['CreateDate'].replace(tzinfo=None)) - key['CreateDate'].replace(tzinfo=None)
                if key['Status'] == 'Active' and age.days > 90:
                    self.alerts.append({
                        'severity': 'MEDIUM',
                        'type': 'old_access_key',
                        'detail': f"用户 {user['UserName']} 有超过90天的活跃密钥",
                        'time': key['CreateDate']
                    })
                if key['Status'] == 'Active' and age.days < 1:
                    self.alerts.append({
                        'severity': 'HIGH',
                        'type': 'new_access_key',
                        'detail': f"用户 {user['UserName']} 新建了AccessKey，需确认是否合法",
                        'time': key['CreateDate']
                    })

    def check_security_group_changes(self, hours_back=24):
        """检测安全组变更"""
        start = datetime.now() - timedelta(hours=hours_back)
        sg_events = self.cloudtrail.lookup_events(
            LookupAttributes=[{
                'AttributeKey': 'EventName',
                'AttributeValue': 'AuthorizeSecurityGroupIngress'
            }],
            StartTime=start,
            EndTime=datetime.now()
        )
        
        for event in sg_events.get('Events', []):
            ct_event = eval(event['CloudTrailEvent'])
            if ct_event.get('requestParameters'):
                self.alerts.append({
                    'severity': 'HIGH',
                    'type': 'sg_rule_added',
                    'detail': f"{event['Username']} 添加了安全组入站规则",
                    'time': event['EventTime']
                })

    def check_s3_bucket_access(self):
        """检测S3桶的异常访问"""
        buckets = self.s3.list_buckets()
        for bucket in buckets['Buckets']:
            try:
                policy = self.s3.get_bucket_policy_status(Bucket=bucket['Name'])
                if policy['PolicyStatus']['isPublic']:
                    self.alerts.append({
                        'severity': 'CRITICAL',
                        'type': 'public_s3_bucket',
                        'detail': f"S3桶 {bucket['Name']} 是公开的",
                        'time': bucket['CreationDate']
                    })
            except Exception:
                pass

    def report(self):
        """生成报告"""
        print(f"\n{'='*60}")
        print("云API审计报告")
        print(f"{'='*60}")
        for alert in self.alerts:
            print(f"\n[{alert['severity']}] {alert['type']}")
            print(f"  {alert['detail']}")
            print(f"  时间: {alert['time']}")
        print(f"\n总计: {len(self.alerts)} 个告警")

# 使用
auditor = CloudAPIAuditor()
auditor.check_privilege_escalation(hours_back=48)
auditor.check_unauthorized_access_keys()
auditor.check_security_group_changes(hours_back=48)
auditor.check_s3_bucket_access()
auditor.report()
```
