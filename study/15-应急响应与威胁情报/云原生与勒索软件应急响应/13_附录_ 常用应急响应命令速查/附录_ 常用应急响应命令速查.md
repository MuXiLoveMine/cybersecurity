# 附录: 常用应急响应命令速查

```bash
# === 容器 ===
docker pause <id>              # 冻结容器
docker export <id> -o file.tar # 导出文件系统
docker commit <id> img:tag     # 创建快照镜像
docker inspect <id>            # 容器详情
docker top <id>                # 容器进程
docker logs <id>               # 容器日志
docker diff <id>               # 修改层差异
docker network disconnect      # 断开网络

# === Kubernetes ===
kubectl get pods -A            # 所有Pod
kubectl get events -A         # 集群事件
kubectl describe pod <name>   # Pod详情
kubectl logs <pod> -n <ns>    # Pod日志
kubectl exec <pod> -- <cmd>   # 在Pod中执行命令
kubectl cordon <node>         # 节点不可调度
kubectl drain <node>          # 驱逐节点

# === 勒索软件 ===
# 检查加密文件
find / -name "*.lockbit" -o -name "*.encrypted"
# 检查VSS删除
grep -i "shadow" /var/log/syslog
# 检查数据外传
tcpdump -i any -w capture.pcap 'tcp and not port 22'
# 检查勒索说明
find / -name "HOW_TO_DECRYPT*" -o -name "README*.txt"
```
