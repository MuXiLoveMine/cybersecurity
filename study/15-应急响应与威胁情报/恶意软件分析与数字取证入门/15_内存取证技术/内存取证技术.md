# 内存取证技术

### 6.1 内存取证基础

```
内存取证的重要性：

1. 易失性数据保存
   - 运行中的进程
   - 网络连接状态
   - 加载的模块和驱动
   - 注册表缓存
   - 解密的密钥和密码
   - 剪贴板内容

2. 反取证对抗
   - 无文件恶意软件检测
   - 进程注入检测
   - Rootkit检测
   - 隐藏进程发现

3. 证据完整性
   - 内存中难以篡改
   - 可检测hook和patch
   - 可还原运行时状态

内存获取方法：

方法              工具               优点               缺点
──────────────────────────────────────────────────────────
内核驱动获取      WinPmem            快速，完整          需要驱动
                  DumpIt
                  Memoryze

硬件获取          PCILeech           最可靠，不依赖OS    需要专用硬件
                  Quarksc

虚拟化获取        VMware内存导出      简单方便            仅限虚拟机
                  VirtualBox内存

崩溃转储          BlueScreenView      可分析崩溃          需要触发崩溃
                  WinDbg

休眠文件          Hiberfil.sys       可离线分析          可能不完整
                  Windows休眠文件

崩溃转储          MEMORY.DMP         完整内存            需要配置
```

### 6.2 Volatility框架详解

```python
#!/usr/bin/env python3
# Volatility 3 内存分析自动化脚本
# 自动化内存取证分析流程

import subprocess
import json
import os
from datetime import datetime

class VolatilityAutomator:
    """Volatility自动化分析器"""
    
    def __init__(self, memory_dump, volatility_path="vol"):
        self.memory_dump = memory_dump
        self.volatility_path = volatility_path
        self.results = {}
        
    def run_plugin(self, plugin_name, extra_args=""):
        """运行Volatility插件"""
        cmd = f'{self.volatility_path} -f "{self.memory_dump}" {plugin_name} {extra_args}'
        print(f"运行: {cmd}")
        
        result = subprocess.run(
            cmd, shell=True, capture_output=True, text=True, timeout=300
        )
        
        if result.returncode == 0:
            return result.stdout
        else:
            print(f"错误: {result.stderr}")
            return None
    
    def run_json_plugin(self, plugin_name, extra_args=""):
        """运行Volatility插件并获取JSON输出"""
        extra_args += " --output-format=json"
        output = self.run_plugin(plugin_name, extra_args)
        if output:
            try:
                return json.loads(output)
            except json.JSONDecodeError:
                return None
        return None
    
    def identify_os(self):
        """识别操作系统信息"""
        print("\n[1] 识别操作系统")
        result = self.run_plugin("banners.Banners")
        if result:
            self.results['os_info'] = result
            print(result[:500])
    
    def analyze_processes(self):
        """分析进程"""
        print("\n[2] 分析进程列表")
        
        # 获取进程列表
        procs = self.run_json_plugin("windows.pslist.PsList")
        if procs:
            self.results['processes'] = procs
            print(f"发现 {len(procs.get('rows', []))} 个进程")
            
            # 查找可疑进程
            suspicious_names = [
                'cmd.exe', 'powershell.exe', 'wscript.exe', 'cscript.exe',
                'mshta.exe', 'regsvr32.exe', 'rundll32.exe', 'certutil.exe',
                'bitsadmin.exe', 'msiexec.exe', 'wmic.exe',
                'unknown.exe', 'temp.exe', 'update.exe', 'svchost.exe (可疑PID)'
            ]
            
            for proc in procs.get('rows', []):
                name = proc[0] if isinstance(proc, list) else proc.get('Name', '')
                pid = proc[2] if isinstance(proc, list) else proc.get('PID', '')
                ppid = proc[3] if isinstance(proc, list) else proc.get('PPID', '')
                
                if any(s in str(name).lower() for s in ['temp', 'unknown', 'suspicious']):
                    print(f"  [!] 可疑进程: {name} (PID:{pid}, PPID:{ppid})")
    
    def analyze_network_connections(self):
        """分析网络连接"""
        print("\n[3] 分析网络连接")
        
        connections = self.run_json_plugin("windows.netstat.NetStat")
        if connections:
            self.results['network'] = connections
            print(f"发现 {len(connections.get('rows', []))} 个网络连接")
            
            for conn in connections.get('rows', []):
                if isinstance(conn, list):
                    proto = conn[0]
                    local = conn[1]
                    remote = conn[2]
                    state = conn[3]
                    pid = conn[4]
                    
                    # 标记ESTABLISHED的外部连接
                    if state == 'ESTABLISHED' and remote != '0.0.0.0:0':
                        print(f"  [!] 活动连接: {local} -> {remote} (PID:{pid})")
    
    def analyze_dlls(self):
        """分析加载的DLL"""
        print("\n[4] 分析加载的DLL")
        
        dlls = self.run_json_plugin("windows.dlllist.DllList")
        if dlls:
            self.results['dlls'] = dlls
            
            # 查找可疑DLL
            suspicious_dlls = [
                'inject', 'hook', 'patch', 'crack',
                'temp', 'appdata', 'programdata'
            ]
            
            for dll in dlls.get('rows', []):
                path = dll[3] if isinstance(dll, list) else dll.get('Path', '')
                
                if any(s in str(path).lower() for s in suspicious_dlls):
                    print(f"  [!] 可疑DLL: {path}")
    
    def analyze_handles(self):
        """分析进程句柄"""
        print("\n[5] 分析互斥体")
        
        # 查找互斥体
        handles = self.run_plugin("windows.handles.Handles --pid 0")
        if handles:
            self.results['handles'] = handles
    
    def analyze_malware(self):
        """恶意软件检测"""
        print("\n[6] 恶意软件检测")
        
        # 检测进程注入
        print("  检测进程注入...")
        malfind = self.run_json_plugin("windows.malfind.Malfind")
        if malfind:
            injections = malfind.get('rows', [])
            self.results['injections'] = injections
            if injections:
                print(f"  [!] 发现 {len(injections)} 个可能的注入区域")
                for inj in injections[:5]:
                    print(f"    {inj}")
            else:
                print("  未检测到注入")
        
        # 检测隐藏进程
        print("  检测隐藏进程...")
        psscan = self.run_json_plugin("windows.psscan.PsScan")
        if psscan:
            scanned_procs = set()
            for proc in psscan.get('rows', []):
                pid = proc[2] if isinstance(proc, list) else proc.get('PID', '')
                scanned_procs.add(pid)
            
            if self.results.get('processes'):
                listed_pids = set()
                for proc in self.results['processes'].get('rows', []):
                    pid = proc[2] if isinstance(proc, list) else proc.get('PID', '')
                    listed_pids.add(pid)
                
                hidden_pids = scanned_procs - listed_pids
                if hidden_pids:
                    print(f"  [!] 发现 {len(hidden_pids)} 个隐藏进程: {hidden_pids}")
                else:
                    print("  未检测到隐藏进程")
    
    def analyze_registry(self):
        """注册表分析"""
        print("\n[7] 注册表分析")
        
        # 检查启动项
        print("  检查启动项...")
        run_keys = self.run_plugin(
            "windows.registry.hivelist.HiveList --hive=SOFTWARE "
            "--key=Microsoft\\Windows\\CurrentVersion\\Run"
        )
        if run_keys:
            self.results['startup'] = run_keys
            print(f"  启动项信息已保存")
    
    def extract_credentials(self):
        """提取凭证"""
        print("\n[8] 凭证提取")
        
        # 提取LSASS进程内存
        print("  提取LSASS进程内存...")
        lsass = self.run_plugin("windows.lsalsassist.LsaLsass")
        if lsass:
            self.results['credentials'] = lsass
            print("  凭证信息已提取")
    
    def analyze_timelines(self):
        """时间线分析"""
        print("\n[9] 时间线分析")
        
        timeline = self.run_plugin("windows.timeliner.Timeliner")
        if timeline:
            self.results['timeline'] = timeline
            print("  时间线已生成")
    
    def run_full_analysis(self):
        """运行完整分析"""
        print("=" * 60)
        print("Volatility内存分析自动化报告")
        print(f"内存转储: {self.memory_dump}")
        print(f"分析时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        print("=" * 60)
        
        self.identify_os()
        self.analyze_processes()
        self.analyze_network_connections()
        self.analyze_dlls()
        self.analyze_handles()
        self.analyze_malware()
        self.analyze_registry()
        self.extract_credentials()
        self.analyze_timelines()
        
        self.generate_report()
    
    def generate_report(self):
        """生成最终报告"""
        print("\n" + "=" * 60)
        print("分析报告摘要")
        print("=" * 60)
        
        if self.results.get('processes'):
            proc_count = len(self.results['processes'].get('rows', []))
            print(f"进程数: {proc_count}")
        
        if self.results.get('network'):
            conn_count = len(self.results['network'].get('rows', []))
            print(f"网络连接数: {conn_count}")
        
        if self.results.get('injections'):
            inj_count = len(self.results['injections'])
            print(f"检测到注入: {inj_count}个")
        
        print(f"\n完整结果已保存")

# 使用示例
if __name__ == "__main__":
    analyzer = VolatilityAutomator("memory.dmp")
    analyzer.run_full_analysis()
```

### 6.4 内存中的恶意代码检测

```python
#!/usr/bin/env python3
# 内存恶意代码检测器
# 检测内存中的恶意软件特征

import re
import struct
import math
from collections import defaultdict

class MemoryMalwareDetector:
    """内存恶意代码检测器"""
    
    # 恶意软件代码特征
    MALWARE_PATTERNS = {
        'shellcode_x86': [
            # x86 shellcode特征
            b'\xfc\xe8\x89\x00\x00\x00\x60\x89\xe5\x31\xd2\x64\x8b\x52',
            b'\x64\xa1\x30\x00\x00\x00',  # PEB访问
            b'\xeb\x06\x68\x33\x32\x00\x00\x68\x77\x73\x32\x5f',
            b'\x68\xcb\x6c\x48\x00',  # 常见shellcode
        ],
        'api_hoook': [
            # API hook特征
            b'\x8b\xff\x55\x8b\xec\x83',  # 函数序言
            b'\xff\x25',  # JMP [addr]
            b'\xe8',  # CALL
            b'\xe9',  # JMP
        ],
        'rootkit': [
            # Rootkit特征
            b'\x8b\x35\x80\x74\x02\x08',  # SSDT hook
            b'\x8b\x4d\x08\x8b\x55\x0c',  # DKOM
        ],
        'injected_dll': [
            # 反射DLL注入特征
            b'4D\x5A',  # MZ头
            b'\x72\x73\x2e\x64\x6c\x6c',  # rs.dll
        ],
    }
    
    # 可疑API调用模式
    SUSPICIOUS_API_PATTERNS = [
        b'VirtualAllocEx',
        b'WriteProcessMemory',
        b'CreateRemoteThread',
        b'NtMapViewOfSection',
        b'QueueUserAPC',
        b'SetWindowsHookEx',
        b'GetAsyncKeyState',
        b'NtQueryInformationProcess',
        b'IsDebuggerPresent',
        b'CheckRemoteDebuggerPresent',
        b'ZwSetInformationThread',
    ]
    
    # C2通信模式
    C2_PATTERNS = [
        # HTTP C2 beacon模式
        b'GET /',
        b'POST /',
        b'User-Agent:',
        b'Cookie:',
        b'X-Session:',
        # 自定义协议特征
        b'\x00\x00\x00\x00\x00\x00\x00\x00',  # 填充
        b'\xde\xad\xbe\xef',  # 魔数
    ]
    
    def __init__(self, memory_dump):
        with open(memory_dump, 'rb') as f:
            self.data = f.read()
        self.findings = []
    
    def calculate_entropy(self, data):
        """计算数据块熵值"""
        if not data:
            return 0
        freq = defaultdict(int)
        for b in data:
            freq[b] += 1
        entropy = 0
        for count in freq.values():
            p = count / len(data)
            entropy -= p * math.log2(p)
        return entropy
    
    def scan_patterns(self):
        """扫描恶意模式"""
        print("扫描恶意代码模式...")
        
        for category, patterns in self.MALWARE_PATTERNS.items():
            for pattern in patterns:
                positions = self.find_all(pattern)
                if positions:
                    self.findings.append({
                        'type': category,
                        'pattern': pattern.hex(),
                        'positions': positions,
                        'count': len(positions),
                    })
                    print(f"  [!] 发现 {category} 特征: {len(positions)} 处")
    
    def scan_suspicious_apis(self):
        """扫描可疑API"""
        print("扫描可疑API引用...")
        
        for api in self.SUSPICIOUS_API_PATTERNS:
            positions = self.find_all(api)
            if positions:
                self.findings.append({
                    'type': 'suspicious_api',
                    'pattern': api.decode('ascii', errors='ignore'),
                    'positions': positions,
                    'count': len(positions),
                })
                print(f"  [!] 发现API {api.decode()}: {len(positions)} 处")
    
    def scan_c2_patterns(self):
        """扫描C2通信模式"""
        print("扫描C2通信模式...")
        
        for pattern in self.C2_PATTERNS:
            positions = self.find_all(pattern)
            if positions:
                self.findings.append({
                    'type': 'c2_pattern',
                    'pattern': pattern.decode('ascii', errors='ignore'),
                    'positions': positions,
                    'count': len(positions),
                })
    
    def find_all(self, pattern):
        """查找所有匹配位置"""
        positions = []
        start = 0
        while True:
            pos = self.data.find(pattern, start)
            if pos == -1:
                break
            positions.append(hex(pos))
            start = pos + 1
        return positions[:100]  # 限制结果数量
    
    def scan_high_entropy_regions(self):
        """扫描高熵值区域（可能的加密/压缩数据）"""
        print("扫描高熵值区域...")
        
        block_size = 4096
        high_entropy_regions = []
        
        for offset in range(0, len(self.data), block_size):
            block = self.data[offset:offset + block_size]
            entropy = self.calculate_entropy(block)
            
            if entropy > 7.5:  # 高熵值阈值
                high_entropy_regions.append({
                    'offset': hex(offset),
                    'size': len(block),
                    'entropy': round(entropy, 4),
                })
        
        if high_entropy_regions:
            self.findings.append({
                'type': 'high_entropy',
                'regions': high_entropy_regions[:50],
                'count': len(high_entropy_regions),
            })
            print(f"  [!] 发现 {len(high_entropy_regions)} 个高熵值区域")
    
    def scan_pe_headers_in_memory(self):
        """扫描内存中的PE头（注入的DLL/EXE）"""
        print("扫描内存中的PE头...")
        
        mz_pattern = b'\x4D\x5A'  # MZ
        pe_positions = []
        start = 0
        
        while True:
            pos = self.data.find(mz_pattern, start)
            if pos == -1:
                break
            
            # 验证PE头
            if pos + 64 < len(self.data):
                e_lfanew = struct.unpack('<I', self.data[pos + 0x3C:pos + 0x40])[0]
                if 0 < e_lfanew < 0x1000 and pos + e_lfanew + 4 <= len(self.data):
                    pe_sig = self.data[pos + e_lfanew:pos + e_lfanew + 4]
                    if pe_sig == b'PE\x00\x00':
                        pe_positions.append(hex(pos))
            
            start = pos + 1
        
        if pe_positions:
            self.findings.append({
                'type': 'pe_in_memory',
                'positions': pe_positions[:50],
                'count': len(pe_positions),
            })
            print(f"  [!] 在内存中发现 {len(pe_positions)} 个PE文件")
    
    def generate_report(self):
        """生成报告"""
        print("\n" + "=" * 60)
        print("内存恶意代码检测报告")
        print("=" * 60)
        
        if not self.findings:
            print("未检测到恶意代码特征")
            return
        
        print(f"发现 {len(self.findings)} 类异常\n")
        
        for finding in self.findings:
            ftype = finding['type']
            count = finding.get('count', 0)
            print(f"【{ftype}】 ({count}个)")
            if 'positions' in finding:
                for pos in finding['positions'][:5]:
                    print(f"  位置: {pos}")
                if len(finding['positions']) > 5:
                    print(f"  ... 共{len(finding['positions'])}个")
            elif 'regions' in finding:
                for region in finding['regions'][:5]:
                    print(f"  偏移: {region['offset']}, "
                          f"熵值: {region['entropy']}")
            print()
    
    def run(self):
        """运行完整扫描"""
        self.scan_patterns()
        self.scan_suspicious_apis()
        self.scan_c2_patterns()
        self.scan_high_entropy_regions()
        self.scan_pe_headers_in_memory()
        self.generate_report()

# 使用示例
if __name__ == "__main__":
    detector = MemoryMalwareDetector("memory.dmp")
    detector.run()
```
