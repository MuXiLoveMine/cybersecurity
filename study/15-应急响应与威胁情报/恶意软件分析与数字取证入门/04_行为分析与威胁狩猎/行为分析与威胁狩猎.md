### 4.2 攻击指标提取

```python
#!/usr/bin/env python3
# IOC提取与标准化工具
# 从恶意软件分析结果中提取入侵指标

import re
import json
import hashlib
import ipaddress
from datetime import datetime

class IOCExtractor:
    """入侵指标提取器"""
    
    def __init__(self):
        self.iocs = {
            'file': [],
            'network': [],
            'registry': [],
            'process': [],
            'mutex': [],
        }
    
    def add_file_ioc(self, path, md5=None, sha1=None, sha256=None, 
                     size=None, filetype=None, description=""):
        """添加文件IOC"""
        ioc = {
            'type': 'file',
            'value': path,
            'md5': md5,
            'sha1': sha1,
            'sha256': sha256,
            'size': size,
            'filetype': filetype,
            'description': description,
            'first_seen': datetime.now().isoformat(),
        }
        self.iocs['file'].append(ioc)
    
    def add_network_ioc(self, indicator_type, value, port=None, 
                        protocol=None, description=""):
        """添加网络IOC"""
        ioc = {
            'type': indicator_type,  # ip, domain, url, email
            'value': value,
            'port': port,
            'protocol': protocol,
            'description': description,
            'first_seen': datetime.now().isoformat(),
        }
        self.iocs['network'].append(ioc)
    
    def add_registry_ioc(self, key, value_name=None, value_data=None, 
                         action="modified", description=""):
        """添加注册表IOC"""
        ioc = {
            'type': 'registry',
            'key': key,
            'value_name': value_name,
            'value_data': value_data,
            'action': action,
            'description': description,
            'first_seen': datetime.now().isoformat(),
        }
        self.iocs['registry'].append(ioc)
    
    def add_process_ioc(self, name, command_line=None, parent_process=None, 
                        description=""):
        """添加进程IOC"""
        ioc = {
            'type': 'process',
            'name': name,
            'command_line': command_line,
            'parent_process': parent_process,
            'description': description,
            'first_seen': datetime.now().isoformat(),
        }
        self.iocs['process'].append(ioc)
    
    def add_mutex_ioc(self, name, description=""):
        """添加互斥体IOC"""
        ioc = {
            'type': 'mutex',
            'name': name,
            'description': description,
            'first_seen': datetime.now().isoformat(),
        }
        self.iocs['mutex'].append(ioc)
    
    def validate_ip(self, ip_str):
        """验证IP地址"""
        try:
            ipaddress.ip_address(ip_str)
            return True
        except ValueError:
            return False
    
    def validate_domain(self, domain):
        """验证域名格式"""
        pattern = re.compile(
            r'^(?:[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?\.)*'
            r'[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?$'
        )
        return bool(pattern.match(domain)) and len(domain) <= 253
    
    def validate_hash(self, hash_str, hash_type):
        """验证哈希值"""
        lengths = {'md5': 32, 'sha1': 40, 'sha256': 64}
        if hash_type not in lengths:
            return False
        if len(hash_str) != lengths[hash_type]:
            return False
        return all(c in '0123456789abcdef' for c in hash_str.lower())
    
    def export_stix(self):
        """导出STIX格式"""
        # 简化的STIX 2.1输出
        stix_objects = []
        
        for file_ioc in self.iocs['file']:
            if file_ioc.get('sha256'):
                stix_objects.append({
                    "type": "indicator",
                    "spec_version": "2.1",
                    "id": f"indicator--{hashlib.sha256(file_ioc['sha256'].encode()).hexdigest()[:36]}",
                    "created": file_ioc['first_seen'],
                    "modified": file_ioc['first_seen'],
                    "pattern": f"[file:hashes.'SHA-256' = '{file_ioc['sha256']}']",
                    "pattern_type": "stix",
                    "name": file_ioc.get('description', 'File IOC'),
                    "valid_from": file_ioc['first_seen'],
                })
        
        for net_ioc in self.iocs['network']:
            pattern = None
            if net_ioc['type'] == 'ip':
                pattern = f"[network-traffic:dst_ref.value = '{net_ioc['value']}']"
            elif net_ioc['type'] == 'domain':
                pattern = f"[domain-name:value = '{net_ioc['value']}']"
            elif net_ioc['type'] == 'url':
                pattern = f"[url:value = '{net_ioc['value']}']"
            
            if pattern:
                stix_objects.append({
                    "type": "indicator",
                    "spec_version": "2.1",
                    "id": f"indicator--{hashlib.sha256(pattern.encode()).hexdigest()[:36]}",
                    "created": net_ioc['first_seen'],
                    "modified": net_ioc['first_seen'],
                    "pattern": pattern,
                    "pattern_type": "stix",
                    "name": net_ioc.get('description', 'Network IOC'),
                    "valid_from": net_ioc['first_seen'],
                })
        
        return {
            "type": "bundle",
            "id": f"bundle--{hashlib.sha256(str(datetime.now()).encode()).hexdigest()[:36]}",
            "objects": stix_objects
        }
    
    def export_csv(self):
        """导出CSV格式"""
        csv_lines = ["type,value,description,first_seen"]
        
        for category in self.iocs.values():
            for ioc in category:
                if ioc['type'] in ['ip', 'domain', 'url']:
                    csv_lines.append(
                        f"network,{ioc['value']},{ioc.get('description','')},{ioc['first_seen']}"
                    )
                elif ioc['type'] == 'file':
                    csv_lines.append(
                        f"file,{ioc.get('sha256', ioc.get('md5', ioc['value']))},"
                        f"{ioc.get('description','')},{ioc['first_seen']}"
                    )
                elif ioc['type'] == 'registry':
                    csv_lines.append(
                        f"registry,{ioc['key']},{ioc.get('description','')},{ioc['first_seen']}"
                    )
                elif ioc['type'] == 'mutex':
                    csv_lines.append(
                        f"mutex,{ioc['name']},{ioc.get('description','')},{ioc['first_seen']}"
                    )
        
        return '\n'.join(csv_lines)
    
    def print_summary(self):
        """打印IOC摘要"""
        print("\n" + "=" * 60)
        print("入侵指标 (IOC) 摘要")
        print("=" * 60)
        
        for category, iocs in self.iocs.items():
            if iocs:
                print(f"\n【{category.upper()}】 ({len(iocs)}个)")
                for ioc in iocs[:10]:
                    if category == 'network':
                        print(f"  {ioc['type']}: {ioc['value']}")
                    elif category == 'file':
                        print(f"  {ioc['value']} (SHA256: {ioc.get('sha256', 'N/A')[:16]}...)")
                    elif category == 'registry':
                        print(f"  {ioc['key']}\\{ioc.get('value_name', '')}")
                    elif category == 'process':
                        print(f"  {ioc['name']} ({ioc.get('command_line', '')[:50]})")
                    elif category == 'mutex':
                        print(f"  {ioc['name']}")
                if len(iocs) > 10:
                    print(f"  ... 共{len(iocs)}个")

# 使用示例
if __name__ == "__main__":
    extractor = IOCExtractor()
    
    # 添加文件IOC
    extractor.add_file_ioc(
        path="C:\\Users\\Public\\malware.exe",
        sha256="a" * 64,
        size=45056,
        filetype="PE32",
        description="恶意软件主程序"
    )
    
    # 添加网络IOC
    extractor.add_network_ioc("ip", "192.168.1.100", port=443, 
                               protocol="HTTPS", description="C2服务器")
    extractor.add_network_ioc("domain", "evil.example.com", 
                               description="C2域名")
    extractor.add_network_ioc("url", "http://evil.example.com/payload.dll", 
                             description="载荷下载URL")
    
    # 添加注册表IOC
    extractor.add_registry_ioc(
        "HKLM\\SOFTWARE\\Microsoft\\Windows\\CurrentVersion\\Run\\Updater",
        "Updater", "C:\\Users\\Public\\malware.exe",
        "created", "启动项持久化"
    )
    
    # 添加进程IOC
    extractor.add_process_ioc("cmd.exe", 
                              "/c powershell -enc SGVsbG8=",
                              "malware.exe", "命令执行")
    
    # 添加互斥体IOC
    extractor.add_mutex_ioc("Global\\MyMalwareMutex", "互斥体检测")
    
    extractor.print_summary()
    
    # 导出STIX
    stix = extractor.export_stix()
    print(f"\nSTIX对象数: {len(stix['objects'])}")
```
