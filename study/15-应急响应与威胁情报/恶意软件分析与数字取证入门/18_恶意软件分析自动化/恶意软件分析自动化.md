# 恶意软件分析自动化

### 8.1 自动化分析管道

```python
#!/usr/bin/env python3
# 恶意软件自动化分析管道
# 整合静态分析、动态分析和报告生成

import os
import json
import hashlib
import subprocess
import time
from datetime import datetime
from pathlib import Path

class MalwareAnalysisPipeline:
    """恶意软件自动化分析管道"""
    
    def __init__(self, sample_path, output_dir="./analysis_results"):
        self.sample_path = sample_path
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(parents=True, exist_ok=True)
        
        # 计算文件哈希
        self.sample_data = open(sample_path, 'rb').read()
        self.hashes = {
            'md5': hashlib.md5(self.sample_data).hexdigest(),
            'sha1': hashlib.sha1(self.sample_data).hexdigest(),
            'sha256': hashlib.sha256(self.sample_data).hexdigest(),
        }
        
        self.results = {
            'sample': os.path.basename(sample_path),
            'hashes': self.hashes,
            'size': len(self.sample_data),
            'analysis_time': datetime.now().isoformat(),
            'stages': {}
        }
    
    def stage1_static_analysis(self):
        """阶段1: 静态分析"""
        print("[阶段1] 静态分析")
        stage_results = {}
        
        # 1.1 文件类型识别
        stage_results['file_type'] = self.detect_file_type()
        
        # 1.2 字符串提取
        stage_results['strings'] = self.extract_strings()
        
        # 1.3 PE结构分析（如果是PE文件）
        if stage_results['file_type'].get('type', '').startswith('PE'):
            stage_results['pe_info'] = self.analyze_pe()
        
        # 1.4 加壳检测
        stage_results['packed'] = self.detect_packing()
        
        # 1.5 YARA规则扫描
        stage_results['yara'] = self.scan_yara()
        
        self.results['stages']['static'] = stage_results
    
    def detect_file_type(self):
        """检测文件类型"""
        # 检查文件魔数
        magic_numbers = {
            b'\x4D\x5A': 'PE (Windows可执行文件)',
            b'\x7F\x45\x4C\x46': 'ELF (Linux可执行文件)',
            b'\xFE\xED\xFA\xCE': 'Mach-O (32位)',
            b'\xFE\xED\xFA\xCF': 'Mach-O (64位)',
            b'\xCA\xFE\xBA\xBE': 'Java类文件',
            b'\x50\x4B\x03\x04': 'ZIP/JAR/APK',
            b'\x25\x50\x44\x46': 'PDF',
            b'\xD0\xCF\x11\xE0': 'OLE (Office文档)',
            b'\x52\x61\x72\x21': 'RAR压缩包',
            b'\x1F\x8B': 'GZIP压缩',
        }
        
        file_type = '未知'
        for magic, description in magic_numbers.items():
            if self.sample_data[:len(magic)] == magic:
                file_type = description
                break
        
        return {
            'type': file_type,
            'magic_bytes': self.sample_data[:8].hex(),
            'size': len(self.sample_data),
        }
    
    def extract_strings(self, min_length=4):
        """提取字符串"""
        import re
        strings = []
        
        # ASCII字符串
        for match in re.finditer(rb'[\x20-\x7e]{4,}', self.sample_data):
            strings.append({
                'type': 'ASCII',
                'value': match.group().decode('ascii'),
                'offset': hex(match.start()),
            })
        
        # Unicode字符串
        for match in re.finditer(rb'(?:[\x20-\x7e]\x00){4,}', self.sample_data):
            strings.append({
                'type': 'Unicode',
                'value': match.group().decode('utf-16-le'),
                'offset': hex(match.start()),
            })
        
        # 过滤有意义的字符串
        interesting = []
        for s in strings:
            value = s['value']
            if any(keyword in value.lower() for keyword in [
                'http', 'https', 'ftp', 'cmd', 'powershell',
                'create', 'open', 'write', 'virtual',
                'socket', 'connect', 'internet', 'url',
                'password', 'key', 'encrypt', 'decrypt',
                'mutex', 'event', 'service', 'registry',
                'kill', 'delete', 'download', 'upload',
            ]):
                interesting.append(s)
        
        return {
            'total': len(strings),
            'interesting': interesting[:50],
        }
    
    def analyze_pe(self):
        """分析PE结构"""
        try:
            import pefile
            pe = pefile.PE(data=self.sample_data)
            
            return {
                'entry_point': hex(pe.OPTIONAL_HEADER.AddressOfEntryPoint),
                'image_base': hex(pe.OPTIONAL_HEADER.ImageBase),
                'sections': [
                    {
                        'name': s.Name.decode('utf-8', errors='ignore').strip('\x00'),
                        'virtual_address': hex(s.VirtualAddress),
                        'virtual_size': hex(s.Misc_VirtualSize),
                        'characteristics': hex(s.Characteristics),
                    }
                    for s in pe.sections
                ],
                'imports': self.extract_imports(pe),
                'has_exports': hasattr(pe, 'DIRECTORY_ENTRY_EXPORT'),
                'has_resources': hasattr(pe, 'DIRECTORY_ENTRY_RESOURCE'),
            }
        except Exception as e:
            return {'error': str(e)}
    
    def extract_imports(self, pe):
        """提取导入表"""
        imports = {}
        if hasattr(pe, 'DIRECTORY_ENTRY_IMPORT'):
            for entry in pe.DIRECTORY_ENTRY_IMPORT:
                dll = entry.dll.decode('utf-8', errors='ignore')
                imports[dll] = [
                    func.name.decode('utf-8', errors='ignore') if func.name else f'#{func.ordinal}'
                    for func in entry.imports
                ]
        return imports
    
    def detect_packing(self):
        """检测加壳"""
        packer_indicators = []
        
        # 检查节区名
        packer_sections = [
            b'UPX0', b'UPX1', b'.aspack', b'.adata',
            b'.vmp0', b'.vmp1', b'.MPRESS', b'.enigma',
            b'PEC2', b'.nsp0', b'.nsp1', b'.themida',
        ]
        
        for section in packer_sections:
            if section in self.sample_data:
                packer = section.decode('ascii', errors='ignore').strip('\x00')
                packer_indicators.append(f"检测到加壳器: {packer}")
        
        # 检查熵值
        import math
        from collections import Counter
        
        freq = Counter(self.sample_data)
        entropy = -sum((count / len(self.sample_data)) * 
                       math.log2(count / len(self.sample_data)) 
                       for count in freq.values())
        
        if entropy > 7.0:
            packer_indicators.append(f"文件熵值过高: {entropy:.2f}")
        
        return {
            'packed': len(packer_indicators) > 0,
            'indicators': packer_indicators,
            'entropy': round(entropy, 4),
        }
    
    def scan_yara(self):
        """YARA规则扫描"""
        # 模拟YARA扫描
        yara_rules = {
            'suspicious_api': [
                'VirtualAllocEx', 'WriteProcessMemory', 'CreateRemoteThread',
            ],
            'network_api': [
                'WSAStartup', 'InternetOpen', 'HttpSendRequest',
            ],
            'crypto_api': [
                'CryptAcquireContext', 'BCryptOpenAlgorithm',
            ],
            'anti_debug': [
                'IsDebuggerPresent', 'CheckRemoteDebuggerPresent',
            ],
        }
        
        matches = []
        for rule_name, patterns in yara_rules.items():
            for pattern in patterns:
                if pattern.encode() in self.sample_data:
                    matches.append({
                        'rule': rule_name,
                        'pattern': pattern,
                    })
        
        return {
            'matches': matches,
            'total': len(matches),
        }
    
    def stage2_dynamic_analysis(self):
        """阶段2: 动态分析"""
        print("[阶段2] 动态分析")
        stage_results = {}
        
        # 2.1 沙箱分析（模拟）
        stage_results['sandbox'] = self.run_sandbox()
        
        # 2.2 API监控（模拟）
        stage_results['api_monitor'] = self.monitor_apis()
        
        self.results['stages']['dynamic'] = stage_results
    
    def run_sandbox(self):
        """运行沙箱分析（模拟）"""
        return {
            'status': 'completed',
            'duration': '60s',
            'processes_created': [
                {'name': 'malware.exe', 'pid': 1234, 'parent': 'explorer.exe'},
                {'name': 'cmd.exe', 'pid': 1235, 'parent': 'malware.exe'},
            ],
            'files_created': [
                'C:\\Users\\Public\\temp.dll',
                'C:\\ProgramData\\update.exe',
            ],
            'registry_modified': [
                {'key': 'HKLM\\...\\Run\\Update', 'value': 'C:\\ProgramData\\update.exe'},
            ],
            'network_activity': [
                {'protocol': 'HTTPS', 'destination': 'evil.example.com:443'},
            ],
        }
    
    def monitor_apis(self):
        """API监控结果（模拟）"""
        return {
            'suspicious_calls': [
                {'api': 'VirtualAllocEx', 'count': 5},
                {'api': 'WriteProcessMemory', 'count': 3},
                {'api': 'CreateRemoteThread', 'count': 1},
            ],
        }
    
    def stage3_ioc_extraction(self):
        """阶段3: IOC提取"""
        print("[阶段3] IOC提取")
        iocs = {
            'files': [],
            'network': [],
            'registry': [],
        }
        
        # 从静态分析结果提取IOC
        static = self.results['stages'].get('static', {})
        
        # 从字符串提取网络IOC
        if 'strings' in static:
            for s in static['strings'].get('interesting', []):
                value = s['value']
                if value.startswith('http'):
                    iocs['network'].append({
                        'type': 'url',
                        'value': value,
                    })
        
        # 从动态分析结果提取IOC
        dynamic = self.results['stages'].get('dynamic', {})
        
        if 'sandbox' in dynamic:
            sandbox = dynamic['sandbox']
            
            for file in sandbox.get('files_created', []):
                iocs['files'].append(file)
            
            for reg in sandbox.get('registry_modified', []):
                iocs['registry'].append(reg)
            
            for net in sandbox.get('network_activity', []):
                iocs['network'].append({
                    'type': 'connection',
                    'destination': net['destination'],
                    'protocol': net['protocol'],
                })
        
        self.results['iocs'] = iocs
    
    def stage4_report_generation(self):
        """阶段4: 生成报告"""
        print("[阶段4] 生成报告")
        
        report = self.generate_json_report()
        report_file = self.output_dir / f"report_{self.hashes['sha256'][:16]}.json"
        
        with open(report_file, 'w', encoding='utf-8') as f:
            json.dump(report, f, indent=2, ensure_ascii=False)
        
        # 生成文本报告
        text_report = self.generate_text_report()
        text_file = self.output_dir / f"report_{self.hashes['sha256'][:16]}.txt"
        
        with open(text_file, 'w', encoding='utf-8') as f:
            f.write(text_report)
        
        print(f"报告已保存: {report_file}")
        print(f"文本报告: {text_file}")
    
    def generate_json_report(self):
        """生成JSON格式报告"""
        return self.results
    
    def generate_text_report(self):
        """生成文本格式报告"""
        report = "=" * 60 + "\n"
        report += "恶意软件分析报告\n"
        report += "=" * 60 + "\n\n"
        
        report += f"文件名: {self.results['sample']}\n"
        report += f"大小: {self.results['size']} bytes\n"
        report += f"MD5: {self.hashes['md5']}\n"
        report += f"SHA1: {self.hashes['sha1']}\n"
        report += f"SHA256: {self.hashes['sha256']}\n"
        report += f"分析时间: {self.results['analysis_time']}\n\n"
        
        # 静态分析结果
        static = self.results['stages'].get('static', {})
        report += "【静态分析】\n"
        report += f"  文件类型: {static.get('file_type', {}).get('type', 'N/A')}\n"
        report += f"  字符串数: {static.get('strings', {}).get('total', 0)}\n"
        report += f"  加壳: {static.get('packed', {}).get('packed', False)}\n"
        report += f"  YARA匹配: {static.get('yara', {}).get('total', 0)}\n\n"
        
        # 动态分析结果
        dynamic = self.results['stages'].get('dynamic', {})
        report += "【动态分析】\n"
        sandbox = dynamic.get('sandbox', {})
        report += f"  创建进程数: {len(sandbox.get('processes_created', []))}\n"
        report += f"  创建文件数: {len(sandbox.get('files_created', []))}\n"
        report += f"  修改注册表: {len(sandbox.get('registry_modified', []))}\n"
        report += f"  网络活动: {len(sandbox.get('network_activity', []))}\n\n"
        
        # IOC
        iocs = self.results.get('iocs', {})
        report += "【入侵指标 (IOC)】\n"
        report += f"  文件IOC: {len(iocs.get('files', []))}\n"
        report += f"  网络IOC: {len(iocs.get('network', []))}\n"
        report += f"  注册表IOC: {len(iocs.get('registry', []))}\n\n"
        
        return report
    
    def run(self):
        """运行完整分析管道"""
        print("=" * 60)
        print("恶意软件自动化分析管道")
        print("=" * 60)
        print(f"样本: {self.sample_path}")
        print(f"SHA256: {self.hashes['sha256']}")
        print()
        
        start_time = time.time()
        
        self.stage1_static_analysis()
        self.stage2_dynamic_analysis()
        self.stage3_ioc_extraction()
        self.stage4_report_generation()
        
        elapsed = time.time() - start_time
        print(f"\n分析完成，耗时: {elapsed:.2f}秒")

# 使用示例
if __name__ == "__main__":
    pipeline = MalwareAnalysisPipeline("malware_sample.exe")
    pipeline.run()
```

### 8.3 YARA规则引擎

```
YARA规则编写指南：

规则结构：
  rule 规则名 {
      meta:
          description = "描述"
          author = "作者"
          date = "日期"
      
      strings:
          $字符串 = "字符串内容"
          $十六进制 = { AA BB CC DD }
          $正则 = /正则表达式/ nocase
      
      condition:
          $字符串 and $十六进制
  }

条件运算符：
  and, or, not         逻辑运算
  any of ($str*)        匹配任意
  all of ($str*)        匹配全部
  $str at offset        指定偏移
  $str in (start..end)  范围匹配
  filesize < 100KB      文件大小
  pe.entry_point       PE入口点
  pe.imphash()         导入哈希
```

3. 应用场景
