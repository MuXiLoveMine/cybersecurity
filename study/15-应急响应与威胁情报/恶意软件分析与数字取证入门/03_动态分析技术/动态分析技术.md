```python
#!/usr/bin/env python3
# Cuckoo沙箱自定义分析模块
# 用于扩展Cuckoo的恶意软件分析能力

from cuckoo.processing.signature import Signature

class CustomMalwareSignatures(Signature):

            'patterns': [

                'CheckRemoteDebuggerPresent',
                'NtQueryInformationProcess',
                'OutputDebugString',
                'FindWindow',
            ],
            'severity': 2,
            'category': 'anti_analysis'
        },
        # 检测持久化
        {
            'name': 'persistence_mechanism',
            'description': '检测到持久化机制',
            'patterns': [
                'RegSetValue.*Run',
                'CreateService',
                'schtasks',
                'schtask',
                'at ',
                'WMI Event Subscription',
            ],
            'severity': 4,
            'category': 'persistence'
        },
    ]
    
    def run(self):
        """执行签名检测"""
        results = []
        
        # 检查API调用
        if 'behavior' in self.results:
            for process in self.results['behavior'].get('processes', []):
                for call in process.get('calls', []):
                    api_name = call.get('api', '')
                    
                    for indicator in self.indicators:
                        for pattern in indicator['patterns']:
                            if pattern.lower() in api_name.lower():
                                results.append({
                                    'indicator': indicator['name'],
                                    'description': indicator['description'],
                                    'api': api_name,
                                    'severity': indicator['severity'],
                                    'category': indicator['category'],
                                })
        
        # 检查网络通信
        if 'network' in self.results:
            for domain in self.results['network'].get('domains', []):
                # 检查DGA域名
                if self.is_dga_domain(domain['domain']):
                    results.append({
                        'indicator': 'c2_communication',
                        'description': f'检测到疑似DGA域名: {domain["domain"]}',
                        'severity': 4,
                        'category': 'c2',
                    })
        
        # 检查文件系统操作
        if 'behavior' in self.results:
            for summary in self.results['behavior'].get('summary', {}).values():
                for entry in summary:
                    for indicator in self.indicators:
                        for pattern in indicator['patterns']:
                            if pattern.lower() in str(entry).lower():
                                results.append({
                                    'indicator': indicator['name'],
                                    'description': indicator['description'],
                                    'evidence': entry,
                                    'severity': indicator['severity'],
                                    'category': indicator['category'],
                                })
        
        if results:
            self.severity = max(r['severity'] for r in results)
            for result in results:
                self.add_match(result['category'], result)
        
        return results
    
    def is_dga_domain(self, domain):
        """简单的DGA域名检测"""
        # 检查域名长度和字符分布
        parts = domain.split('.')
        if len(parts) < 2:
            return False
        sld = parts[-2]
        # DGA域名通常较长且随机
        if len(sld) > 12:
            vowels = sum(1 for c in sld if c.lower() in 'aeiou')
            consonants = len(sld) - vowels
            # 元音辅音比例异常
            if vowels == 0 or (consonants > 0 and vowels / consonants < 0.2):
                return True
        return False
```

### 3.3 API监控与Hook

```python
#!/usr/bin/env python3
# API监控工具 - 使用Frida进行动态API监控
# 用于监控恶意软件的API调用行为

import frida
import sys
import json
from datetime import datetime

class APIMonitor:
    """API调用监控器"""
    
    # JavaScript Hook脚本
    HOOK_SCRIPT = """
    // API监控脚本
    var logEntries = [];
    
    // Hook进程相关API
    var CreateProcessW = Module.findExportByName('kernel32.dll', 'CreateProcessW');
    if (CreateProcessW) {
        Interceptor.attach(CreateProcessW, {
            onEnter: function(args) {
                this.cmdline = args[1].readUtf16String();
                logEntry('CreateProcessW', { commandLine: this.cmdline });
            }
        });
    }
    
    // Hook文件相关API
    var CreateFileW = Module.findExportByName('kernel32.dll', 'CreateFileW');
    if (CreateFileW) {
        Interceptor.attach(CreateFileW, {
            onEnter: function(args) {
                var filename = args[0].readUtf16String();
                if (filename) {
                    logEntry('CreateFileW', { filename: filename });
                }
            }
        });
    }
    
    // Hook注册表API
    var RegSetValueExW = Module.findExportByName('advapi32.dll', 'RegSetValueExW');
    if (RegSetValueExW) {
        Interceptor.attach(RegSetValueExW, {
            onEnter: function(args) {
                var valueName = args[1].readUtf16String();
                var data = args[5].readUtf16String();
                logEntry('RegSetValueExW', { valueName: valueName, data: data });
            }
        });
    }
    
    // Hook网络API
    var InternetOpenW = Module.findExportByName('wininet.dll', 'InternetOpenW');
    if (InternetOpenW) {
        Interceptor.attach(InternetOpenW, {
            onEnter: function(args) {
                var agent = args[0].readUtf16String();
                logEntry('InternetOpenW', { userAgent: agent });
            }
        });
    }
    
    var socket = Module.findExportByName('ws2_32.dll', 'socket');
    if (socket) {
        Interceptor.attach(socket, {
            onEnter: function(args) {
                logEntry('socket', { 
                    domain: args[0], 
                    type: args[1], 
                    protocol: args[2] 
                });
            }
        });
    }
    
    var connect = Module.findExportByName('ws2_32.dll', 'connect');
    if (connect) {
        Interceptor.attach(connect, {
            onEnter: function(args) {
                var sockaddr = args[1];
                var family = sockaddr.readU16();
                if (family === 2) { // AF_INET
                    var port = (sockaddr.add(2).readU8() << 8) | sockaddr.add(3).readU8();
                    var ip = sockaddr.add(4).readU8() + '.' +
                             sockaddr.add(5).readU8() + '.' +
                             sockaddr.add(6).readU8() + '.' +
                             sockaddr.add(7).readU8();
                    logEntry('connect', { ip: ip, port: port });
                }
            }
        });
    }
    
    // Hook内存操作API
    var VirtualAllocEx = Module.findExportByName('kernel32.dll', 'VirtualAllocEx');
    if (VirtualAllocEx) {
        Interceptor.attach(VirtualAllocEx, {
            onEnter: function(args) {
                logEntry('VirtualAllocEx', {
                    processHandle: args[0],
                    address: args[1],
                    size: args[2].toInt32(),
                    protection: args[3].toInt32()
                });
            }
        });
    }
    
    var WriteProcessMemory = Module.findExportByName('kernel32.dll', 'WriteProcessMemory');
    if (WriteProcessMemory) {
        Interceptor.attach(WriteProcessMemory, {
            onEnter: function(args) {
                logEntry('WriteProcessMemory', {
                    processHandle: args[0],
                    baseAddress: args[1],
                    buffer: args[2],
                    size: args[3].toInt32()
                });
            }
        });
    }
    
    var CreateRemoteThread = Module.findExportByName('kernel32.dll', 'CreateRemoteThread');
    if (CreateRemoteThread) {
        Interceptor.attach(CreateRemoteThread, {
            onEnter: function(args) {
                logEntry('CreateRemoteThread', {
                    processHandle: args[0],
                    startAddress: args[3]
                });
            }
        });
    }
    
    // Hook加密API
    var CryptAcquireContextW = Module.findExportByName('advapi32.dll', 'CryptAcquireContextW');
    if (CryptAcquireContextW) {
        Interceptor.attach(CryptAcquireContextW, {
            onEnter: function(args) {
                var provider = args[2].readUtf16String();
                logEntry('CryptAcquireContextW', { provider: provider });
            }
        });
    }
    
    function logEntry(api, params) {
        var entry = {
            timestamp: new Date().toISOString(),
            api: api,
            params: params,
            pid: Process.id,
            processName: Process.name
        };
        send(JSON.stringify(entry));
    }
    """
    
    def __init__(self, target_process):
        self.target_process = target_process
        self.api_logs = []
        
    def on_message(self, message, data):
        """处理Frida消息"""
        if message['type'] == 'send':
            entry = json.loads(message['payload'])
            self.api_logs.append(entry)
            self.format_log(entry)
    
    def format_log(self, entry):
        """格式化日志输出"""
        timestamp = entry['timestamp']
        api = entry['api']
        params = entry['params']
        pid = entry['pid']
        
        print(f"[{timestamp}] PID:{pid} {api}({params})")
    
    def start_monitoring(self):
        """开始监控"""
        print(f"开始监控进程: {self.target_process}")
        
        try:
            session = frida.attach(self.target_process)
            script = session.create_script(self.HOOK_SCRIPT)
            script.on('message', self.on_message)
            script.load()
            
            print("监控已启动，按Ctrl+C停止...")
            sys.stdin.read()
            
        except KeyboardInterrupt:
            print("\n监控已停止")
            self.generate_report()
            
        except Exception as e:
            print(f"错误: {e}")
    
    def generate_report(self):
        """生成报告"""
        if not self.api_logs:
            print("无API调用记录")
            return
        
        print("\n" + "=" * 60)
        print("API监控报告")
        print("=" * 60)
        
        # 按API分类统计
        api_stats = {}
        for entry in self.api_logs:
            api = entry['api']
            if api not in api_stats:
                api_stats[api] = []
            api_stats[api].append(entry)
        
        print(f"\n总调用数: {len(self.api_logs)}")
        print(f"API种类: {len(api_stats)}\n")
        
        for api, entries in sorted(api_stats.items(), key=lambda x: len(x[1]), reverse=True):
            print(f"  {api}: {len(entries)}次")
            for entry in entries[:3]:
                params = entry['params']
                param_str = ', '.join(f'{k}={v}' for k, v in params.items())
                print(f"    - {param_str}")
            if len(entries) > 3:
                print(f"    ... 共{len(entries)}次")
            print()

# 使用示例
if __name__ == "__main__":
    monitor = APIMonitor("malware.exe")
    monitor.start_monitoring()
```

### 3.4 网络行为分析

```python
#!/usr/bin/env python3
# 网络行为分析工具
# 解析PCAP文件，提取恶意软件网络通信特征

import struct
from collections import defaultdict, Counter
from datetime import datetime

class NetworkAnalyzer:
    """网络行为分析器"""
    
    def __init__(self, pcap_file):
        self.pcap_file = pcap_file
        self.packets = []
        self.connections = defaultdict(list)
        self.dns_queries = []
        self.http_requests = []
        self.tls_connections = []
        
    def parse_pcap(self):
        """解析PCAP文件（简化版）"""
        with open(self.pcap_file, 'rb') as f:
            # 读取全局头
            magic = struct.unpack('<I', f.read(4))[0]
            if magic == 0xa1b2c3d4:
                endian = '<'
            elif magic == 0xd4c3b2a1:
                endian = '>'
            else:
                raise ValueError("非标准PCAP文件")
            
            f.read(20)  # 跳过全局头剩余部分
            
            # 读取数据包
            while True:
                header = f.read(16)
                if len(header) < 16:
                    break
                
                ts_sec, ts_usec, incl_len, orig_len = struct.unpack(
                    endian + 'IIII', header
                )
                
                packet_data = f.read(incl_len)
                if len(packet_data) < incl_len:
                    break
                
                self.packets.append({
                    'timestamp': ts_sec + ts_usec / 1e6,
                    'data': packet_data,
                    'length': incl_len
                })
        
        print(f"解析完成: {len(self.packets)}个数据包")
    
    def analyze_packet(self, packet):
        """分析单个数据包"""
        data = packet['data']
        
        # 以太网头 (14字节)
        if len(data) < 14:
            return
        
        eth_type = struct.unpack('>H', data[12:14])[0]
        
        if eth_type == 0x0800:  # IPv4
            self.parse_ipv4(data[14:], packet['timestamp'])
        elif eth_type == 0x86DD:  # IPv6
            self.parse_ipv6(data[14:], packet['timestamp'])
        elif eth_type == 0x0806:  # ARP
            self.parse_arp(data[14:], packet['timestamp'])
    
    def parse_ipv4(self, data, timestamp):
        """解析IPv4数据包"""
        if len(data) < 20:
            return
        
        version_ihl = data[0]
        ihl = (version_ihl & 0x0F) * 4
        protocol = data[9]
        
        src_ip = '.'.join(str(b) for b in data[12:16])
        dst_ip = '.'.join(str(b) for b in data[16:20])
        
        conn_key = f"{src_ip} -> {dst_ip}"
        
        if protocol == 6:  # TCP
            self.parse_tcp(data[ihl:], src_ip, dst_ip, timestamp)
        elif protocol == 17:  # UDP
            self.parse_udp(data[ihl:], src_ip, dst_ip, timestamp)
        
        self.connections[conn_key].append({
            'protocol': protocol,
            'timestamp': timestamp,
            'length': len(data)
        })
    
    def parse_tcp(self, data, src_ip, dst_ip, timestamp):
        """解析TCP段"""
        if len(data) < 20:
            return
        
        src_port = struct.unpack('>H', data[0:2])[0]
        dst_port = struct.unpack('>H', data[2:4])[0]
        
        flags = data[13]
        
        conn = {
            'src_ip': src_ip,
            'src_port': src_port,
            'dst_ip': dst_ip,
            'dst_port': dst_port,
            'flags': flags,
            'timestamp': timestamp
        }
        
        # 检测特定端口活动
        if dst_port == 80 or src_port == 80:
            self.parse_http(data[20:], conn)
        elif dst_port == 443 or src_port == 443:
            self.tls_connections.append(conn)
        elif dst_port in [445, 139]:
            self.log_smb_activity(conn)
        elif dst_port == 3389:
            self.log_rdp_activity(conn)
    
    def parse_udp(self, data, src_ip, dst_ip, timestamp):
        """解析UDP数据报"""
        if len(data) < 8:
            return
        
        src_port = struct.unpack('>H', data[0:2])[0]
        dst_port = struct.unpack('>H', data[2:4])[0]
        
        if dst_port == 53 or src_port == 53:
            self.parse_dns(data[8:], timestamp)
    
    def parse_dns(self, data, timestamp):
        """解析DNS查询"""
        if len(data) < 12:
            return
        
        # 解析DNS问题部分
        transaction_id = struct.unpack('>H', data[0:2])[0]
        flags = struct.unpack('>H', data[2:4])[0]
        qdcount = struct.unpack('>H', data[4:6])[0]
        
        offset = 12
        for _ in range(qdcount):
            domain = self.parse_dns_name(data, offset)
            if domain:
                self.dns_queries.append({
                    'domain': domain,
                    'timestamp': timestamp,
                    'transaction_id': transaction_id
                })
                offset = self.skip_dns_name(data, offset)
                offset += 4  # 跳过类型和类
    
    def parse_dns_name(self, data, offset):
        """解析DNS域名"""
        labels = []
        while offset < len(data) and data[offset] != 0:
            length = data[offset]
            if length & 0xC0 == 0xC0:  # 指针
                break
            offset += 1
            if offset + length > len(data):
                break
            labels.append(data[offset:offset + length].decode('ascii', errors='ignore'))
            offset += length
        return '.'.join(labels)
    
    def skip_dns_name(self, data, offset):
        """跳过DNS名称"""
        while offset < len(data) and data[offset] != 0:
            if data[offset] & 0xC0 == 0xC0:
                return offset + 2
            offset += data[offset] + 1
        return offset + 1
    
    def parse_http(self, data, conn):
        """解析HTTP请求"""
        try:
            text = data.decode('utf-8', errors='ignore')
            if text.startswith('GET ') or text.startswith('POST ') or text.startswith('PUT '):
                lines = text.split('\r\n')
                request_line = lines[0] if lines else ''
                headers = {}
                for line in lines[1:]:
                    if ':' in line:
                        key, value = line.split(':', 1)
                        headers[key.strip()] = value.strip()
                
                self.http_requests.append({
                    'method': request_line.split(' ')[0] if ' ' in request_line else '',
                    'url': request_line,
                    'host': headers.get('Host', ''),
                    'user_agent': headers.get('User-Agent', ''),
                    'connection': conn
                })
        except:
            pass
    
    def log_smb_activity(self, conn):
        """记录SMB活动"""
        print(f"  [SMB] {conn['src_ip']}:{conn['src_port']} -> {conn['dst_ip']}:{conn['dst_port']}")
    
    def log_rdp_activity(self, conn):
        """记录RDP活动"""
        print(f"  [RDP] {conn['src_ip']}:{conn['src_port']} -> {conn['dst_ip']}:{conn['dst_port']}")
    
    def analyze(self):
        """执行完整分析"""
        self.parse_pcap()
        
        for packet in self.packets:
            self.analyze_packet(packet)
        
        self.generate_report()
    
    def generate_report(self):
        """生成分析报告"""
        print("\n" + "=" * 60)
        print("网络行为分析报告")
        print("=" * 60)
        
        # 连接统计
        print(f"\n【连接统计】")
        print(f"  总数据包数: {len(self.packets)}")
        print(f"  唯一连接: {len(self.connections)}")
        
        # DNS查询
        print(f"\n【DNS查询】 ({len(self.dns_queries)}个)")
        unique_domains = set(q['domain'] for q in self.dns_queries)
        for domain in sorted(unique_domains):
            count = sum(1 for q in self.dns_queries if q['domain'] == domain)
            print(f"  {domain} ({count}次)")
        
        # HTTP请求
        print(f"\n【HTTP请求】 ({len(self.http_requests)}个)")
        for req in self.http_requests[:20]:
            print(f"  {req['method']} {req['host']} - UA: {req['user_agent'][:50]}")
        
        # TLS连接
        print(f"\n【TLS连接】 ({len(self.tls_connections)}个)")
        tls_targets = defaultdict(int)
        for conn in self.tls_connections:
            key = f"{conn['dst_ip']}:{conn['dst_port']}"
            tls_targets[key] += 1
        for target, count in sorted(tls_targets.items(), key=lambda x: x[1], reverse=True):
            print(f"  {target} ({count}个数据包)")
```
