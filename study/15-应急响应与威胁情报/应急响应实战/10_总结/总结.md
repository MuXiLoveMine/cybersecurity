# 总结

应急响应实战是检验应急响应能力的关键。只有通过不断的实践和总结，才能提高应急响应水平。

**核心要点回顾：**

1. **Web 入侵响应**：掌握 Web 入侵的检测、分析、处置流程，熟悉 Webshell 检测、SQL 注入分析等技术

2. **Windows 入侵排查**：掌握进程、网络、日志、启动项、用户、服务等全方位排查方法，熟练使用 Sysinternals 工具集

3. **Linux 入侵排查**：掌握 Linux 系统下的进程、网络、用户、日志、计划任务、服务等排查方法，熟悉常用命令

4. **数据库入侵排查**：掌握常见数据库（MySQL、SQL Server、Oracle、Redis、MongoDB）的入侵排查方法

5. **勒索软件响应**：掌握勒索软件的应急响应流程，重点是快速遏制和数据恢复，重视备份的重要性

6. **数据泄露响应**：掌握数据泄露的应急响应流程，了解法律合规要求，重视通知和补救措施

7. **横向移动检测**：掌握横向移动的常见技术和检测方法，通过登录日志、网络连接、进程命令等多维度检测

8. **溯源分析**：了解溯源分析的方法和流程，掌握 IOC 关联、TTP 分析等技术，认识溯源的局限性

9. **取证与证据保全**：掌握数字取证的原则和流程，了解常用取证工具，学会内存取证的基本方法

> "纸上得来终觉浅，绝知此事要躬行。" 应急响应是一门实践性很强的技能，只有通过大量的实战和练习，才能真正掌握。建议多做实验、多分析案例、多参加演练，在实践中不断提升自己的能力。

### 三、应急响应 Python 实战脚本

#### 1. 快速信息收集脚本（Linux）

```python
#!/usr/bin/env python3
"""
应急响应快速信息收集脚本
收集系统状态、进程、网络、用户等信息
"""
import os
import subprocess
import datetime
import platform

OUTPUT_DIR = f"/tmp/ir_collect_{datetime.datetime.now().strftime('%Y%m%d_%H%M%S')}"

def run_cmd(cmd, output_file=None):
    """执行命令并保存输出"""
    try:
        result = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=30)
        output = result.stdout + result.stderr
        if output_file:
            filepath = os.path.join(OUTPUT_DIR, output_file)
            with open(filepath, 'w', encoding='utf-8', errors='ignore') as f:
                f.write(f"Command: {cmd}\n")
                f.write(f"Time: {datetime.datetime.now()}\n")
                f.write("=" * 60 + "\n\n")
                f.write(output)
        return output
    except Exception as e:
        return f"Error: {str(e)}"

def main():
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    print(f"[*] 开始收集信息，输出目录: {OUTPUT_DIR}")
    
    # 系统信息
    print("[+] 收集系统信息...")
    run_cmd("uname -a", "01_system_info.txt")
    run_cmd("cat /etc/os-release", "01_system_info.txt")
    run_cmd("hostname", "01_system_info.txt")
    run_cmd("uptime", "01_system_info.txt")
    
    # 进程信息
    print("[+] 收集进程信息...")
    run_cmd("ps auxf", "02_process.txt")
    run_cmd("ps -efH", "02_process.txt")
    run_cmd("top -bn1 | head -30", "02_process.txt")
    
    # 网络信息
    print("[+] 收集网络信息...")
    run_cmd("netstat -tulnp 2>/dev/null || ss -tulnp", "03_network.txt")
    run_cmd("netstat -anp 2>/dev/null || ss -anp", "03_network.txt")
    run_cmd("ip addr", "03_network.txt")
    run_cmd("ip route", "03_network.txt")
    run_cmd("arp -a", "03_network.txt")
    run_cmd("cat /etc/hosts", "03_network.txt")
    
    # 用户信息
    print("[+] 收集用户信息...")
    run_cmd("cat /etc/passwd", "04_users.txt")
    run_cmd("cat /etc/shadow 2>/dev/null", "04_users.txt")
    run_cmd("last", "04_users.txt")
    run_cmd("lastlog", "04_users.txt")
    run_cmd("who", "04_users.txt")
    run_cmd("w", "04_users.txt")
    
    # 启动项
    print("[+] 收集启动项信息...")
    run_cmd("crontab -l 2>/dev/null; echo '---'; ls -la /etc/cron.d/ 2>/dev/null", "05_persistence.txt")
    run_cmd("systemctl list-unit-files --type=service 2>/dev/null", "05_persistence.txt")
    run_cmd("ls -la /etc/rc.d/rc*.d/ 2>/dev/null", "05_persistence.txt")
    
    # 可疑文件
    print("[+] 收集可疑文件信息...")
    run_cmd("find /tmp -type f -mtime -1 2>/dev/null", "06_suspicious_files.txt")
    run_cmd("find /dev/shm -type f 2>/dev/null", "06_suspicious_files.txt")
    run_cmd("find / -perm -4000 -type f 2>/dev/null | head -50", "06_suspicious_files.txt")
    
    # 日志
    print("[+] 收集日志摘要...")
    run_cmd("tail -100 /var/log/messages 2>/dev/null || tail -100 /var/log/syslog 2>/dev/null", "07_logs.txt")
    run_cmd("tail -100 /var/log/secure 2>/dev/null || tail -100 /var/log/auth.log 2>/dev/null", "07_logs.txt")
    run_cmd("grep 'Failed password' /var/log/secure 2>/dev/null | tail -20", "07_logs.txt")
    
    # 计算哈希
    print("[+] 计算收集文件的哈希值...")
```

#### 2. Web入侵痕迹快速扫描脚本

```python
#!/usr/bin/env python3
"""
Web目录入侵痕迹快速扫描工具
扫描Web目录中的可疑文件和特征
"""
import os
import re
import sys
from datetime import datetime, timedelta

# Webshell 特征模式
SUSPICIOUS_PATTERNS = {
    'php': [
        r'eval\s*\(',
        r'assert\s*\(',
        r'system\s*\(',
        r'shell_exec',
        r'passthru\s*\(',
        r'exec\s*\(',
        r'base64_decode\s*\(',
        r'gzinflate\s*\(',
        r'str_rot13\s*\(',
        r'move_uploaded_file',
        r'\$_(POST|GET|REQUEST)\[.*\]\s*\(',
        r'preg_replace\s*\(.*\/e',
        r'create_function\s*\(',
        r'file_put_contents.*\$_',
    ],
    'asp': [
        r'eval\s*\(',
        r'execute\s*\(',
        r'request\s*\(',
        r'cmd\.exe',
        r'wscript\.shell',
        r'shell\.application',
    ],
    'aspx': [
        r'eval\s*\(',
        r'system\.diagnostics\.process',
        r'cmd\.exe',
        r'server\.createobject',
    ],
    'jsp': [
        r'Runtime\.getRuntime',
        r'ProcessBuilder',
        r'FileOutputStream',
        r'getParameter.*exec',
    ],
}

def scan_file(filepath):
    """扫描单个文件"""
    ext = filepath.rsplit('.', 1)[-1].lower() if '.' in filepath else ''
    patterns = SUSPICIOUS_PATTERNS.get(ext, [])
    if not patterns:
        return []
    
    findings = []
    try:
        with open(filepath, 'r', encoding='utf-8', errors='ignore') as f:
            content = f.read()
            for pattern in patterns:
                matches = re.findall(pattern, content, re.IGNORECASE)
                if matches:
                    findings.append(f"匹配模式: {pattern} (次数: {len(matches)})")
    except Exception as e:
        findings.append(f"读取错误: {str(e)}")
    
    return findings

def scan_directory(webroot, days=None):
    """扫描目录"""
    results = []
    now = datetime.now()
    
    for root, dirs, files in os.walk(webroot):
        # 跳过某些目录
        dirs[:] = [d for d in dirs if d not in ['.git', '.svn', 'node_modules']]
        
        for filename in files:
            filepath = os.path.join(root, filename)
            
            # 按修改时间过滤
            if days:
                try:
                    mtime = datetime.fromtimestamp(os.path.getmtime(filepath))
                    if (now - mtime).days > days:
                        continue
                except:
                    pass
            
            findings = scan_file(filepath)
            if findings:
                results.append({
                    'file': filepath,
                    'findings': findings,
                    'mtime': datetime.fromtimestamp(os.path.getmtime(filepath)),
                    'size': os.path.getsize(filepath),
                })
    
    return results

def main():
    if len(sys.argv) < 2:
        print("用法: python web_scan.py <web根目录> [天数]")
        print("示例: python web_scan.py /var/www/html 7")
        sys.exit(1)
    
    webroot = sys.argv[1]
    days = int(sys.argv[2]) if len(sys.argv) > 2 else None
    
    print(f"[*] 扫描目录: {webroot}")
    if days:
        print(f"[*] 只扫描最近 {days} 天修改的文件")
    print("=" * 60)
    
    results = scan_directory(webroot, days)
    
    if not results:
        print("[+] 未发现可疑文件")
        return
    
    print(f"[!] 发现 {len(results)} 个可疑文件：\n")
    for i, result in enumerate(results, 1):
        print(f"  [{i}] {result['file']}")
        print(f"      修改时间: {result['mtime']}")
        print(f"      文件大小: {result['size']} bytes")
        for finding in result['findings']:
            print(f"      - {finding}")
        print()

if __name__ == "__main__":
    main()
```
