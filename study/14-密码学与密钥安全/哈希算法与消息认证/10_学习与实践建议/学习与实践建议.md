### 一、命令行哈希计算方法速查表

#### OpenSSL 哈希计算

```bash
# ===== MD5（不推荐，仅用于学习/校验）=====
openssl dgst -md5 file.txt
# 输出: MD5(file.txt)= d41d8cd98f00b204e9800998ecf8427e

# ===== SHA-1（不推荐用于安全场景）=====
openssl dgst -sha1 file.txt

# ===== SHA-256（推荐）=====
openssl dgst -sha256 file.txt

# ===== SHA-512 =====
openssl dgst -sha512 file.txt

# ===== SHA-3 系列 =====
openssl dgst -sha3-256 file.txt
openssl dgst -sha3-512 file.txt

# ===== 只输出哈希值（不带文件名）=====
openssl dgst -sha256 -binary file.txt | xxd -p
# -binary: 输出二进制，再用xxd转hex

# ===== 从标准输入计算哈希 =====
echo -n "hello world" | openssl dgst -sha256
# 注意：echo默认加换行，用-n去掉

# ===== 批量计算文件哈希 =====
find /path/to/dir -type f -exec openssl dgst -sha256 {} \; > hashes.txt

# ===== 列出所有支持的摘要算法 =====
openssl dgst -list
```

### 二、Python 哈希算法实现示例

#### 基础哈希计算

```python
import hashlib

# ===== MD5 =====
def md5_hash(data: str) -> str:
    """计算MD5哈希（不推荐用于安全场景）"""
    return hashlib.md5(data.encode()).hexdigest()

# ===== SHA-1 =====
def sha1_hash(data: str) -> str:
    """计算SHA-1哈希"""
    return hashlib.sha1(data.encode()).hexdigest()

# ===== SHA-256 =====
def sha256_hash(data: str) -> str:
    """计算SHA-256哈希"""
    return hashlib.sha256(data.encode()).hexdigest()

# ===== SHA-512 =====
def sha512_hash(data: str) -> str:
    """计算SHA-512哈希"""
    return hashlib.sha512(data.encode()).hexdigest()

# ===== SHA-3 系列 =====
def sha3_256_hash(data: str) -> str:
    """计算SHA3-256哈希"""
    return hashlib.sha3_256(data.encode()).hexdigest()

# ===== 文件哈希计算 =====
def file_sha256(filepath: str, chunk_size: int = 8192) -> str:
    """计算大文件的SHA-256哈希（分块读取，节省内存）"""
    sha256 = hashlib.sha256()
    with open(filepath, 'rb') as f:
        while chunk := f.read(chunk_size):
            sha256.update(chunk)
    return sha256.hexdigest()

# ===== 使用示例 =====
if __name__ == "__main__":
    text = "Hello, Hash!"
    print(f"MD5:     {md5_hash(text)}")
    print(f"SHA-1:   {sha1_hash(text)}")
    print(f"SHA-256: {sha256_hash(text)}")
    print(f"SHA-512: {sha512_hash(text)}")
    print(f"SHA3-256: {sha3_256_hash(text)}")
    # print(f"文件哈希: {file_sha256('test.txt')}")
```

#### HMAC 消息认证码

```python
import hmac
import hashlib
import base64

# ===== HMAC-SHA256 计算 =====
def hmac_sha256(key: bytes, message: bytes) -> str:
    """计算HMAC-SHA256，返回十六进制字符串"""
    return hmac.new(key, message, hashlib.sha256).hexdigest()

# ===== HMAC-SHA512 计算 =====
def hmac_sha512(key: bytes, message: bytes) -> str:
    """计算HMAC-SHA512"""
    return hmac.new(key, message, hashlib.sha512).hexdigest()

# ===== HMAC 验证 =====
def hmac_verify(key: bytes, message: bytes, expected_hmac: str) -> bool:
    """
    安全地验证HMAC（使用恒定时间比较，防止时序攻击）
    """
    computed = hmac.new(key, message, hashlib.sha256).digest()
    expected = bytes.fromhex(expected_hmac)
    return hmac.compare_digest(computed, expected)

# ===== Base64 编码的 HMAC（常用于API签名）=====
def hmac_sha256_base64(key: str, message: str) -> str:
    """计算HMAC-SHA256并返回Base64编码"""
    signature = hmac.new(
        key.encode(),
        message.encode(),
        hashlib.sha256
    ).digest()
    return base64.b64encode(signature).decode()

# 使用示例
if __name__ == "__main__":
    key = b"my_secret_key_123"
    message = b"GET /api/data?user=admin&timestamp=1700000000"
    
    mac = hmac_sha256(key, message)
    print(f"HMAC-SHA256: {mac}")
    
    is_valid = hmac_verify(key, message, mac)
    print(f"HMAC验证: {'通过' if is_valid else '失败'}")
    
    # 模拟被篡改的消息
    tampered_msg = b"GET /api/data?user=admin&timestamp=1700000000&role=admin"
    is_valid_tampered = hmac_verify(key, tampered_msg, mac)
    print(f"篡改后验证: {'通过' if is_valid_tampered else '失败（正确拦截）'}")
```

#### 密码哈希（bcrypt / scrypt / Argon2）

```python
# pip install bcrypt argon2-cffi
import bcrypt
import hashlib

# ===== bcrypt 哈希密码 =====
def bcrypt_hash_password(password: str) -> str:
    """使用bcrypt哈希密码，返回字符串格式"""
    # 生成盐并哈希，cost factor=12（迭代次数=2^12=4096）
    salt = bcrypt.gensalt(rounds=12)
    hashed = bcrypt.hashpw(password.encode(), salt)
    return hashed.decode()

def bcrypt_verify_password(password: str, hashed: str) -> bool:
    """验证bcrypt密码哈希"""
    return bcrypt.checkpw(password.encode(), hashed.encode())

# ===== scrypt 哈希密码（标准库）=====
def scrypt_hash_password(password: str, salt: bytes = None) -> tuple:
    """
    使用scrypt哈希密码
    返回 (hashed_hex, salt_hex)
    """
    if salt is None:
        salt = hashlib.scrypt(b'', salt=b'', n=1, dklen=16)  # 占位，下面重新生成
    salt = hashlib.scrypt(b'', salt=b'', n=1, dklen=16)  # 实际应该用 os.urandom(16)
    
    import os
    salt = os.urandom(16)
    hashed = hashlib.scrypt(
        password.encode(),
        salt=salt,
        n=2**14,   # CPU/内存成本因子
        r=8,       # 块大小
        p=1,       # 并行化因子
        dklen=32   # 输出长度
    )
    return hashed.hex(), salt.hex()

def scrypt_verify_password(password: str, hashed_hex: str, salt_hex: str) -> bool:
    """验证scrypt密码哈希"""
    import os
    hashed = hashlib.scrypt(
        password.encode(),
        salt=bytes.fromhex(salt_hex),
        n=2**14,
        r=8,
        p=1,
        dklen=32
    )
    return hashed.hex() == hashed_hex

# ===== Argon2（推荐用于新系统）=====
from argon2 import PasswordHasher
from argon2.exceptions import VerifyMismatchError

def argon2_hash_password(password: str) -> str:
    """使用Argon2id哈希密码（推荐）"""
    ph = PasswordHasher(
        time_cost=3,       # 迭代次数
        memory_cost=65536, # 内存使用（KB）
        parallelism=4,     # 并行度
        hash_len=32,       # 哈希长度
        type=argon2.Type.ID  # Argon2id（抗侧信道+抗GPU）
    )
    return ph.hash(password)

def argon2_verify_password(password: str, hashed: str) -> bool:
    """验证Argon2密码哈希"""
    ph = PasswordHasher()
    try:
        return ph.verify(hashed, password)
    except VerifyMismatchError:
        return False

# 使用示例
if __name__ == "__main__":
    password = "MyStr0ngP@ssw0rd!"
    
    # bcrypt 测试
    print("=== bcrypt ===")
    bcrypt_hashed = bcrypt_hash_password(password)
    print(f"哈希结果: {bcrypt_hashed}")
    print(f"验证正确密码: {bcrypt_verify_password(password, bcrypt_hashed)}")
    print(f"验证错误密码: {bcrypt_verify_password('wrong', bcrypt_hashed)}")
    
    # Argon2 测试
    print("\n=== Argon2 ===")
    argon2_hashed = argon2_hash_password(password)
    print(f"哈希结果: {argon2_hashed}")
    print(f"验证正确密码: {argon2_verify_password(password, argon2_hashed)}")
    print(f"验证错误密码: {argon2_verify_password('wrong', argon2_hashed)}")
```

```python
        padding += struct.pack('>Q', msg_len * 8)  # 长度以bit为单位
        return padding
    
    padding = sha256_padding(original_data_len)
    total_len = original_data_len + len(padding)
    
    # 从已知状态继续哈希追加数据
    # 注意：这需要调用底层SHA-256实现，标准库hashlib不直接支持
    # 实际攻击工具：hash_extender, length_extender 等

# ===== 彩虹表原理演示 =====
def simple_rainbow_table(passwords: list) -> dict:
    """
    构建简单的彩虹表（密码 -> 哈希）
    实际彩虹表使用更复杂的链结构减少存储空间
    """
    table = {}
    for pwd in passwords:
        h = hashlib.md5(pwd.encode()).hexdigest()
        table[h] = pwd
    return table

def crack_with_rainbow_table(hash_to_crack: str, rainbow_table: dict) -> str:
    """使用彩虹表破解哈希"""
    return rainbow_table.get(hash_to_crack, "未找到")
```

### 三、哈希算法安全选型速查表

| 用途 | 推荐算法 | 不推荐 | 说明 |
|------|---------|--------|------|
| 文件完整性校验 | SHA-256 / SHA-512 | MD5 / SHA-1 | 校验用MD5也行，但安全场景用SHA-2 |
| 数字签名 | SHA-256 / SHA-512 | MD5 / SHA-1 | 配合RSA/ECDSA使用 |
| 消息认证（HMAC） | HMAC-SHA256 | - | 用HMAC而不是裸哈希 |
| 密码存储 | Argon2id / bcrypt | MD5 / SHA-1 / SHA-256 | 必须用慢哈希算法 |
| 区块链/抗量子 | SHA-3 / SHA-256 | - | SHA-3抗长度扩展攻击 |
| 性能敏感场景 | BLAKE2 / SHA-256 | - | BLAKE2比MD5还快且安全 |

> **安全红线**：
> - 绝对不要用普通哈希（MD5/SHA-1/SHA-256）存储密码
> - 密码存储必须用 bcrypt/scrypt/Argon2 等慢哈希算法
> - MD5 和 SHA-1 已被攻破，不要用于任何安全认证场景
> - HMAC 比 "密钥+消息" 直接哈希更安全，不要自己发明MAC算法
> - SHA-256 存在长度扩展攻击风险，敏感场景用 HMAC 或 SHA-3
