# 国密算法实现与工具

### 9.1 GmSSL工具

**GmSSL** 是支持国密算法的OpenSSL分支。

```bash
# GmSSL基本命令

# 1. 生成SM2密钥对
gmssl ecparam -genkey -name sm2p256v1 -out sm2_key.pem

# 2. 生成SM2自签名证书
gmssl req -new -x509 -key sm2_key.pem -out sm2_cert.pem \
    -sm3 -subj "/C=CN/O=MyOrg/CN=www.example.com" -days 365

# 3. SM3哈希
gmssl dgst -sm3 message.txt

# 4. SM2签名
gmssl dgst -sm3 -sign sm2_key.pem -out signature.bin message.txt

# 5. SM2验证签名
gmssl dgst -sm3 -verify sm2_pub.pem -signature signature.bin message.txt

# 6. SM4加密（ECB）
gmssl enc -sm4-ecb -in plaintext.txt -out ciphertext.bin \
    -K 0123456789ABCDEFFEDCBA9876543210

# 7. SM4加密（CBC）
gmssl enc -sm4-cbc -in plaintext.txt -out ciphertext.bin \
    -K 0123456789ABCDEFFEDCBA9876543210 \
    -iv 000102030405060708090a0b0c0d0e0f

# 8. SM4解密
gmssl enc -d -sm4-cbc -in ciphertext.bin -out plaintext.txt \
    -K 0123456789ABCDEFFEDCBA9876543210 \
    -iv 000102030405060708090a0b0c0d0e0f

# 9. 查看证书
gmssl x509 -in sm2_cert.pem -text -noout

# 10. TLCP服务器
gmssl s_server -accept 443 \
    -cert server_sign.pem -key server_sign_key.pem \
    -cert2 server_enc.pem -key2 server_enc_key.pem \
    -CAfile ca.pem
```

### 9.2 Python国密库

```python
"""
Python国密算法库汇总
"""

# 1. gmssl - 纯Python实现
# pip install gmssl
# 提供: SM2, SM3, SM4

# 2. pycryptodome + 国密扩展
# pip install pycryptodome
# 部分版本支持SM2/SM3/SM4

# 3. cryptography + 国密支持
# 新版cryptography库开始支持SM2/SM3

# 4. 国密HSM SDK
# 各HSM厂商提供的SDK

# 使用gmssl库的综合示例
try:
    from gmssl import sm2, sm3, sm4, func
    
    # === SM3 哈希 ===
    data = b"Hello SM3"
    hash_value = sm3.sm3_hash(data)
    print(f"SM3('{data.decode()}') = {hash_value}")
    
    # === SM4 加密 ===
    key = b'0123456789abcdef'  # 16字节密钥
    iv = b'\x00' * 16
    
    sm4_crypt = sm4.CryptSM4()
    sm4_crypt.set_key(key, sm4.SM4_ENCRYPT)
    ciphertext = sm4_crypt.crypt_cbc(iv, b"Hello SM4 CBC!")
    
    sm4_crypt.set_key(key, sm4.SM4_DECRYPT)
    plaintext = sm4_crypt.crypt_cbc(iv, ciphertext)
    
    print(f"SM4 CBC加密: {ciphertext.hex()}")
    print(f"SM4 CBC解密: {plaintext}")
    
    # === SM2 加密/签名 ===
    privkey = func.random_hex(64)
    crypt = sm2.CryptSM2(pubkey='', privkey=privkey)
    pub = crypt._kg(int(privkey, 16), crypt.pointG)
    pubkey = '%064x%064x' % pub
    
    sm2_crypt = sm2.CryptSM2(pubkey=pubkey, privkey=privkey)
    
    # 加密
    plaintext = b"Hello SM2!"
    ciphertext = sm2_crypt.encrypt(plaintext)
    decrypted = sm2_crypt.decrypt(ciphertext)
    print(f"SM2加密/解密: {decrypted == plaintext}")
    
    # 签名
    user_id = "1234567812345678"
    signature = sm2_crypt.sign(plaintext, user_id.encode())
    valid = sm2_crypt.verify(signature, plaintext, user_id.encode())
    print(f"SM2签名验证: {valid}")
    
except ImportError:
    print("请安装gmssl库: pip install gmssl")
```

```java
import org.bouncycastle.crypto.digests.SM3Digest;

public class SM3Example {
    public static byte[] hash(byte[] data) {
```

### 9.4 C语言国密库（GmSSL C库）

```c
// GmSSL C API 示例

#include <gmssl/sm2.h>
#include <gmssl/sm3.h>
#include <gmssl/sm4.h>

// SM3哈希
void sm3_hash_example() {
    unsigned char data[] = "Hello SM3";
    unsigned char hash[32];
    size_t data_len = strlen((char*)data);
    
    sm3(data, data_len, hash);
    
    printf("SM3: ");
    for (int i = 0; i < 32; i++) {
        printf("%02x", hash[i]);
    }
    printf("\n");
}

// SM4-CBC加密
void sm4_cbc_example() {
    unsigned char key[16] = {0x01, 0x23, ...};
    unsigned char iv[16] = {0x00, 0x01, ...};
    unsigned char plaintext[16] = "Hello SM4!!!!!";
    unsigned char ciphertext[16];
    
    sm4_cbc_encrypt(plaintext, ciphertext, 16, key, iv);
}

// SM2签名
void sm2_sign_example() {
    uint8_t privateKey[32];
    uint8_t publicKey[64];
    uint8_t message[] = "Message to sign";
    uint8_t signature[64]; // r || s
    
    // 生成密钥对
    sm2_keygen(privateKey, publicKey);
    
    // 签名
    sm2_sign(message, strlen((char*)message),
             privateKey, publicKey,
             signature, NULL);
}
```
