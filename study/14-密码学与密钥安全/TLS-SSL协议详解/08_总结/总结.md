# 总结

TLS/SSL 协议是网络通信安全的基石，本文覆盖的关键要点：

- **演进方向**：全面迁移到 TLS 1.3，彻底淘汰 SSL 3.0/TLS 1.0/1.1
- **握手核心**：TLS 1.3 的 1-RTT 握手和 0-RTT 恢复大幅降低延迟
- **PKI 体系**：证书链验证、CT 日志、ACME 自动化是信任链的保障
- **安全配置**：优先 ECDHE + AEAD 密码套件，启用 HSTS，禁用 CBC/RC4/3DES
- **历史教训**：Heartbleed/POODLE/CRIME 等漏洞推动协议持续改进
- **调试工具**：OpenSSL s_client、SSL Labs、testssl.sh 是必备检测手段
- **mTLS**：服务网格零信任的关键技术，Sidecar 代理自动处理证书管理

安全工程师应定期用 testssl.sh 和 SSL Labs 扫描所有对外服务，确保 TLS 配置符合最新安全标准。

### 一、OpenSSL TLS 调试命令速查表

#### s_client 连接测试

```bash
# ===== 基本连接测试 =====
openssl s_client -connect example.com:443
# 连接到目标服务器，输出证书链、握手过程、密码套件等信息
# 连接成功后可以手动发送HTTP请求（如 GET / HTTP/1.0 回车两次）

# ===== 指定 SNI（服务器名称指示）=====
openssl s_client -connect example.com:443 -servername example.com
# 虚拟主机场景必须指定servername，否则可能拿到默认证书

# ===== 显示完整证书链 =====
openssl s_client -connect example.com:443 -showcerts
# -showcerts: 显示所有证书（包括中间CA证书）

# ===== 测试特定 TLS 版本 =====
openssl s_client -connect example.com:443 -tls1_2       # TLS 1.2
openssl s_client -connect example.com:443 -tls1_3       # TLS 1.3
openssl s_client -connect example.com:443 -tls1_1       # TLS 1.1（旧版）
openssl s_client -connect example.com:443 -no_tls1_3    # 禁用 TLS 1.3

# ===== 测试特定密码套件 =====
openssl s_client -connect example.com:443 -cipher 'ECDHE-RSA-AES256-GCM-SHA384'
# 测试服务器是否支持指定的密码套件

# ===== 测试 TLS 1.3 密码套件 =====
openssl s_client -connect example.com:443 -tls1_3 -ciphersuites 'TLS_AES_256_GCM_SHA384'

# ===== 静默模式（只输出证书）=====
openssl s_client -connect example.com:443 -quiet </dev/null

# ===== 测试会话复用 =====
# 第一次连接保存会话
openssl s_client -connect example.com:443 -sess_out session.pem </dev/null
# 第二次连接复用会话
openssl s_client -connect example.com:443 -sess_in session.pem </dev/null
# 查看输出中 "Reused, TLSv1.3" 表示会话复用成功

# ===== 测试客户端证书认证（mTLS）=====
openssl s_client -connect example.com:443 -cert client.crt -key client.key
# 使用客户端证书进行双向认证
```

### 二、证书生成与验证命令速查表

#### 自签名证书生成

```bash
# ===== 生成自签名证书（一键生成）=====
openssl req -x509 -newkey rsa:2048 -keyout server.key -out server.crt -days 365 -nodes
# -x509: 生成自签名证书而不是CSR
# -newkey rsa:2048: 生成2048位RSA新密钥
# -nodes: 私钥不加密（否则需要输入密码）
# -days 365: 有效期365天

# ===== 生成自签名证书（指定主题，免交互）=====
openssl req -x509 -newkey rsa:2048 -keyout server.key -out server.crt -days 365 -nodes \
  -subj "/C=CN/ST=Beijing/L=Beijing/O=MyCompany/OU=IT/CN=example.com"
# -subj: 指定证书主题，免交互
# C=国家, ST=省份, L=城市, O=组织, OU=部门, CN=通用名(域名)

# ===== 生成带 SAN（主题备用名称）的自签名证书 =====
# 1. 创建配置文件 san.cnf
cat > san.cnf << 'EOF'
[req]
distinguished_name = req_distinguished_name
req_extensions = v3_req
prompt = no

[req_distinguished_name]
C = CN
ST = Beijing
L = Beijing
O = MyCompany
CN = example.com

[v3_req]
keyUsage = critical, digitalSignature, keyEncipherment
extendedKeyUsage = serverAuth
subjectAltName = @alt_names

[alt_names]
DNS.1 = example.com
DNS.2 = www.example.com
DNS.3 = api.example.com
IP.1 = 192.168.1.1
EOF

# 2. 使用配置文件生成
openssl req -x509 -newkey rsa:2048 -keyout server.key -out server.crt -days 365 -nodes -config san.cnf -extensions v3_req

# ===== 使用 ECC 证书 =====
openssl req -x509 -newkey ec:<(openssl ecparam -name prime256v1) -keyout server_ecc.key -out server_ecc.crt -days 365 -nodes -subj "/CN=example.com"
# 生成P-256曲线的ECC自签名证书
```

#### CSR（证书签名请求）生成

```bash
# ===== 生成RSA私钥 + CSR =====
# 1. 先生成私钥
openssl genrsa -out server.key 2048

# 2. 生成CSR
openssl req -new -key server.key -out server.csr
# 交互式填写信息

# ===== 免交互生成CSR =====
openssl req -new -key server.key -out server.csr \
  -subj "/C=CN/ST=Beijing/L=Beijing/O=MyCompany/CN=example.com"

# ===== 生成带SAN的CSR =====
openssl req -new -key server.key -out server.csr -config san.cnf

# ===== 查看CSR内容 =====
openssl req -in server.csr -text -noout

# ===== 验证CSR中的公钥与私钥匹配 =====
openssl req -in server.csr -noout -modulus | md5sum
openssl rsa -in server.key -noout -modulus | md5sum
# 两个md5应该一致
```

#### CA 证书与证书签发

```bash
# ===== 生成CA根证书 =====
# 1. 生成CA私钥
openssl genrsa -out ca.key 4096

# 2. 生成CA自签名证书
openssl req -x509 -new -key ca.key -out ca.crt -days 3650 \
  -subj "/C=CN/ST=Beijing/O=MyCA/CN=My Root CA"

# ===== 使用CA签发服务器证书 =====
# 1. 服务器生成CSR（同上）
# 2. CA签发证书
openssl x509 -req -in server.csr -CA ca.crt -CAkey ca.key -CAcreateserial \
  -out server.crt -days 365 -extensions v3_req -extfile san.cnf
# -CAcreateserial: 自动创建序列号文件
# -extensions: 指定扩展配置段

# ===== 查看签发的证书 =====
openssl x509 -in server.crt -text -noout

# ===== 查看证书有效期 =====
openssl x509 -in server.crt -noout -dates
# 输出 notBefore 和 notAfter

# ===== 查看证书序列号 =====
openssl x509 -in server.crt -noout -serial
```

#### 证书验证

```bash
# ===== 验证证书链 =====
openssl verify -CAfile ca.crt server.crt
# 输出 server.crt: OK 表示验证通过

# ===== 验证包含中间CA的证书链 =====
openssl verify -CAfile root_ca.crt -untrusted intermediate_ca.crt server.crt

# ===== 验证证书与私钥匹配 =====
# 方法1：比较模数的哈希
openssl x509 -in server.crt -noout -modulus | md5sum
openssl rsa -in server.key -noout -modulus | md5sum
# 两个值必须相同

# 方法2：比较公钥
openssl x509 -in server.crt -pubkey -noout | md5sum
openssl rsa -in server.key -pubout 2>/dev/null | md5sum

# ===== 验证证书与CSR匹配 =====
openssl x509 -in server.crt -noout -modulus | md5sum
openssl req -in server.csr -noout -modulus | md5sum

# ===== 检查证书用途 =====
openssl x509 -in server.crt -noout -purpose
# 查看证书是否可用于SSL服务器、客户端、代码签名等

# ===== 检查证书是否在有效期内 =====
openssl x509 -in server.crt -noout -checkend 0
# 检查当前是否有效

openssl x509 -in server.crt -noout -checkend 2592000
# 检查30天（2592000秒）内是否过期
```

#### 证书格式转换

```bash
# ===== PEM 转 DER（二进制格式）=====
openssl x509 -in cert.pem -outform DER -out cert.der

# ===== DER 转 PEM =====
openssl x509 -in cert.der -inform DER -out cert.pem

# ===== PEM 转 PKCS#12（.pfx/.p12，包含证书+私钥）=====
openssl pkcs12 -export -out cert.pfx -inkey server.key -in server.crt -certfile ca.crt
# -certfile: 添加CA证书链
# Windows/IIS常用PFX格式

# ===== PKCS#12 提取私钥 =====
openssl pkcs12 -in cert.pfx -nocerts -out private.key -nodes

# ===== PKCS#12 提取证书 =====
openssl pkcs12 -in cert.pfx -clcerts -nokeys -out certificate.crt

# ===== 证书指纹计算 =====
openssl x509 -in cert.pem -noout -fingerprint -sha256  # SHA-256指纹
openssl x509 -in cert.pem -noout -fingerprint -sha1    # SHA-1指纹
openssl x509 -in cert.pem -noout -fingerprint -md5     # MD5指纹
```

#### 证书撤销与OCSP

```bash
# ===== 生成CRL（证书撤销列表）=====
# 1. 创建CRL配置
cat > crl.cnf << 'EOF'
[ca]
default_ca = CA_default

[CA_default]
database = index.txt
crlnumber = crlnumber
default_crl_days = 30
default_md = sha256
EOF

# 2. 初始化数据库
touch index.txt
echo 01 > crlnumber

# 3. 生成CRL
openssl ca -gencrl -out ca.crl -config crl.cnf -keyfile ca.key -cert ca.crt

# ===== 查看CRL内容 =====
openssl crl -in ca.crl -text -noout

# ===== 验证证书是否被撤销 =====
openssl verify -CAfile ca.crt -crl_check -CRLfile ca.crl server.crt

# ===== OCSP 查询 =====
openssl ocsp -issuer ca.crt -cert server.crt -url http://ocsp.example.com -CAfile ca.crt
# 查询指定证书的OCSP状态
```

### 三、TLS 安全配置生成器（推荐配置）

#### Nginx 安全配置

```nginx
# /etc/nginx/conf.d/ssl.conf

# 证书配置
ssl_certificate /etc/ssl/certs/server.crt;
ssl_certificate_key /etc/ssl/private/server.key;

# 协议版本（只启用TLS 1.2和1.3）
ssl_protocols TLSv1.2 TLSv1.3;

# 密码套件（TLS 1.2）
ssl_ciphers 'ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305:DHE-RSA-AES128-GCM-SHA256:DHE-RSA-AES256-GCM-SHA384';

# TLS 1.3 密码套件
ssl_conf_command Ciphersuites TLS_AES_128_GCM_SHA256:TLS_AES_256_GCM_SHA384:TLS_CHACHA20_POLY1305_SHA256;

# 优先使用服务器端密码套件
ssl_prefer_server_ciphers on;

# ECDH 曲线
ssl_ecdh_curve X25519:prime256v1:secp384r1;

# 会话缓存
ssl_session_cache shared:SSL:10m;
ssl_session_timeout 10m;
ssl_session_tickets off;

# OCSP 装订
ssl_stapling on;
ssl_stapling_verify on;
resolver 8.8.8.8 8.8.4.4 valid=300s;
resolver_timeout 5s;

# HSTS（强制HTTPS）
add_header Strict-Transport-Security "max-age=63072000; includeSubDomains; preload" always;

# 安全头
add_header X-Frame-Options DENY;
add_header X-Content-Type-Options nosniff;
add_header X-XSS-Protection "1; mode=block";
```

#### Apache 安全配置

```apache
# /etc/apache2/mods-available/ssl.conf

SSLEngine on
SSLCertificateFile /etc/ssl/certs/server.crt
SSLCertificateKeyFile /etc/ssl/private/server.key
SSLCertificateChainFile /etc/ssl/certs/ca-bundle.crt

# 协议版本
SSLProtocol -all +TLSv1.2 +TLSv1.3

# 密码套件
SSLCipherSuite ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305
SSLHonorCipherOrder on

# OCSP 装订
SSLUseStapling on
SSLStaplingCache shmcb:/tmp/stapling_cache(128000)

# HSTS
Header always set Strict-Transport-Security "max-age=63072000; includeSubDomains; preload"
```

> - 生产环境只启用 TLS 1.2 和 TLS 1.3
