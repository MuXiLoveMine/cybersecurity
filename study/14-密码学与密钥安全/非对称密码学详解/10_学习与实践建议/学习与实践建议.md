### 一、OpenSSL RSA 命令速查表

#### RSA 密钥生成与管理

```bash
# ===== 生成 RSA 私钥（2048位）=====
openssl genrsa -out private_key.pem 2048
# 生成PKCS#1格式的RSA私钥

# ===== 生成 RSA 私钥（4096位，推荐生产环境）=====
openssl genrsa -out private_key.pem 4096

# ===== 生成带密码保护的 RSA 私钥 =====
openssl genrsa -aes256 -out private_encrypted.pem 2048
# 使用AES-256加密私钥文件，需输入密码

# ===== 从私钥提取公钥 =====
openssl rsa -in private_key.pem -pubout -out public_key.pem

# ===== 查看私钥信息 =====
openssl rsa -in private_key.pem -text -noout
# 显示模数、指数、素数等详细参数

# ===== 查看公钥信息 =====
openssl rsa -pubin -in public_key.pem -text -noout

# ===== 私钥格式转换（PKCS#1 -> PKCS#8）=====
openssl pkcs8 -topk8 -in private_key.pem -out private_pkcs8.pem -nocrypt
# Java等语言常用PKCS#8格式

# ===== PKCS#8 转 PKCS#1 =====
openssl rsa -in private_pkcs8.pem -out private_pkcs1.pem

# ===== 检查私钥是否完好 =====
openssl rsa -in private_key.pem -check -noout

# ===== 修改私钥密码 =====
openssl rsa -in private_key.pem -aes256 -out private_new.pem
```

#### RSA 加密/解密

```bash
# ===== 使用公钥加密（OAEP填充，推荐）=====
openssl pkeyutl -encrypt -in plain.txt -out cipher.enc -inkey public_key.pem -pubin -pkeyopt rsa_padding_mode:oaep -pkeyopt rsa_oaep_md:sha256
# pkeyutl是新版通用公钥工具
# OAEP填充比PKCS#1 v1.5更安全

# ===== 使用私钥解密（OAEP填充）=====
openssl pkeyutl -decrypt -in cipher.enc -out decrypted.txt -inkey private_key.pem -pkeyopt rsa_padding_mode:oaep -pkeyopt rsa_oaep_md:sha256

# ===== 使用公钥加密（PKCS#1 v1.5 填充，兼容旧系统）=====
openssl rsautl -encrypt -in plain.txt -out cipher.enc -inkey public_key.pem -pubin
# rsautl是旧版RSA专用工具

# ===== 使用私钥解密（PKCS#1 v1.5 填充）=====
openssl rsautl -decrypt -in cipher.enc -out decrypted.txt -inkey private_key.pem

# ===== RSA加密文件大小限制说明 =====
# RSA不适合加密大文件，最大加密数据 = 密钥长度 - 填充开销
# 2048位密钥 + OAEP-SHA256 = 最多加密 245 字节
# 2048位密钥 + PKCS#1 v1.5 = 最多加密 245 字节
# 大文件请使用"混合加密"：RSA加密AES密钥 + AES加密文件
```

#### RSA 数字签名

```bash
# ===== 私钥签名（SHA256 + RSA，推荐）=====
openssl dgst -sha256 -sign private_key.pem -out signature.bin data.txt
# 对data.txt生成SHA256摘要后用RSA私钥签名

# ===== 公钥验签 =====
openssl dgst -sha256 -verify public_key.pem -signature signature.bin data.txt
# 验证成功输出 "Verified OK"

# ===== 生成 Base64 编码的签名 =====
openssl dgst -sha256 -sign private_key.pem data.txt | base64 > signature.b64

# ===== 验证 Base64 编码的签名 =====
base64 -d signature.b64 | openssl dgst -sha256 -verify public_key.pem -signature /dev/stdin data.txt

# ===== PSS 填充签名（更安全）=====
openssl dgst -sha256 -sign private_key.pem -sigopt rsa_padding_mode:pss -out signature.pss data.txt

# ===== PSS 填充验签 =====
openssl dgst -sha256 -verify public_key.pem -sigopt rsa_padding_mode:pss -signature signature.pss data.txt
```

#### RSA 混合加密实战

```bash
# ===== 步骤1：生成随机AES密钥 =====
openssl rand -base64 32 > aes_key.txt

# ===== 步骤2：用RSA公钥加密AES密钥 =====
openssl pkeyutl -encrypt -in aes_key.txt -out aes_key.enc -inkey public_key.pem -pubin -pkeyopt rsa_padding_mode:oaep -pkeyopt rsa_oaep_md:sha256

# ===== 步骤3：用AES密钥加密大文件 =====
openssl enc -aes-256-cbc -salt -in bigfile.zip -out bigfile.enc -pass file:aes_key.txt

# ===== 解密步骤 =====
# 1. 用RSA私钥解密AES密钥
openssl pkeyutl -decrypt -in aes_key.enc -out aes_key_decrypted.txt -inkey private_key.pem -pkeyopt rsa_padding_mode:oaep -pkeyopt rsa_oaep_md:sha256

# 2. 用AES密钥解密文件
openssl enc -aes-256-cbc -d -in bigfile.enc -out bigfile_decrypted.zip -pass file:aes_key_decrypted.txt
```

### 二、OpenSSL ECDSA/ECC 命令速查表

#### EC 密钥生成与管理

```bash
# ===== 列出所有支持的椭圆曲线 =====
openssl ecparam -list_curves
# 常用曲线: prime256v1 (P-256), secp384r1 (P-384), secp521r1 (P-521)

# ===== 生成 P-256 曲线的 EC 私钥 =====
openssl ecparam -name prime256v1 -genkey -noout -out ec_private.pem

# ===== 生成 P-384 曲线的 EC 私钥 =====
openssl ecparam -name secp384r1 -genkey -noout -out ec_private_384.pem

# ===== 另一种方式生成EC密钥（新版命令）=====
openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:prime256v1 -out ec_private.pem

# ===== 从私钥提取公钥 =====
openssl ec -in ec_private.pem -pubout -out ec_public.pem

# ===== 查看EC私钥详情 =====
openssl ec -in ec_private.pem -text -noout

# ===== 查看EC公钥详情 =====
openssl ec -pubin -in ec_public.pem -text -noout

# ===== 私钥加密保护 =====
openssl ec -in ec_private.pem -aes256 -out ec_private_enc.pem
```

#### ECDSA 数字签名

```bash
# ===== ECDSA 签名（SHA256 + P-256）=====
openssl dgst -sha256 -sign ec_private.pem -out ecdsa_sig.bin data.txt

# ===== ECDSA 验签 =====
openssl dgst -sha256 -verify ec_public.pem -signature ecdsa_sig.bin data.txt

# ===== ECDSA 签名（SHA384 + P-384）=====
openssl dgst -sha384 -sign ec_private_384.pem -out ecdsa_sig_384.bin data.txt

# ===== 验证 ECDSA 签名 =====
openssl dgst -sha384 -verify ec_public_384.pem -signature ecdsa_sig_384.bin data.txt
```

#### ECDH 密钥交换

```bash
# ===== 双方各自生成密钥对 =====
openssl ecparam -name prime256v1 -genkey -noout -out alice_priv.pem
openssl ec -in alice_priv.pem -pubout -out alice_pub.pem
openssl ecparam -name prime256v1 -genkey -noout -out bob_priv.pem
openssl ec -in bob_priv.pem -pubout -out bob_pub.pem

# ===== Alice用自己私钥 + Bob公钥 计算共享密钥 =====
openssl pkeyutl -derive -inkey alice_priv.pem -peerkey bob_pub.pem -out shared_secret_alice.bin

# ===== Bob用自己私钥 + Alice公钥 计算共享密钥 =====
openssl pkeyutl -derive -inkey bob_priv.pem -peerkey alice_pub.pem -out shared_secret_bob.bin

# ===== 验证共享密钥一致 =====
diff shared_secret_alice.bin shared_secret_bob.bin
# 两个文件应该完全相同
```

### 三、Python 非对称加密代码示例

#### RSA 加密解密与签名

```python
# pip install cryptography
from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.backends import default_backend
import base64

# ===== 生成 RSA 密钥对 =====
def generate_rsa_keypair(key_size: int = 2048) -> tuple:
    """生成RSA密钥对，返回(私钥, 公钥)"""
    private_key = rsa.generate_private_key(
        public_exponent=65537,
        key_size=key_size,
        backend=default_backend()
    )
    public_key = private_key.public_key()
    return private_key, public_key

# ===== 保存私钥到文件 =====
def save_private_key(private_key, filename: str, password: bytes = None):
    """保存私钥到PEM文件"""
    encryption_algorithm = serialization.NoEncryption()
    if password:
        encryption_algorithm = serialization.BestAvailableEncryption(password)
    
    pem = private_key.private_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PrivateFormat.PKCS8,
        encryption_algorithm=encryption_algorithm
    )
    with open(filename, 'wb') as f:
        f.write(pem)

# ===== 保存公钥到文件 =====
def save_public_key(public_key, filename: str):
    """保存公钥到PEM文件"""
    pem = public_key.public_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PublicFormat.SubjectPublicKeyInfo
    )
    with open(filename, 'wb') as f:
        f.write(pem)

# ===== 加载私钥 =====
def load_private_key(filename: str, password: bytes = None):
    """从PEM文件加载私钥"""
    with open(filename, 'rb') as f:
        return serialization.load_pem_private_key(
            f.read(),
            password=password,
            backend=default_backend()
        )

# ===== 加载公钥 =====
def load_public_key(filename: str):
    """从PEM文件加载公钥"""
    with open(filename, 'rb') as f:
        return serialization.load_pem_public_key(
            f.read(),
            backend=default_backend()
        )

# ===== RSA-OAEP 加密 =====
def rsa_encrypt(plaintext: bytes, public_key) -> bytes:
    """使用RSA-OAEP加密（推荐）"""
    ciphertext = public_key.encrypt(
        plaintext,
        padding.OAEP(
            mgf=padding.MGF1(algorithm=hashes.SHA256()),
            algorithm=hashes.SHA256(),
            label=None
        )
    )
    return ciphertext

# ===== RSA-OAEP 解密 =====
def rsa_decrypt(ciphertext: bytes, private_key) -> bytes:
    """使用RSA-OAEP解密"""
    plaintext = private_key.decrypt(
        ciphertext,
        padding.OAEP(
            mgf=padding.MGF1(algorithm=hashes.SHA256()),
            algorithm=hashes.SHA256(),
            label=None
        )
    )
    return plaintext

# ===== RSA-PSS 签名 =====
def rsa_sign(message: bytes, private_key) -> bytes:
    """使用RSA-PSS签名"""
    signature = private_key.sign(
        message,
        padding.PSS(
            mgf=padding.MGF1(hashes.SHA256()),
            salt_length=padding.PSS.MAX_LENGTH
        ),
        hashes.SHA256()
    )
    return signature

# ===== RSA-PSS 验签 =====
def rsa_verify(message: bytes, signature: bytes, public_key) -> bool:
    """验证RSA-PSS签名，成功返回True"""
    try:
        public_key.verify(
            signature,
            message,
            padding.PSS(
                mgf=padding.MGF1(hashes.SHA256()),
                salt_length=padding.PSS.MAX_LENGTH
            ),
            hashes.SHA256()
        )
        return True
    except Exception:
        return False

# ===== 使用示例 =====
if __name__ == "__main__":
    # 生成密钥对
    priv, pub = generate_rsa_keypair(2048)
    print("RSA密钥对生成成功")
    
    # 加密解密测试
    message = b"Hello, RSA非对称加密！"
    encrypted = rsa_encrypt(message, pub)
    decrypted = rsa_decrypt(encrypted, priv)
    print(f"\nRSA加密测试:")
    print(f"密文(hex): {encrypted.hex()[:64]}...")
    print(f"解密结果: {decrypted.decode()}")
    assert decrypted == message, "解密失败！"
    
    # 签名验签测试
    signature = rsa_sign(message, priv)
    is_valid = rsa_verify(message, signature, pub)
    print(f"\nRSA签名测试:")
    print(f"签名(hex): {signature.hex()[:64]}...")
    print(f"验签结果: {'通过' if is_valid else '失败'}")
```

#### ECDSA 签名与 ECDH 密钥交换

```python
from cryptography.hazmat.primitives.asymmetric import ec
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives.kdf.hkdf import HKDF
import os

# ===== 生成 ECDSA 密钥对 =====
def generate_ec_keypair(curve=ec.SECP256R1()):
    """生成椭圆曲线密钥对"""
    private_key = ec.generate_private_key(curve, default_backend())
    public_key = private_key.public_key()
    return private_key, public_key

# ===== ECDSA 签名 =====
def ecdsa_sign(message: bytes, private_key) -> bytes:
    """ECDSA签名"""
    signature = private_key.sign(
        message,
        ec.ECDSA(hashes.SHA256())
    )
    return signature

# ===== ECDSA 验签 =====
def ecdsa_verify(message: bytes, signature: bytes, public_key) -> bool:
    """验证ECDSA签名"""
    try:
        public_key.verify(signature, message, ec.ECDSA(hashes.SHA256()))
        return True
    except Exception:
        return False

# ===== ECDH 密钥交换 + HKDF 派生会话密钥 =====
def ecdh_derive_key(private_key, peer_public_key, info: bytes = b"") -> bytes:
    """
    ECDH计算共享密钥，然后用HKDF派生出对称密钥
    """
    # ECDH 计算共享秘密
    shared_secret = private_key.exchange(ec.ECDH(), peer_public_key)
    
    # HKDF 派生密钥
    derived_key = HKDF(
        algorithm=hashes.SHA256(),
        length=32,  # 派生256位密钥
        salt=None,
        info=info,
        backend=default_backend()
    ).derive(shared_secret)
    
    return derived_key

# ===== 使用示例 =====
if __name__ == "__main__":
    # ECDSA 签名测试
    priv, pub = generate_ec_keypair(ec.SECP256R1())
    message = b"Hello, ECDSA!"
    sig = ecdsa_sign(message, priv)
    valid = ecdsa_verify(message, sig, pub)
    print(f"ECDSA签名长度: {len(sig)} 字节")
    print(f"ECDSA验签结果: {'通过' if valid else '失败'}")
    
    # ECDH 密钥交换测试
    alice_priv, alice_pub = generate_ec_keypair()
    bob_priv, bob_pub = generate_ec_keypair()
    
    alice_shared = ecdh_derive_key(alice_priv, bob_pub, b"session_key")
    bob_shared = ecdh_derive_key(bob_priv, alice_pub, b"session_key")
    
    print(f"\nECDH密钥交换测试:")
    print(f"Alice计算的密钥: {alice_shared.hex()[:32]}...")
    print(f"Bob计算的密钥:   {bob_shared.hex()[:32]}...")
    print(f"密钥一致: {alice_shared == bob_shared}")
```

#### 混合加密完整示例（RSA + AES）

```python
from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.primitives import hashes, padding as sym_padding
from cryptography.hazmat.backends import default_backend
import os
import struct

def hybrid_encrypt(plaintext: bytes, rsa_public_key) -> bytes:
    """
    混合加密：RSA加密AES密钥 + AES-GCM加密数据
    返回格式: [encrypted_key_len(2字节)][encrypted_aes_key][nonce(12字节)][tag(16字节)][ciphertext]
    """
    # 1. 生成随机AES-256密钥
    aes_key = os.urandom(32)
    
    # 2. 用RSA公钥加密AES密钥
    encrypted_aes_key = rsa_public_key.encrypt(
        aes_key,
        padding.OAEP(
            mgf=padding.MGF1(algorithm=hashes.SHA256()),
            algorithm=hashes.SHA256(),
            label=None
        )
    )
    
    # 3. 用AES-GCM加密明文
    nonce = os.urandom(12)
    cipher = Cipher(algorithms.AES(aes_key), modes.GCM(nonce), backend=default_backend())
    encryptor = cipher.encryptor()
    ciphertext = encryptor.update(plaintext) + encryptor.finalize()
    tag = encryptor.tag
    
    # 4. 打包所有数据
    key_len = len(encrypted_aes_key)
    result = struct.pack('!H', key_len) + encrypted_aes_key + nonce + tag + ciphertext
    return result

def hybrid_decrypt(encrypted_data: bytes, rsa_private_key) -> bytes:
    """混合解密"""
    # 1. 解析数据
    offset = 0
    key_len = struct.unpack('!H', encrypted_data[offset:offset+2])[0]
    offset += 2
    
    encrypted_aes_key = encrypted_data[offset:offset+key_len]
    offset += key_len
    
    nonce = encrypted_data[offset:offset+12]
    offset += 12
    
    tag = encrypted_data[offset:offset+16]
    offset += 16
    
    ciphertext = encrypted_data[offset:]
    
    # 2. 用RSA私钥解密AES密钥
    aes_key = rsa_private_key.decrypt(
        encrypted_aes_key,
        padding.OAEP(
            mgf=padding.MGF1(algorithm=hashes.SHA256()),
            algorithm=hashes.SHA256(),
            label=None
        )
    )
    
    # 3. 用AES-GCM解密密文
    cipher = Cipher(algorithms.AES(aes_key), modes.GCM(nonce, tag), backend=default_backend())
    decryptor = cipher.decryptor()
    plaintext = decryptor.update(ciphertext) + decryptor.finalize()
    
    return plaintext

# 使用示例
if __name__ == "__main__":
    # 生成RSA密钥对
    private_key = rsa.generate_private_key(65537, 2048, default_backend())
    public_key = private_key.public_key()
    
    # 加密大文件内容（混合加密支持任意大小数据）
    large_data = b"这是一段很长的敏感数据..." * 100
    encrypted = hybrid_encrypt(large_data, public_key)
    decrypted = hybrid_decrypt(encrypted, private_key)
    
    print(f"原始数据大小: {len(large_data)} 字节")
    print(f"加密后大小: {len(encrypted)} 字节")
    print(f"解密成功: {decrypted == large_data}")
```

### 四、非对称加密速查表

| 功能 | RSA | ECDSA/ECC |
|------|-----|-----------|
| 密钥生成 | `openssl genrsa -out key.pem 2048` | `openssl ecparam -name prime256v1 -genkey -out key.pem` |
| 提取公钥 | `openssl rsa -in key.pem -pubout -out pub.pem` | `openssl ec -in key.pem -pubout -out pub.pem` |
| 加密 | `openssl pkeyutl -encrypt ...` | 不直接用于加密（用ECDH派生密钥） |
| 签名 | `openssl dgst -sha256 -sign key.pem ...` | `openssl dgst -sha256 -sign key.pem ...` |
| 验签 | `openssl dgst -sha256 -verify pub.pem ...` | `openssl dgst -sha256 -verify pub.pem ...` |
| 密钥交换 | 不支持（用DH） | 支持（ECDH） |
| 安全强度(2048位/256位) | 112位 | 128位 |
| 签名长度 | 256字节(2048位) | 64-72字节(P-256) |

> **安全建议**：
> - 新系统优先使用ECC（密钥小、速度快、安全强度高）
> - RSA密钥长度至少2048位，推荐4096位
> - 加密使用OAEP填充，签名使用PSS填充
> - 大文件用混合加密（非对称加密对称密钥 + 对称加密数据）
> - 永远不要用ECB模式，不要自己实现密码学算法
