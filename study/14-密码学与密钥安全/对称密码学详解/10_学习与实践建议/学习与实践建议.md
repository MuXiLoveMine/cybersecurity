### 一、OpenSSL 对称加密命令速查表

#### AES 加密/解密

```bash
# ===== AES-256-CBC 加密文件 =====
openssl enc -aes-256-cbc -salt -in plain.txt -out cipher.enc -pass pass:MySecretPassword
# -aes-256-cbc: 使用AES-256-CBC模式
# -salt: 添加盐值增加安全性
# -in/-out: 输入/输出文件
# -pass pass:xxx: 使用密码派生密钥

# ===== AES-256-CBC 解密文件 =====
openssl enc -aes-256-cbc -d -in cipher.enc -out decrypted.txt -pass pass:MySecretPassword
# -d: 解密模式

# ===== AES-256-GCM 加密（认证加密模式）=====
openssl enc -aes-256-gcm -salt -in plain.txt -out cipher.enc -pass pass:MySecretPassword
# GCM模式提供机密性+完整性认证

# ===== 使用 Base64 编码输出 =====
openssl enc -aes-256-cbc -a -salt -in plain.txt -out cipher.b64 -pass pass:MySecretPassword
# -a: 输出Base64编码，便于文本传输

# ===== 使用十六进制密钥（直接指定密钥，不派生）=====
openssl enc -aes-256-cbc -in plain.txt -out cipher.enc -K 00112233445566778899aabbccddeeff00112233445566778899aabbccddeeff -iv 0102030405060708090a0b0c0d0e0f10
# -K: 十六进制密钥（AES-256需要64个hex字符=32字节）
# -iv: 初始化向量（CBC模式需要16字节IV）

# ===== 列出所有支持的加密算法 =====
openssl enc -list
```

#### DES / 3DES 加密/解密

```bash
# ===== DES-CBC 加密（不推荐用于生产，仅用于学习）=====
openssl enc -des-cbc -salt -in plain.txt -out des_cipher.enc -pass pass:password

# ===== 3DES（三重DES）加密 =====
openssl enc -des-ede3-cbc -salt -in plain.txt -out 3des_cipher.enc -pass pass:password
# des-ede3-cbc: 三重DES EDE模式，CBC分组

# ===== DES 解密 =====
openssl enc -des-cbc -d -in des_cipher.enc -out decrypted.txt -pass pass:password
```

#### 对称加密实用技巧

```bash
# ===== 加密目录（先打包再加密）=====
tar czf - /path/to/dir | openssl enc -aes-256-cbc -salt -out backup.tar.gz.enc -pass pass:MyPass

# ===== 解密并解包 =====
openssl enc -aes-256-cbc -d -in backup.tar.gz.enc -pass pass:MyPass | tar xzf -

# ===== 生成随机密钥 =====
openssl rand -hex 32    # 生成256位（32字节）随机密钥，十六进制输出
openssl rand -base64 32 # 生成Base64编码的随机密钥

# ===== 生成随机IV =====
openssl rand -hex 16    # 生成128位（16字节）随机IV

# ===== 加密速度基准测试 =====
openssl speed aes-128-cbc aes-256-cbc des-cbc
# 测试不同算法的加密性能
```

### 二、Python 对称加密代码示例

#### 使用 cryptography 库（推荐）

```python
# pip install cryptography
from cryptography.fernet import Fernet
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.primitives import padding
from cryptography.hazmat.backends import default_backend
import os
import base64

# ===== 方式1：Fernet（高层封装，推荐新手使用）=====
# Fernet内部使用 AES-128-CBC + HMAC-SHA256，安全且易用

def fernet_encrypt(message: bytes, key: bytes) -> bytes:
    """使用Fernet加密"""
    f = Fernet(key)
    return f.encrypt(message)

def fernet_decrypt(token: bytes, key: bytes) -> bytes:
    """使用Fernet解密"""
    f = Fernet(key)
    return f.decrypt(token)

# 生成密钥
fernet_key = Fernet.generate_key()
print(f"Fernet密钥: {fernet_key.decode()}")

# 加密解密测试
plaintext = b"Hello, 对称加密！"
encrypted = fernet_encrypt(plaintext, fernet_key)
decrypted = fernet_decrypt(encrypted, fernet_key)
print(f"加密后: {encrypted}")
print(f"解密后: {decrypted.decode()}")

# ===== 方式2：AES-256-CBC 手动实现（底层操作）=====

def aes_256_cbc_encrypt(plaintext: bytes, key: bytes) -> tuple:
    """
    AES-256-CBC 加密
    返回 (密文, IV)
    """
    # 生成随机IV（16字节）
    iv = os.urandom(16)
    
    # PKCS7 填充
    padder = padding.PKCS7(128).padder()  # AES块大小为128位
    padded_data = padder.update(plaintext) + padder.finalize()
    
    # 创建加密器
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    encryptor = cipher.encryptor()
    
    # 加密
    ciphertext = encryptor.update(padded_data) + encryptor.finalize()
    
    return ciphertext, iv

def aes_256_cbc_decrypt(ciphertext: bytes, key: bytes, iv: bytes) -> bytes:
    """AES-256-CBC 解密"""
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    decryptor = cipher.decryptor()
    
    # 解密
    padded_data = decryptor.update(ciphertext) + decryptor.finalize()
    
    # 移除PKCS7填充
    unpadder = padding.PKCS7(128).unpadder()
    plaintext = unpadder.update(padded_data) + unpadder.finalize()
    
    return plaintext

# 使用示例
key = os.urandom(32)  # 256位密钥
plaintext = b"敏感数据: password123"

ciphertext, iv = aes_256_cbc_encrypt(plaintext, key)
decrypted = aes_256_cbc_decrypt(ciphertext, key, iv)

print(f"\nAES-256-CBC 加密示例:")
print(f"密钥(hex): {key.hex()}")
print(f"IV(hex): {iv.hex()}")
print(f"密文(hex): {ciphertext.hex()}")
print(f"解密结果: {decrypted.decode()}")

# ===== 方式3：AES-256-GCM 认证加密（推荐生产使用）=====
# GCM模式同时提供机密性和完整性认证

def aes_256_gcm_encrypt(plaintext: bytes, key: bytes, associated_data: bytes = b"") -> tuple:
    """
    AES-256-GCM 加密（认证加密）
    返回 (密文, nonce, tag)
    """
    # 生成随机nonce（12字节是GCM推荐值）
    nonce = os.urandom(12)
    
    cipher = Cipher(algorithms.AES(key), modes.GCM(nonce), backend=default_backend())
    encryptor = cipher.encryptor()
    
    # 添加关联数据（认证但不加密）
    encryptor.authenticate_additional_data(associated_data)
    
    # 加密
    ciphertext = encryptor.update(plaintext) + encryptor.finalize()
    
    # 获取认证标签
    tag = encryptor.tag
    
    return ciphertext, nonce, tag

def aes_256_gcm_decrypt(ciphertext: bytes, key: bytes, nonce: bytes, 
                        tag: bytes, associated_data: bytes = b"") -> bytes:
    """AES-256-GCM 解密（验证认证标签）"""
    cipher = Cipher(algorithms.AES(key), modes.GCM(nonce, tag), backend=default_backend())
    decryptor = cipher.decryptor()
    
    # 添加关联数据
    decryptor.authenticate_additional_data(associated_data)
    
    # 解密（tag验证失败会抛出异常）
    plaintext = decryptor.update(ciphertext) + decryptor.finalize()
    
    return plaintext

# 使用示例
key = os.urandom(32)
plaintext = b"重要机密信息"
aad = b"metadata: user=admin,date=2024-01-01"  # 关联数据

ciphertext, nonce, tag = aes_256_gcm_encrypt(plaintext, key, aad)
decrypted = aes_256_gcm_decrypt(ciphertext, key, nonce, tag, aad)

print(f"\nAES-256-GCM 认证加密示例:")
print(f"密文(hex): {ciphertext.hex()}")
print(f"认证标签(hex): {tag.hex()}")
print(f"解密结果: {decrypted.decode()}")

# ===== 方式4：文件加密工具函数 =====

def encrypt_file(input_path: str, output_path: str, password: str):
    """
    基于密码的文件加密（使用PBKDF2派生密钥 + AES-256-CBC）
    文件格式: [salt(16字节)][iv(16字节)][密文]
    """
    from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
    from cryptography.hazmat.primitives import hashes
    
    # 生成随机盐
    salt = os.urandom(16)
    
    # PBKDF2 派生密钥（100000次迭代）
    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),
        length=32,
        salt=salt,
        iterations=100000,
        backend=default_backend()
    )
    key = kdf.derive(password.encode())
    
    # 读取文件
    with open(input_path, 'rb') as f:
        plaintext = f.read()
    
    # 加密
    iv = os.urandom(16)
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    encryptor = cipher.encryptor()
    padder = padding.PKCS7(128).padder()
    padded_data = padder.update(plaintext) + padder.finalize()
    ciphertext = encryptor.update(padded_data) + encryptor.finalize()
    
    # 写入文件：salt + iv + ciphertext
    with open(output_path, 'wb') as f:
        f.write(salt + iv + ciphertext)
    
    print(f"文件加密完成: {output_path}")

def decrypt_file(input_path: str, output_path: str, password: str):
    """基于密码的文件解密"""
    from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
    from cryptography.hazmat.primitives import hashes
    
    # 读取文件
    with open(input_path, 'rb') as f:
        data = f.read()
    
    # 提取 salt 和 iv
    salt = data[:16]
    iv = data[16:32]
    ciphertext = data[32:]
    
    # 派生密钥
    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),
        length=32,
        salt=salt,
        iterations=100000,
        backend=default_backend()
    )
    key = kdf.derive(password.encode())
    
    # 解密
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    decryptor = cipher.decryptor()
    padded_data = decryptor.update(ciphertext) + decryptor.finalize()
    
    # 去填充
    unpadder = padding.PKCS7(128).unpadder()
    plaintext = unpadder.update(padded_data) + unpadder.finalize()
    
    # 写入输出
    with open(output_path, 'wb') as f:
        f.write(plaintext)
    
    print(f"文件解密完成: {output_path}")

# 使用示例
# encrypt_file("secret.txt", "secret.enc", "MyStrongPassword")
# decrypt_file("secret.enc", "secret_decrypted.txt", "MyStrongPassword")
```

#### 使用 PyCryptodome 库

```python
# pip install pycryptodome
from Crypto.Cipher import AES
from Crypto.Util.Padding import pad, unpad
from Crypto.Random import get_random_bytes
import base64

def aes_encrypt(plaintext: str, key: bytes) -> str:
    """AES-256-CBC 加密，返回Base64编码的密文（含IV）"""
    iv = get_random_bytes(16)
    cipher = AES.new(key, AES.MODE_CBC, iv)
    ciphertext = cipher.encrypt(pad(plaintext.encode(), AES.block_size))
    # IV + 密文一起Base64编码
    return base64.b64encode(iv + ciphertext).decode()

def aes_decrypt(encrypted: str, key: bytes) -> str:
    """AES-256-CBC 解密"""
    raw = base64.b64decode(encrypted)
    iv = raw[:16]
    ciphertext = raw[16:]
    cipher = AES.new(key, AES.MODE_CBC, iv)
    plaintext = unpad(cipher.decrypt(ciphertext), AES.block_size)
    return plaintext.decode()

# 使用示例
key = get_random_bytes(32)  # AES-256
encrypted = aes_encrypt("Hello PyCryptodome!", key)
decrypted = aes_decrypt(encrypted, key)
print(f"加密: {encrypted}")
print(f"解密: {decrypted}")
```

### 三、对称加密安全实践速查

| 场景 | 推荐算法/模式 | 密钥长度 | 说明 |
|------|--------------|----------|------|
| 通用文件加密 | AES-256-GCM | 256位 | 认证加密，首选 |
| 数据库字段加密 | AES-256-CBC + HMAC | 256位 | 需单独验证完整性 |
| 磁盘加密 | AES-256-XTS | 256位 | 扇区加密专用模式 |
| 流加密场景 | AES-256-CTR | 256位 | 计数器模式，无填充 |
| 密码派生密钥 | PBKDF2/Argon2 | - | 迭代次数≥100000 |

> **安全提示**：
> - 永远不要使用ECB模式
> - 优先使用GCM等认证加密模式
> - IV必须随机且唯一，但不需要保密
> - 密钥必须使用密码学安全的随机数生成器
> - 密码不能直接作为密钥，必须用KDF派生
