# 数字签名实践应用

### 11.1 软件代码签名

```python
"""
代码签名实践
"""
import hashlib
import os
from cryptography.hazmat.primitives.asymmetric import ec
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric.utils import (
    encode_dss_signature, decode_dss_signature
)
from cryptography.hazmat.primitives.serialization import (
    Encoding, PrivateFormat, NoEncryption, PublicFormat
)
import json
import time

class CodeSigner:
    """代码签名工具"""
    
    def __init__(self):
        self.private_key = ec.generate_private_key(ec.SECP256R1())
        self.public_key = self.private_key.public_key()
    
    def sign_file(self, file_path: str) -> dict:
        """
        对文件进行签名
        
        Args:
            file_path: 文件路径
            
        Returns:
            签名信息
        """
        # 计算文件哈希
        hasher = hashlib.sha256()
        with open(file_path, 'rb') as f:
            while chunk := f.read(8192):
                hasher.update(chunk)
        file_hash = hasher.hexdigest()
        
        # 获取文件信息
        file_size = os.path.getsize(file_path)
        file_name = os.path.basename(file_path)
        
        # 构建签名内容
        sign_content = {
            'file_name': file_name,
            'file_size': file_size,
            'file_hash': file_hash,
            'hash_algorithm': 'SHA-256',
            'timestamp': int(time.time()),
        }
        
        # 签名
        content_bytes = json.dumps(sign_content, 
                                   sort_keys=True).encode()
        signature = self.private_key.sign(
            content_bytes,
            ec.ECDSA(hashes.SHA256())
        )
        
        # 导出公钥
        pub_pem = self.public_key.public_bytes(
            Encoding.PEM,
            PublicFormat.SubjectPublicKeyInfo
        ).decode()
        
        return {
            'file_info': sign_content,
            'signature': signature.hex(),
            'public_key': pub_pem,
            'sign_algorithm': 'ECDSA-SHA256',
            'sign_time': time.strftime(
                '%Y-%m-%d %H:%M:%S',
                time.localtime(sign_content['timestamp'])
            )
        }
    
    def verify_file(self, file_path: str, sig_info: dict) -> dict:
        """
        验证文件签名
        
        Args:
            file_path: 文件路径
            sig_info: 签名信息
            
        Returns:
            验证结果
        """
        result = {
            'valid': False,
            'file_intact': False,
            'signature_valid': False,
            'errors': []
        }
        
        # 1. 验证文件哈希
        hasher = hashlib.sha256()
        with open(file_path, 'rb') as f:
            while chunk := f.read(8192):
                hasher.update(chunk)
        current_hash = hasher.hexdigest()
        
        expected_hash = sig_info['file_info']['file_hash']
        result['file_intact'] = (current_hash == expected_hash)
        
        if not result['file_intact']:
            result['errors'].append('文件已被修改')
        
        # 2. 验证签名
        content_bytes = json.dumps(
            sig_info['file_info'],
            sort_keys=True
        ).encode()
        
        signature = bytes.fromhex(sig_info['signature'])
        
        try:
            self.public_key.verify(
                signature,
                content_bytes,
                ec.ECDSA(hashes.SHA256())
            )
            result['signature_valid'] = True
        except Exception as e:
            result['signature_valid'] = False
            result['errors'].append(f'签名验证失败: {e}')
        
        # 3. 综合结果
        result['valid'] = result['file_intact'] and result['signature_valid']
        
        return result


# 使用示例
signer = CodeSigner()

# 签名文件
test_file = "test_program.exe"
with open(test_file, 'wb') as f:
    f.write(b"executable content here")

sig_info = signer.sign_file(test_file)
print("签名信息:")
print(f"  文件: {sig_info['file_info']['file_name']}")
print(f"  哈希: {sig_info['file_info']['file_hash'][:32]}...")
print(f"  签名: {sig_info['signature'][:32]}...")
print(f"  签名时间: {sig_info['sign_time']}")

# 验证文件
verify_result = signer.verify_file(test_file, sig_info)
print(f"\n验证结果: {verify_result['valid']}")

# 模拟篡改
with open(test_file, 'ab') as f:
    f.write(b"malicious payload")

verify_result = signer.verify_file(test_file, sig_info)
print(f"篡改后验证: {verify_result['valid']}")
print(f"错误: {verify_result['errors']}")
```

### 11.2 电子合同签名

```python
"""
电子合同签名系统
支持多方签名和签名顺序控制
"""

class ElectronicContract:
    """电子合同签名系统"""
    
    def __init__(self, title: str, content: str):
        self.title = title
        self.content = content
        self.content_hash = hashlib.sha256(
            content.encode('utf-8')
        ).hexdigest()
        self.signatures = []
        self.created_at = time.time()
        self.signing_order = []
    
    def add_signer(self, name: str, role: str, 
                   must_sign_before: list = None):
        """添加签名者"""
        self.signing_order.append({
            'name': name,
            'role': role,
            'must_sign_before': must_sign_before or [],
            'signed': False,
            'signature': None,
            'signed_at': None
        })
    
    def sign(self, signer_name: str, private_key, 
             user_id: str = None) -> bool:
        """
        签名合同
        
        Args:
            signer_name: 签名者姓名
            private_key: 私钥
            user_id: 用户ID（SM2用）
            
        Returns:
            签名是否成功
        """
        # 查找签名者
        signer = None
        for s in self.signing_order:
            if s['name'] == signer_name and not s['signed']:
                signer = s
                break
        
        if not signer:
            print(f"签名者 {signer_name} 不在待签名列表中")
            return False
        
        # 检查签名顺序
        for dep_name in signer['must_sign_before']:
            dep = next(
                (s for s in self.signing_order if s['name'] == dep_name),
                None
            )
            if dep and not dep['signed']:
                print(f"需要 {dep_name} 先签名")
                return False
        
        # 签名内容
        sign_content = f"{self.title}\n{self.content}\n{self.content_hash}"
        
        # 使用私钥签名
        if hasattr(private_key, 'sign'):
            # RSA或ECDSA
            signature = private_key.sign(
                sign_content.encode(),
                padding.PSS(
                    mgf=padding.MGF1(hashes.SHA256()),
                    salt_length=padding.PSS.MAX_LENGTH
                ),
                hashes.SHA256()
            )
        else:
            signature = b''
        
        signer['signed'] = True
        signer['signature'] = signature.hex()
        signer['signed_at'] = time.time()
        
        print(f"{signer_name} 签名成功")
        return True
    
    def is_fully_signed(self) -> bool:
        """检查是否所有签名者都已签名"""
        return all(s['signed'] for s in self.signing_order)
    
    def get_signing_status(self) -> list:
        """获取签名状态"""
        return [
            {
                'name': s['name'],
                'role': s['role'],
                'signed': s['signed'],
                'signed_at': s['signed_at']
            }
            for s in self.signing_order
        ]
    
    def to_dict(self) -> dict:
        """导出合同信息"""
        return {
            'title': self.title,
            'content': self.content,
            'content_hash': self.content_hash,
            'created_at': self.created_at,
            'signatures': self.signing_order,
            'fully_signed': self.is_fully_signed()
        }


# 使用示例
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.primitives.asymmetric import padding

# 创建合同
contract = ElectronicContract(
    title="技术服务合同",
    content="甲方委托乙方提供技术服务，服务费用为100万元..."
)

# 添加签名者（设置签名顺序）
contract.add_signer("张三", "甲方代表")
contract.add_signer("李四", "乙方代表", must_sign_before=["张三"])
contract.add_signer("王五", "见证人", must_sign_before=["张三", "李四"])

# 生成密钥对
keys = {name: rsa.generate_private_key(65537, 2048) 
        for name in ["张三", "李四", "王五"]}

# 签名
contract.sign("张三", keys["张三"])
contract.sign("李四", keys["李四"])
contract.sign("王五", keys["王五"])

# 检查状态
print(f"合同已完全签名: {contract.is_fully_signed()}")
for s in contract.get_signing_status():
    print(f"  {s['name']} ({s['role']}): {'已签名' if s['signed'] else '未签名'}")
```

### 11.3 区块链交易签名

```python
"""
区块链交易签名示例
以Bitcoin风格为例
"""

import hashlib
import struct

class BitcoinTransaction:
    """简化的比特币交易结构"""
    
    def __init__(self):
        self.inputs = []
        self.outputs = []
        self.locktime = 0
    
    def add_input(self, txid: str, vout: int, 
                  script_pubkey: bytes):
        """添加输入"""
        self.inputs.append({
            'txid': bytes.fromhex(txid),
            'vout': vout,
            'script_pubkey': script_pubkey,
            'script_sig': b'',
            'sequence': 0xFFFFFFFF
        })
    
    def add_output(self, address: str, value: int):
        """添加输出"""
        self.outputs.append({
            'address': address,
            'value': value,
            'script_pubkey': b''
        })
    
    def serialize_for_signing(self, input_index: int) -> bytes:
        """
        序列化交易用于签名（简化版）
        
        Bitcoin使用SIGHASH标志控制签名范围
        """
        # 在实际实现中需要处理SIGHASH逻辑
        data = b''
        
        # 版本号
        data += struct.pack('<I', 1)
        
        # 输入
        data += struct.pack('<B', len(self.inputs))
        for i, tx_in in enumerate(self.inputs):
            data += tx_in['txid'][::-1]  # 小端序
            data += struct.pack('<I', tx_in['vout'])
            if i == input_index:
                data += struct.pack('<B', len(tx_in['script_pubkey']))
                data += tx_in['script_pubkey']
            else:
                data += b'\x00'  # 空脚本
            data += struct.pack('<I', tx_in['sequence'])
        
        # 输出
        data += struct.pack('<B', len(self.outputs))
        for tx_out in self.outputs:
            data += struct.pack('<Q', tx_out['value'])
            data += struct.pack('<B', len(tx_out['script_pubkey']))
            data += tx_out['script_pubkey']
        
        # 锁定时间
        data += struct.pack('<I', self.locktime)
        
        # SIGHASH标志
        data += struct.pack('<I', 0x01)  # SIGHASH_ALL
        
        return data
    
    def sign_input(self, input_index: int, private_key) -> bytes:
        """签名输入"""
        from cryptography.hazmat.primitives.asymmetric import ec
        from cryptography.hazmat.primitives import hashes
        
        # 序列化
        ser = self.serialize_for_signing(input_index)
        
        # 计算哈希
        tx_hash = hashlib.sha256(hashlib.sha256(ser).digest()).digest()
        
        # 签名（使用ECDSA）
        # Bitcoin使用secp256k1曲线
        der_sig = private_key.sign(tx_hash, ec.ECDSA(hashes.SHA256()))
        
        # 添加SIGHASH标志
        sig_with_hashtype = der_sig + b'\x01'
        
        return sig_with_hashtype
```

### 11.4 JWT签名

```python
"""
JWT（JSON Web Token）签名
JWT是Web应用中常用的令牌格式
"""
import json
import base64
import hmac
import hashlib
from cryptography.hazmat.primitives.asymmetric import rsa, ec
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import padding

class JWTEncoder:
    """JWT编码/解码器"""
    
    @staticmethod
    def base64url_encode(data: bytes) -> str:
        """Base64URL编码"""
        return base64.urlsafe_b64encode(data).rstrip(b'=').decode()
    
    @staticmethod
    def base64url_decode(data: str) -> bytes:
        """Base64URL解码"""
        padding = 4 - len(data) % 4
        if padding != 4:
            data += '=' * padding
        return base64.urlsafe_b64decode(data)
    
    @staticmethod
    def encode_hs256(payload: dict, secret: str) -> str:
        """HMAC-SHA256签名JWT"""
        header = {"alg": "HS256", "typ": "JWT"}
        
        header_b64 = JWTEncoder.base64url_encode(
            json.dumps(header).encode()
        )
        payload_b64 = JWTEncoder.base64url_encode(
            json.dumps(payload).encode()
        )
        
        signing_input = f"{header_b64}.{payload_b64}"
        signature = hmac.new(
            secret.encode(),
            signing_input.encode(),
            hashlib.sha256
        ).digest()
        
        sig_b64 = JWTEncoder.base64url_encode(signature)
        
        return f"{header_b64}.{payload_b64}.{sig_b64}"
    
    @staticmethod
    def encode_rs256(payload: dict, private_key) -> str:
        """RSA-SHA256签名JWT"""
        header = {"alg": "RS256", "typ": "JWT"}
        
        header_b64 = JWTEncoder.base64url_encode(
            json.dumps(header).encode()
        )
        payload_b64 = JWTEncoder.base64url_encode(
            json.dumps(payload).encode()
        )
        
        signing_input = f"{header_b64}.{payload_b64}"
        signature = private_key.sign(
            signing_input.encode(),
            padding.PKCS1v15(),
            hashes.SHA256()
        )
        
        sig_b64 = JWTEncoder.base64url_encode(signature)
        
        return f"{header_b64}.{payload_b64}.{sig_b64}"
    
    @staticmethod
    def encode_es256(payload: dict, private_key) -> str:
        """ECDSA-P256-SHA256签名JWT"""
        header = {"alg": "ES256", "typ": "JWT"}
        
        header_b64 = JWTEncoder.base64url_encode(
            json.dumps(header).encode()
        )
        payload_b64 = JWTEncoder.base64url_encode(
            json.dumps(payload).encode()
        )
        
        signing_input = f"{header_b64}.{payload_b64}"
        
        # ECDSA签名
        der_sig = private_key.sign(
            signing_input.encode(),
            ec.ECDSA(hashes.SHA256())
        )
        
        # 将DER格式转换为原始r||s格式
        from cryptography.hazmat.primitives.asymmetric.utils import (
            decode_dss_signature
        )
        r, s = decode_dss_signature(der_sig)
        
        # r和s各32字节
        sig_bytes = r.to_bytes(32, 'big') + s.to_bytes(32, 'big')
        sig_b64 = JWTEncoder.base64url_encode(sig_bytes)
        
        return f"{header_b64}.{payload_b64}.{sig_b64}"
    
    @staticmethod
    def encode_eddsa(payload: dict, private_key) -> str:
        """EdDSA（Ed25519）签名JWT"""
        header = {"alg": "EdDSA", "typ": "JWT"}
        
        header_b64 = JWTEncoder.base64url_encode(
            json.dumps(header).encode()
        )
        payload_b64 = JWTEncoder.base64url_encode(
            json.dumps(payload).encode()
        )
        
        signing_input = f"{header_b64}.{payload_b64}"
        signature = private_key.sign(signing_input.encode())
        
        sig_b64 = JWTEncoder.base64url_encode(signature)
        
        return f"{header_b64}.{payload_b64}.{sig_b64}"
    
    @staticmethod
    def decode(token: str, verify_key=None, algorithm=None) -> dict:
        """
        解码JWT（可选验证签名）
        
        Args:
            token: JWT令牌
            verify_key: 验证密钥/公钥
            algorithm: 签名算法
            
        Returns:
            payload字典
        """
        parts = token.split('.')
        if len(parts) != 3:
            raise ValueError("无效的JWT格式")
        
        header_b64, payload_b64, sig_b64 = parts
        
        # 解码header和payload
        header = json.loads(
            JWTEncoder.base64url_decode(header_b64)
        )
        payload = json.loads(
            JWTEncoder.base64url_decode(payload_b64)
        )
        signature = JWTEncoder.base64url_decode(sig_b64)
        
        # 验证签名
        if verify_key and algorithm:
            signing_input = f"{header_b64}.{payload_b64}"
            
            if algorithm == "HS256":
                expected = hmac.new(
                    verify_key.encode(),
                    signing_input.encode(),
                    hashlib.sha256
                ).digest()
                if not hmac.compare_digest(signature, expected):
                    raise ValueError("签名验证失败")
            
            elif algorithm == "RS256":
                try:
                    verify_key.verify(
                        signature,
                        signing_input.encode(),
                        padding.PKCS1v15(),
                        hashes.SHA256()
                    )
                except Exception:
                    raise ValueError("签名验证失败")
            
            elif algorithm == "ES256":
                # 将原始格式转换为DER
                r = int.from_bytes(signature[:32], 'big')
                s = int.from_bytes(signature[32:], 'big')
                from cryptography.hazmat.primitives.asymmetric.utils import (
                    encode_dss_signature
                )
                der_sig = encode_dss_signature(r, s)
                try:
                    verify_key.verify(
                        der_sig,
                        signing_input.encode(),
                        ec.ECDSA(hashes.SHA256())
                    )
                except Exception:
                    raise ValueError("签名验证失败")
            
            elif algorithm == "EdDSA":
                try:
                    verify_key.verify(
                        signature,
                        signing_input.encode()
                    )
                except Exception:
                    raise ValueError("签名验证失败")
        
        return payload


# 使用示例
payload = {"user_id": 123, "name": "Alice", "exp": 9999999999}

# HS256
secret = "my-secret-key"
token_hs = JWTEncoder.encode_hs256(payload, secret)
print(f"HS256 JWT: {token_hs[:50]}...")
decoded = JWTEncoder.decode(token_hs, secret, "HS256")
print(f"解码: {decoded}")

# RS256
rsa_key = rsa.generate_private_key(65537, 2048)
token_rs = JWTEncoder.encode_rs256(payload, rsa_key)
print(f"\nRS256 JWT: {token_rs[:50]}...")
decoded = JWTEncoder.decode(token_rs, rsa_key.public_key(), "RS256")
print(f"解码: {decoded}")

# ES256
ec_key = ec.generate_private_key(ec.SECP256R1())
token_es = JWTEncoder.encode_es256(payload, ec_key)
print(f"\nES256 JWT: {token_es[:50]}...")
decoded = JWTEncoder.decode(token_es, ec_key.public_key(), "ES256")
print(f"解码: {decoded}")

# EdDSA
from cryptography.hazmat.primitives.asymmetric.ed25519 import (
    Ed25519PrivateKey
)
ed_key = Ed25519PrivateKey.generate()
token_ed = JWTEncoder.encode_eddsa(payload, ed_key)
print(f"\nEdDSA JWT: {token_ed[:50]}...")
decoded = JWTEncoder.decode(token_ed, ed_key.public_key(), "EdDSA")
print(f"解码: {decoded}")
```
