### 1.1 高级函数与参数集

PowerShell 高级函数提供了参数验证、管道支持和多参数集等强大特性。

**高级函数基础**

```powershell
# 高级函数使用 [CmdletBinding] 特性
function Get-UserInfo {
    <#
    .SYNOPSIS
    获取用户信息
    .DESCRIPTION
    从 Active Directory 或本地获取用户详细信息
    .EXAMPLE
    Get-UserInfo -Username "admin"
    .EXAMPLE
    Get-UserInfo -Filter "active"
    #>
    [CmdletBinding(DefaultParameterSetName='ByName')]
    param (
        # 参数集 1：按用户名查询
        [Parameter(
            ParameterSetName='ByName',
            Mandatory=$true,
            Position=0,
            ValueFromPipeline=$true,
            ValueFromPipelineByPropertyName=$true,
            HelpMessage="输入用户名"
        )]
        [Alias('Name','User')]
        [ValidateNotNullOrEmpty()]
        [string]$Username,

        # 参数集 2：按过滤器查询
        [Parameter(
            ParameterSetName='ByFilter',
            Mandatory=$true
        )]
        [ValidateSet('active','disabled','locked','all')]
        [string]$Filter,

        # 公共参数
        [Parameter()]
        [ValidateRange(1,1000)]
        [int]$ResultLimit = 100,

        [Parameter()]
        [switch]$IncludeDisabled
    )

    BEGIN {
        # 在管道开始前执行一次
        Write-Verbose "开始获取用户信息"
        $results = @()
    }

    PROCESS {
        # 对每个管道输入执行
        foreach ($name in $Username) {
            Write-Verbose "查询用户: $name"

            $user = Get-LocalUser -Name $name -ErrorAction SilentlyContinue
            if ($user) {
                $results += [PSCustomObject]@{
                    Name        = $user.Name
                    Enabled     = $user.Enabled
                    Description = $user.Description
                    LastLogon   = $user.LastLogon
                }
            }
        }

        # 按 Filter 查询
        if ($PSCmdlet.ParameterSetName -eq 'ByFilter') {
            Write-Verbose "按过滤器查询: $Filter"
            $allUsers = Get-LocalUser
            switch ($Filter) {
                'active'   { $filtered = $allUsers | Where-Object Enabled -eq $true }
                'disabled' { $filtered = $allUsers | Where-Object Enabled -eq $false }
                default    { $filtered = $allUsers }
            }
            $results = $filtered | Select-Object Name, Enabled, Description
        }
    }

    END {
        # 管道结束后执行一次
        $results = $results | Select-Object -First $ResultLimit
        Write-Verbose "返回 $($results.Count) 条结果"
        return $results
    }
}

# 使用
Get-UserInfo -Username "admin" -Verbose
Get-UserInfo -Filter "active" -ResultLimit 50
"admin","guest" | Get-UserInfo
```

### 1.2 管道与流式处理

PowerShell 管道传递的是对象，不是文本。

```powershell
# === 管道基础 ===

# 每个命令的输出是一个对象流
Get-Process | Where-Object CPU -gt 10 | Select-Object Name, CPU, WorkingSet | Format-Table

# 等价于
$processes = Get-Process
$filtered = $processes | Where-Object { $_.CPU -gt 10 }
$selected = $filtered | Select-Object Name, CPU, @{Name='Memory(MB)';Expression={$_.WorkingSet/1MB}}
$selected | Format-Table -AutoSize

# === 管道中的 $_ 变量 ===
# $_ 或 $PSItem 代表当前管道对象
Get-Service | Where-Object { $_.Status -eq 'Running' -and $_.Name -like 'win*' }

# === 流式处理：逐条处理而非全部加载 ===
# 使用管道可以处理大量数据而不会耗尽内存
Get-Content C:\logs\large.log -ReadCount 1000 | ForEach-Object {
    foreach ($line in $_) {
        if ($line -match 'ERROR') {
            $line
        }
    }
} | Measure-Object  # 统计错误行数

# === 自定义管道函数 ===
function Select-Even {
    [CmdletBinding()]
    param(
        [Parameter(ValueFromPipeline=$true)]
        [int[]]$InputObject
    )
    process {
        foreach ($num in $InputObject) {
            if ($num % 2 -eq 0) {
                Write-Output $num
            }
        }
    }
}

1..100 | Select-Even

# === ForEach-Object 的三种脚本块 ===
1..5 | ForEach-Object {
    # process 块
    $_ * 2
} -Begin {
    # begin 块（管道开始前）
    Write-Host "开始处理"
} -End {
    # end 块（管道结束后）
    Write-Host "处理完成"
}

# === 管道对象属性映射 ===
$services = Get-Service | Select-Object -First 5 |
    Select-Object Name,
           Status,
           @{Name='StartupType';Expression={(Get-CimInstance Win32_Service -Filter "Name='$($_.Name)'").StartMode}},
           @{Name='DisplayName';Expression={$_.DisplayName}}

# === Group-Object 分组 ===
Get-Process | Group-Object -Property Name | Sort-Object Count -Descending | Select-Object -First 10

# === Tee-Object：分支管道 ===
Get-Process | Tee-Object -FilePath processes.txt | Where-Object CPU -gt 10
# 同时保存到文件和继续管道

# === Measure-Object 统计 ===
Get-Process | Measure-Object -Property WorkingSet -Sum -Average -Max -Min
```

### 1.3 Runspace 多线程

Runspace 是 PowerShell 的轻量级线程实现，比 PowerShell jobs 更高效。

```powershell
# === 基础 Runspace ===
$script = {
    param($computer)
    # 每个线程独立运行
    Test-Connection -ComputerName $computer -Count 1 -Quiet
}

# 创建 Runspace
$runspace = [runspacefactory]::CreateRunspace()
$runspace.Open()

# 创建 PowerShell 实例
$ps = [powershell]::Create()
$ps.Runspace = $runspace
$ps.AddScript($script).AddArgument("127.0.0.1")

# 异步执行
$handle = $ps.BeginInvoke()
$result = $ps.EndInvoke($handle)

$ps.Dispose()
$runspace.Close()
$runspace.Dispose()

# === Runspace 池（多线程并发）===
function Invoke-Parallel {
    <#
    .SYNOPSIS
    并行执行脚本块
    #>
    [CmdletBinding()]
    param(
        [Parameter(Mandatory=$true, ValueFromPipeline=$true)]
        [object[]]$InputObject,

        [Parameter(Mandatory=$true)]
        [scriptblock]$ScriptBlock,

        [Parameter()]
        [int]$ThrottleLimit = 10,

        [Parameter()]
        [int]$Timeout = 300
    )

    BEGIN {
        # 创建 Runspace 池
        $runspacePool = [runspacefactory]::CreateRunspacePool(1, $ThrottleLimit)
        $runspacePool.Open()

        $jobs = @()
        $startTime = Get-Date
    }

    PROCESS {
        foreach ($item in $InputObject) {
            # 检查超时
            if ((Get-Date) - $startTime -gt (New-TimeSpan -Seconds $Timeout)) {
                Write-Warning "执行超时"
                break
            }

            # 创建 PowerShell 实例
            $ps = [powershell]::Create()
            $ps.RunspacePool = $runspacePool
            [void]$ps.AddScript($ScriptBlock)
            [void]$ps.AddArgument($item)

            # 异步启动
            $job = [PSCustomObject]@{
                Instance = $ps
                Handle   = $ps.BeginInvoke()
                Item     = $item
            }
            $jobs += $job
        }
    }

    END {
        # 等待所有作业完成
        $results = @()
        foreach ($job in $jobs) {
            try {
                $result = $job.Instance.EndInvoke($job.Handle)
                $results += $result
            } catch {
                Write-Error "执行失败: $($_.Exception.Message)"
            } finally {
                $job.Instance.Dispose()
            }
        }

        # 清理 Runspace 池
        $runspacePool.Close()
        $runspacePool.Dispose()

        return $results
    }
}

# 使用：并行扫描多个主机
$computers = 1..254 | ForEach-Object { "192.168.1.$_" }
$results = $computers | Invoke-Parallel -ScriptBlock {
    param($ip)
    if (Test-Connection -ComputerName $ip -Count 1 -Quiet -Timeout 2) {
        [PSCustomObject]@{
            IP    = $ip
            Alive = $true
        }
    }
} -ThrottleLimit 50 -Timeout 120

$results | Format-Table

# === 线程安全的变量共享 ===
$sharedData = [Hashtable]::Synchronized(@{
    Count   = 0
    Results = @()
})

$script = {
    param($target, $shared)
    # 线程安全地更新共享变量
    [System.Threading.Monitor]::Enter($shared.SyncRoot)
    try {
        $shared.Count++
        $shared.Results += $target
    } finally {
        [System.Threading.Monitor]::Exit($shared.SyncRoot)
    }
}

# 创建带共享变量的 Runspace
$runspace = [runspacefactory]::CreateRunspace()
$runspace.Open()
$runspace.SessionStateProxy.SetVariable("shared", $sharedData)
```

### 1.4 正则表达式与 Select-String

```powershell
# === Select-String：PowerShell 的 grep ===

# 基本搜索

# 带上下文

Select-String -Path *.log -Pattern "warning" -CaseSensitive:$false

Get-ChildItem -Path C:\project -Recurse -Filter *.ps1 |
    Select-String -Pattern "password\s*="

# === 正则匹配 ===

# -match 返回布尔值，$matches 存储捕获组
if ("admin@example.com" -match '(\w+)@(\w+\.\w+)') {
    $matches[1]  # admin
    $matches[2]  # example.com
}

# -replace 替换
"hello world" -replace 'world', 'PowerShell'   # hello PowerShell

$log = '192.168.1.1 - - [10/Jan/2024:13:55:36] "GET /index.html"'

$logs = Get-Content C:\logs\access.log

# 替换回调（使用 ScriptBlock）
$encoded = "hello world" -replace '\w', { $args[0].Value.ToUpper() }
# HELLO WORLD

# -split 用正则分割
"one,two;three|four" -split '[,;|]'
# one, two, three, four

# 正则选项
$text = "Line1`nLine2`nLine3"
[regex]::Matches($text, "Line\d", "Multiline")  # 多行模式
```

### 1.7 错误处理与 $ErrorActionPreference

```powershell
# === $ErrorActionPreference ===
# 控制非终止性错误的默认行为
$ErrorActionPreference = 'Continue'   # 默认：显示错误继续
# 'SilentlyContinue'：静默继续
# 'Stop'：错误转为终止性错误（抛出异常）
# 'Inquire'：询问用户
# 'Ignore'：忽略（不记录到 $Error）

# 临时修改
$ErrorActionPreference = 'Stop'
try {
    Get-Item "C:\nonexistent" -ErrorAction Stop  # 强制终止
} catch {
    Write-Host "捕获错误: $($_.Exception.Message)"
} finally {
    $ErrorActionPreference = 'Continue'  # 恢复
}

# === try/catch/finally ===
try {
    # 终止性错误才能被 catch 捕获
    $result = Get-Content "C:\config.txt" -ErrorAction Stop
} catch [System.OutOfMemoryException] {
    # 特定异常类型
    Write-Host "内存不足"
} catch [System.IO.FileNotFoundException] {
    Write-Host "文件不存在"
} catch {
    # 通用捕获
    Write-Host "未知错误: $($_.Exception.Message)"
    Write-Host "位置: $($_.InvocationInfo.ScriptName):$($_.InvocationInfo.ScriptLineNumber)"
} finally {
    # 总是执行（清理资源）
    Write-Host "清理完成"
}

# === $Error 变量 ===
# $Error 是一个数组，保存最近的错误（最后一个在前）
Get-Item "C:\nonexistent" -ErrorAction SilentlyContinue
if ($Error[0]) {
    $Error[0].Exception.Message
    $Error[0].CategoryInfo
    $Error[0].InvocationInfo.PositionMessage
}

# 清除错误
$Error.Clear()

# === 自定义错误 ===
# Write-Error：非终止性错误
function Process-Data {
    param([string]$Input)
    if (-not $Input) {
        Write-Error "输入不能为空" -Category InvalidArgument
        return
    }
    # 处理...
}

# Throw：终止性错误
function Connect-Server {
    param([string]$Server)
    if (-not (Test-Connection $Server -Quiet)) {
        throw "无法连接到服务器 $Server"
    }
}

# === $? 和 $LASTEXITCODE ===
# $? : 最后一个命令是否成功（布尔）
Get-Process
$?  # True

Get-Item "C:\nonexistent" -ErrorAction SilentlyContinue
$?  # False

# $LASTEXITCODE : 最后一个外部程序的退出码
ping.exe -n 1 127.0.0.1
$LASTEXITCODE  # 0（成功）

# === 错误记录的自定义处理 ===
function Invoke-WithRetry {
    param(
        [scriptblock]$Action,
        [int]$MaxRetries = 3,
        [int]$Delay = 5
    )
    $attempt = 0
    while ($attempt -lt $MaxRetries) {
        try {
            $attempt++
            return & $Action
        } catch {
            Write-Warning "第 $attempt 次尝试失败: $($_.Exception.Message)"
            if ($attempt -ge $MaxRetries) {
                throw
            }
            Start-Sleep -Seconds $Delay
        }
    }
}

# 使用
Invoke-WithRetry -Action { Invoke-RestMethod "https://api.example.com" } -MaxRetries 5
```

### 2.1 PowerShell 信息收集全套脚本

```powershell
<#
.SYNOPSIS
    Windows 系统信息收集脚本（用于授权安全评估）
.DESCRIPTION
    收集系统配置、网络信息、用户账户、服务、共享等安全相关信息
#>

[CmdletBinding()]
param(
    [string]$OutputDir = ".\recon_$(Get-Date -Format 'yyyyMMdd_HHmmss')"
)

# 创建输出目录
New-Item -Path $OutputDir -ItemType Directory -Force | Out-Null

# 1. 系统基本信息
function Get-SystemInfo {
    Write-Host "[*] 收集系统基本信息..."
    $os = Get-CimInstance -ClassName Win32_OperatingSystem
    $cs = Get-CimInstance -ClassName Win32_ComputerSystem
    $cpu = Get-CimInstance -ClassName Win32_Processor

    $info = [PSCustomObject]@{
        HostName         = $env:COMPUTERNAME
        Domain           = $cs.Domain
        Manufacturer     = $cs.Manufacturer
        Model            = $cs.Model
        OSName           = $os.Caption
        OSVersion        = $os.Version
        OSBuild          = $os.BuildNumber
        InstallDate      = $os.InstallDate
        LastBoot         = $os.LastBootUpTime
        UptimeHours      = ((Get-Date) - $os.LastBootUpTime).TotalHours
        Architecture     = $os.OSArchitecture
        CPUName          = $cpu.Name
        CPUCores         = $cpu.NumberOfCores
        TotalMemoryGB    = [math]::Round($cs.TotalPhysicalMemory / 1GB, 2)
        FreeMemoryGB     = [math]::Round($os.FreePhysicalMemory / 1MB, 2)
    }
    $info | Export-Csv "$OutputDir\system_info.csv" -NoTypeInformation
    return $info
}
```

```powershell
function Find-SuspiciousAutoruns {
    $paths = @(
        "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run",
        "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run",
        "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\RunOnce",
        "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\RunOnce",
        "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Run"
    )

    $suspicious = @()
    foreach ($path in $paths) {
        if (Test-Path $path) {
            $props = Get-ItemProperty $path

                    # 检查可疑特征
                    $cmd = $_.Value
                    $flags = @()
                    if ($cmd -match 'powershell|cmd\.exe') { $flags += 'PowerShell/CMD' }
                    if ($cmd -match '-enc|-EncodedCommand') { $flags += 'Base64编码' }
                    if ($cmd -match '-w hidden|-WindowStyle') { $flags += '隐藏窗口' }

                    if ($cmd -match 'IEX|Invoke-Expression|DownloadString') { $flags += '远程下载' }
                    if ($cmd -match 'temp|appdata') { $flags += '可疑路径' }

                    if ($flags.Count -gt 0) {
                        $suspicious += [PSCustomObject]@{
                            Location = $path
                            Name     = $_.Name
                            Command = $cmd
                            Flags   = ($flags -join ', ')
                        }
                    }
                }
        }
    }
    return $suspicious
}

# 2. 计划任务后门
function New-ScheduledTaskBackdoor {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]
        [string]$TaskName,

        [Parameter(Mandatory)]
        [string]$Command,

        [int]$IntervalMinutes = 5,

        [switch]$Remove
    )

    if ($Remove) {
        Unregister-ScheduledTask -TaskName $TaskName -Confirm:$false -ErrorAction SilentlyContinue
        Write-Verbose "已移除计划任务: $TaskName"
        return
    }

    # 创建触发器（每 N 分钟执行）
    $trigger = New-ScheduledTaskTrigger -Once -At (Get-Date) `
        -RepetitionInterval (New-TimeSpan -Minutes $IntervalMinutes) `
        -RepetitionDuration (New-TimeSpan -Days 365)

    # 创建动作
    $action = New-ScheduledTaskAction -Execute "powershell.exe" `
        -Argument "-WindowStyle Hidden -Command $Command"

    # 创建设置（隐藏、高权限）
    $settings = New-ScheduledTaskSettingsSet `
        -AllowStartIfOnBatteries `
        -DontStopIfGoingOnBatteries `
        -StartWhenAvailable `
        -Hidden

    # 注册任务
    Register-ScheduledTask -TaskName $TaskName `
        -Trigger $trigger -Action $action -Settings $settings -Force

    Write-Verbose "已创建计划任务: $TaskName"
}

# 检测可疑计划任务
function Find-SuspiciousTasks {
    $tasks = Get-ScheduledTask | Where-Object State -ne 'Disabled'
    $suspicious = @()

    foreach ($task in $tasks) {
        foreach ($action in $task.Actions) {
            $cmd = "$($action.Execute) $($action.Arguments)"
            if ($cmd -match 'powershell|cmd\.exe|cscript|wscript' -and
                $cmd -match '-enc|hidden|bypass|IEX|DownloadString') {
                $suspicious += [PSCustomObject]@{
                    TaskName = $task.TaskName
                    Path     = $task.TaskPath
                    Command  = $cmd
                    State    = $task.State
                    Author   = $task.Author
                }
            }
        }
    }
    return $suspicious
}

# 3. WMI 事件订阅（无文件持久化）
function New-WmiEventSubscription {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]
        [string]$Name,

        [Parameter(Mandatory)]
        [string]$Command,

        [int]$IntervalSeconds = 60,

        [switch]$Remove
    )

    if ($Remove) {
        # 移除事件消费者
        Get-WmiObject -Namespace root\subscription -Class CommandLineEventConsumer |
            Where-Object { $_.Name -eq $Name } |
            ForEach-Object { $_.Delete() }
        # 移除事件过滤器
        Get-WmiObject -Namespace root\subscription -Class __EventFilter |
            Where-Object { $_.Name -eq $Name } |
            ForEach-Object { $_.Delete() }
        Write-Verbose "已移除 WMI 事件订阅: $Name"
        return
    }

    # 创建事件过滤器（定时触发）
    $query = "SELECT * FROM __InstanceModificationEvent WITHIN $IntervalSeconds WHERE TargetInstance ISA 'Win32_PerfFormattedData_PerfOS_System'"
    $filter = Set-WmiInstance -Namespace root\subscription -Class __EventFilter -Arguments @{
        Name = $Name
        Query = $query
        QueryLanguage = "WQL"
    }

    # 创建命令行消费者
    $consumer = Set-WmiInstance -Namespace root\subscription -Class CommandLineEventConsumer -Arguments @{
        Name = $Name
        CommandLineTemplate = "powershell.exe -WindowStyle Hidden -Command `"$Command`""
    }

    # 绑定过滤器和消费者
    Set-WmiInstance -Namespace root\subscription -Class __FilterToConsumerBinding -Arguments @{
        Filter = $filter
        Consumer = $consumer
    }

    Write-Verbose "已创建 WMI 事件订阅: $Name"
}

# 检测 WMI 事件订阅
function Find-SuspiciousWmiSubscriptions {
    $filters = Get-CimInstance -Namespace root\subscription -ClassName __EventFilter
    $consumers = Get-CimInstance -Namespace root\subscription -ClassName CommandLineEventConsumer
    $bindings = Get-CimInstance -Namespace root\subscription -ClassName __FilterToConsumerBinding

    $results = @()
    foreach ($binding in $bindings) {
        $filter = $filters | Where-Object { $_.__PATH -eq $binding.Filter }
        $consumer = $consumers | Where-Object { $_.__PATH -eq $binding.Consumer }

        if ($consumer.CommandLineTemplate -match 'powershell|cmd|cscript') {
            $results += [PSCustomObject]@{
                Filter   = $filter.Name
                Consumer = $consumer.Name
                Command  = $consumer.CommandLineTemplate
                Query    = $filter.Query
            }
        }
    }
    return $results
}
```

### 2.3 PowerShell AMSI 绕过原理与检测

AMSI (Anti-Malware Scan Interface) 是 Windows 10+ 的恶意脚本检测机制。

```powershell
# === AMSI 工作原理 ===
# AMSI 在脚本执行前将内容发送给杀毒软件（如 Windows Defender）
# PowerShell 5.0+ 集成了 AMSI
# 当 PowerShell 执行脚本时，amsi.dll 中的 AmsiScanBuffer 函数被调用

# 检查 AMSI 是否启用
function Test-AmsiEnabled {
    try {
        $amsi = [System.Reflection.Assembly]::LoadWithPartialName("System.Management.Automation")
        $amsiObj = [AMSIUtils]
        # 如果 AMSI 可用，尝试触发
        $content = "AMSI Test"
        $result = [System.Runtime.InteropServices.Marshal]::SecureStringToBSTR(
            (ConvertTo-SecureString $content -AsPlainText -Force)
        )
        Write-Host "AMSI 已启用" -ForegroundColor Green
        return $true
    } catch {
        Write-Host "AMSI 可能被禁用" -ForegroundColor Red
        return $false
    }
}

# === AMSI 绕过原理（仅理论分析，不提供可执行绕过）===

# 原理 1：修改 amsi.dll 中的 AmsiScanBuffer 函数
# 攻击者通过修改内存中的函数指令使其直接返回 S_OK
# 这需要管理员权限

# 原理 2：反射加载 .NET 程序集绕过
# 直接在内存中加载程序集，不经过 PowerShell 的脚本解析

# 原理 3：混淆脚本内容
# 修改脚本特征使 AMSI 无法匹配

# === 蓝队：检测 AMSI 绕过 ===

function Find-AmsiBypass {
    Write-Host "[*] 检测 AMSI 绕过痕迹..."

    # 1. 检查 amsi.dll 是否被修改
    $amsiPath = Join-Path $env:windir "System32\amsi.dll"
    if (Test-Path $amsiPath) {
        $signature = Get-AuthenticodeSignature $amsiPath
        if ($signature.Status -ne 'Valid') {
            Write-Warning "amsi.dll 签名无效: $($signature.Status)"
        }
        # 检查文件完整性
        $hash = Get-FileHash $amsiPath -Algorithm SHA256
        Write-Host "  amsi.dll SHA256: $($hash.Hash)"
    }

    # 2. 检查 AMSI 注册表是否被禁用
    $amsiKeys = @(
        "HKLM:\SOFTWARE\Microsoft\AMSI\Providers",
        "HKCU:\SOFTWARE\Microsoft\AMSI"
    )
    foreach ($key in $amsiKeys) {
        if (Test-Path $key) {
            $props = Get-ItemProperty $key
            Write-Host "  $key 存在"
            $props.PSObject.Properties | Where-Object { $_.Name -notmatch "^PS" } | ForEach-Object {
                Write-Host "    $($_.Name) = $($_.Value)"
            }
        }
    }

    # 3. 检查 PowerShell 进程中的 amsi.dll 加载状态
    $procId = $PID
    $modules = (Get-Process -Id $procId).Modules | Where-Object { $_.ModuleName -eq 'amsi.dll' }
    if ($modules) {
        Write-Host "  AMSI 已加载到当前进程"
    } else {
        Write-Warning "  AMSI 未加载到当前进程（可能被绕过）"
    }

    # 4. 检查事件日志中的 AMSI 相关事件
    # AMSI 事件记录在 Microsoft-Windows-AMSI/Operational
    try {
        $events = Get-WinEvent -LogName "Microsoft-Windows-AMSI/Operational" -MaxEvents 50 -ErrorAction Stop
        $detections = $events | Where-Object { $_.Id -eq 1116 -or $_.Id -eq 1117 }
        if ($detections) {
            Write-Host "`n  AMSI 检测记录:"
            $detections | Select-Object TimeCreated, Id, Message | Format-Table -AutoSize
        }
    } catch {
        Write-Host "  无法读取 AMSI 事件日志"
    }
}
```

### 2.4 PowerShell 日志分析

```powershell
# === PowerShell 日志分析 ===

# PowerShell 有四种重要日志：
# 1. Windows PowerShell (经典)  -> Event ID 400, 403, 500, 800
# 2. Microsoft-Windows-PowerShell/Operational -> Event ID 4100-4104
# 3. Microsoft-Windows-PowerShell/Analytic
# 4. Script Block Logging (Event ID 4104)

function Get-PowerShellLogs {
    [CmdletBinding()]
    param(
        [int]$Hours = 24,
        [switch]$SuspiciousOnly
    )

    $startTime = (Get-Date).AddHours(-$Hours)

    # 1. 脚本块日志（Script Block Logging）
    # Event ID 4104: 脚本执行
    # Event ID 4103: 模块日志
    Write-Host "[*] 获取脚本块日志..."
    $scriptBlocks = Get-WinEvent -FilterHashtable @{
        LogName   = "Microsoft-Windows-PowerShell/Operational"
        Id        = 4104
        StartTime = $startTime
    } -ErrorAction SilentlyContinue

    # 2. 经典 PowerShell 日志
    Write-Host "[*] 获取经典 PowerShell 日志..."
    $classic = Get-WinEvent -FilterHashtable @{
        LogName   = "Windows PowerShell"
        StartTime = $startTime
    } -ErrorAction SilentlyContinue

    # 3. 可疑模式
    $suspiciousPatterns = @(
        'IEX|Invoke-Expression',
        'DownloadString|DownloadFile',
        'Net\.WebClient|System\.Net',
        'FromBase64String',
        '-enc|-EncodedCommand',
        'Bypass|Unrestricted',
        'Hidden|WindowStyle',
        'Invoke-Mimikatz|Invoke-Shellcode',
        'powersploit|PowerSploit',
        'Set-MpPreference|Add-MpPreference',
        'DisableRealtimeMonitoring',
        'AmsiScanBuffer|amsi\.dll',
        'System\.Reflection\.Assembly',
        'Runtime\.InteropServices',
        'token_elevation|SeDebugPrivilege',
        'AdjustTokenPrivileges',
        'CreateRemoteThread|VirtualAllocEx',
        'WriteProcessMemory'
    )

    Write-Host "[*] 分析可疑活动..."

    # 分析脚本块
    $allEvents = @($scriptBlocks) + @($classic)

    $suspicious = @()
    foreach ($event in $allEvents) {
        $msg = $event.Message

        foreach ($pattern in $suspiciousPatterns) {
            if ($msg -match $pattern) {
                # 提取脚本内容（4104 事件在 Message 中包含脚本）
                $scriptContent = if ($event.Properties) {
                    $event.Properties[0].Value
                } else {
                    $msg
                }

                $suspicious += [PSCustomObject]@{
                    TimeCreated   = $event.TimeCreated
                    EventId       = $event.Id
                    Pattern       = $pattern
                    ComputerName  = $event.MachineName
                    User          = ($event.UserId -replace 'S-1-.*\\','')
                    ScriptPreview = if ($scriptContent) {
                        $scriptContent.Substring(0, [Math]::Min(200, $scriptContent.Length))
```

```powershell
function Send-HttpRequest {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]
        [string]$Url,

        [Parameter()]
        [string]$Method = "GET",

        [Parameter()]
        [hashtable]$Headers = @{},

        [Parameter()]
        [string]$Body,

        [Parameter()]
        [int]$Timeout = $Script:Config.Timeout
    )

    try {
        $request = [System.Net.HttpWebRequest]::Create($Url)
        $request.Method = $Method
        $request.Timeout = $Timeout * 1000
        $request.UserAgent = $Script:Config.UserAgent

        foreach ($key in $Headers.Keys) {
            $request.Headers[$key] = $Headers[$key]
        }

        if ($Body -and $Method -ne "GET") {
            $bytes = [System.Text.Encoding]::UTF8.GetBytes($Body)
            $request.ContentLength = $bytes.Length
            $stream = $request.GetRequestStream()
            $stream.Write($bytes, 0, $bytes.Length)
            $stream.Close()
        }

        $response = $request.GetResponse()
        $reader = New-Object System.IO.StreamReader($response.GetResponseStream())
        $content = $reader.ReadToEnd()
        $reader.Close()
        $response.Close()

        return [PSCustomObject]@{
            StatusCode = [int]$response.StatusCode
            Headers    = $response.Headers
            Body       = $content
        }
    } catch [System.Net.WebException] {
        if ($_.Exception.Response) {
            $reader = New-Object System.IO.StreamReader($_.Exception.Response.GetResponseStream())
            return [PSCustomObject]@{
                StatusCode = [int]$_.Exception.Response.StatusCode
                Body       = $reader.ReadToEnd()
                Error      = $_.Exception.Message
            }
        }
        Write-Log "请求失败: $($_.Exception.Message)" "ERROR"
        return $null
    }
}

function Invoke-PortScan {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]
        [string]$ComputerName,

        [int[]]$Ports = @(21, 22, 23, 25, 53, 80, 110, 143, 443, 445,
                          993, 995, 1433, 1521, 3306, 3389, 5432,
                          5900, 6379, 8080, 8443, 9200, 27017),

        [int]$Timeout = 2000,

        [int]$Threads = 50
    )

    Write-Log "端口扫描: $ComputerName 端口: $($Ports -join ',')" "INFO"

    # 使用 Runspace 池并行扫描
    $pool = [runspacefactory]::CreateRunspacePool(1, $Threads)
    $pool.Open()

    $jobs = @()
    foreach ($port in $Ports) {
        $ps = [powershell]::Create()
        $ps.RunspacePool = $pool
        [void]$ps.AddScript({
            param($host, $port, $timeout)
            $tcp = New-Object System.Net.Sockets.TcpClient
            $iar = $tcp.BeginConnect($host, $port, $null, $null)
            $success = $iar.AsyncWaitHandle.WaitOne($timeout, $false)
            if ($success -and $tcp.Connected) {
                $tcp.Close()
                return [PSCustomObject]@{
                    Port    = $port
                    State   = "Open"
                    Service = switch ($port) {
                        21 {"FTP"} 22 {"SSH"} 23 {"Telnet"} 25 {"SMTP"}
                        80 {"HTTP"} 443 {"HTTPS"} 3306 {"MySQL"}
                        5432 {"PostgreSQL"} 6379 {"Redis"} 8080 {"HTTP-Alt"}
                        default {"unknown"}
                    }
                }
            } else {
                $tcp.Close()
                return $null
            }
        })
        [void]$ps.AddArgument($ComputerName)
        [void]$ps.AddArgument($port)
        [void]$ps.AddArgument($Timeout)

        $jobs += [PSCustomObject]@{
            Instance = $ps
            Handle   = $ps.BeginInvoke()
            Port     = $port
        }
    }

    # 收集结果
    $results = @()
    foreach ($job in $jobs) {
        $result = $job.Instance.EndInvoke($job.Handle)
        if ($result) { $results += $result }
        $job.Instance.Dispose()
    }

    $pool.Close()
    $pool.Dispose()

    Write-Log "扫描完成: 发现 $($results.Count) 个开放端口" "INFO"
    return $results
}
```

```powershell
# === 蓝队：PowerShell 攻击检测 ===

function Invoke-PowerShellAttackDetection {
    <#
    .SYNOPSIS
    检测系统上的 PowerShell 攻击痕迹
    #>
    [CmdletBinding()]
    param(
        [int]$Days = 7
    )

    $startTime = (Get-Date).AddDays(-$Days)
    $findings = @()

    # 1. 检测可疑的 PowerShell 进程创建
    Write-Host "[*] 检测可疑 PowerShell 进程..." -ForegroundColor Cyan
    $procEvents = Get-WinEvent -FilterHashtable @{
        LogName   = "Security"
        Id         = 4688
        StartTime = $startTime
    } -ErrorAction SilentlyContinue

    $suspiciousPatterns = @(
        @{Pattern='powershell.*-enc';       Desc='Base64 编码命令'; Severity='High'},
        @{Pattern='powershell.*-EncodedCommand'; Desc='Base64 编码命令'; Severity='High'},
        @{Pattern='powershell.*bypass';      Desc='绕过执行策略'; Severity='Medium'},
        @{Pattern='powershell.*-w hidden';   Desc='隐藏窗口'; Severity='High'},
        @{Pattern='powershell.*WindowStyle.*Hidden'; Desc='隐藏窗口'; Severity='High'},
        @{Pattern='powershell.*IEX';         Desc='远程执行（IEX）'; Severity='Critical'},
        @{Pattern='powershell.*DownloadString'; Desc='远程下载'; Severity='High'},
        @{Pattern='powershell.*FromBase64String'; Desc='Base64 解码'; Severity='High'},
        @{Pattern='powershell.*Invoke-Mimikatz'; Desc='凭据提取'; Severity='Critical'},
        @{Pattern='powershell.*Set-MpPreference'; Desc='修改 Defender 配置'; Severity='High'},
        @{Pattern='powershell.*DisableRealtimeMonitoring'; Desc='禁用实时保护'; Severity='Critical'},
        @{Pattern='cmd.*/c.*powershell';      Desc='CMD 调用 PowerShell'; Severity='Medium'},
        @{Pattern='wscript.*powershell';      Desc='脚本调用 PowerShell'; Severity='Medium'}
    )

    foreach ($event in $procEvents) {
        $cmdLine = $event.Message
        foreach ($sp in $suspiciousPatterns) {
            if ($cmdLine -match $sp.Pattern) {
                $findings += [PSCustomObject]@{
                    Time     = $event.TimeCreated
                    Type     = "可疑进程"
                    Pattern  = $sp.Pattern
                    Desc     = $sp.Desc
                    Severity = $sp.Severity
                    Details  = $cmdLine.Substring(0, [Math]::Min(200, $cmdLine.Length))
                    EventId  = $event.Id
                }
            }
        }
    }

    # 2. 检测 PowerShell 脚本块中的可疑内容
    Write-Host "[*] 检测可疑脚本块..." -ForegroundColor Cyan
    $scriptBlocks = Get-WinEvent -FilterHashtable @{
        LogName   = "Microsoft-Windows-PowerShell/Operational"
        Id        = 4104
        StartTime = $startTime
    } -ErrorAction SilentlyContinue

    $scriptPatterns = @(
        @{Pattern='Invoke-Mimikatz';          Desc='Mimikatz'; Severity='Critical'},
        @{Pattern='Invoke-Shellcode';          Desc='Shellcode 注入'; Severity='Critical'},
        @{Pattern='Invoke-ReflectivePEInjection'; Desc='反射 DLL 注入'; Severity='Critical'},
        @{Pattern='Invoke-NinjaCopy';          Desc='文件复制（绕权限）'; Severity='High'},
        @{Pattern='Get-Keystrokes';            Desc='键盘记录'; Severity='Critical'},
        @{Pattern='Set-MpPreference.*Disable'; Desc='禁用 Defender'; Severity='Critical'},
        @{Pattern='netsh.*advfirewall';        Desc='修改防火墙'; Severity='High'},
        @{Pattern='reg.*add';                  Desc='修改注册表'; Severity='Medium'},
        @{Pattern='schtasks.*create';          Desc='创建计划任务'; Severity='Medium'},
        @{Pattern='System\.Reflection\.Assembly.*Load'; Desc='内存加载程序集'; Severity='High'},
        @{Pattern='Invoke-WebRequest.*http';   Desc='远程下载'; Severity='Medium'},
        @{Pattern='FromBase64String';          Desc='Base64 解码'; Severity='Medium'}
    )

    foreach ($event in $scriptBlocks) {
        $script = if ($event.Properties) { $event.Properties[0].Value } else { $event.Message }
        foreach ($sp in $scriptPatterns) {
            if ($script -match $sp.Pattern) {
                $findings += [PSCustomObject]@{
                    Time     = $event.TimeCreated
                    Type     = "可疑脚本"
                    Pattern  = $sp.Pattern
                    Desc     = $sp.Desc
                    Severity = $sp.Severity
                    Details  = if ($script) { $script.Substring(0, [Math]::Min(200, $script.Length)) }
                    EventId  = $event.Id
                }
            }
        }
    }

    # 3. 检测 PowerShell 远程会话
    Write-Host "[*] 检测远程会话..." -ForegroundColor Cyan
    $remoteSessions = Get-WinEvent -FilterHashtable @{
        LogName   = "Microsoft-Windows-WinRM/Operational"
        StartTime = $startTime
    } -ErrorAction SilentlyContinue | Select-Object -First 50

    # 4. 检测 WMI 滥用
    Write-Host "[*] 检测 WMI 滥用..." -ForegroundColor Cyan
    $wmiSubs = Get-CimInstance -Namespace root\subscription -ClassName __FilterToConsumerBinding -ErrorAction SilentlyContinue
    if ($wmiSubs) {
        foreach ($sub in $wmiSubs) {
            $findings += [PSCustomObject]@{
                Time     = "Unknown"
                Type     = "WMI 事件订阅"
                Pattern  = "WMI Persistence"
                Desc     = "WMI 持久化机制"
                Severity = "High"
                Details  = "$($sub.Filter) -> $($sub.Consumer)"
                EventId  = "WMI"
            }
        }
    }

    # 5. 检测 AMSI 绕过
    Write-Host "[*] 检测 AMSI 绕过..." -ForegroundColor Cyan
    $amsiEvents = Get-WinEvent -FilterHashtable @{
        LogName   = "Microsoft-Windows-AMSI/Operational"
        StartTime = $startTime
    } -ErrorAction SilentlyContinue

    # 生成报告
    Write-Host "`n=== PowerShell 攻击检测报告 ===" -ForegroundColor Cyan
    Write-Host "扫描时间范围: 最近 $Days 天"
    Write-Host "发现可疑事件: $($findings.Count) 个"

    if ($findings.Count -gt 0) {
        # 按严重度排序
        $severityOrder = @{ 'Critical' = 0; 'High' = 1; 'Medium' = 2; 'Low' = 3 }
        $sorted = $findings | Sort-Object { $severityOrder[$_.Severity] }

        $sorted | Format-Table Time, Type, Desc, Severity, Details -AutoSize

        # 严重度统计
        Write-Host "`n--- 严重度统计 ---" -ForegroundColor Yellow
        $sorted | Group-Object Severity | Sort-Object { $severityOrder[$_.Name] } |
            ForEach-Object {
                Write-Host "  $($_.Name): $($_.Count) 个"
            }

        # 输出 JSON
        $findings | ConvertTo-Json -Depth 3 | Out-File ".\ps_attack_detection_$(Get-Date -Format 'yyyyMMdd').json"
        Write-Host "`n报告已保存到 JSON 文件"
    }

    return $findings
}

Invoke-PowerShellAttackDetection -Days 7
```

```powershell
# 1. 启用所有 PowerShell 日志
function Enable-FullPowerShellLogging {
    # 需要管理员权限

    # 脚本块日志
    $path = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell\ScriptBlockLogging"
    New-Item -Path $path -Force | Out-Null
    Set-ItemProperty -Path $path -Name "EnableScriptBlockLogging" -Value 1 -Type DWord

    # 模块日志
    $path = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell\ModuleLogging"
    New-Item -Path $path -Force | Out-Null
    Set-ItemProperty -Path $path -Name "EnableModuleLogging" -Value 1 -Type DWord
    Set-ItemProperty -Path $path -Name "ModuleNames" -Value "*" -Type MultiString

    # 转写日志
    $path = "HKLM:\SOFTWARE\Policies\Microsoft\Windows\PowerShell\Transcription"
    New-Item -Path $path -Force | Out-Null
    Set-ItemProperty -Path $path -Name "EnableTranscripting" -Value 1 -Type DWord
    Set-ItemProperty -Path $path -Name "OutputDirectory" -Value "C:\Logs\PS_Transcripts" -Type String
    Set-ItemProperty -Path $path -Name "EnableInvocationHeader" -Value 1 -Type DWord

    # 启用进程创建审计
    auditpol /set /subcategory:"Process Creation" /success:enable

    # 启用命令行审计
    $path = "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Policies\System\AuditProcessCreation"
    New-Item -Path $path -Force | Out-Null
    Set-ItemProperty -Path $path -Name "IncludeCommandLine" -Value 1 -Type DWord

    Write-Host "PowerShell 完整日志已启用" -ForegroundColor Green
    Write-Host "  - 脚本块日志"
    Write-Host "  - 模块日志"
    Write-Host "  - 转写日志"
```
