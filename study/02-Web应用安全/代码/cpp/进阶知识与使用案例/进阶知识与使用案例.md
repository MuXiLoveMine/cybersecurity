### 1.1 智能指针（unique_ptr、shared_ptr、weak_ptr）

C++11 引入智能指针，自动管理堆内存，避免内存泄漏和悬垂指针。

```cpp
#include <iostream>
#include <memory>
#include <vector>
#include <string>

// ===== unique_ptr：独占所有权 =====
// 同一时刻只能有一个 unique_ptr 拥有对象
// 不可拷贝，可以移动
void uniquePtrDemo() {
    // 创建 unique_ptr
    std::unique_ptr<int> p1 = std::make_unique<int>(42);
    std::cout << "p1 = " << *p1 << std::endl;

    // 移动语义转移所有权
    std::unique_ptr<int> p2 = std::move(p1);
    // p1 现在为 nullptr
    std::cout << "p2 = " << *p2 << std::endl;
    std::cout << "p1 是否为空: " << (p1 == nullptr) << std::endl;

    // 自定义删除器
    std::unique_ptr<FILE, decltype(&fclose)> file(
        fopen("test.txt", "w"), &fclose);
    if (file) {
        fprintf(file.get(), "Hello from unique_ptr");
    }
    // file 析构时自动调用 fclose

    // 数组版本
    std::unique_ptr<int[]> arr = std::make_unique<int[]>(10);
    for (int i = 0; i < 10; i++) {
        arr[i] = i * i;
    }
    std::cout << "arr[5] = " << arr[5] << std::endl;
}

// ===== shared_ptr：共享所有权 =====
// 多个 shared_ptr 可以共享同一对象
// 内部引用计数，最后一个 shared_ptr 销毁时释放对象
void sharedPtrDemo() {
    auto p1 = std::make_shared<std::string>("Hello");
    std::cout << "p1 引用计数: " << p1.use_count() << std::endl; // 1

    {
        auto p2 = p1; // 拷贝，引用计数 +1
        std::cout << "p1 引用计数: " << p1.use_count() << std::endl; // 2
        std::cout << "p2 = " << *p2 << std::endl;
    } // p2 离开作用域，引用计数 -1

    std::cout << "p1 引用计数: " << p1.use_count() << std::endl; // 1

    // 注意：shared_ptr 的控制块是原子操作，有性能开销
    // 不要用裸指针构造多个 shared_ptr（会导致 double free）
    // 错误示例：
    // int* raw = new int(42);
    // std::shared_ptr<int> sp1(raw);
    // std::shared_ptr<int> sp2(raw); // 灾难！两个独立引用计数

    // 自定义删除器
    std::shared_ptr<int> sp(new int(100), [](int* p) {
        std::cout << "自定义删除器释放: " << *p << std::endl;
        delete p;
    });
}

// ===== weak_ptr：弱引用，不增加引用计数 =====
// 解决 shared_ptr 循环引用问题
void weakPtrDemo() {
    auto shared = std::make_shared<int>(42);
    std::weak_ptr<int> weak = shared; // 不增加引用计数

    std::cout << "shared 引用计数: " << shared.use_count() << std::endl; // 1

    // 检查 weak_ptr 是否过期
    if (auto sp = weak.lock()) { // lock() 返回 shared_ptr
        std::cout << "weak_ptr 有效: " << *sp << std::endl;
    } else {
        std::cout << "weak_ptr 已过期" << std::endl;
    }

    shared.reset(); // 释放对象

    // 再次检查
    if (auto sp = weak.lock()) {
        std::cout << "weak_ptr 有效" << std::endl;
    } else {
        std::cout << "weak_ptr 已过期（对象已释放）" << std::endl;
    }
}

// ===== 循环引用问题示例 =====
struct Node {
    std::string name;
    // 如果使用 shared_ptr 会导致循环引用内存泄漏
    std::weak_ptr<Node> next;  // 正确：使用 weak_ptr 打破循环
    std::weak_ptr<Node> prev;
};

void circularRefDemo() {
    auto n1 = std::make_shared<Node>();
    auto n2 = std::make_shared<Node>();
    n1->name = "Node1";
    n2->name = "Node2";

    n1->next = n2; // weak_ptr 不增加引用计数
    n2->prev = n1; // weak_ptr 不增加引用计数

    std::cout << "n1 引用计数: " << n1.use_count() << std::endl; // 1
    std::cout << "n2 引用计数: " << n2.use_count() << std::endl; // 1
    // 两个对象都能正常释放
}

int main() {
    std::cout << "=== unique_ptr ===" << std::endl;
    uniquePtrDemo();

    std::cout << "\n=== shared_ptr ===" << std::endl;
    sharedPtrDemo();

    std::cout << "\n=== weak_ptr ===" << std::endl;
    weakPtrDemo();

    std::cout << "\n=== 循环引用 ===" << std::endl;
    circularRefDemo();

    return 0;
}
```

### 1.2 移动语义与右值引用

移动语义避免不必要的深拷贝，显著提升性能。

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <utility>

// ===== 左值与右值 =====
// 左值：有名字、有地址，可以取地址（变量名、引用）
// 右值：没有名字的临时值（字面量、表达式结果、函数返回值）

void lvalueRvalueDemo() {
    int a = 10;        // a 是左值，10 是右值
    int& lref = a;     // 左值引用
    int&& rref = 20;   // 右值引用（绑定到右值）

    // int& bad = 42;  // 错误：左值引用不能绑定右值
    const int& cref = 42; // 但 const 左值引用可以绑定右值

    std::cout << a << " " << lref << " " << rref << " " << cref << std::endl;
}

// ===== 移动构造函数和移动赋值 =====
class StringList {
private:
    std::string* data;
    size_t size;
public:
    // 构造函数
    explicit StringList(size_t n) : size(n), data(new std::string[n]) {}

    // 析构函数
    ~StringList() { delete[] data; }

    // 拷贝构造（深拷贝，开销大）
    StringList(const StringList& other) : size(other.size), data(new std::string[other.size]) {
        std::cout << "拷贝构造（深拷贝）" << std::endl;
        for (size_t i = 0; i < size; i++) {
            data[i] = other.data[i];
        }
    }

    // 移动构造（转移所有权，零拷贝）
    StringList(StringList&& other) noexcept : data(other.data), size(other.size) {
        std::cout << "移动构造（零拷贝）" << std::endl;
        other.data = nullptr; // 清空源对象
        other.size = 0;
    }

    // 拷贝赋值
    StringList& operator=(const StringList& other) {
        if (this != &other) {
            delete[] data;
            size = other.size;
            data = new std::string[size];
            for (size_t i = 0; i < size; i++) {
                data[i] = other.data[i];
            }
        }
        return *this;
    }

    // 移动赋值
    StringList& operator=(StringList&& other) noexcept {
        if (this != &other) {
            delete[] data;
            data = other.data;
            size = other.size;
            other.data = nullptr;
            other.size = 0;
        }
        return *this;
    }
};

// ===== std::move：将左值转换为右值引用 =====
void moveDemo() {
    std::string s1 = "Hello World";
    std::string s2 = std::move(s1); // 移动而非拷贝

    std::cout << "s2 = " << s2 << std::endl;
    std::cout << "s1 移动后: " << (s1.empty() ? "(空)" : s1) << std::endl;

    // 注意：std::move 本身不移动任何东西
    // 它只是做了一个 static_cast<T&&>，真正的移动在移动构造函数中
}

// ===== std::forward：完美转发 =====
// 在模板函数中保持参数的值类别
template <typename T>
void wrapper(T&& arg) {
    // std::forward 保持 arg 的左值/右值属性
    process(std::forward<T>(arg));
}

void process(std::string& s) {
    std::cout << "左值版本: " << s << std::endl;
}

void process(std::string&& s) {
    std::cout << "右值版本: " << s << std::endl;
}

// ===== emplace_back vs push_back =====
void emplaceVsPush() {
    std::vector<std::string> vec;

    // push_back：先构造临时对象，再移动或拷贝
    vec.push_back(std::string("hello")); // 构造 + 移动

    // emplace_back：直接在容器内存中构造，无拷贝无移动
    vec.emplace_back("world"); // 直接构造

    // emplace_back 更高效
}

int main() {
    lvalueRvalueDemo();

    std::cout << "\n=== 移动语义 ===" << std::endl;
    StringList a(1000);
    StringList b = a;            // 调用拷贝构造
    StringList c = std::move(a); // 调用移动构造

    moveDemo();

    return 0;
}
```

### 1.3 模板编程进阶（SFINAE、type traits）

```cpp
#include <iostream>
#include <type_traits>
#include <vector>
#include <string>
#include <algorithm>

// ===== type_traits：编译时类型查询 =====
void typeTraitsDemo() {
    std::cout << "int 是否为整数: " << std::is_integral<int>::value << std::endl;
    std::cout << "double 是否为浮点: " << std::is_floating_point<double>::value << std::endl;
    std::cout << "int* 是否为指针: " << std::is_pointer<int*>::value << std::endl;
    std::cout << "std::string 是否为类: " << std::is_class<std::string>::value << std::endl;

    // 类型关系
    std::cout << "int 和 unsigned 是否相同基础类型: "
              << std::is_same<std::remove_cv<const int>::type, int>::value << std::endl;

    // 条件编译
    if constexpr (std::is_integral_v<int>) { // C++17
        std::cout << "int 是整数类型" << std::endl;
    }
}

// ===== SFINAE：替换失败不是错误 =====
// 当模板替换失败时，编译器不会报错，而是跳过该重载

// enable_if 方式1：函数参数
template <typename T,
          typename = typename std::enable_if<std::is_integral<T>::value>::type>
void processNumber(T value) {
    std::cout << "处理整数: " << value << std::endl;
}

// enable_if 方式2：返回类型
template <typename T>
typename std::enable_if<std::is_floating_point<T>::value, void>::type
processNumber(T value) {
    std::cout << "处理浮点数: " << value << std::endl;
}

// C++14 的 enable_if_t 简化写法
template <typename T,
          std::enable_if_t<std::is_pointer<T>::value, int> = 0>
void processNumber(T value) {
    std::cout << "处理指针" << std::endl;
}

// C++17 的 if constexpr 简化模板
template <typename T>
void smartProcess(T value) {
    if constexpr (std::is_integral_v<T>) {
        std::cout << "整数: " << value << std::endl;
    } else if constexpr (std::is_floating_point_v<T>) {
        std::cout << "浮点: " << value << std::endl;
    } else if constexpr (std::is_same_v<T, std::string>) {
        std::cout << "字符串: " << value << std::endl;
    } else {
        std::cout << "未知类型" << std::endl;
    }
}

// ===== 可变参数模板 =====
// 递归终止
void print() {}

template <typename T, typename... Args>
void print(T first, Args... rest) {
    std::cout << first << " ";
    print(rest...); // 递归展开
}

// C++17 折叠表达式
template <typename... Args>
auto sum(Args... args) {
    return (... + args); // 折叠表达式
}

template <typename... Args>
void printAll(Args... args) {
    ((std::cout << args << " "), ...); // 逗号折叠
    std::cout << std::endl;
}

// ===== 模板特化 =====
template <typename T>
struct TypeName {
    static std::string get() { return "Unknown"; }
};

template <>
struct TypeName<int> {
    static std::string get() { return "int"; }
};

template <>
struct TypeName<double> {
    static std::string get() { return "double"; }
};

template <typename T>
struct TypeName<T*> {
    static std::string get() { return "pointer"; }
};

int main() {
    typeTraitsDemo();

    std::cout << "\n=== SFINAE ===" << std::endl;
    processNumber(42);       // 整数版本
    processNumber(3.14);     // 浮点版本
    smartProcess(100);
    smartProcess(2.71);
    smartProcess(std::string("hello"));

    std::cout << "\n=== 可变参数模板 ===" << std::endl;
    print(1, 2.5, "hello", 'c');
    std::cout << "sum = " << sum(1, 2, 3, 4, 5) << std::endl;
    printAll("a", "b", "c");

    std::cout << "\n=== 模板特化 ===" << std::endl;
    std::cout << TypeName<int>::get() << std::endl;
    std::cout << TypeName<double>::get() << std::endl;
    std::cout << TypeName<int*>::get() << std::endl;

    return 0;
}
```

### 1.4 Lambda 表达式与 STL 算法

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <functional>
#include <string>

// Lambda 基本语法:
// [capture](parameters) -> return_type { body }

void lambdaDemo() {
    int x = 10;
    int y = 20;

    // ===== 捕获方式 =====
    // [] 不捕获任何变量
    auto noCapture = []() { return 42; };

    // [x] 值捕获 x（拷贝）
    auto valCapture = [x]() { return x + 1; };

    // [&x] 引用捕获 x
    auto refCapture = [&x]() { x++; return x; };

    // [=] 值捕获所有外部变量
    auto captureAllVal = [=]() { return x + y; };

    // [&] 引用捕获所有外部变量
    auto captureAllRef = [&]() { x++; y++; };

    // [x, &y] 混合捕获
    auto mixedCapture = [x, &y]() { return x + y; };

    // [=, &x] 值捕获所有，但 x 用引用
    auto mixedCapture2 = [=, &x]() { x++; return x + y; };

    // [this] 捕获 this 指针（类成员中）
    // [=, *this] C++17：值捕获 *this 对象

    std::cout << "noCapture: " << noCapture() << std::endl;
    std::cout << "valCapture: " << valCapture() << std::endl;
    refCapture();
    std::cout << "x after refCapture: " << x << std::endl;

    // ===== 泛型 Lambda（C++14）=====
    auto generic = [](auto a, auto b) { return a + b; };
    std::cout << "generic int: " << generic(1, 2) << std::endl;
    std::cout << "generic string: " << generic(std::string("hello"), std::string("world")) << std::endl;

    // ===== mutable Lambda =====
    int counter = 0;
    auto increment = [counter]() mutable {
        counter++; // 修改捕获的副本
        return counter;
    };
    std::cout << "increment: " << increment() << std::endl;
    std::cout << "increment: " << increment() << std::endl;
    std::cout << "original counter: " << counter << std::endl; // 不变
}

void stlAlgorithmDemo() {
    std::vector<int> vec = {5, 3, 1, 4, 2, 8, 6, 7, 9, 0};

    // ===== 排序 =====
    std::sort(vec.begin(), vec.end());
    // 使用 Lambda 自定义比较
    std::sort(vec.begin(), vec.end(), [](int a, int b) {
        return a > b; // 降序
    });

    // ===== 查找 =====
    auto it = std::find_if(vec.begin(), vec.end(), [](int n) {
        return n > 5;
    });
    if (it != vec.end()) {
        std::cout << "找到第一个大于5的: " << *it << std::endl;
    }

    // ===== 遍历 =====
    std::for_each(vec.begin(), vec.end(), [](int n) {
        std::cout << n << " ";
    });
    std::cout << std::endl;

    // ===== 转换 =====
    std::vector<int> squared(vec.size());
    std::transform(vec.begin(), vec.end(), squared.begin(),
        [](int n) { return n * n; });

    // ===== 过滤 + 转换 =====
    std::vector<int> evens;
    std::copy_if(vec.begin(), vec.end(), std::back_inserter(evens),
        [](int n) { return n % 2 == 0; });

    // ===== 聚合 =====
    int sum = std::accumulate(vec.begin(), vec.end(), 0);
    int product = std::accumulate(vec.begin(), vec.end(), 1,
        [](int acc, int n) { return acc * n; });

    // ===== min/max =====
    auto maxIt = std::max_element(vec.begin(), vec.end());
    auto minIt = std::min_element(vec.begin(), vec.end());

    // ===== 计数 =====
    int count = std::count_if(vec.begin(), vec.end(),
        [](int n) { return n > 5; });

    // ===== all_of / any_of / none_of =====
    bool allPositive = std::all_of(vec.begin(), vec.end(),
        [](int n) { return n >= 0; });
    bool anyNegative = std::any_of(vec.begin(), vec.end(),
        [](int n) { return n < 0; });

    // ===== remove-erase 惯用法 =====
    vec.erase(std::remove_if(vec.begin(), vec.end(),
        [](int n) { return n < 5; }), vec.end());

    // ===== 字符串操作 =====
    std::vector<std::string> words = {"banana", "apple", "cherry", "date"};
    std::sort(words.begin(), words.end()); // 字典序
    std::sort(words.begin(), words.end(),
        [](const std::string& a, const std::string& b) {
            return a.size() < b.size(); // 按长度
        });

    // ===== partition 分区 =====
    std::vector<int> nums = {1, 2, 3, 4, 5, 6, 7, 8};
    auto pivot = std::partition(nums.begin(), nums.end(),
        [](int n) { return n % 2 == 0; });
}

int main() {
    lambdaDemo();
    stlAlgorithmDemo();
    return 0;
}
```

### 1.5 多线程（thread、mutex、condition_variable）

```cpp
#include <iostream>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <atomic>
#include <vector>
#include <queue>
#include <functional>
#include <future>
#include <chrono>

// ===== std::thread 基本使用 =====
void threadDemo() {
    // Lambda 方式创建线程
    int x = 10;
    std::thread t1([&x]() {
        x += 5;
        std::cout << "线程1: x = " << x << std::endl;
    });

    // 函数指针方式
    std::thread t2([]() {
        std::cout << "线程2 运行中" << std::endl;
    });

    // 带参数
    std::thread t3([](int a, int b) {
        std::cout << "线程3: " << a + b << std::endl;
    }, 3, 4);

    t1.join(); // 等待线程完成
    t2.join();
    t3.join();

    // detach：分离线程，后台运行
    std::thread t4([]() {
        std::cout << "后台线程运行" << std::endl;
    });
    t4.detach(); // 不需要 join

    // 获取硬件并发数
    std::cout << "硬件并发数: " << std::thread::hardware_concurrency() << std::endl;
}

// ===== std::mutex 互斥锁 =====
std::mutex mtx;
int sharedCounter = 0;

void incrementWithMutex() {
    for (int i = 0; i < 10000; i++) {
        // lock_guard：RAII 锁，构造时加锁，析构时解锁
        std::lock_guard<std::mutex> lock(mtx);
        sharedCounter++;
    }
}

void mutexDemo() {
    std::vector<std::thread> threads;
    for (int i = 0; i < 10; i++) {
        threads.emplace_back(incrementWithMutex);
    }
    for (auto& t : threads) {
        t.join();
    }
    std::cout << "最终计数: " << sharedCounter << std::endl;
}

// ===== unique_lock：更灵活的锁 =====
std::mutex mtx2;

        std::this_thread::sleep_for(std::chrono::milliseconds(100));

        std::lock_guard<std::mutex> lock(queueMtx);
        finished = true;
    }
    cv.notify_all(); // 通知所有等待者
}

void consumer() {
    while (true) {
        std::unique_lock<std::mutex> lock(queueMtx);
        // 等待条件满足（防止虚假唤醒需要循环判断）
        cv.wait(lock, []() {
            return !dataQueue.empty() || finished;
        });

        if (dataQueue.empty() && finished) {
            break;
        }

        int value = dataQueue.front();
        dataQueue.pop();
        lock.unlock();

        std::cout << "消费: " << value << std::endl;
    }
}

// ===== 线程池 =====
class ThreadPool {
private:
    std::vector<std::thread> workers;
    std::queue<std::function<void()>> tasks;
    std::mutex mtx;
    std::condition_variable cv;
    bool stop;

public:
    explicit ThreadPool(size_t numThreads) : stop(false) {
        for (size_t i = 0; i < numThreads; i++) {
            workers.emplace_back([this]() {
                while (true) {
                    std::function<void()> task;
                    {
                        std::unique_lock<std::mutex> lock(mtx);
                        cv.wait(lock, [this]() {
                            return !tasks.empty() || stop;
                        });

                        if (stop && tasks.empty()) return;

                        task = std::move(tasks.front());
                        tasks.pop();
                    }
                    task(); // 执行任务
                }
            });
        }
    }

    ~ThreadPool() {
        {
            std::lock_guard<std::mutex> lock(mtx);
            stop = true;
        }
        cv.notify_all();
        for (auto& w : workers) {
            w.join();
        }
    }

    // 提交任务
    template <typename F, typename... Args>
    auto submit(F&& f, Args&&... args)
        -> std::future<typename std::invoke_result<F, Args...>::type> {
        using RetType = typename std::invoke_result<F, Args...>::type;
        auto task = std::make_shared<std::packaged_task<RetType()>>(
            std::bind(std::forward<F>(f), std::forward<Args>(args)...)
        );
        std::future<RetType> result = task->get_future();
        {
            std::lock_guard<std::mutex> lock(mtx);
            tasks.emplace([task]() { (*task)(); });
        }
        cv.notify_one();
        return result;
    }
};
```

### 2.1 缓冲区溢出漏洞分析（栈溢出）

```cpp
#include <iostream>
#include <cstring>
#include <cstdio>
#include <string>

// ===== 漏洞示例：不安全的 strcpy =====
// 这个函数存在栈缓冲区溢出漏洞
void vulnerableFunction(const char* input) {
    char buffer[64]; // 栈上 64 字节缓冲区
    // 漏洞：不检查输入长度，直接拷贝
    strcpy(buffer, input); // 如果 input > 64 字节，溢出！
    printf("缓冲区内容: %s\n", buffer);

// ===== 漏洞利用原理 =====
// 栈帧布局（从高地址到低地址）：
// [函数参数]
// [返回地址 return address]  <-- 覆盖目标
// [保存的 EBP/RBP]
// [局部变量 buffer[64]]     <-- 溢出起点
//
// 攻击者输入超长数据，覆盖返回地址为 shellcode 地址

void stackOverflowAnalysis() {

    // [64字节填充buffer] + [8字节覆盖EBP] + [8字节覆盖返回地址]
    // 其中返回地址指向 shellcode
}
```

```cpp
// ===== Use-After-Free（UAF）=====
// 释放内存后继续使用指向该内存的指针

// 漏派示例
class UserAccount {
public:
    char name[32];
    int privileges;
    UserAccount(const char* n, int p) : privileges(p) {
        strncpy(name, n, 31);

    }

    UserAccount* user = new UserAccount("guest", 0);
    user->printInfo();

    delete user; // 释放内存

    // 漏洞：释放后继续使用！
    // user->printInfo(); // UAF：内存已被释放，行为未定义
    // 攻击者可以在这之间分配新对象到同一内存位置

    // 更危险的场景：
    // delete user;
    // ... 其他操作可能重新分配这块内存 ...
    // user->privileges = 1; // 攻击者可能写入精心构造的对象
}

// ===== UAF 攻击原理 =====
// 1. 分配对象 A（大小 S）
// 2. 释放对象 A
// 3. 分配对象 B（大小接近 S，可能复用 A 的内存）
// 4. 通过 A 的悬垂指针操作，实际修改 B 的数据
// 这可能导致类型混淆、权限提升等

void uafAttackDemo() {
    // 步骤1：分配 victim
    char* victim = (char*)malloc(32);
    strcpy(victim, "original data");

    // 步骤2：释放 victim
    free(victim);
    // victim 现在是悬垂指针
```

```cpp
bool detectFormatStringAttack(const char* input) {

        "%x", "%p", "%s", "%n", "%hn", "%hhn",
        "%d", "%u", "%c", "%lx", "%lln", "%*d"

            return true;

        while (*p >= '0' && *p <= '9') p++;
        if (*p == '$') {

        }
    }

    return false;
```

```cpp
class SecureTCPClient {
private:
    SOCKET sock;
    std::string host;
    int port;

public:
    SecureTCPClient(const std::string& h, int p) : host(h), port(p), sock(INVALID_SOCKET) {}

    ~SecureTCPClient() {
        if (sock != INVALID_SOCKET) {
            closesocket(sock);
        }
    }

    bool connect() {
        // 创建 socket
        sock = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
        if (sock == INVALID_SOCKET) {
            std::cerr << "创建 socket 失败" << std::endl;
            return false;
        }

        // 设置超时
#ifdef _WIN32
        DWORD timeout = 5000; // 5 秒
        setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, (const char*)&timeout, sizeof(timeout));
        setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO, (const char*)&timeout, sizeof(timeout));
#else
        struct timeval tv;
        tv.tv_sec = 5;
        tv.tv_usec = 0;
        setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
        setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
#endif

        // 连接
        struct sockaddr_in addr;
        addr.sin_family = AF_INET;
        addr.sin_port = htons(port);
        inet_pton(AF_INET, host.c_str(), &addr.sin_addr);

        if (::connect(sock, (struct sockaddr*)&addr, sizeof(addr)) == SOCKET_ERROR) {
            std::cerr << "连接失败" << std::endl;
            closesocket(sock);
            sock = INVALID_SOCKET;
            return false;
        }

        std::cout << "已连接到 " << host << ":" << port << std::endl;
        return true;
    }

    // 安全发送：检查返回值
    bool sendData(const std::string& data) {
        int totalSent = 0;
        int remaining = data.size();

        while (remaining > 0) {
            int sent = send(sock, data.c_str() + totalSent, remaining, 0);
            if (sent == SOCKET_ERROR) {
                std::cerr << "发送失败" << std::endl;
                return false;
            }
            totalSent += sent;
            remaining -= sent;
        }
        return true;
    }

    // 安全接收：指定最大长度防溢出
    std::string receiveData(size_t maxLen = 4096) {
        std::vector<char> buffer(maxLen);
        int received = recv(sock, buffer.data(), maxLen, 0);

        if (received == SOCKET_ERROR) {
            std::cerr << "接收失败" << std::endl;
            return "";
        }
        if (received == 0) {
            std::cout << "连接已关闭" << std::endl;
            return "";
        }

        return std::string(buffer.data(), received);
    }
};

    explicit SecureTCPServer(int p) : port(p), listenSock(INVALID_SOCKET), clientSock(INVALID_SOCKET) {}

    ~SecureTCPServer() {
        if (clientSock != INVALID_SOCKET) closesocket(clientSock);
        if (listenSock != INVALID_SOCKET) closesocket(listenSock);
    }

    bool start() {
        listenSock = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
        if (listenSock == INVALID_SOCKET) return false;

        // 地址重用
        int opt = 1;
        setsockopt(listenSock, SOL_SOCKET, SO_REUSEADDR, (const char*)&opt, sizeof(opt));

        struct sockaddr_in addr;
        addr.sin_family = AF_INET;
        addr.sin_port = htons(port);
        addr.sin_addr.s_addr = htonl(INADDR_ANY);

        if (bind(listenSock, (struct sockaddr*)&addr, sizeof(addr)) == SOCKET_ERROR) {
            std::cerr << "bind 失败" << std::endl;
            return false;
        }

        if (listen(listenSock, 5) == SOCKET_ERROR) {
            std::cerr << "listen 失败" << std::endl;
            return false;
        }

        std::cout << "服务器监听端口 " << port << std::endl;
        return true;
    }

    bool acceptClient() {
        clientSock = accept(listenSock, nullptr, nullptr);
        return clientSock != INVALID_SOCKET;
    }

    std::string receive(size_t maxLen = 4096) {
        std::vector<char> buffer(maxLen);
        int received = recv(clientSock, buffer.data(), maxLen, 0);
        if (received <= 0) return "";
        return std::string(buffer.data(), received);
    }

    bool send(const std::string& data) {
        int total = 0;
        while (total < (int)data.size()) {
            int sent = ::send(clientSock, data.c_str() + total, data.size() - total, 0);
            if (sent == SOCKET_ERROR) return false;
            total += sent;
        }
        return true;
    }
};

class PortScanner {
public:
    static bool scanPort(const std::string& host, int port, int timeoutMs = 2000) {
        SOCKET sock = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
        if (sock == INVALID_SOCKET) return false;

        // 设置非阻塞
#ifdef _WIN32
        u_long mode = 1;
        ioctlsocket(sock, FIONBIO, &mode);
#else
        int flags = fcntl(sock, F_GETFL, 0);
        fcntl(sock, F_SETFL, flags | O_NONBLOCK);
#endif

        struct sockaddr_in addr;
        addr.sin_family = AF_INET;
        addr.sin_port = htons(port);
        inet_pton(AF_INET, host.c_str(), &addr.sin_addr);

        connect(sock, (struct sockaddr*)&addr, sizeof(addr)); // 非阻塞，立即返回

        // 使用 select 等待连接完成
        fd_set writeSet;
        FD_ZERO(&writeSet);
        FD_SET(sock, &writeSet);

        struct timeval tv;
        tv.tv_sec = timeoutMs / 1000;
        tv.tv_usec = (timeoutMs % 1000) * 1000;

        bool isOpen = false;
        int result = select(sock + 1, nullptr, &writeSet, nullptr, &tv);
        if (result > 0) {
#ifdef _WIN32
            char optVal;
            int optLen = sizeof(optVal);
            getsockopt(sock, SOL_SOCKET, SO_ERROR, &optVal, &optLen);
            if (optVal == 0) isOpen = true;
#else
            int error = 0;
            socklen_t len = sizeof(error);
            getsockopt(sock, SOL_SOCKET, SO_ERROR, &error, &len);
            if (error == 0) isOpen = true;
#endif
        }

        closesocket(sock);
        return isOpen;
    }
};
```

```cpp
// 安全拷贝
bool safeCopy(char* dst, size_t dstSize, const char* src, size_t srcLen = 0) {
    if (dst == nullptr || src == nullptr || dstSize == 0) {
        return false;
    }

    // 如果未指定源长度，计算它
    if (srcLen == 0) {
        srcLen = strnlen(src, SIZE_MAX);
    }

    size_t copyLen = (srcLen < dstSize - 1) ? srcLen : dstSize - 1;
    memcpy(dst, src, copyLen);
    dst[copyLen] = '\0';

    return copyLen == srcLen; // 返回是否完整拷贝
}

// 安全拼接
bool safeConcat(char* dst, size_t dstSize, const char* src) {
    if (dst == nullptr || src == nullptr || dstSize == 0) {
        return false;
    }

    size_t dstLen = strnlen(dst, dstSize);
    size_t srcLen = strnlen(src, SIZE_MAX);

    if (dstLen + srcLen >= dstSize) {
        // 截断
        size_t copyLen = dstSize - dstLen - 1;
        memcpy(dst + dstLen, src, copyLen);
        dst[dstLen + copyLen] = '\0';
        return false; // 不完整
    }

    memcpy(dst + dstLen, src, srcLen);
    dst[dstLen + srcLen] = '\0';
    return true;
}

// 安全字符串比较（常量时间，防时序攻击）
bool safeCompare(const char* a, const char* b) {
    if (a == nullptr || b == nullptr) return false;

    size_t lenA = strlen(a);
    size_t lenB = strlen(b);

    // 先比较长度（注意：长度泄露信息，但必须做）
    if (lenA != lenB) {
        // 仍然比较所有字节，防止时序攻击
        volatile int result = 0;
        const char* longer = lenA > lenB ? a : b;
        size_t maxLen = lenA > lenB ? lenA : lenB;
        for (size_t i = 0; i < maxLen; i++) {
            result |= longer[i];
        }
        return false;
    }

    // 常量时间比较
    volatile int result = 0;
    for (size_t i = 0; i < lenA; i++) {
        result |= a[i] ^ b[i];
    }

    return result == 0;
}
```

```cpp
DWORD FindProcessId(const char* processName) {

BOOL InjectShellcode(DWORD pid, unsigned char* shellcode, size_t scSize) {
    // 1. 打开目标进程
    HANDLE hProcess = OpenProcess(PROCESS_ALL_ACCESS, FALSE, pid);
    if (!hProcess) {
        printf("OpenProcess 失败: %d\n", GetLastError());
        return FALSE;
    }
    
    // 2. 在远程进程分配内存
    LPVOID remoteMem = VirtualAllocEx(hProcess, NULL, scSize, 
        MEM_COMMIT | MEM_RESERVE, PAGE_EXECUTE_READWRITE);
    if (!remoteMem) {
        printf("VirtualAllocEx 失败: %d\n", GetLastError());
        CloseHandle(hProcess);
        return FALSE;
    }
    
    // 3. 写入 shellcode
    SIZE_T written;
    if (!WriteProcessMemory(hProcess, remoteMem, shellcode, scSize, &written)) {
        printf("WriteProcessMemory 失败: %d\n", GetLastError());
        VirtualFreeEx(hProcess, remoteMem, 0, MEM_RELEASE);
        CloseHandle(hProcess);
        return FALSE;

    // 4. 创建远程线程执行 shellcode
    HANDLE hThread = CreateRemoteThread(hProcess, NULL, 0,

    if (!hThread) {
        printf("CreateRemoteThread 失败: %d\n", GetLastError());

        return FALSE;
    }
    
    // 等待执行完成（可选）
    WaitForSingleObject(hThread, 5000);

    CloseHandle(hProcess);
    return TRUE;

BOOL ProcessHollowing(const char* targetPath, unsigned char* shellcode, size_t scSize) {
```

### 4.2 自定义 PE 解析器

手动解析 PE 文件结构，用于恶意代码分析。

```cpp
#include <windows.h>
#include <stdio.h>
#include <vector>
#include <string>

class PEParser {
private:
    BYTE* fileBuffer;
    size_t fileSize;
    PIMAGE_DOS_HEADER pDosHeader;
    PIMAGE_NT_HEADERS pNtHeaders;
    PIMAGE_SECTION_HEADER pSectionHeaders;
    
public:
    PEParser(const char* filePath) {
        FILE* f = fopen(filePath, "rb");
        if (!f) throw "无法打开文件";
        
        fseek(f, 0, SEEK_END);
        fileSize = ftell(f);
        fseek(f, 0, SEEK_SET);
        
        fileBuffer = new BYTE[fileSize];
        fread(fileBuffer, 1, fileSize, f);
        fclose(f);
        
        // 验证 DOS 头
        pDosHeader = (PIMAGE_DOS_HEADER)fileBuffer;
        if (pDosHeader->e_magic != IMAGE_DOS_SIGNATURE) {
            delete[] fileBuffer;
            throw "无效的 DOS 头";
        }
        
        // 获取 NT 头
        pNtHeaders = (PIMAGE_NT_HEADERS)(fileBuffer + pDosHeader->e_lfanew);
        if (pNtHeaders->Signature != IMAGE_NT_SIGNATURE) {
            delete[] fileBuffer;
            throw "无效的 NT 签名";
        }
        
        // 获取节表头
        pSectionHeaders = IMAGE_FIRST_SECTION(pNtHeaders);
    }
    
    ~PEParser() {
        delete[] fileBuffer;
    }
    
    // 打印基本信息
    void PrintBasicInfo() {
        printf("=== PE 文件基本信息 ===\n");
        printf("机器类型: 0x%04X", pNtHeaders->FileHeader.Machine);
        if (pNtHeaders->FileHeader.Machine == IMAGE_FILE_MACHINE_AMD64)
            printf(" (x64)");
        else if (pNtHeaders->FileHeader.Machine == IMAGE_FILE_MACHINE_I386)
            printf(" (x86)");
        printf("\n");
        
        printf("节区数量: %d\n", pNtHeaders->FileHeader.NumberOfSections);
        printf("时间戳: 0x%08X\n", pNtHeaders->FileHeader.TimeDateStamp);
        printf("入口点: 0x%08X\n", pNtHeaders->OptionalHeader.AddressOfEntryPoint);
        printf("镜像基址: 0x%08llX\n", (ULONGLONG)pNtHeaders->OptionalHeader.ImageBase);
        printf("节区对齐: 0x%08X\n", pNtHeaders->OptionalHeader.SectionAlignment);
        printf("文件对齐: 0x%08X\n", pNtHeaders->OptionalHeader.FileAlignment);
        printf("子系统: %d\n", pNtHeaders->OptionalHeader.Subsystem);
    }
    
    // 打印节区信息
    void PrintSections() {
        printf("\n=== 节区信息 ===\n");
        printf("%-10s %-10s %-10s %-10s %-10s\n", 
            "名称", "虚拟大小", "虚拟地址", "原始大小", "原始偏移");
        
        for (int i = 0; i < pNtHeaders->FileHeader.NumberOfSections; i++) {
            char name[9] = {0};
            memcpy(name, pSectionHeaders[i].Name, 8);
            
            printf("%-10s 0x%08X 0x%08X 0x%08X 0x%08X  ",
                name,
                pSectionHeaders[i].Misc.VirtualSize,
                pSectionHeaders[i].VirtualAddress,
                pSectionHeaders[i].SizeOfRawData,
                pSectionHeaders[i].PointerToRawData);
            
            // 节区属性
            if (pSectionHeaders[i].Characteristics & IMAGE_SCN_MEM_EXECUTE)
                printf("X");
            if (pSectionHeaders[i].Characteristics & IMAGE_SCN_MEM_READ)
                printf("R");
            if (pSectionHeaders[i].Characteristics & IMAGE_SCN_MEM_WRITE)
                printf("W");
            printf("\n");
        }
    }
    
    // 打印导入表
    void PrintImports() {
        printf("\n=== 导入表 ===\n");
        
        DWORD importRVA = pNtHeaders->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_IMPORT].VirtualAddress;
        if (!importRVA) {
            printf("无导入表\n");
            return;
        }
        
        PIMAGE_IMPORT_DESCRIPTOR pImportDesc = 
            (PIMAGE_IMPORT_DESCRIPTOR)(RvaToOffset(importRVA) + fileBuffer);
        
        while (pImportDesc->Name) {
            char* dllName = (char*)(RvaToOffset(pImportDesc->Name) + fileBuffer);
            printf("DLL: %s\n", dllName);
            
            PIMAGE_THUNK_DATA pThunk = 
                (PIMAGE_THUNK_DATA)(RvaToOffset(pImportDesc->OriginalFirstThunk) + fileBuffer);
            
            while (pThunk->u1.AddressOfData) {
                if (!(pThunk->u1.Ordinal & IMAGE_ORDINAL_FLAG)) {
                    PIMAGE_IMPORT_BY_NAME pImportByName = 
                        (PIMAGE_IMPORT_BY_NAME)(RvaToOffset(pThunk->u1.AddressOfData) + fileBuffer);
                    printf("  - %s\n", pImportByName->Name);
                } else {
                    printf("  - Ordinal: %d\n", IMAGE_ORDINAL(pThunk->u1.Ordinal));
                }
                pThunk++;
            }
            pImportDesc++;
        }
    }
    
    // 检测可疑特征
    std::vector<std::string> DetectSuspicious() {
        std::vector<std::string> findings;
        
        // 1. 检查非标准节区
        const char* standardSections[] = {".text", ".data", ".rdata", ".bss", 
            ".idata", ".edata", ".rsrc", ".reloc", ".tls", ".pdata"};
        int standardCount = sizeof(standardSections) / sizeof(standardSections[0]);
        
        for (int i = 0; i < pNtHeaders->FileHeader.NumberOfSections; i++) {
            char name[9] = {0};
            memcpy(name, pSectionHeaders[i].Name, 8);
            
            bool isStandard = false;
            for (int j = 0; j < standardCount; j++) {
                if (strcmp(name, standardSections[j]) == 0) {
                    isStandard = true;
                    break;
                }
            }
            
            if (!isStandard) {
                char buf[256];
                sprintf(buf, "非标准节区: %s", name);
                findings.push_back(buf);
            }
            
            // 2. 可写可执行节区
            if ((pSectionHeaders[i].Characteristics & IMAGE_SCN_MEM_EXECUTE) &&
                (pSectionHeaders[i].Characteristics & IMAGE_SCN_MEM_WRITE)) {
                char buf[256];
                sprintf(buf, "可写可执行节区: %s (W+X)", name);
                findings.push_back(buf);
            }
            
            // 3. 异常大的节区
            if (pSectionHeaders[i].Misc.VirtualSize > 10 * 1024 * 1024) {
                char buf[256];
                sprintf(buf, "异常大的节区: %s (%d MB)", name, 
                    pSectionHeaders[i].Misc.VirtualSize / (1024*1024));
                findings.push_back(buf);
            }
        }
        
        // 4. 检查入口点不在 .text 节
        DWORD entryPoint = pNtHeaders->OptionalHeader.AddressOfEntryPoint;
        bool entryInText = false;
        for (int i = 0; i < pNtHeaders->FileHeader.NumberOfSections; i++) {
            if (entryPoint >= pSectionHeaders[i].VirtualAddress &&
                entryPoint < pSectionHeaders[i].VirtualAddress + pSectionHeaders[i].Misc.VirtualSize) {
                char name[9] = {0};
                memcpy(name, pSectionHeaders[i].Name, 8);
                if (strcmp(name, ".text") == 0) entryInText = true;
                break;
            }
        }
        if (!entryInText) {
            findings.push_back("入口点不在 .text 节区（可能被加壳/混淆）");
        }
        
        return findings;
    }
    
private:
    DWORD RvaToOffset(DWORD rva) {
        for (int i = 0; i < pNtHeaders->FileHeader.NumberOfSections; i++) {
            if (rva >= pSectionHeaders[i].VirtualAddress &&
                rva < pSectionHeaders[i].VirtualAddress + pSectionHeaders[i].Misc.VirtualSize) {
                return rva - pSectionHeaders[i].VirtualAddress + pSectionHeaders[i].PointerToRawData;
            }
        }
        return 0;
    }
};

int main(int argc, char* argv[]) {
    if (argc < 2) {
        printf("用法: %s <pe文件路径>\n", argv[0]);
        return 1;
    }
    
    try {
        PEParser parser(argv[1]);
        parser.PrintBasicInfo();
        parser.PrintSections();
        parser.PrintImports();
        
        printf("\n=== 可疑特征检测 ===\n");
        auto findings = parser.DetectSuspicious();
        if (findings.empty()) {
            printf("未发现明显可疑特征\n");
        } else {
            for (auto& f : findings) {
                printf("  [!] %s\n", f.c_str());
            }
        }
    } catch (const char* err) {
        printf("错误: %s\n", err);
        return 1;
    }
    
    return 0;
}
```
