### 1.1 所有权与借用进阶（生命周期标注）

Rust 的所有权系统是其内存安全的核心。理解生命周期标注是掌握进阶 Rust 的关键。

**所有权基础回顾**

```rust
// 所有权三规则：
// 1. 每个值有且只有一个所有者
// 2. 当所有者离开作用域，值被丢弃
// 3. 赋值/传参时默认是"移动"（move），不是复制

fn main() {
    let s1 = String::from("hello");
    let s2 = s1;              // s1 的所有权移动到 s2
    // println!("{}", s1);    // 编译错误：s1 已被移动

    let s3 = s2.clone();      // 深拷贝，s2 仍然有效
    println!("{}", s2);       // 正常
    println!("{}", s3);       // 正常

    // 基本类型（i32, bool 等）实现了 Copy trait，赋值时复制而非移动
    let x = 5;
    let y = x;                // x 被复制，x 仍然有效
    println!("x={}, y={}", x, y);
}
```

**借用规则**

```rust
fn main() {
    let mut data = String::from("hello");

    // 同一时刻：
    // - 可以有多个不可变引用（&T）
    // - 或一个可变引用（&mut T）
    // - 但不能同时存在可变和不可变引用

    let r1 = &data;           // 不可变借用
    let r2 = &data;           // 不可变借用（可以多个）
    println!("{} {}", r1, r2);
    // r1, r2 在此之后不再使用（NLL: Non-Lexical Lifetimes）

    let r3 = &mut data;       // 可变借用（此时前面不可变引用已结束）
    r3.push_str(" world");
    println!("{}", r3);

    // 数据竞争在编译期被阻止
    // let r4 = &data;         // 编译错误：已经有可变借用
}
```

**生命周期标注**

当编译器无法自动推断引用的生命周期时，需要手动标注。

```rust
// 基本生命周期标注语法：'a
// 'a 是一个泛型生命周期参数，表示"某个生命周期"

// 函数返回的引用生命周期必须与输入引用有关联
fn longest<'a>(x: &'a str, y: &'a str) -> &'a str {
    // 'a 表示：返回值的生命周期 = x 和 y 中较短的那个
    if x.len() > y.len() {
        x
    } else {
        y
    }
}

fn main() {
    let s1 = String::from("long string");
    let s2 = String::from("short");
    let result = longest(s1.as_str(), s2.as_str());
    println!("最长的: {}", result);

    // 生命周期保证返回的引用不会悬空
    // let r;
    // {
    //     let s = String::from("inner");
    //     r = longest(s.as_str(), s1.as_str());
    // }  // s 在这里被释放
    // println!("{}", r);  // 编译错误：s 已失效
}

// 结构体中的生命周期
struct Parser<'a> {
    input: &'a str,   // 结构体持有引用，必须标注生命周期
    pos: usize,
}

impl<'a> Parser<'a> {
    fn new(input: &'a str) -> Self {
        Parser { input, pos: 0 }
    }

    // 返回的引用与 Parser 的生命周期相同
    fn next_token(&mut self) -> Option<&'a str> {
        let rest = &self.input[self.pos..];
        match rest.find(char::is_whitespace) {
            Some(end) => {
                let token = &rest[..end];
                self.pos += end + 1;
                Some(token)
            }
            None if !rest.is_empty() => {
                self.pos = self.input.len();
                Some(rest)
            }
            None => None,
        }
    }
}

// 静态生命周期：'static 表示存活整个程序运行期
// 所有字符串字面量都是 &'static str
fn static_example() {
    let s: &'static str = "我活在整个程序期间";
    // 字符串字面量直接编译进二进制文件
}
```

**生命周期消除规则**

```rust
// 编译器有三条消除规则，大多数情况不需要手写标注

// 规则 1：每个引用参数都获得自己的生命周期
fn foo(x: &str, y: &str) {}  // 编译器看作 fn foo<'a, 'b>(x: &'a str, y: &'b str)

// 规则 2：如果只有一个输入生命周期参数，它被赋给所有输出
fn bar(x: &str) -> &str {}   // 编译器看作 fn bar<'a>(x: &'a str) -> &'a str

// 规则 3：如果有 &self 或 &mut self，self 的生命周期被赋给所有输出
impl Foo {
    fn method(&self, x: &str) -> &str {}  // 输出生命周期 = &self
}
```

### 1.2 智能指针（Box、Rc、RefCell、Arc、Mutex）

**Box：堆分配**

```rust
fn main() {
    // Box 将数据分配到堆上
    let b = Box::new(5);
    println!("b = {}", b);   // 自动解引用

    // 用途 1：递归类型（大小在编译期未知）
    #[derive(Debug)]
    enum List {
        Cons(i32, Box<List>),
        Nil,
    }
    let list = List::Cons(1, Box::new(List::Cons(2, Box::new(List::Nil))));
    println!("{:?}", list);

    // 用途 2：大型数据转移所有权时避免拷贝
    let large_array = Box::new([0u8; 1_000_000]);
    let moved = large_array;  // 只拷贝一个指针

    // 用途 3：trait 对象（动态分发）
    trait Animal {
        fn sound(&self) -> String;
    }
    struct Dog;
    impl Animal for Dog {
        fn sound(&self) -> String { "Woof".to_string() }
    }
    struct Cat;
    impl Animal for Cat {
        fn sound(&self) -> String { "Meow".to_string() }
    }

    // 编译期不知道具体类型，用 Box<dyn Trait> 做动态分发
    let animals: Vec<Box<dyn Animal>> = vec![
        Box::new(Dog),
        Box::new(Cat),
    ];
    for a in &animals {
        println!("{}", a.sound());
    }
}
```

**Rc：引用计数（单线程）**

```rust
use std::rc::Rc;

fn main() {
    // Rc 允许多个所有者共享数据（不可变）
    // 通过引用计数跟踪，当计数为 0 时释放
    let data = Rc::new(String::from("shared data"));

    let a = Rc::clone(&data);    // 引用计数 +1
    let b = Rc::clone(&data);    // 引用计数 +1

    println!("计数: {}", Rc::strong_count(&data));  // 3
    println!("a={}, b={}", a, b);

    // Rc 是不可变的：不能通过 Rc 修改内部值
    // 要修改需要 Rc<RefCell<T>>

    // 典型用途：图数据结构、共享配置
    let config = Rc::new(vec!["admin", "user"]);
    let handler1 = Rc::clone(&config);
    let handler2 = Rc::clone(&config);
    // 三个 handler 共享同一份配置
}
```

**RefCell：内部可变性（单线程）**

```rust
use std::cell::RefCell;

fn main() {
    // RefCell 在运行时检查借用规则（而非编译时）
    // 配合 Rc 实现多所有者 + 可变

    let data = Rc::new(RefCell::new(vec![1, 2, 3]));

    let d1 = Rc::clone(&data);
    let d2 = Rc::clone(&data);

    // 通过 borrow_mut 获取可变借用
    d1.borrow_mut().push(4);
    d2.borrow_mut().push(5);

    // 通过 borrow 获取不可变借用
    println!("数据: {:?}", data.borrow());

    // 运行时借用检查：同时只能有一个 mutable borrow
    // let r1 = data.borrow_mut();
    // let r2 = data.borrow_mut();  // panic: 已被可变借用
}
```

**Arc：原子引用计数（多线程）**

```rust
use std::sync::Arc;
use std::thread;

fn main() {
    // Arc 是线程安全版本的 Rc
    // 使用原子操作实现引用计数，开销比 Rc 大

    let data = Arc::new(vec![1, 2, 3, 4, 5]);

    let mut handles = vec![];
    for i in 0..3 {
        let data = Arc::clone(&data);    // 原子计数 +1
        handles.push(thread::spawn(move || {
            println!("线程 {} 看到的数据: {:?}", i, data);
        }));
    }

    for h in handles {
        h.join().unwrap();
    }

    println!("最终引用计数: {}", Arc::strong_count(&data));
}
```

**Mutex：互斥锁**

```rust
use std::sync::{Arc, Mutex};
use std::thread;

fn main() {
    // Mutex 保证同一时刻只有一个线程访问数据
    let counter = Arc::new(Mutex::new(0));

    let mut handles = vec![];
    for _ in 0..10 {
        let counter = Arc::clone(&counter);
        handles.push(thread::spawn(move || {
            // lock() 获取锁，返回 MutexGuard
            let mut num = counter.lock().unwrap();
            *num += 1;
            // MutexGuard 离开作用域时自动释放锁
        }));
    }

    for h in handles {
        h.join().unwrap();
    }

    println!("最终计数: {}", *counter.lock().unwrap());  // 10

    // Arc<Mutex<T>> 是最常见的多线程共享可变数据模式
    // 类似于 "Rc<RefCell<T>>" 的多线程版本

    // RwLock：允许多个读或一个写
    use std::sync::RwLock;
    let data = Arc::new(RwLock::new(0i32));

    // 多个读可以并发
    let read_handles: Vec<_> = (0..5).map(|_| {
        let data = Arc::clone(&data);
        thread::spawn(move || {
            let r = data.read().unwrap();
            println!("读到: {}", *r);
        })
    }).collect();
    for h in read_handles { h.join().unwrap(); }

    // 写锁独占
    {
        let mut w = data.write().unwrap();
        *w = 42;
    }
    println!("写后: {}", *data.read().unwrap());
}
```

### 1.3 trait 进阶（trait object vs 泛型、关联类型）

**trait object（动态分发）vs 泛型（静态分发）**

```rust
// === 泛型（静态分发 / 单态化） ===
// 编译期为每种类型生成一份代码，零运行时开销
trait Shape {
    fn area(&self) -> f64;
}

struct Circle { radius: f64 }
struct Square { side: f64 }

impl Shape for Circle {
    fn area(&self) -> f64 { std::f64::consts::PI * self.radius * self.radius }
}
impl Shape for Square {
    fn area(&self) -> f64 { self.side * self.side }
}

// 泛型函数：编译时为每种类型生成特化代码
fn print_area<T: Shape>(shape: &T) {
    println!("面积: {}", shape.area());
}

fn main() {
    let c = Circle { radius: 2.0 };
    let s = Square { side: 3.0 };
    print_area(&c);     // 生成 print_area::<Circle>
    print_area(&s);     // 生成 print_area::<Square>
    // 优点：零开销，内联优化
    // 缺点：二进制体积增大，不同类型不能放同一个集合
}

// === trait object（动态分发） ===
// 通过 vtable 在运行时分发，有一个间接跳转开销
fn print_area_dyn(shape: &dyn Shape) {
    println!("面积: {}", shape.area());
}

fn main_dyn() {
    let c = Circle { radius: 2.0 };
    let s = Square { side: 3.0 };
    print_area_dyn(&c);
    print_area_dyn(&s);

    // trait object 可以放同一个集合
    let shapes: Vec<Box<dyn Shape>> = vec![
        Box::new(Circle { radius: 2.0 }),
        Box::new(Square { side: 3.0 }),
    ];
    for s in &shapes {
        println!("面积: {}", s.area());
    }
    // 优点：灵活，不同类型可共存
    // 缺点：不能内联，有间接调用开销
}
```

**关联类型**

```rust
use std::iter::Iterator;

// 关联类型让 trait 更灵活，一个类型只实现一次
trait Database {
    type Error;            // 关联类型：实现时确定
    type Record;

    fn get(&self, id: &str) -> Result<Self::Record, Self::Error>;
    fn insert(&self, record: Self::Record) -> Result<(), Self::Error>;
}

struct User { id: String, name: String }
struct UserDB;

impl Database for UserDB {
    type Error = String;
    type Record = User;

    fn get(&self, id: &str) -> Result<User, String> {
        Ok(User { id: id.to_string(), name: "Alice".to_string() })
    }
    fn insert(&self, record: User) -> Result<(), String> {
        println!("插入用户: {}", record.name);
        Ok(())
    }
}

fn main() {
    let db = UserDB;
    let user = db.get("1").unwrap();
    println!("获取到: {}", user.name);
}

// 对比泛型参数 trait：
// trait Database<Record, Error> { ... }  // 每次调用要指定类型参数
// 关联类型更简洁：db.get("1") 直接返回 User
```

### 1.4 迭代器与闭包

**迭代器**

```rust
fn main() {
    let nums = vec![1, 2, 3, 4, 5];

    // 迭代器适配器（lazy：不消费就不执行）
    let result: Vec<i32> = nums
        .iter()                       // 创建迭代器
        .map(|x| x * 2)              // 映射（惰性）
        .filter(|x| *x > 4)          // 过滤（惰性）
        .collect();                  // 消费迭代器，收集结果

    println!("{:?}", result);        // [6, 8, 10]

    // 常用适配器
    let doubled: Vec<_> = nums.iter().map(|x| x * 2).collect();
    let evens: Vec<_> = nums.iter().filter(|x| *x % 2 == 0).copied().collect();
    let sum: i32 = nums.iter().sum();                    // 求和
    let max: Option<&i32> = nums.iter().max();           // 最大值
    let count = nums.iter().filter(|x| **x > 2).count(); // 计数
    let any = nums.iter().any(|x| *x > 3);               // 任意匹配
    let all = nums.iter().all(|x| *x > 0);               // 全部匹配

    // enumerate：带索引
    for (i, v) in nums.iter().enumerate() {
        println!("[{}] = {}", i, v);
    }

    // zip：并行迭代
    let names = vec!["Alice", "Bob", "Carol"];
    let ages = vec![30, 25, 28];
    let pairs: Vec<_> = names.iter().zip(ages.iter()).collect();
    // [("Alice", 30), ("Bob", 25), ("Carol", 28)]

    // flat_map：映射后扁平化
    let words = vec!["hello world", "foo bar"];
    let all_words: Vec<_> = words.iter()
        .flat_map(|s| s.split_whitespace())
        .collect();
    // ["hello", "world", "foo", "bar"]

    // chunks：分块处理
    let big = vec![1, 2, 3, 4, 5, 6, 7, 8];
    for chunk in big.chunks(3) {
        println!("{:?}", chunk);  // [1,2,3] [4,5,6] [7,8]
    }

    // 自定义迭代器
    struct Counter { count: u32, max: u32 }
    impl Counter {
        fn new(max: u32) -> Self { Counter { count: 0, max } }
    }
    impl Iterator for Counter {
        type Item = u32;
        fn next(&mut self) -> Option<Self::Item> {
            if self.count < self.max {
                self.count += 1;
                Some(self.count)
            } else {
                None
            }
        }
    }

    for n in Counter::new(5) {
        println!("{}", n);  // 1 2 3 4 5
    }
}
```

**闭包**

```rust
fn main() {
    // 闭包语法
    let add = |a, b| a + b;              // 类型推断
    let add: fn(i32, i32) -> i32 = |a, b| a + b;
    let add = |a: i32, b: i32| -> i32 { a + b };

    println!("{}", add(1, 2));

    // 闭包捕获环境变量（三种方式）
    let name = String::from("Alice");

    // 1. 不可变借用（Fn）
    let greet = || println!("Hello, {}", name);
    greet();

    // 2. 可变借用（FnMut）
    let mut count = 0;
    let mut increment = || {
        count += 1;     // 可变借用外部的 count
    };
    increment();
    increment();
    println!("count = {}", count);  // 2

    // 3. 移动所有权（FnOnce）
    let data = vec![1, 2, 3];
    let consume = move || {
        println!("消费了 {:?}", data);   // data 被移入闭包
    };
    consume();
    // println!("{:?}", data);  // 编译错误：data 已被移动

    // 闭包作为参数
    fn apply<F: Fn(i32) -> i32>(f: F, x: i32) -> i32 {
        f(x)
    }
    println!("{}", apply(|n| n * 2, 5));   // 10

    // 闭包作为返回值
    fn make_multiplier(factor: i32) -> impl Fn(i32) -> i32 {
        move |x| x * factor
    }
    let double = make_multiplier(2);
    let triple = make_multiplier(3);
    println!("{} {}", double(5), triple(5));  // 10 15

    // 函数指针 vs 闭包
    // fn 类型是函数指针（不接受捕获环境）
    fn process(f: fn(i32) -> i32, x: i32) -> i32 { f(x) }
    fn square(x: i32) -> i32 { x * x }
    println!("{}", process(square, 5));  // 25
    // process(|x| x * 2, 5);  // 编译错误：闭包不是 fn 类型
}
```

### 1.5 错误处理进阶（自定义错误类型、thiserror/anyhow）

**Result 与 ? 操作符**

```rust
use std::fs::File;
use std::io::{self, Read};

// ? 操作符：出错时提前返回，成功时解包
fn read_config(path: &str) -> Result<String, io::Error> {
    let mut file = File::open(path)?;     // 打开失败则 return Err
    let mut contents = String::new();
    file.read_to_string(&mut contents)?;  // 读取失败则 return Err
    Ok(contents)
}

// ? 也可以做错误类型转换（From trait）
fn read_and_parse(path: &str) -> Result<i32, io::Error> {
    let contents = read_config(path)?;    // io::Error
    // 以下会自动将 ParseIntError 转为 io::Error（通过 From）
    let num: i32 = contents.trim().parse()
        .map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))?;
    Ok(num)
}

fn main() {
    match read_and_parse("config.txt") {
        Ok(n) => println!("配置值: {}", n),
        Err(e) => eprintln!("错误: {}", e),
    }
}
```

**自定义错误类型**

```rust
use std::fmt;
use std::error::Error;

// 定义应用错误枚举
#[derive(Debug)]
enum AppError {
    Io(io::Error),
    Parse(std::num::ParseIntError),
    NotFound(String),
    Unauthorized,
    Custom(String),
}

// 实现 Display（必需）
impl fmt::Display for AppError {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        match self {
            AppError::Io(e) => write!(f, "IO 错误: {}", e),
            AppError::Parse(e) => write!(f, "解析错误: {}", e),
            AppError::NotFound(msg) => write!(f, "未找到: {}", msg),
            AppError::Unauthorized => write!(f, "未授权"),
            AppError::Custom(msg) => write!(f, "自定义错误: {}", msg),
        }
    }
}

// 实现 std::error::Error（推荐）
impl Error for AppError {
    fn source(&self) -> Option<&(dyn Error + 'static)> {
        match self {
            AppError::Io(e) => Some(e),
            AppError::Parse(e) => Some(e),
            _ => None,
        }
    }
}

// 实现 From 以支持 ? 自动转换
impl From<io::Error> for AppError {
    fn from(e: io::Error) -> Self { AppError::Io(e) }
}
impl From<std::num::ParseIntError> for AppError {
    fn from(e: std::num::ParseIntError) -> Self { AppError::Parse(e) }
}

// 使用：? 自动将底层错误转为 AppError
fn load_config(path: &str) -> Result<i32, AppError> {
    let content = std::fs::read_to_string(path)?;   // io::Error -> AppError
    let num: i32 = content.trim().parse()?;           // ParseIntError -> AppError
    if num < 0 {
        return Err(AppError::Custom("值不能为负".into()));
    }
    Ok(num)
}
```

### 2.2 Rust 编写 Web 端口扫描器

```rust
use std::net::{TcpStream, ToSocketAddrs};
use std::time::Duration;
use std::sync::Arc;
use std::sync::mpsc;
use std::thread;

// 异步多线程端口扫描器
struct PortScanner {
    target: String,
    timeout: Duration,
    threads: usize,
}

impl PortScanner {
    fn new(target: &str) -> Self {
        PortScanner {
            target: target.to_string(),
            timeout: Duration::from_millis(500),
            threads: 100,
        }
    }

    // 扫描单个端口
    fn scan_port(&self, port: u16) -> bool {
        let addr = format!("{}:{}", self.target, port);
        match addr.to_socket_addrs() {
            Ok(mut addrs) => {
                if let Some(addr) = addrs.next() {
                    TcpStream::connect_timeout(&addr, self.timeout).is_ok()
                } else {
                    false
                }
            }
            Err(_) => false,
        }
    }

    // 扫描端口范围
    fn scan_range(&self, start: u16, end: u16) -> Vec<(u16, String)> {
        let (tx, rx) = mpsc::channel();
        let rx = Arc::new(rx);   // 注意：channel 不是简单的多消费者

        // 使用工作窃取模式：把所有任务放入队列，多个线程取任务
        let ports: Vec<u16> = (start..=end).collect();
        let ports = Arc::new(ports);
        let mut handles = vec![];

        // 多线程扫描
        for worker_id in 0..self.threads {
            let ports = Arc::clone(&ports);
            let target = self.target.clone();
            let timeout = self.timeout;
            let tx = tx.clone();

            handles.push(thread::spawn(move || {
                // 每个线程处理一部分端口
                for &port in ports.iter().step_by(worker_id + 1) {
                    let addr = format!("{}:{}", target, port);
                    if let Ok(socket_addrs) = addr.to_socket_addrs() {
                        if let Some(addr) = socket_addrs.next() {
                            if TcpStream::connect_timeout(&addr, timeout).is_ok() {
                                let service = identify_service(port);
                                tx.send((port, service)).unwrap();
                            }
                        }
                    }
                }
            }));
        }

        drop(tx);   // 关闭发送端，rx 才能结束
        rx.iter().collect()
    }

    // 扫描常见端口
    fn scan_common(&self) -> Vec<(u16, String)> {
        let common_ports: Vec<u16> = vec![
            21, 22, 23, 25, 53, 80, 110, 143, 443, 445,
            993, 995, 1433, 1521, 3306, 3389, 5432,
            5900, 6379, 8080, 8443, 9200, 27017
        ];

        let (tx, rx) = mpsc::channel();
        let mut handles = vec![];

        for port in common_ports {
            let target = self.target.clone();
            let timeout = self.timeout;
            let tx = tx.clone();

            handles.push(thread::spawn(move || {
                let addr = format!("{}:{}", target, port);
                if let Ok(mut socket_addrs) = addr.to_socket_addrs() {
                    if let Some(addr) = socket_addrs.next() {
                        if TcpStream::connect_timeout(&addr, timeout).is_ok() {
                            // 尝试获取服务横幅
                            let banner = grab_banner(&addr, port);
                            tx.send((port, banner)).unwrap();
                        }
                    }
                }
            }));
        }

        drop(tx);
        rx.iter().collect()
    }
}

    match port {

        22 => "SSH".to_string(),
        23 => "Telnet".to_string(),
        25 => "SMTP".to_string(),
        53 => "DNS".to_string(),
        80 => "HTTP".to_string(),
        443 => "HTTPS".to_string(),
        3306 => "MySQL".to_string(),
        5432 => "PostgreSQL".to_string(),
        6379 => "Redis".to_string(),
        8080 => "HTTP-Alt".to_string(),

    }

// 获取服务横幅（用于版本识别）
fn grab_banner(addr: &std::net::SocketAddr, port: u16) -> String {
    use std::io::{Read, Write};
    use std::net::TcpStream;

    if let Ok(mut stream) = TcpStream::connect_timeout(addr, Duration::from_millis(1000)) {
        stream.set_read_timeout(Some(Duration::from_millis(1000))).ok();

        // 某些服务需要先发送数据（如 HTTP）
        if port == 80 || port == 8080 {
            let _ = stream.write(b"GET / HTTP/1.0\r\nHost: localhost\r\n\r\n");
        }

        let mut buf = [0u8; 1024];
        if let Ok(n) = stream.read(&mut buf) {
            if n > 0 {
                let banner = String::from_utf8_lossy(&buf[..n]);
                // 提取 Server 头
                for line in banner.lines() {
                    if line.to_lowercase().starts_with("server:") {
                        return line.to_string();
                    }
                }
                // 返回前几行
                return banner.lines().take(1).collect::<String>();
            }
        }
    }
    identify_service(port)
}

    }

    let results = scanner.scan_range(1, 1000);
```

```rust
// === 4. 数据竞争：编译期阻止 ===
fn data_race_safety() {
    use std::sync::{Arc, Mutex};
    use std::thread;

    // Rust 的借用检查器在编译期阻止数据竞争
    // 规则：同一数据同一时刻只能有一个可变引用

    // 以下代码编译错误：
    // let mut data = 0;
    // let r1 = &mut data;
    // let r2 = &mut data;  // 编译错误：已有可变借用
    // thread::spawn(move || { *r1 = 1; });
    // *r2 = 2;  // 数据竞争！

    // 正确做法：使用 Arc<Mutex<T>>
    let data = Arc::new(Mutex::new(0));
    let mut handles = vec![];

    for _ in 0..10 {
        let data = Arc::clone(&data);
        handles.push(thread::spawn(move || {
            let mut num = data.lock().unwrap();
            *num += 1;
        }));
    }

    for h in handles { h.join().unwrap(); }
    println!("最终值: {}", *data.lock().unwrap());   // 10
```

```rust
            id: 2003, name: "XSS - javascript: 伪协议".into(),
            pattern: Regex::new(r"(?i)javascript:").unwrap(),
            severity: Severity::High, action: Action::Block,
        });

        // 路径穿越
        rules.push(WafRule {
            id: 3001, name: "路径穿越".into(),
            pattern: Regex::new(r"\.\.[\\/]").unwrap(),
            severity: Severity::High, action: Action::Block,
        });

        // 命令注入
        rules.push(WafRule {
            id: 4001, name: "命令注入 - 管道符".into(),
            pattern: Regex::new(r"[|;&`$()]").unwrap(),
            severity: Severity::Medium, action: Action::Log,
        });

        // SSRF
        rules.push(WafRule {
            id: 5001, name: "SSRF - 内网 IP".into(),
            pattern: Regex::new(r"(?:127\.0\.0\.1|localhost|169\.254\.|10\.|192\.168\.)").unwrap(),
            severity: Severity::High, action: Action::Block,
        });

        WafEngine { rules }
    }

    // 检查请求
    pub fn check_request(
        &self,
        method: &str,
        path: &str,
        query: &HashMap<String, String>,
        headers: &HashMap<String, String>,
        body: &str,
    ) -> Vec<WafAlert> {
        let mut alerts = vec![];

        // 检查 URL 路径
        for rule in &self.rules {
            if rule.pattern.is_match(path) {
                alerts.push(WafAlert {
                    rule_id: rule.id, rule_name: rule.name.clone(),
                    matched_in: "path".into(), matched_value: path.into(),
                    severity: rule.severity.clone(), action: rule.action.clone(),
                });
            }
        }

        // 检查查询参数
        for (key, value) in query {
            for rule in &self.rules {
                if rule.pattern.is_match(value) {
                    alerts.push(WafAlert {
                        rule_id: rule.id, rule_name: rule.name.clone(),
                        matched_in: format!("query:{}", key),
                        matched_value: value.clone(),
                        severity: rule.severity.clone(), action: rule.action.clone(),
                    });
                }
            }
        }

        // 检查请求头
        for (key, value) in headers {
            for rule in &self.rules {
                if rule.pattern.is_match(value) {
                    alerts.push(WafAlert {
                        rule_id: rule.id, rule_name: rule.name.clone(),
                        matched_in: format!("header:{}", key),
                        matched_value: value.clone(),
                        severity: rule.severity.clone(), action: rule.action.clone(),
                    });
                }
            }
        }

        // 检查请求体
        for rule in &self.rules {
            if rule.pattern.is_match(body) {
                alerts.push(WafAlert {
                    rule_id: rule.id, rule_name: rule.name.clone(),
                    matched_in: "body".into(), matched_value: body[..body.len().min(200)].into(),
                    severity: rule.severity.clone(), action: rule.action.clone(),
                });
            }
        }

        alerts
    }

    // 判断请求是否应该被阻止
    pub fn should_block(&self, alerts: &[WafAlert]) -> bool {
        alerts.iter().any(|a| a.action == Action::Block)
    }
}

struct WafAlert {
    rule_id: u32,
    rule_name: String,
    matched_in: String,
    matched_value: String,
    severity: Severity,
    action: Action,
}

fn main() {
    let waf = WafEngine::new();

    // 模拟恶意请求
    let mut query = HashMap::new();
    query.insert("id".to_string(), "1 OR 1=1".to_string());
    query.insert("search".to_string(), "<script>alert(1)</script>".to_string());

    let mut headers = HashMap::new();
    headers.insert("User-Agent".to_string(), "sqlmap/1.5".to_string());

    let body = r#"{"name":"'; DROP TABLE users; --"}"#;

    let alerts = waf.check_request("GET", "/api/users", &query, &headers, body);

    println!("=== WAF 检测结果 ===");
    for alert in &alerts {
        println!("[{}] 规则 {} - {} (位置: {})",
            match alert.severity {
                Severity::Critical => "严重", Severity::High => "高危",
                Severity::Medium => "中危", Severity::Low => "低危",
            },
            alert.rule_id, alert.rule_name, alert.matched_in);
    }

        println!("\n[!] 请求已拦截");
    } else {
        println!("\n[OK] 请求放行");
    }
}
```

### 2.5 Rust 文件系统安全操作

```rust
use std::fs;
use std::path::{Path, PathBuf, Component};
use std::io;

// 安全的文件操作工具
pub struct SafeFileOps {
    base_dir: PathBuf,
}

impl SafeFileOps {
    pub fn new(base: &str) -> Self {
        SafeFileOps { base_dir: PathBuf::from(base) }
    }

    // 防止路径穿越：确保最终路径在 base_dir 内
    fn sanitize_path(&self, user_input: &str) -> io::Result<PathBuf> {
        // 1. 拼接路径
        let joined = self.base_dir.join(user_input);

        // 2. 规范化路径（解析 . 和 ..）
        let canonical = joined.canonicalize()?;

        // 3. 检查规范化后的路径是否仍以 base_dir 开头
        if !canonical.starts_with(&self.base_dir) {
            return Err(io::Error::new(
                io::ErrorKind::PermissionDenied,
                "路径穿越攻击被阻止",
            ));
        }

        // 4. 检查路径组件中不包含意外内容
        let path = Path::new(user_input);
        for component in path.components() {
            match component {
                Component::Normal(_) => {},    // 正常路径组件
                Component::CurDir => {},       // .
                _ => return Err(io::Error::new(
                    io::ErrorKind::InvalidInput,
                    "路径包含非法组件",
                )),
            }
        }

        Ok(canonical)
    }

    // 安全读取文件
    pub fn read_file(&self, filename: &str) -> io::Result<String> {
        let path = self.sanitize_path(filename)?;
        fs::read_to_string(path)
    }

    // 安全写入文件
    pub fn write_file(&self, filename: &str, content: &str) -> io::Result<()> {
        let path = self.sanitize_path(filename)?;
        fs::write(path, content)
    }

    // 安全删除文件
    pub fn delete_file(&self, filename: &str) -> io::Result<()> {
        let path = self.sanitize_path(filename)?;
        let metadata = fs::metadata(&path)?;

        // 防止删除目录（除非显式要求）
        if metadata.is_dir() {
            return Err(io::Error::new(
                io::ErrorKind::InvalidInput,
                "目标是目录，请使用 delete_dir",
            ));
        }

        // 检查文件权限
        #[cfg(unix)]
        {
            use std::os::unix::fs::PermissionsExt;
            let perms = metadata.permissions();
            if perms.mode() & 0o4000 != 0 {
                return Err(io::Error::new(
                    io::ErrorKind::PermissionDenied,
                    "不能删除 setuid 文件",
                ));
            }
        }

        fs::remove_file(path)
    }

    // 安全创建临时文件（防止 TOCTOU 攻击）
    pub fn create_temp_file(&self) -> io::Result<(PathBuf, fs::File)> {
        // 使用随机文件名防止预测
        use std::time::{SystemTime, UNIX_EPOCH};
        let timestamp = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .unwrap()
            .as_nanos();

        let random_part: u64 = rand::random();
        let temp_name = format!("temp_{}_{}", timestamp, random_part);

        let temp_path = self.base_dir.join(&temp_name);

        // 使用 OpenOptions 确保文件不存在才创建（O_CREAT | O_EXCL）
        let file = fs::OpenOptions::new()
            .write(true)
            .create_new(true)       // 文件已存在则失败（防 TOCTOU）
            .mode(0o600)            // 仅 owner 可读写（Unix）
            .open(&temp_path)?;

        Ok((temp_path, file))
    }

    // 安全地遍历目录（限制深度）
    pub fn walk_dir(&self, dir: &str, max_depth: usize) -> io::Result<Vec<PathBuf>> {
        let base = self.sanitize_path(dir)?;
        let mut results = vec![];
        self.walk_dir_recursive(&base, &base, 0, max_depth, &mut results)?;
        Ok(results)
    }

    fn walk_dir_recursive(
        &self,
        current: &Path,
        base: &Path,
        depth: usize,
        max_depth: usize,
        results: &mut Vec<PathBuf>,
    ) -> io::Result<()> {
        if depth > max_depth { return Ok(()); }

        // 防止符号链接攻击
        let metadata = fs::symlink_metadata(current)?;
        if metadata.file_type().is_symlink() {
            // 可选：跳过或解析符号链接
            return Ok(());
        }

        if metadata.is_dir() {
            for entry in fs::read_dir(current)? {
                let entry = entry?;
                let path = entry.path();

                // 验证每个路径都在 base 内
                if path.starts_with(base) {
                    results.push(path.clone());
                    if fs::symlink_metadata(&path)?.is_dir() {
                        self.walk_dir_recursive(&path, base, depth + 1, max_depth, results)?;
                    }
                }
            }
        }

        Ok(())
    }
}

fn main() -> io::Result<()> {
    let ops = SafeFileOps::new("/var/www/uploads");

    // 正常使用
    ops.write_file("report.txt", "安全内容")?;
    let content = ops.read_file("report.txt")?;
    println!("读取: {}", content);

    // 路径穿越攻击被阻止
    match ops.read_file("../../etc/passwd") {
        Ok(s) => println!("读到: {}", s),
        Err(e) => println!("[!] 安全拦截: {}", e),  // 阻止
    }

    // 安全临时文件
    let (temp_path, mut file) = ops.create_temp_file()?;
    use std::io::Write;
    file.write_all(b"临时数据")?;
    println!("临时文件: {:?}", temp_path);

    Ok(())
}
```

### 2.6 Rust 调用 C 库（FFI）安全实践

```rust
use std::ffi::{CString, CStr};
use std::os::raw::{c_char, c_int};
use std::ptr;

// Rust 调用 C 库需要使用 FFI（Foreign Function Interface）
// 以下示例模拟调用 C 的系统函数

// 1. 声明外部 C 函数
extern "C" {
    // 注意：调用 C 函数本身是 unsafe 的
    fn puts(s: *const c_char) -> c_int;
    fn strlen(s: *const c_char) -> usize;
}

// 2. 安全封装：将 unsafe 隐藏在安全 API 后面
fn safe_puts(msg: &str) -> Result<(), std::io::Error> {
    // CString 确保字符串以 null 结尾，且不含内部 null 字节
    let c_str = CString::new(msg)
        .map_err(|_| std::io::Error::new(
            std::io::ErrorKind::InvalidInput,
            "字符串包含 null 字节",
        ))?;

    // unsafe 块：调用 C 函数
    unsafe {
        puts(c_str.as_ptr());
    }
    Ok(())
}

fn safe_strlen(s: &str) -> Result<usize, std::io::Error> {
    let c_str = CString::new(s)
        .map_err(|_| std::io::Error::new(
            std::io::ErrorKind::InvalidInput,
            "字符串包含 null 字节",
        ))?;

    let len = unsafe { strlen(c_str.as_ptr()) };
    Ok(len)
}

// 3. 从 C 字符串安全转换为 Rust 字符串
fn c_str_to_rust(ptr: *const c_char) -> Option<String> {
    if ptr.is_null() {
        return None;
    }

    // 安全地从 C 字符串创建 Rust String
    // CStr 检查 UTF-8 有效性
    unsafe {
        let c_str = CStr::from_ptr(ptr);
        match c_str.to_str() {
            Ok(s) => Some(s.to_owned()),
            Err(_) => {
                // 无效 UTF-8，使用 lossy 转换（替换无效字节）
                Some(c_str.to_string_lossy().into_owned())
            }
        }
    }
}

// 4. 回调函数模式
extern "C" fn callback(data: *const c_char) -> c_int {
    if let Some(msg) = c_str_to_rust(data) {
        println!("C 回调: {}", msg);
        0
    } else {
        -1
    }
}

// 5. 声明需要回调的 C 函数（模拟）
extern "C" {
    #[link_name = "register_callback"]
    fn register_callback(cb: extern "C" fn(*const c_char) -> c_int);
}

// 6. Rust 库导出给 C 调用（#[no_mangle] + extern "C"）
#[no_mangle]              // 禁止 Rust 名称修饰
pub extern "C" fn rust_add(a: c_int, b: c_int) -> c_int {
    a + b
}

#[no_mangle]
pub extern "C" fn rust_process_string(input: *const c_char) -> *mut c_char {
    if input.is_null() {
        return ptr::null_mut();
    }

    // 安全地读取 C 字符串
    let rust_str = unsafe {
        match CStr::from_ptr(input).to_str() {
            Ok(s) => s.to_owned(),
            Err(_) => return ptr::null_mut(),
        }
    };

    // 处理（示例：转大写）
    let result = rust_str.to_uppercase();

    // 转回 C 字符串（需要分配内存，C 端负责释放）
    let c_result = CString::new(result).unwrap();
    c_result.into_raw()    // 交出所有权，C 端需调用 free
}

// 7. 内存安全：提供释放函数
#[no_mangle]
pub extern "C" fn rust_free_string(ptr: *mut c_char) {
    if ptr.is_null() { return; }
    unsafe {
        // CString::from_raw 回收内存
        let _ = CString::from_raw(ptr);
    }
}

fn main() {
    // 安全调用 C 函数
    safe_puts("Hello from Rust via C FFI!").unwrap();

    let len = safe_strlen("Hello").unwrap();
    println!("字符串长度: {}", len);

    // 上面定义的函数也可以在 C 中调用
    // 对应 C 头文件:
    // int rust_add(int a, int b);
    // char* rust_process_string(const char* input);
    // void rust_free_string(char* ptr);
}
```

```rust
use uuid::Uuid;
use bcrypt::{hash, verify, DEFAULT_COST};

#[derive(Clone)]
struct AppState {
    jwt_secret: String,
    db_pool: sqlx::SqlitePool,
}

// ===== 用户数据结构 =====
#[derive(Debug, Serialize, Deserialize)]
struct User {
    id: Uuid,
    username: String,
    #[serde(skip_serializing)]  // 不序列化密码哈希
    password_hash: String,
    role: String,
    created_at: chrono::NaiveDateTime,
}

#[derive(Debug, Deserialize)]
struct RegisterRequest {
    username: String,
    password: String,
}

#[derive(Debug, Deserialize)]
struct LoginRequest {
    username: String,
    password: String,
}

#[derive(Debug, Serialize)]
struct LoginResponse {
    token: String,
    user: UserInfo,
}

#[derive(Debug, Serialize)]
struct UserInfo {
    id: Uuid,
    username: String,
    role: String,
}

// ===== JWT Claims =====
#[derive(Debug, Serialize, Deserialize)]
struct Claims {
    sub: String,     // 用户 ID
    username: String,
    role: String,
    exp: usize,      // 过期时间
    iat: usize,      // 签发时间
}

// ===== 输入验证 =====
fn validate_username(username: &str) -> Result<(), String> {
    if username.len() < 3 || username.len() > 32 {
        return Err("用户名长度必须在 3-32 之间".into());
    }
    if !username.chars().all(|c| c.is_alphanumeric() || c == '_' || c == '-') {
        return Err("用户名只能包含字母、数字、下划线和连字符".into());
    }
    Ok(())
}

fn validate_password(password: &str) -> Result<(), String> {
    if password.len() < 8 {
        return Err("密码至少 8 位".into());
    }
    if !password.chars().any(|c| c.is_uppercase()) {
        return Err("密码必须包含大写字母".into());
    }
    if !password.chars().any(|c| c.is_lowercase()) {
        return Err("密码必须包含小写字母".into());
    }
    if !password.chars().any(|c| c.is_numeric()) {
        return Err("密码必须包含数字".into());
    }
    Ok(())
}

// ===== 密码哈希 =====
fn hash_password(password: &str) -> Result<String, String> {
    hash(password, DEFAULT_COST).map_err(|e| format!("密码哈希失败: {}", e))
}

fn verify_password(password: &str, hash: &str) -> Result<bool, String> {
    verify(password, hash).map_err(|e| format!("密码验证失败: {}", e))
}

// ===== JWT 工具 =====
fn generate_token(user: &User, secret: &str) -> Result<String, String> {
    let now = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap()
        .as_secs() as usize;
    
    let claims = Claims {
        sub: user.id.to_string(),
        username: user.username.clone(),
        role: user.role.clone(),
        exp: now + 3600,  // 1 小时过期
        iat: now,
    };
    
    encode(
        &Header::default(),
        &claims,
        &EncodingKey::from_secret(secret.as_bytes()),
    ).map_err(|e| format!("JWT 生成失败: {}", e))
}

fn validate_token(token: &str, secret: &str) -> Result<Claims, String> {
    let token_data = decode::<Claims>(
        token,
        &DecodingKey::from_secret(secret.as_bytes()),
        &Validation::default(),
    ).map_err(|e| format!("JWT 验证失败: {}", e))?;
    
    Ok(token_data.claims)
}

// 用户注册
async fn register(
    State(state): State<Arc<AppState>>,
    Json(req): Json<RegisterRequest>,
) -> Result<impl IntoResponse, (StatusCode, String)> {
    // 输入验证
    validate_username(&req.username)
        .map_err(|e| (StatusCode::BAD_REQUEST, e))?;
    validate_password(&req.password)
        .map_err(|e| (StatusCode::BAD_REQUEST, e))?;
    
    // 检查用户名是否已存在
    let exists: Option<i64> = sqlx::query_scalar(
        "SELECT COUNT(*) FROM users WHERE username = ?"
    )
    .bind(&req.username)
    .fetch_one(&state.db_pool)
    .await
    .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("数据库错误: {}", e)))?;
    
    if exists.unwrap_or(0) > 0 {
        return Err((StatusCode::CONFLICT, "用户名已存在".into()));
    }
    
    // 密码哈希
    let password_hash = hash_password(&req.password)
        .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, e))?;
    
    // 创建用户
    let user_id = Uuid::new_v4();
    sqlx::query(
        "INSERT INTO users (id, username, password_hash, role) VALUES (?, ?, ?, 'user')"
    )
    .bind(user_id)
    .bind(&req.username)
    .bind(&password_hash)
    .execute(&state.db_pool)
    .await
    .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("数据库错误: {}", e)))?;
    
    Ok((StatusCode::CREATED, Json(serde_json::json!({
        "message": "注册成功",
        "user_id": user_id
    }))))
}

// 用户登录
async fn login(
    State(state): State<Arc<AppState>>,
    Json(req): Json<LoginRequest>,
) -> Result<impl IntoResponse, (StatusCode, String)> {
    // 查找用户
    let row: Option<(String, String, String)> = sqlx::query_as(
        "SELECT id, username, password_hash FROM users WHERE username = ?"
    )
    .bind(&req.username)
    .fetch_optional(&state.db_pool)
    .await
    .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, format!("数据库错误: {}", e)))?;
    
    let (id_str, username, password_hash) = match row {
        Some(r) => r,
        None => {
            // 防止用户名枚举，返回相同错误信息
            return Err((StatusCode::UNAUTHORIZED, "用户名或密码错误".into()));
        }
    };
    
    // 验证密码
    let valid = verify_password(&req.password, &password_hash)
        .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, e))?;
    
    if !valid {
        return Err((StatusCode::UNAUTHORIZED, "用户名或密码错误".into()));
    }
    
    let user_id = Uuid::parse_str(&id_str).unwrap();
    
    // 生成 JWT
    let user = User {
        id: user_id,
        username: username.clone(),
        password_hash,
        role: "user".into(),
        created_at: chrono::Utc::now().naive_utc(),
    };
    
    let token = generate_token(&user, &state.jwt_secret)
        .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, e))?;
    
    Ok(Json(LoginResponse {
        token,
        user: UserInfo {
            id: user.id,
            username: user.username,
            role: user.role,
        },
    }))
}

// 受保护的路由
async fn protected_route(
    State(state): State<Arc<AppState>>,
    headers: axum::http::HeaderMap,
) -> Result<impl IntoResponse, (StatusCode, String)> {
    // 从 header 提取 token
    let auth_header = headers.get("Authorization")
        .and_then(|h| h.to_str().ok())
        .ok_or((StatusCode::UNAUTHORIZED, "缺少 Authorization header".to_string()))?;
    
    let token = auth_header.strip_prefix("Bearer ")
        .ok_or((StatusCode::UNAUTHORIZED, "无效的 Authorization 格式".to_string()))?;
    
    // 验证 token
    let claims = validate_token(token, &state.jwt_secret)
        .map_err(|e| (StatusCode::UNAUTHORIZED, e))?;
    
    Ok(Json(serde_json::json!({
        "message": "访问成功",
        "user_id": claims.sub,
        "username": claims.username,
        "role": claims.role
    })))
}

// ===== 主函数 =====
#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    // 初始化数据库
    let db_pool = sqlx::SqlitePool::connect("sqlite:./app.db").await?;
    
    sqlx::query(
        r#"CREATE TABLE IF NOT EXISTS users (
            id TEXT PRIMARY KEY,
            username TEXT UNIQUE NOT NULL,
            password_hash TEXT NOT NULL,
            role TEXT NOT NULL DEFAULT 'user',
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        )"#
    ).execute(&db_pool).await?;
    
    let state = Arc::new(AppState {
        jwt_secret: std::env::var("JWT_SECRET")
            .unwrap_or_else(|_| "your-secret-key-change-in-production".into()),
        db_pool,
    });
    
    // 构建路由
    let app = Router::new()
        .route("/api/register", post(register))
        .route("/api/login", post(login))
        .route("/api/protected", get(protected_route))
        .with_state(state);
    
    // 启动服务
    let listener = tokio::net::TcpListener::bind("127.0.0.1:8080").await?;
    println!("服务器运行在 http://127.0.0.1:8080");
    axum::serve(listener, app).await?;
    
    Ok(())
}
```

```rust
// ===== AES-GCM 对称加密 =====
pub struct AesEncryptor {
    cipher: Aes256Gcm,
}

impl AesEncryptor {
    pub fn new(key: &[u8; 32]) -> Self {
        let key = GenericArray::from_slice(key);
        Self {
            cipher: Aes256Gcm::new(key),
        }
    }
    
    pub fn encrypt(&self, plaintext: &[u8], associated_data: &[u8]) -> Result<Vec<u8>, String> {
        // 生成随机 nonce (12 bytes)
        let mut nonce_bytes = [0u8; 12];
        OsRng.fill_bytes(&mut nonce_bytes);
        let nonce = Nonce::from_slice(&nonce_bytes);
        
        // 加密
        let ciphertext = self.cipher
            .encrypt(nonce, plaintext)
            .map_err(|e| format!("加密失败: {}", e))?;
        
        // 返回 nonce + ciphertext
        let mut result = Vec::with_capacity(nonce_bytes.len() + ciphertext.len());
        result.extend_from_slice(&nonce_bytes);
        result.extend_from_slice(&ciphertext);
        Ok(result)
    }
    
    pub fn decrypt(&self, data: &[u8], associated_data: &[u8]) -> Result<Vec<u8>, String> {
        if data.len() < 12 {
            return Err("数据太短".into());
        }
        
        // 分离 nonce 和密文
        let (nonce_bytes, ciphertext) = data.split_at(12);
        let nonce = Nonce::from_slice(nonce_bytes);
        
        // 解密
        self.cipher
            .decrypt(nonce, ciphertext)
            .map_err(|e| format!("解密失败: {}", e))
    }
}

pub fn sha256_hash(data: &[u8]) -> Vec<u8> {
    let mut hasher = Sha256::new();
    hasher.update(data);
    hasher.finalize().to_vec()
}

pub fn sha256_hex(data: &[u8]) -> String {
    let hash = sha256_hash(data);
    hash.iter().map(|b| format!("{:02x}", b)).collect()
}

// ===== HMAC =====
pub fn hmac_sha256(key: &[u8], data: &[u8]) -> Result<Vec<u8>, String> {
    let mut mac = HmacSha256::new_from_slice(key)
        .map_err(|e| format!("HMAC 初始化失败: {}", e))?;
    mac.update(data);
    Ok(mac.finalize().into_bytes().to_vec())
}

pub fn hmac_verify(key: &[u8], data: &[u8], expected: &[u8]) -> Result<bool, String> {
    let mut mac = HmacSha256::new_from_slice(key)
        .map_err(|e| format!("HMAC 初始化失败: {}", e))?;
    mac.update(data);
    Ok(mac.verify_slice(expected).is_ok())
}

pub fn hash_password(password: &str) -> Result<String, String> {
    let salt = SaltString::generate(&mut OsRng);
    let argon2 = Argon2::default();
    
    let password_hash = argon2
        .hash_password(password.as_bytes(), &salt)
        .map_err(|e| format!("密码哈希失败: {}", e))?;
    
    Ok(password_hash.to_string())
}

pub fn verify_password(password: &str, hash: &str) -> Result<bool, String> {
    let parsed_hash = PasswordHash::new(hash)
        .map_err(|e| format!("解析哈希失败: {}", e))?;
    
    Ok(Argon2::default()
        .verify_password(password.as_bytes(), &parsed_hash)
        .is_ok())
}

pub struct Ed25519KeyPair {
    signing_key: SigningKey,
}

impl Ed25519KeyPair {
    pub fn generate() -> Self {
        let mut csprng = OsRng;
        let signing_key = SigningKey::generate(&mut csprng);
        Self { signing_key }
    }
    
    pub fn from_seed(seed: &[u8; 32]) -> Self {
        Self {
            signing_key: SigningKey::from_bytes(seed),
        }
    }
    
    pub fn public_key_bytes(&self) -> [u8; 32] {
        self.signing_key.verifying_key().to_bytes()
    }
    
    pub fn sign(&self, message: &[u8]) -> Vec<u8> {
        let signature = self.signing_key.sign(message);
        signature.to_bytes().to_vec()
    }
}

pub fn verify_ed25519(public_key_bytes: &[u8; 32], message: &[u8], signature_bytes: &[u8; 64]) -> bool {
    let verifying_key = match VerifyingKey::from_bytes(public_key_bytes.into()) {
        Ok(k) => k,
        Err(_) => return false,
    };
    
    let signature = Signature::from_bytes(signature_bytes.into());
    verifying_key.verify(message, &signature).is_ok()
}
```
