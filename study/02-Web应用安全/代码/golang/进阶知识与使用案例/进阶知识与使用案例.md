### 1.1 goroutine 调度器原理（GMP 模型）

Go 的并发调度基于 **GMP 模型**，这是理解 Go 并发性能的核心：

- **G（Goroutine）**：每个 goroutine 对应一个 G 结构体，存储 goroutine 的栈、状态、调度信息。
- **M（Machine）**：操作系统线程，真正执行代码的载体。M 必须绑定 P 才能运行 G。
- **P（Processor）**：逻辑处理器，持有本地 G 队列（LRQ）。P 的数量由 `GOMAXPROCS` 决定。

```
调度流程：
  1. 新建 goroutine 优先放入当前 P 的本地队列
  2. M 从绑定的 P 的本地队列获取 G 执行
  3. 本地队列为空时，执行 work-stealing（从其他 P 偷取 G）
  4. 所有本地队列空时，从全局队列（GRQ）获取
  5. 仍无 G 时，M 进入自旋或休眠
```

```go
package main

import (
	"fmt"
	"runtime"
	"sync"
)

func main() {
	// 设置 GOMAXPROCS，决定并行使用的 P 数量
	// Go 1.5+ 默认等于 CPU 核数
	runtime.GOMAXPROCS(runtime.NumCPU())

	fmt.Printf("CPU 核数: %d\n", runtime.NumCPU())
	fmt.Printf("GOMAXPROCS: %d\n", runtime.GOMAXPROCS(0))

	var wg sync.WaitGroup

	// 启动多个 goroutine 演示并发
	for i := 0; i < 10; i++ {
		wg.Add(1)
		go func(id int) {
			defer wg.Done()
			// Gosched 让出当前 P，给其他 goroutine 执行机会
			runtime.Gosched()
			fmt.Printf("goroutine %d 运行中\n", id)
		}(i)
	}

	wg.Wait()

	// 查看当前 goroutine 数量
	fmt.Printf("当前 goroutine 数: %d\n", runtime.NumGoroutine())
}
```

**抢占式调度**（Go 1.14+）：基于信号的抢占式调度，解决长时间运行的 goroutine 饿死其他 goroutine 的问题。

```go
package main

import (
	"fmt"
	"time"
)

// 演示 Go 调度器的抢占机制
// Go 1.14 之前，死循环 goroutine 会导致垃圾回收等无法运行
func main() {
	// 一个死循环 goroutine
	go func() {
		for {
			// Go 1.14+ 会被抢占式调度打断
			// 但仍然是尽力而为，不建议写死循环
		}
	}()

	time.Sleep(100 * time.Millisecond)
	fmt.Println("主 goroutine 正常结束")
}
```

### 1.2 channel 进阶（select、缓冲通道、方向控制）

channel 是 Go 中 goroutine 间通信的核心机制。

```go
package main

import (
	"fmt"
	"time"
)

func main() {
	// ===== 无缓冲通道：同步通信 =====
	// 发送和接收必须同时准备好（同步）
	unch := make(chan int)
	go func() {
		unch <- 42 // 发送方阻塞，直到有人接收
	}()
	fmt.Println("接收:", <-unch) // 接收方阻塞，直到有人发送

	// ===== 缓冲通道：异步通信 =====
	// 缓冲区满之前发送不阻塞，缓冲区空之前接收不阻塞
	bufch := make(chan string, 3)
	bufch <- "msg1" // 不阻塞，缓冲区未满
	bufch <- "msg2"
	bufch <- "msg3"
	// bufch <- "msg4" // 此时缓冲区已满，会阻塞

	fmt.Println(len(bufch), cap(bufch)) // 3 3

	// ===== select 多路复用 =====
	// select 让 goroutine 同时等待多个 channel 操作
	ch1 := make(chan string)
	ch2 := make(chan string)

	go func() {
		time.Sleep(100 * time.Millisecond)
		ch1 <- "来自 ch1"
	}()

	go func() {
		time.Sleep(200 * time.Millisecond)
		ch2 <- "来自 ch2"
	}()

	// select 会等待第一个准备好的 case
	select {
	case msg := <-ch1:
		fmt.Println("收到:", msg)
	case msg := <-ch2:
		fmt.Println("收到:", msg)
	}

	// ===== select + default 实现非阻塞操作 =====
	select {
	case msg := <-ch1:
		fmt.Println("收到:", msg)
	default:
		fmt.Println("无消息，非阻塞返回") // 如果 ch1 无数据，走 default
	}

	// ===== select 超时控制 =====
	timeoutCh := make(chan string, 1)
	go func() {
		time.Sleep(2 * time.Second)
		timeoutCh <- "慢操作完成"
	}()

	select {
	case msg := <-timeoutCh:
		fmt.Println(msg)
	case <-time.After(500 * time.Millisecond):
		fmt.Println("超时！操作耗时过长")
	}

	// ===== channel 方向控制 =====
	// 只发送通道：chan<- int
	// 只接收通道：<-chan int
	producer := make(chan int)
	consumer := make(chan int)

	// 只发送的参数
	go func(ch chan<- int) {
		for i := 0; i < 5; i++ {
			ch <- i
		}
		close(ch)
	}(producer)

	// 只接收的参数
	go func(ch <-chan int, out chan<- int) {
		for v := range ch {
			out <- v * 2
		}
		close(out)
	}(producer, consumer)

	for v := range consumer {
		fmt.Printf("处理结果: %d ", v)
	}
	fmt.Println()

	// ===== 用 channel 实现工作池模式 =====
	results := workerPool(100, 5)
	for r := range results {
		fmt.Printf("工作池结果: %d\n", r)
	}
}

// workerPool 工作池模式：N 个 worker 并行处理 M 个任务
func workerPool(tasks, workers int) <-chan int {
	jobs := make(chan int, tasks)
	results := make(chan int, tasks)

	// 启动 worker
	for w := 0; w < workers; w++ {
		go worker(w, jobs, results)
	}

	// 分发任务
	for j := 0; j < tasks; j++ {
		jobs <- j
	}
	close(jobs)

	// 等待所有 worker 完成后关闭 results
	go func() {
		// 这里需要 WaitGroup 来确保所有 worker 完成
		// 简化演示，实际需要更严谨的控制
		time.Sleep(500 * time.Millisecond)
		close(results)
	}()

	return results
}

func worker(id int, jobs <-chan int, results chan<- int) {
	for j := range jobs {
		// 模拟工作
		time.Sleep(10 * time.Millisecond)
		results <- j * j
	}
}
```

### 1.3 context 包使用与超时控制

`context` 是 Go 中传递截止时间、取消信号和请求作用域值的标准方式。

```go
package main

import (
	"context"
	"fmt"
	"time"
)

func main() {
	// ===== context.WithCancel：手动取消 =====
	ctx, cancel := context.WithCancel(context.Background())
	go workerWithCancel(ctx)

	time.Sleep(2 * time.Second)
	cancel() // 发送取消信号
	time.Sleep(500 * time.Millisecond)
	fmt.Println("--- cancel 演示结束 ---")

	// ===== context.WithTimeout：超时取消 =====
	ctx2, cancel2 := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel2() // 始终调用 cancel 释放资源
	go workerWithTimeout(ctx2)

	time.Sleep(4 * time.Second)
	fmt.Println("--- timeout 演示结束 ---")

	// ===== context.WithDeadline：截止时间 =====
	deadline := time.Now().Add(2 * time.Second)
	ctx3, cancel3 := context.WithDeadline(context.Background(), deadline)
	defer cancel3()
	go workerWithDeadline(ctx3)

	time.Sleep(3 * time.Second)
	fmt.Println("--- deadline 演示结束 ---")

	// ===== context.WithValue：传递请求作用域值 =====
	// 注意：WithValue 不应传递函数参数，而应传递请求作用域的数据
	// 如 trace ID、用户身份等
	ctx4 := context.WithValue(context.Background(), "userID", 12345)
	ctx4 = context.WithValue(ctx4, "traceID", "abc-xyz-123")
	processRequest(ctx4)
}

func workerWithCancel(ctx context.Context) {
	for {
		select {
		case <-ctx.Done():
			fmt.Println("worker 收到取消信号:", ctx.Err())
			return
		default:
			fmt.Println("worker 工作中...")
			time.Sleep(500 * time.Millisecond)
		}
	}
}

func workerWithTimeout(ctx context.Context) {
	select {
	case <-time.After(5 * time.Second):
		fmt.Println("工作完成")
	case <-ctx.Done():
		fmt.Println("工作因超时被取消:", ctx.Err())
	}
}

func workerWithDeadline(ctx context.Context) {
	select {
	case <-time.After(5 * time.Second):
		fmt.Println("工作完成")
	case <-ctx.Done():
		fmt.Println("工作因截止时间到达被取消:", ctx.Err())
	}
}

func processRequest(ctx context.Context) {
	// 从 context 中取出传递的值
	// 类型断言是必须的
	userID, ok := ctx.Value("userID").(int)
	if ok {
		fmt.Printf("处理用户 %d 的请求\n", userID)
	}
	traceID, ok := ctx.Value("traceID").(string)
	if ok {
		fmt.Printf("链路追踪 ID: %s\n", traceID)
	}
}
```

**HTTP 请求中使用 context 超时控制：**

```go
package main

import (
	"context"
	"fmt"
	"net/http"
	"time"
)

func main() {
	// 创建带超时的 HTTP 请求
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	req, err := http.NewRequestWithContext(ctx, "GET", "https://httpbin.org/delay/3", nil)
	if err != nil {
		fmt.Println("创建请求失败:", err)
		return
	}

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		fmt.Println("请求失败:", err)
		return
	}
	defer resp.Body.Close()

	fmt.Println("响应状态:", resp.StatusCode)
}
```

### 1.4 interface 底层原理（eface vs iface）

Go 的 interface 底层由两个指针组成，分为两种结构：

```go
package main

import (
	"fmt"
	"unsafe"
)

// eface: 空接口 interface{} 的底层结构
// type eface struct {
//     _type *_type       // 类型信息
//     data  unsafe.Pointer // 指向实际数据
// }

// iface: 带方法的接口的底层结构
// type iface struct {
//     tab  *itab          // 类型信息 + 方法表
//     data unsafe.Pointer // 指向实际数据
// }

// itab 结构：
// type itab struct {
//     inter *interfacetype // 接口类型信息
//     _type *_type         // 实际类型信息
//     hash  uint32         // 类型 hash
//     _     [4]byte
//     fun   [1]uintptr     // 方法表（实际是变长数组）
// }

// ===== 空接口示例 =====
func describeEmpty(i interface{}) {
	// i 是 eface
	fmt.Printf("空接口类型: %T, 值: %v\n", i, i)
}

// ===== 带方法的接口示例 =====
type Animal interface {
	Speak() string
	Name() string
}

type Dog struct {
	name string
}

func (d Dog) Speak() string {
	return "汪汪汪"
}

func (d Dog) Name() string {
	return d.name
}

// 接口类型断言与 type switch
func identify(a Animal) {
	switch v := a.(type) {
	case Dog:
		fmt.Printf("这是一只狗，名字: %s\n", v.name)
	default:
		fmt.Println("未知动物")
	}
}

// nil 接口 vs nil 值的陷阱
type errorer interface {
	Error() string
}

type MyError struct{}

func (m *MyError) Error() string {
	return "自定义错误"
}

func returnError() errorer {
	// 注意：这里返回了一个非 nil 的接口
	// 因为接口 = (类型指针, 值指针)
	// 类型指针不为 nil，所以接口不为 nil
	var err *MyError = nil
	return err
}

func main() {
	// 空接口 eface 示例
	describeEmpty(42)
	describeEmpty("hello")
	describeEmpty(3.14)

	// 带方法接口 iface 示例
	d := Dog{name: "旺财"}
	var a Animal = d // 此时创建 iface，包含 itab 和 data 指针
	fmt.Println(a.Speak())
	fmt.Println(a.Name())
	identify(d)

	// 查看接口大小（两个指针的大小）
	fmt.Printf("空接口大小: %d 字节\n", unsafe.Sizeof(a))
	fmt.Printf("Animal 接口大小: %d 字节\n", unsafe.Sizeof(d))

	// nil 接口陷阱
	var e errorer
	fmt.Println("nil 接口:", e == nil) // true

	e = returnError()
	// 接口类型指针非 nil 但值为 nil，接口 != nil
	fmt.Println("陷阱: 返回的接口 == nil?", e == nil) // false！这是一个常见的 bug
	// 正确做法：函数签名返回具体的 error 类型，或显式返回 nil

	// 接口类型断言
	var i interface{} = "hello world"
	s, ok := i.(string)
	fmt.Println(ok, s)

	n, ok := i.(int)
	fmt.Println(ok, n) // false 0
}
```

### 1.5 反射（reflect 包）

反射允许在运行时检查类型信息和操作值。

```go
package main

import (
	"fmt"
	"reflect"
)

type User struct {
	ID    int    `json:"id" validate:"required"`
	Name  string `json:"name" validate:"required,min=2,max=50"`
	Email string `json:"email" validate:"required,email"`
	Age   int    `json:"age" validate:"min=0,max=150"`
}

func main() {
	user := User{ID: 1, Name: "Alice", Email: "alice@example.com", Age: 25}

	// ===== 基本反射操作 =====
	// reflect.TypeOf 获取类型信息
	// reflect.ValueOf 获取值信息
	t := reflect.TypeOf(user)
	v := reflect.ValueOf(user)

	fmt.Println("类型名:", t.Name())
	fmt.Println("类型种类:", t.Kind()) // struct

	// ===== 遍历结构体字段 =====
	for i := 0; i < t.NumField(); i++ {
		field := t.Field(i)
		value := v.Field(i)
		tag := field.Tag.Get("json")
		validateTag := field.Tag.Get("validate")

		fmt.Printf("字段: %s, 类型: %s, 值: %v, json标签: %s, validate: %s\n",
			field.Name, field.Type, value.Interface(), tag, validateTag)
	}

	// ===== 通过反射修改值 =====
	// 必须传入指针才能修改
	userPtr := &User{ID: 1, Name: "Bob", Email: "bob@test.com", Age: 30}
	modifyStruct(userPtr)
	fmt.Println("修改后:", *userPtr)

	// ===== 通过反射调用方法 =====
	dog := Dog{name: "旺财"}
	callMethod(dog)

	// ===== 简易验证器：利用结构体标签 =====
	validUser := User{ID: 1, Name: "Al", Email: "bad", Age: 200}
	invalidUser := User{ID: 0, Name: "", Email: "", Age: -1}
	fmt.Println("验证结果:", validateStruct(validUser))
	fmt.Println("验证结果:", validateStruct(invalidUser))
}

// modifyStruct 通过反射修改结构体字段
func modifyStruct(ptr interface{}) {
	v := reflect.ValueOf(ptr)
	if v.Kind() != reflect.Ptr {
		fmt.Println("必须传入指针")
		return
	}
	// Elem() 获取指针指向的值
	v = v.Elem()

	// 修改 Name 字段
	nameField := v.FieldByName("Name")
	if nameField.IsValid() && nameField.CanSet() {
		nameField.SetString("ModifiedName")
	}

	// 修改 Age 字段
	ageField := v.FieldByName("Age")
	if ageField.IsValid() && ageField.CanSet() {
		ageField.SetInt(99)
	}
}

type Dog struct {
	name string
}

func (d Dog) Speak() string {
	return fmt.Sprintf("%s: 汪汪汪", d.name)
}

// callMethod 通过反射调用方法
func callMethod(obj interface{}) {
	v := reflect.ValueOf(obj)
	method := v.MethodByName("Speak")
	if method.IsValid() {
		// Call 的参数是 []reflect.Value，这里无参数
		results := method.Call(nil)
		if len(results) > 0 {
			fmt.Println("方法返回:", results[0].String())
		}
	}
}

// validateStruct 简易结构体验证器（利用反射读取 validate 标签）
func validateStruct(obj interface{}) []string {
	var errors []string
	t := reflect.TypeOf(obj)
	v := reflect.ValueOf(obj)

	for i := 0; i < t.NumField(); i++ {
		field := t.Field(i)
		val := v.Field(i)
		tag := field.Tag.Get("validate")
		if tag == "" {
			continue
		}

		fieldName := field.Name

		switch val.Kind() {
		case reflect.String:
			s := val.String()
			if contains(tag, "required") && s == "" {
				errors = append(errors, fieldName+"不能为空")
			}
		case reflect.Int:
			n := val.Int()
			if contains(tag, "required") && n == 0 {
				errors = append(errors, fieldName+"不能为0")
			}
			if n < 0 || n > 150 {
				errors = append(errors, fieldName+"超出范围")
			}
		}
	}

	if len(errors) == 0 {
		return nil
	}
	return errors
}

func contains(s, substr string) bool {
	for i := 0; i <= len(s)-len(substr); i++ {
		if s[i:i+len(substr)] == substr {
			return true
		}
	}
	return false
}
```

### 1.6 内存管理与逃逸分析

Go 使用**堆**和**栈**两种内存分配方式。编译器通过**逃逸分析**决定变量分配在栈还是堆：

- **栈分配**：速度快，函数返回时自动回收，无需 GC 介入。
- **堆分配**：需要 GC 回收，有额外开销。

```go
package main

import "fmt"

// 栈分配：变量不逃逸
func stackAllocation() int {
	x := 42 // x 在栈上分配，函数返回后自动回收
	return x
}

// 堆分配：变量逃逸（返回局部变量的指针）
func heapAllocation() *int {
	x := 42 // x 逃逸到堆，因为返回了指针
	return &x
}

// interface{} 导致逃逸
func escapeByInterface() {
	x := 42
	// 发送空接口会逃逸（因为编译器无法确定大小）
	fmt.Println(x) // fmt.Println 的参数是 interface{}，导致 x 逃逸
}

// 闭包导致逃逸
func escapeByClosure() func() int {
	x := 42
	// 闭包捕获 x，导致 x 逃逸到堆
	return func() int {
		return x
	}
}

// 切片容量不确定导致逃逸
func escapeBySlice() {
	size := 10 // 编译器可能无法确定大小
	s := make([]int, size)
	_ = s
}

// 不逃逸的优化：返回值而非指针
func noEscape() [1000]int {
	var arr [1000]int
	return arr // 返回值拷贝，不逃逸
}

func main() {
	fmt.Println(stackAllocation())
	fmt.Println(*heapAllocation())
	escapeByInterface()
	_ = escapeByClosure()
	escapeBySlice()
```

```go
package main

import (
	"crypto/tls"
	"fmt"
	"log"
	"net/http"
	"time"

// 安全中间件链
type Middleware func(http.Handler) http.Handler

// ChainMiddleware 将多个中间件链接起来
func ChainMiddleware(middlewares ...Middleware) Middleware {
	return func(final http.Handler) http.Handler {
		for i := len(middlewares) - 1; i >= 0; i-- {
			final = middlewares[i](final)
		}
		return final
	}
}

// SecurityHeaders 添加安全响应头
func SecurityHeaders(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("X-Content-Type-Options", "nosniff")
		w.Header().Set("X-Frame-Options", "DENY")
		w.Header().Set("X-XSS-Protection", "1; mode=block")
		w.Header().Set("Strict-Transport-Security", "max-age=31536000; includeSubDomains")
		w.Header().Set("Content-Security-Policy", "default-src 'self'")
		w.Header().Set("Referrer-Policy", "strict-origin-when-cross-origin")
		next.ServeHTTP(w, r)
	})
}

// RateLimiter 简易限流中间件
type visitor struct {
	lastAccess time.Time
	count      int
}

var visitors = make(map[string]*visitor)

func RateLimiter(limit int, window time.Duration) Middleware {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			ip := r.RemoteAddr
			v, exists := visitors[ip]
			now := time.Now()

			if !exists || now.Sub(v.lastAccess) > window {
				visitors[ip] = &visitor{lastAccess: now, count: 1}
			} else {
				v.count++
				v.lastAccess = now
				if v.count > limit {
					http.Error(w, "请求过于频繁", http.StatusTooManyRequests)
					return
				}
			}
			next.ServeHTTP(w, r)
		})
	}
}

// CORS 中间件
func CORS(allowedOrigins []string) Middleware {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			origin := r.Header.Get("Origin")
			allowed := false
			for _, o := range allowedOrigins {
				if origin == o {
					allowed = true
					break
				}
			}
			if allowed {
				w.Header().Set("Access-Control-Allow-Origin", origin)
				w.Header().Set("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS")
				w.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization")
				w.Header().Set("Access-Control-Max-Age", "86400")
			}
			if r.Method == "OPTIONS" {
				w.WriteHeader(http.StatusOK)
				return
			}
			next.ServeHTTP(w, r)
		})
	}
}

// RequestLogger 请求日志中间件
func RequestLogger(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		start := time.Now()
		next.ServeHTTP(w, r)
		log.Printf("%s %s %s %v", r.Method, r.URL.Path, r.RemoteAddr, time.Since(start))
	})
}

// AuthMiddleware 简易认证中间件
func AuthMiddleware(token string) Middleware {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			auth := r.Header.Get("Authorization")
			if auth != "Bearer "+token {
				http.Error(w, "未授权", http.StatusUnauthorized)
				return
			}
			next.ServeHTTP(w, r)
		})
	}
}

func main() {
	mux := http.NewServeMux()
	mux.HandleFunc("/api/public", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprintf(w, `{"message": "公开 API"}`)
	})
	mux.HandleFunc("/api/protected", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprintf(w, `{"message": "受保护的 API", "user": "admin"}`)
	})

	// 构建中间件链
	handler := ChainMiddleware(
		RequestLogger,
		SecurityHeaders,
		CORS([]string{"https://example.com", "http://localhost:3000"}),
		RateLimiter(100, time.Minute),
	)(mux)

	// 受保护路由使用认证中间件
	protectedHandler := AuthMiddleware("secret-token-123")(handler)
	mux.Handle("/api/protected", protectedHandler)

	// TLS 配置
	tlsConfig := &tls.Config{
		MinVersion:               tls.VersionTLS12, // 强制 TLS 1.2+
		MaxVersion:               tls.VersionTLS13,
		PreferServerCipherSuites: true,
		CurvePreferences: []tls.CurveID{
			tls.X25519, tls.CurveP256,
		},
	}

	server := &http.Server{
		Addr:         ":8443",
		Handler:      handler,
		TLSConfig:    tlsConfig,
		ReadTimeout:  10 * time.Second,
		WriteTimeout: 10 * time.Second,
		IdleTimeout:  120 * time.Second,
	}

	fmt.Println("安全服务器启动在 https://localhost:8443")
	log.Fatal(server.ListenAndServeTLS("server.crt", "server.key"))
}
```

### 2.2 Go 实现反向代理

```go
package main

import (
	"fmt"
	"log"
	"net/http"
	"net/http/httputil"
	"net/url"
	"strings"
	"time"
)

// ReverseProxy 反向代理结构体
type ReverseProxy struct {
	target  *url.URL
	proxy   *httputil.ReverseProxy
	routes  map[string]string // 路径前缀 -> 后端地址
}

// NewReverseProxy 创建反向代理
func NewReverseProxy() *ReverseProxy {
	return &ReverseProxy{
		routes: make(map[string]string),
	}
}

// AddRoute 添加路由规则
func (rp *ReverseProxy) AddRoute(prefix, target string) error {
	targetURL, err := url.Parse(target)
	if err != nil {
		return fmt.Errorf("解析目标 URL 失败: %w", err)
	}

	rp.routes[prefix] = target
	rp.target = targetURL

	rp.proxy = httputil.NewSingleHostReverseProxy(targetURL)

	// 自定义 Director：修改请求
	originalDirector := rp.proxy.Director
	rp.proxy.Director = func(req *http.Request) {
		originalDirector(req)
		// 添加自定义请求头
		req.Header.Set("X-Proxy-By", "Go-ReverseProxy")
		req.Header.Set("X-Forwarded-Host", req.Host)
		req.Header.Set("X-Real-IP", req.RemoteAddr)
	}

	// 自定义 ModifyResponse：修改响应
	rp.proxy.ModifyResponse = func(resp *http.Response) error {
		resp.Header.Set("X-Proxied", "true")
		return nil
	}

	// 自定义错误处理
	rp.proxy.ErrorHandler = func(w http.ResponseWriter, r *http.Request, err error) {
		log.Printf("代理错误: %s %s -> %v", r.Method, r.URL.Path, err)
		http.Error(w, "后端服务不可用", http.StatusBadGateway)
	}

	return nil
}

// ServeHTTP 实现多路由反向代理
func (rp *ReverseProxy) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	for prefix, target := range rp.routes {
		if strings.HasPrefix(r.URL.Path, prefix) {
			targetURL, _ := url.Parse(target)
			proxy := httputil.NewSingleHostReverseProxy(targetURL)

			// 自定义 Director
			proxy.Director = func(req *http.Request) {
				req.URL.Scheme = targetURL.Scheme
				req.URL.Host = targetURL.Host
				req.URL.Path = strings.TrimPrefix(req.URL.Path, prefix)
				req.Host = targetURL.Host
			}

			log.Printf("代理: %s %s -> %s%s", r.Method, r.URL.Path, target, r.URL.Path)
			proxy.ServeHTTP(w, r)
			return
		}
	}

	http.NotFound(w, r)
}

// LoadBalancer 简易负载均衡器
type LoadBalancer struct {
	backends []*url.URL
	current  int
}

func NewLoadBalancer(backends []string) (*LoadBalancer, error) {
	lb := &LoadBalancer{}
	for _, b := range backends {
		u, err := url.Parse(b)
		if err != nil {
			return nil, err
		}
		lb.backends = append(lb.backends, u)
	}
	return lb, nil
}

// GetNext 轮询获取下一个后端
func (lb *LoadBalancer) GetNext() *url.URL {
	if len(lb.backends) == 0 {
		return nil
	}
	backend := lb.backends[lb.current%len(lb.backends)]
	lb.current++
	return backend
}

// ServeHTTP 负载均衡代理
func (lb *LoadBalancer) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	backend := lb.GetNext()
	if backend == nil {
		http.Error(w, "无可用后端", http.StatusServiceUnavailable)
		return
	}

	proxy := httputil.NewSingleHostReverseProxy(backend)
	proxy.Director = func(req *http.Request) {
		req.URL.Scheme = backend.Scheme
		req.URL.Host = backend.Host
		req.Host = backend.Host
	}

	// 健康检查 + 重试逻辑
	proxy.ErrorHandler = func(w http.ResponseWriter, r *http.Request, err error) {
		log.Printf("后端 %s 不可用: %v", backend.Host, err)
		// 可以在这里实现重试到下一个后端的逻辑
		http.Error(w, "服务不可用", http.StatusBadGateway)
	}

	proxy.ServeHTTP(w, r)
}

func main() {
	// ===== 单目标反向代理 =====
	rp := NewReverseProxy()
	err := rp.AddRoute("/api/", "http://localhost:8081")
	if err != nil {
		log.Fatal(err)
	}

	// ===== 多路由反向代理 =====
	mux := http.NewServeMux()
	mux.Handle("/api/", rp)

	// ===== 负载均衡器 =====
	lb, err := NewLoadBalancer([]string{
		"http://localhost:8081",
		"http://localhost:8082",
		"http://localhost:8083",
	})
	if err != nil {
		log.Fatal(err)
	}
	mux.Handle("/lb/", lb)

	// 健康检查端点
	mux.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusOK)
		fmt.Fprintln(w, `{"status": "ok"}`)
	})

	server := &http.Server{
		Addr:         ":8080",
		Handler:      mux,
		ReadTimeout:  10 * time.Second,
		WriteTimeout: 10 * time.Second,
	}

	fmt.Println("反向代理启动在 http://localhost:8080")
```

```go
package main

import (

// ===== 不安全的写法（有 SQL 注入风险） =====
// 永远不要这样写！
func unsafeQuery(db *sql.DB, username string) {
	// 危险：直接拼接用户输入
	query := fmt.Sprintf("SELECT id, username, email FROM users WHERE username = '%s'", username)

	}

	// ...

// ===== 安全的写法：使用预处理语句 =====
func safeQuery(db *sql.DB, username string) (sql.Result, error) {
	// 使用占位符 ?（MySQL/SQLite）或 $1（PostgreSQL）
	// database/sql 会自动进行参数化查询，防止注入
	query := "SELECT id, username, email FROM users WHERE username = $1"
	rows, err := db.Query(query, username)
	if err != nil {
		return nil, fmt.Errorf("查询失败: %w", err)
	}
	defer rows.Close()

	for rows.Next() {
		var id int
		var uname, email string
		if err := rows.Scan(&id, &uname, &email); err != nil {
			return nil, err
		}
		fmt.Printf("ID: %d, Username: %s, Email: %s\n", id, uname, email)
	}
	return nil, nil
}

// ===== 使用 Prepared Statement =====
func preparedQuery(db *sql.DB, username string) error {
	// 预编译 SQL，只解析一次
	stmt, err := db.Prepare("SELECT id, username FROM users WHERE username = $1")
	if err != nil {
		return err
	}
	defer stmt.Close()

	var id int
	var uname string
	err = stmt.QueryRow(username).Scan(&id, &uname)
	return err
}

// ===== 批量安全插入 =====
func batchInsert(db *sql.DB, users []struct {
	Username string
	Email   string
}) error {
	// 使用事务 + 预处理语句进行批量安全插入
	tx, err := db.Begin()
	if err != nil {
		return err
	}
	defer tx.Rollback() // 出错时回滚

	stmt, err := tx.Prepare("INSERT INTO users(username, email) VALUES($1, $2)")
	if err != nil {
		return err
	}
	defer stmt.Close()

	for _, u := range users {
		_, err = stmt.Exec(u.Username, u.Email) // 参数化，安全
		if err != nil {
			return err
		}
	}

	return tx.Commit()
}

// ===== 动态查询的安全构建 =====
// 当需要根据条件动态构建 SQL 时的安全做法
func dynamicQuery(db *sql.DB, filters map[string]interface{}) ([]map[string]interface{}, error) {
	conditions := []string{}
	args := []interface{}{}
	argIndex := 1

	// 安全地动态构建 WHERE 子句
	if username, ok := filters["username"].(string); ok && username != "" {
		conditions = append(conditions, fmt.Sprintf("username = $%d", argIndex))
		args = append(args, username)
		argIndex++
	}

	if email, ok := filters["email"].(string); ok && email != "" {
		conditions = append(conditions, fmt.Sprintf("email = $%d", argIndex))
		args = append(args, email)
		argIndex++
	}

	if minAge, ok := filters["minAge"].(int); ok {
		conditions = append(conditions, fmt.Sprintf("age >= $%d", argIndex))
		args = append(args, minAge)
		argIndex++
	}

	query := "SELECT id, username, email FROM users"
	if len(conditions) > 0 {
		query += " WHERE " + strings.Join(conditions, " AND ")
	}

	// 所有用户输入都通过占位符传入，安全
	rows, err := db.Query(query, args...)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	// 安全的 LIKE 查询也需要参数化
	// search := r.URL.Query().Get("search")
	// db.Query("SELECT * FROM products WHERE name LIKE $1", "%"+search+"%")

	return nil, nil
}

// ===== SQL 注入检测函数 =====
// 用于检测输入中是否包含可能的 SQL 注入载荷
func detectSQLi(input string) bool {
	input = strings.ToLower(input)
	sqliPatterns := []string{
		"' or '1'='1",
		"' or 1=1",
		"' or '1'='1' --",
		"' or '1'='1' #",
		"union select",
		"'; drop table",
		"' or ''='",
		"1=1",
		"1=2",
		"exec(",
		"exec (",
		"xp_cmdshell",
		"sp_executesql",
		"--",
		"/*",
	}

	for _, pattern := range sqliPatterns {
		if strings.Contains(input, pattern) {
			return true
		}
	}

	// 检查单引号数量
	quoteCount := strings.Count(input, "'")
	if quoteCount >= 2 {
		return true
	}

	return false
}

	}
```

```go
	scanners []Scanner

// NewScannerEngine 创建扫描引擎

	if config == nil {
		config = DefaultScannerConfig()
	}

	// 跳过 TLS 验证（扫描时需要）
	transport := &http.Transport{
		TLSClientConfig: &tls.Config{InsecureSkipVerify: true},
	}

	client := &http.Client{
		Timeout:   config.Timeout,
		Transport: transport,
		CheckRedirect: func(req *http.Request, via []*http.Request) error {
			if !config.FollowRedirect {
				return http.ErrUseLastResponse
			}
			return nil
		},
	}

	engine := &ScannerEngine{
		config: config,
		client: client,
	}

	// 注册内置扫描器
	engine.scanners = []Scanner{
		&SQLiScanner{client: client},
		&XSSScanner{client: client},
		&HeaderScanner{client: client},
		&DirectoryScanner{client: client},
	}

	return engine
}

// Run 执行扫描
func (e *ScannerEngine) Run(target string) []Vulnerability {
	// 验证 URL
	u, err := url.Parse(target)
	if err != nil || u.Scheme == "" {
		fmt.Println("[!] 无效的目标 URL")
		return nil
	}

	fmt.Printf("[*] 开始扫描: %s\n", target)

	var results []Vulnerability
	var mu sync.Mutex
	var wg sync.WaitGroup

	// 并发运行所有扫描器
	for _, scanner := range e.scanners {
		wg.Add(1)
		go func(s Scanner) {
			defer wg.Done()
			fmt.Printf("[*] 运行扫描器: %s\n", s.Name())
			vulns := s.Scan(target)
			mu.Lock()
			results = append(results, vulns...)
			mu.Unlock()
		}(scanner)
	}

	wg.Wait()

	// 打印结果
	fmt.Printf("\n[+] 扫描完成，发现 %d 个漏洞\n", len(results))
	for _, v := range results {
		fmt.Printf("  [%s] %s - %s\n", v.Severity, v.Name, v.URL)
	}

	return results
}

				Timestamp:   time.Now(),

		}
```

```go
package main

import (
	"context"
	"fmt"
	"net"
	"sync"
	"time"

// PortResult 端口扫描结果
type PortResult struct {
	Port    int
	State   string // "open", "closed", "filtered"
	Service string
	Banner  string
}

// PortScanner 端口扫描器
type PortScanner struct {
	target     string
	timeout    time.Duration
	concurrent int
	results    []PortResult
	mu         sync.Mutex
}

// NewPortScanner 创建扫描器
func NewPortScanner(target string, timeout time.Duration, concurrent int) *PortScanner {
	return &PortScanner{
		target:     target,
		timeout:    timeout,
		concurrent: concurrent,
	}
}

// ScanPort 扫描单个端口
func (ps *PortScanner) ScanPort(port int) PortResult {
	address := fmt.Sprintf("%s:%d", ps.target, port)
	conn, err := net.DialTimeout("tcp", address, ps.timeout)
	if err != nil {
		return PortResult{Port: port, State: "closed"}
	}
	defer conn.Close()

	result := PortResult{Port: port, State: "open"}

	// 尝试获取 Banner
	conn.SetReadDeadline(time.Now().Add(2 * time.Second))
	buffer := make([]byte, 1024)
	n, _ := conn.Read(buffer)
	if n > 0 {
		result.Banner = string(buffer[:n])
	} else {
		// 主动发送探测
		conn.Write([]byte("GET / HTTP/1.0\r\nHost: " + ps.target + "\r\n\r\n"))
		n, _ = conn.Read(buffer)
		if n > 0 {
			result.Banner = string(buffer[:n])
		}
	}

	// 识别服务
	result.Service = identifyService(port, result.Banner)

	return result
}

// identifyService 根据端口号和 Banner 识别服务
func identifyService(port int, banner string) string {
	// 常见端口映射
	portServices := map[int]string{
		21:   "FTP",
		22:   "SSH",
		23:   "Telnet",
		25:   "SMTP",
		53:   "DNS",
		80:   "HTTP",
		110:  "POP3",
		143:  "IMAP",
		443:  "HTTPS",
		3306: "MySQL",
		3389: "RDP",
		5432: "PostgreSQL",
		6379: "Redis",
		8080: "HTTP-Proxy",
		8443: "HTTPS-Alt",
		27017: "MongoDB",
		9200: "Elasticsearch",
	}

	if service, ok := portServices[port]; ok {
		return service
	}

	// 根据 Banner 识别
	bannerLower := strings.ToLower(banner)
	if strings.Contains(bannerLower, "ssh") {
		return "SSH"
	}
	if strings.Contains(bannerLower, "http") {
		return "HTTP"
	}
	if strings.Contains(bannerLower, "ftp") {
		return "FTP"
	}
	if strings.Contains(bannerLower, "smtp") {
		return "SMTP"
	}

	return "unknown"
}

// Scan 使用 worker pool 扫描端口范围
func (ps *PortScanner) Scan(startPort, endPort int) []PortResult {
	// 使用 channel 控制并发数
	ports := make(chan int, ps.concurrent)
	var wg sync.WaitGroup

	// 启动 worker
	for i := 0; i < ps.concurrent; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for port := range ports {
				result := ps.ScanPort(port)
				if result.State == "open" {
					ps.mu.Lock()
					ps.results = append(ps.results, result)
					ps.mu.Unlock()
				}
			}
		}()
	}

	// 分发端口
	for port := startPort; port <= endPort; port++ {
		ports <- port
	}
	close(ports)

	wg.Wait()
	return ps.results
}

// ScanWithContext 带超时控制的扫描
func (ps *PortScanner) ScanWithContext(ctx context.Context, ports []int) []PortResult {
	portCh := make(chan int, ps.concurrent)
	resultCh := make(chan PortResult, len(ports))
	var wg sync.WaitGroup

	// 启动 worker
	for i := 0; i < ps.concurrent; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for port := range portCh {
				select {
				case <-ctx.Done():
					return
				default:
					result := ps.ScanPort(port)
					if result.State == "open" {
						resultCh <- result
					}
				}
			}
		}()
	}

	// 分发
	go func() {
		for _, port := range ports {
			select {
			case <-ctx.Done():
				break
			case portCh <- port:
			}
		}
		close(portCh)
	}()

	// 等待完成
	go func() {
		wg.Wait()
		close(resultCh)
	}()

	var results []PortResult
	for r := range resultCh {
		results = append(results, r)
	}
	return results
}
```

```go
	}
}

// GenerateSelfSigned 生成自签名证书
func (cg *CertificateGenerator) GenerateSelfSigned(commonName string, dnsNames []string, ips []string) (certPEM, keyPEM []byte, err error) {
	// 使用 ECDSA P-256 生成密钥对（比 RSA 更高效）
	priv, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if err != nil {
		return nil, nil, fmt.Errorf("生成密钥对失败: %w", err)
	}

	// 证书序列号
	serialNumberLimit := new(big.Int).Lsh(big.NewInt(1), 128)
	serialNumber, err := rand.Int(rand.Reader, serialNumberLimit)
	if err != nil {
		return nil, nil, fmt.Errorf("生成序列号失败: %w", err)
	}

	// 证书模板
	template := x509.Certificate{
		SerialNumber: serialNumber,
		Subject: pkix.Name{
			Organization:  []string{cg.org},
			CommonName:    commonName,
			Country:       []string{"CN"},
			Province:      []string{"Beijing"},
			Locality:      []string{"Beijing"},
		},
		NotBefore: time.Now().Add(-1 * time.Hour),
		NotAfter:  time.Now().Add(cg.validFor),

		KeyUsage:              x509.KeyUsageDigitalSignature | x509.KeyUsageKeyEncipherment,
		ExtKeyUsage:           []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth, x509.ExtKeyUsageClientAuth},
		BasicConstraintsValid: true,
		IsCA:                  true,
		MaxPathLen:            1,
	}

	// 添加 DNS 名称（SAN）
	if len(dnsNames) > 0 {
		template.DNSNames = dnsNames
	}

	// 添加 IP 地址
	for _, ipStr := range ips {
		template.IPAddresses = append(template.IPAddresses, parseIP(ipStr))
	}

	// 自签名
	derBytes, err := x509.CreateCertificate(rand.Reader, &template, &template, &priv.PublicKey, priv)
	if err != nil {
		return nil, nil, fmt.Errorf("创建证书失败: %w", err)
	}

	// 编码为 PEM
	certPEM = pem.EncodeToMemory(&pem.Block{
		Type:  "CERTIFICATE",
		Bytes: derBytes,
	})

	// 编码私钥为 PEM
	keyBytes, err := x509.MarshalECPrivateKey(priv)
	if err != nil {
		return nil, nil, fmt.Errorf("编码私钥失败: %w", err)
	}

	keyPEM = pem.EncodeToMemory(&pem.Block{
		Type:  "EC PRIVATE KEY",
		Bytes: keyBytes,
	})

	return certPEM, keyPEM, nil
}
```

```go
package main

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/base64"
	"encoding/json"

	"net/http"

	"time"

// Claims JWT 负载
type Claims struct {
	UserID   int    `json:"user_id"`
	Username string `json:"username"`
	Role     string `json:"role"`
	ExpiresAt int64 `json:"exp"`
	IssuedAt  int64 `json:"iat"`
}

// JWTManager JWT 管理器
type JWTManager struct {
	secretKey     []byte
	tokenDuration time.Duration
	issuer        string
}

// NewJWTManager 创建 JWT 管理器
func NewJWTManager(secret string, duration time.Duration) *JWTManager {
	return &JWTManager{
		secretKey:     []byte(secret),
		tokenDuration: duration,
		issuer:        "security-lab",
	}
}

// base64Encode Base64 URL 编码（不含 padding）
func base64Encode(data []byte) string {
	return base64.URLEncoding.EncodeToString(data)
}

// base64Decode Base64 URL 解码
func base64Decode(s string) ([]byte, error) {
	return base64.URLEncoding.DecodeString(s)
}

// GenerateToken 生成 JWT
func (jm *JWTManager) GenerateToken(userID int, username, role string) (string, error) {
	// Header
	header := map[string]interface{}{
		"alg": "HS256",
		"typ": "JWT",
	}
	headerJSON, _ := json.Marshal(header)
	headerEncoded := base64Encode(headerJSON)

	// Payload
	claims := Claims{
		UserID:    userID,
		Username:  username,
		Role:      role,
		ExpiresAt: time.Now().Add(jm.tokenDuration).Unix(),
		IssuedAt:  time.Now().Unix(),
	}
	payloadJSON, _ := json.Marshal(claims)
	payloadEncoded := base64Encode(payloadJSON)

	// Signature
	signingInput := headerEncoded + "." + payloadEncoded
	h := hmac.New(sha256.New, jm.secretKey)
	h.Write([]byte(signingInput))
	signature := base64Encode(h.Sum(nil))

	// 组合
	token := signingInput + "." + signature
	return token, nil
}

// ValidateToken 验证 JWT
func (jm *JWTManager) ValidateToken(token string) (*Claims, error) {
	parts := strings.Split(token, ".")
	if len(parts) != 3 {
		return nil, fmt.Errorf("无效的 token 格式")
	}

	// 验证签名
	signingInput := parts[0] + "." + parts[1]
	h := hmac.New(sha256.New, jm.secretKey)
	h.Write([]byte(signingInput))
	expectedSig := base64Encode(h.Sum(nil))

	if !hmac.Equal([]byte(parts[2]), []byte(expectedSig)) {
		return nil, fmt.Errorf("签名验证失败")
	}

	// 解析 Payload
	payloadBytes, err := base64Decode(parts[1])
	if err != nil {
		return nil, fmt.Errorf("解码 payload 失败: %w", err)
	}

	var claims Claims
	if err := json.Unmarshal(payloadBytes, &claims); err != nil {
		return nil, fmt.Errorf("解析 claims 失败: %w", err)
	}

	// 验证过期时间
	if time.Now().Unix() > claims.ExpiresAt {
		return nil, fmt.Errorf("token 已过期")
	}

	return &claims, nil
}

// RefreshToken 刷新 token
func (jm *JWTManager) RefreshToken(token string) (string, error) {
	claims, err := jm.ValidateToken(token)
	if err != nil {
		return "", err
	}
	return jm.GenerateToken(claims.UserID, claims.Username, claims.Role)
}

// JWTMiddleware JWT 认证中间件
func (jm *JWTManager) JWTMiddleware(next http.HandlerFunc) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		// 从 Authorization 头获取 token
		authHeader := r.Header.Get("Authorization")
		if authHeader == "" {
			http.Error(w, `{"error": "缺少认证头"}`, http.StatusUnauthorized)
			return
		}

		// 解析 Bearer token
		parts := strings.SplitN(authHeader, " ", 2)
		if len(parts) != 2 || parts[0] != "Bearer" {
			http.Error(w, `{"error": "认证头格式错误"}`, http.StatusUnauthorized)
			return
		}

		// 验证 token
		claims, err := jm.ValidateToken(parts[1])
		if err != nil {
			http.Error(w, fmt.Sprintf(`{"error": "%s"}`, err.Error()), http.StatusUnauthorized)
			return
		}

		// 将用户信息存入 context
		ctx := r.Context()
		// 在实际项目中可以用 context.WithValue 传递
		_ = ctx
		_ = claims

		next.ServeHTTP(w, r)
	}
}

// RoleMiddleware 角色检查中间件
func (jm *JWTManager) RoleMiddleware(requiredRole string, next http.HandlerFunc) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		authHeader := r.Header.Get("Authorization")
		parts := strings.SplitN(authHeader, " ", 2)
		if len(parts) != 2 {
			http.Error(w, `{"error": "认证失败"}`, http.StatusUnauthorized)
			return
		}

		claims, err := jm.ValidateToken(parts[1])
		if err != nil {
			http.Error(w, `{"error": "token无效"}`, http.StatusUnauthorized)
			return
		}

		if claims.Role != requiredRole {
			http.Error(w, `{"error": "权限不足"}`, http.StatusForbidden)
			return
		}

		next.ServeHTTP(w, r)
	}
}
```

```go
// CheckInput 检查输入是否匹配规则
func (w *WAF) CheckInput(input, fieldName string) []WAFFinding {
	var findings []WAFFinding

	for i := range w.rules {
		rule := &w.rules[i]
		if rule.Pattern.MatchString(input) {
			match := rule.Pattern.FindString(input)
			findings = append(findings, WAFFinding{
				Rule:        rule,
				MatchString: match,
				InputField:  fieldName,
				Timestamp:   time.Now(),
			})
		}
	}

	return findings
}

// Middleware WAF 中间件
func (w *WAF) Middleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(rw http.ResponseWriter, r *http.Request) {
		// 检查白名单
		for _, ip := range w.config.WhitelistIPs {
			if r.RemoteAddr == ip {
				next.ServeHTTP(rw, r)
				return
			}
		}

		// 检查 URL 路径
		if findings := w.CheckInput(r.URL.Path, "URL.Path"); len(findings) > 0 {
			w.logFindings(findings, r)
			if w.config.BlockMode {
				http.Error(rw, `{"error": "请求被WAF拦截", "reason": "恶意输入检测"}`, http.StatusForbidden)
				return
			}
		}

		// 检查查询参数
		for param, values := range r.URL.Query() {
			for _, value := range values {
				if findings := w.CheckInput(value, "Query."+param); len(findings) > 0 {
					w.logFindings(findings, r)
					if w.config.BlockMode {
						http.Error(rw, `{"error": "请求被WAF拦截"}`, http.StatusForbidden)
						return
					}
				}
			}
		}

		// 检查 Cookie
		for _, cookie := range r.Cookies() {
			if findings := w.CheckInput(cookie.Value, "Cookie."+cookie.Name); len(findings) > 0 {
				w.logFindings(findings, r)
				if w.config.BlockMode {
					http.Error(rw, `{"error": "请求被WAF拦截"}`, http.StatusForbidden)
					return
				}
			}
		}

		// 检查请求头
		for header, values := range r.Header {
			for _, value := range values {
				if findings := w.CheckInput(value, "Header."+header); len(findings) > 0 {
					w.logFindings(findings, r)
					if w.config.BlockMode {
						http.Error(rw, `{"error": "请求被WAF拦截"}`, http.StatusForbidden)
						return
					}
				}
			}
		}

		// 检查请求体（仅 POST/PUT）
		if r.Method == "POST" || r.Method == "PUT" {
			r.ParseForm()
			for param, values := range r.PostForm {
				for _, value := range values {
					if findings := w.CheckInput(value, "Body."+param); len(findings) > 0 {
						w.logFindings(findings, r)
						if w.config.BlockMode {
							http.Error(rw, `{"error": "请求被WAF拦截"}`, http.StatusForbidden)
							return
						}
					}
				}
			}
		}

		next.ServeHTTP(rw, r)
	})
}
```

### 4.1 并发端口扫描器

利用 Go 的 goroutine 实现高性能并发端口扫描。

```go
package main

import (
    "fmt"
    "net"
    "sort"
    "sync"
    "time"
)

// 端口扫描器
type PortScanner struct {
    Host        string
    Timeout     time.Duration
    Concurrency int
}

func NewPortScanner(host string) *PortScanner {
    return &PortScanner{
        Host:        host,
        Timeout:     800 * time.Millisecond,
        Concurrency: 100,
    }
}

// 扫描单个端口
func (ps *PortScanner) ScanPort(port int) bool {
    conn, err := net.DialTimeout("tcp", fmt.Sprintf("%s:%d", ps.Host, port), ps.Timeout)
    if err != nil {
        return false
    }
    conn.Close()
    return true
}

// 批量扫描（带并发控制）
func (ps *PortScanner) ScanRange(start, end int) []int {
    var wg sync.WaitGroup
    var mu sync.Mutex
    openPorts := []int{}
    
    // 信号量模式：使用缓冲通道控制并发
    semaphore := make(chan struct{}, ps.Concurrency)
    
    for port := start; port <= end; port++ {
        wg.Add(1)
        go func(p int) {
            defer wg.Done()
            semaphore <- struct{}{}        // 获取信号量
            defer func() { <-semaphore }() // 释放信号量
            
            if ps.ScanPort(p) {
                mu.Lock()
                openPorts = append(openPorts, p)
                mu.Unlock()
            }
        }(port)
    }
    
    wg.Wait()
    sort.Ints(openPorts)
    return openPorts
}

// 服务版本探测
func (ps *PortScanner) BannerGrab(port int) string {
    conn, err := net.DialTimeout("tcp", fmt.Sprintf("%s:%d", ps.Host, port), ps.Timeout)
    if err != nil {
        return ""
    }
    defer conn.Close()
    
    // 发送一些数据触发响应
    conn.Write([]byte("HEAD / HTTP/1.0\r\n\r\n"))
    
    buf := make([]byte, 1024)
    conn.SetReadDeadline(time.Now().Add(ps.Timeout))
    n, _ := conn.Read(buf)
    
    return string(buf[:n])
}

func main() {
    scanner := NewPortScanner("127.0.0.1")
    scanner.Concurrency = 200
    
    fmt.Println("开始扫描...")
    start := time.Now()
    
    openPorts := scanner.ScanRange(1, 1000)
    
    fmt.Printf("扫描完成，用时 %v\n", time.Since(start))
    fmt.Printf("发现 %d 个开放端口:\n", len(openPorts))
    
    for _, port := range openPorts {
        banner := scanner.BannerGrab(port)
        if len(banner) > 50 {
            banner = banner[:50] + "..."
        }
        fmt.Printf("  端口 %d: %s\n", port, banner)
    }
}
```

### 4.2 Web 漏洞扫描器

使用 Go 标准库实现的轻量级 Web 漏洞扫描器。

```go
package main

import (
    "crypto/tls"
    "fmt"
    "io"
    "net/http"
    "net/url"
    "strings"
    "sync"
    "time"
)

// 漏洞信息
type Vulnerability struct {
    Name        string
    Severity    string // critical/high/medium/low/info
    URL         string
    Description string
    Evidence    string
}

// Web 扫描器
type WebScanner struct {
    BaseURL string
    Client  *http.Client
    vulns   []Vulnerability
    mu      sync.Mutex
}

func NewWebScanner(baseURL string) *WebScanner {
    return &WebScanner{
        BaseURL: strings.TrimRight(baseURL, "/"),
        Client: &http.Client{
            Timeout: 5 * time.Second,
            Transport: &http.Transport{
                TLSClientConfig: &tls.Config{InsecureSkipVerify: true},
            },
            CheckRedirect: func(req *http.Request, via []*http.Request) error {
                return http.ErrUseLastResponse // 不跟随重定向
            },
        },
    }
}

// 发送请求
func (ws *WebScanner) SendRequest(method, path string) (*http.Response, string, error) {
    fullURL := ws.BaseURL + path
    req, err := http.NewRequest(method, fullURL, nil)
    if err != nil {
        return nil, "", err
    }
    
    req.Header.Set("User-Agent", "Mozilla/5.0 (compatible; GoScanner/1.0)")
    
    resp, err := ws.Client.Do(req)
    if err != nil {
        return nil, "", err
    }
    defer resp.Body.Close()
    
    body, _ := io.ReadAll(resp.Body)
    return resp, string(body), nil
}

// SQL 注入检测插件
func (ws *WebScanner) CheckSQLi(path string, param string) {
    payloads := []string{
        "'",
        "' OR '1'='1",
        "' AND 1=2 UNION SELECT 1,2,3--",
        "' AND SLEEP(3)--",
    }
    
    u, _ := url.Parse(ws.BaseURL + path)
    q := u.Query()
    
    for _, payload := range payloads {
        q.Set(param, payload)
        u.RawQuery = q.Encode()
        
        resp, body, err := ws.SendRequest("GET", "?"+u.RawQuery)
        if err != nil {
            continue
        }
        
        // 检查 SQL 错误特征
        errors := []string{
            "MySQL syntax", "SQL syntax", "ORA-",
            "PostgreSQL", "Microsoft SQL Server",
            "Warning: mysql_", "unclosed quotation mark",
        }
        
        for _, errMsg := range errors {
            if strings.Contains(strings.ToLower(body), strings.ToLower(errMsg)) {
                ws.addVuln(Vulnerability{
                    Name:        "SQL Injection",
                    Severity:    "critical",
                    URL:         u.String(),
                    Description: fmt.Sprintf("参数 %s 可能存在 SQL 注入", param),
                    Evidence:    errMsg,
                })
                break
            }
        }
        
        _ = resp
    }
}

// XSS 检测插件
func (ws *WebScanner) CheckXSS(path string, param string) {
    payload := "<script>alert('xss_test')</script>"
    
    u, _ := url.Parse(ws.BaseURL + path)
    q := u.Query()
    q.Set(param, payload)
    u.RawQuery = q.Encode()
    
    _, body, err := ws.SendRequest("GET", "?"+u.RawQuery)
    if err != nil {
        return
    }
    
    if strings.Contains(body, payload) {
        ws.addVuln(Vulnerability{
            Name:        "Reflected XSS",
            Severity:    "high",
            URL:         u.String(),
            Description: fmt.Sprintf("参数 %s 存在反射型 XSS", param),
            Evidence:    payload,
        })
    }
}

// 目录爆破
func (ws *WebScanner) DirBrute(wordlist []string) []string {
    var found []string
    var wg sync.WaitGroup
    var mu sync.Mutex
    
    semaphore := make(chan struct{}, 30)
    
    for _, path := range wordlist {
        wg.Add(1)
        go func(p string) {
            defer wg.Done()
            semaphore <- struct{}{}
            defer func() { <-semaphore }()
            
            resp, _, err := ws.SendRequest("GET", "/"+p)
            if err != nil {
                return
            }
            
            if resp.StatusCode != 404 {
                mu.Lock()
                found = append(found, fmt.Sprintf("[%d] /%s", resp.StatusCode, p))
                mu.Unlock()
            }
        }(path)
    }
    
    wg.Wait()
    return found
}

func (ws *WebScanner) addVuln(v Vulnerability) {
    ws.mu.Lock()
    ws.vulns = append(ws.vulns, v)
    ws.mu.Unlock()
}

func (ws *WebScanner) Vulnerabilities() []Vulnerability {
    ws.mu.Lock()
    defer ws.mu.Unlock()
    return ws.vulns
}

func main() {
    scanner := NewWebScanner("http://example.com")
    
    // 检查 SQL 注入
    scanner.CheckSQLi("/page", "id")
    
    // 检查 XSS
    scanner.CheckXSS("/search", "q")
    
    // 输出结果
    for _, v := range scanner.Vulnerabilities() {
        fmt.Printf("[%s] %s: %s\n", v.Severity, v.Name, v.URL)
    }
}
```

```go
package main

import (
    "crypto/aes"
    "crypto/cipher"
    "crypto/ecdsa"
    "crypto/elliptic"
    "crypto/rand"
    "crypto/rsa"
    "crypto/sha256"
    "crypto/tls"
    "crypto/x509"
    "crypto/x509/pkix"
    "encoding/base64"
    "encoding/pem"
    "fmt"
    "math/big"
    "net/http"
    "os"
    "time"
)

// ===== 加密工具类 =====

// AES 加密（GCM 模式）
func AESGCMEncrypt(plaintext, key []byte) ([]byte, error) {
    block, err := aes.NewCipher(key)
    if err != nil {
        return nil, err
    }
    
    gcm, err := cipher.NewGCM(block)
    if err != nil {
        return nil, err
    }
    
    nonce := make([]byte, gcm.NonceSize())
    if _, err := rand.Read(nonce); err != nil {
        return nil, err
    }
    
    ciphertext := gcm.Seal(nonce, nonce, plaintext, nil)
    return ciphertext, nil
}

func AESGCMDecrypt(ciphertext, key []byte) ([]byte, error) {
    block, err := aes.NewCipher(key)
    if err != nil {
        return nil, err
    }
    
    gcm, err := cipher.NewGCM(block)
    if err != nil {
        return nil, err
    }
    
    nonceSize := gcm.NonceSize()
    nonce, ciphertext := ciphertext[:nonceSize], ciphertext[nonceSize:]
    
    return gcm.Open(nil, nonce, ciphertext, nil)
}

// 哈希计算
func Sha256(data string) string {
    h := sha256.Sum256([]byte(data))
    return fmt.Sprintf("%x", h)
}

// Base64
func B64Encode(data string) string {
    return base64.StdEncoding.EncodeToString([]byte(data))
}

func B64Decode(data string) (string, error) {
    b, err := base64.StdEncoding.DecodeString(data)
    return string(b), err
}

// ===== RSA 非对称加密 =====

func GenerateRSAKey(bits int) (*rsa.PrivateKey, error) {
    return rsa.GenerateKey(rand.Reader, bits)
}

func RSAEncrypt(publicKey *rsa.PublicKey, plaintext []byte) ([]byte, error) {
    return rsa.EncryptOAEP(sha256.New(), rand.Reader, publicKey, plaintext, nil)
}

func RSADecrypt(privateKey *rsa.PrivateKey, ciphertext []byte) ([]byte, error) {
    return privateKey.Decrypt(rand.Reader, ciphertext, &rsa.OAEPOptions{Hash: sha256.New()})
}

// ===== ECDSA 签名 =====

func GenerateECDSAKey() (*ecdsa.PrivateKey, error) {
    return ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
}

func ECDSASign(privateKey *ecdsa.PrivateKey, data []byte) (r, s *big.Int, err error) {
    hash := sha256.Sum256(data)
    return ecdsa.Sign(rand.Reader, privateKey, hash[:])
}

func ECDSAVerify(publicKey *ecdsa.PublicKey, data []byte, r, s *big.Int) bool {
    hash := sha256.Sum256(data)
    return ecdsa.Verify(publicKey, hash[:], r, s)
}

// ===== 自签名证书生成 =====

func GenerateSelfSignedCert() (tls.Certificate, error) {
    // 生成私钥
    priv, err := rsa.GenerateKey(rand.Reader, 2048)
    if err != nil {
        return tls.Certificate{}, err
    }
    
    // 证书模板
    template := x509.Certificate{
        SerialNumber: big.NewInt(1),
        Subject: pkix.Name{
            Organization: []string{"Test Org"},
            CommonName:   "localhost",
        },
        NotBefore: time.Now(),
        NotAfter:  time.Now().Add(365 * 24 * time.Hour),
        KeyUsage:  x509.KeyUsageKeyEncipherment | x509.KeyUsageDigitalSignature,
        ExtKeyUsage: []x509.ExtKeyUsage{
            x509.ExtKeyUsageServerAuth,
        },
        DNSNames: []string{"localhost", "127.0.0.1"},
    }
    
    // 自签名
    certDER, err := x509.CreateCertificate(rand.Reader, &template, &template, &priv.PublicKey, priv)
    if err != nil {
        return tls.Certificate{}, err
    }
    
    // 编码为 PEM
    certPEM := pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: certDER})
    keyPEM := pem.EncodeToMemory(&pem.Block{Type: "RSA PRIVATE KEY", Bytes: x509.MarshalPKCS1PrivateKey(priv)})
    
    return tls.X509KeyPair(certPEM, keyPEM)
}

// ===== 安全 TLS 配置 =====

func secureTLSConfig() *tls.Config {
    return &tls.Config{
        MinVersion: tls.VersionTLS12,
        CipherSuites: []uint16{
            tls.TLS_ECDHE_ECDSA_WITH_AES_256_GCM_SHA384,
            tls.TLS_ECDHE_RSA_WITH_AES_256_GCM_SHA384,
            tls.TLS_ECDHE_ECDSA_WITH_CHACHA20_POLY1305,
            tls.TLS_ECDHE_RSA_WITH_CHACHA20_POLY1305,
        },
        CurvePreferences: []tls.CurveID{
            tls.CurveP256,
            tls.X25519,
        },
        InsecureSkipVerify: false, // 必须验证证书！
    }
}
```
