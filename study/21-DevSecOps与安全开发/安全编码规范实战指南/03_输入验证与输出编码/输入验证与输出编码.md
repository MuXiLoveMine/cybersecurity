# 输入验证与输出编码

### 2.1 输入验证原则

```
白名单验证 > 黑名单过滤

验证维度：
├── 类型验证   → 期望 int 就拒绝 string
├── 长度验证   → 限制最大长度防 DoS
├── 格式验证   → 正则匹配预期格式
├── 范围验证   → 数值/日期在合理区间
└── 字符集验证 → 拒绝控制字符/NULL字节
```

### 2.2 SQL 注入防护

**漏洞代码：**
```java
// Java - 字符串拼接 SQL
String sql = "SELECT * FROM users WHERE name='" + username + "' AND password='" + password + "'";
Statement stmt = conn.createStatement();
ResultSet rs = stmt.executeQuery(sql);  // SQL注入！
```

```python
# Python - 字符串拼接 SQL
cursor.execute(f"SELECT * FROM users WHERE name='{username}'")  # SQL注入！
```

```php
// PHP - 字符串拼接 SQL
$sql = "SELECT * FROM users WHERE id=" . $_GET['id'];
mysqli_query($conn, $sql);  // SQL注入！
```

**安全代码：**
```java
// Java - 参数化查询（PreparedStatement）
String sql = "SELECT * FROM users WHERE name=? AND password=?";
PreparedStatement pstmt = conn.prepareStatement(sql);
pstmt.setString(1, username);
pstmt.setString(2, password);
ResultSet rs = pstmt.executeQuery();  // 安全
```

```python
# Python - 参数化查询
cursor.execute("SELECT * FROM users WHERE name=%s AND password=%s", (username, password))
```

```php
// PHP - PDO 参数化查询
$stmt = $pdo->prepare("SELECT * FROM users WHERE id=:id");
$stmt->bindParam(':id', $_GET['id'], PDO::PARAM_INT);
$stmt->execute();
```

```go
// Go - 参数化查询
row := db.QueryRow("SELECT * FROM users WHERE name=$1 AND password=$2", username, password)
```

```javascript
// JavaScript (Node.js) - 参数化查询
const result = await pool.query('SELECT * FROM users WHERE name=$1 AND password=$2', [username, password]);
```

### 2.3 XSS 防护

**漏洞代码：**
```python
# Python Flask - 直接输出用户输入
@app.route('/search')
def search():
    keyword = request.args.get('q', '')
    return f"<h1>搜索结果: {keyword}</h1>"  # 反射型XSS！
```

```javascript
// JavaScript - innerHTML 注入
document.getElementById('output').innerHTML = userInput;  // DOM型XSS！
```

**安全代码：**
```python
# Python - 自动转义模板引擎
from flask import render_template_string
# Jinja2 默认自动转义
return render_template_string('<h1>搜索结果: {{ keyword }}</h1>', keyword=keyword)
```

```javascript
// JavaScript - 使用 textContent 或框架自动转义
document.getElementById('output').textContent = userInput;
// 或 React JSX（默认转义）
// <div>{userInput}</div>
```

```java
// Java - OWASP Java Encoder
import org.owasp.encoder.Encode;
out.println("<div>" + Encode.forHtml(userInput) + "</div>");
```

```php
// PHP - htmlspecialchars
echo '<div>' . htmlspecialchars($userInput, ENT_QUOTES, 'UTF-8') . '</div>';
```

```go
// Go - html/template 自动转义
import "html/template"
t := template.Must(template.New("page").Parse(`<div>{{.}}</div>`))
t.Execute(w, userInput)
```

### 2.5 路径遍历防护

**漏洞代码：**
```python
# Python - 直接拼接文件路径
filename = request.args.get('file')
with open(f'/var/www/uploads/{filename}') as f:  # 路径遍历！输入 ../../../etc/passwd
    return f.read()
```

**安全代码：**
```python
import os

BASE_DIR = '/var/www/uploads'
filename = request.args.get('file', '')
full_path = os.path.normpath(os.path.join(BASE_DIR, filename))

# 验证最终路径在允许目录内
if not full_path.startswith(BASE_DIR):
    abort(403)
with open(full_path) as f:
    return f.read()
```

```java
// Java - Path 规范化校验
Path basePath = Paths.get("/var/www/uploads").toAbsolutePath();
Path resolved = basePath.resolve(filename).normalize();
if (!resolved.startsWith(basePath)) {
    throw new SecurityException("路径遍历攻击！");
}
```

```go
// Go - filepath.Clean + 前缀检查
import "path/filepath"
absPath := filepath.Join("/var/www/uploads", filename)
absPath, _ = filepath.Abs(absPath)
if !strings.HasPrefix(absPath, "/var/www/uploads/") {
    http.Error(w, "Forbidden", 403)
    return
}
```

### 2.6 SSRF 防护

**漏洞代码：**
```python
# Python - 用户控制 URL 直接请求
import requests
url = request.args.get('url')
resp = requests.get(url)  # SSRF！可请求内网 http://169.254.169.254/
```

**安全代码：**
```python
import ipaddress
import socket
from urllib.parse import urlparse

ALLOWED_SCHEMES = ['http', 'https']

def is_safe_url(url):
    parsed = urlparse(url)
    if parsed.scheme not in ALLOWED_SCHEMES:
        return False
    # 解析 IP，拒绝内网地址
    try:
        ip = ipaddress.ip_address(socket.gethostbyname(parsed.hostname))
        if ip.is_private or ip.is_loopback or ip.is_link_local:
            return False
    except ValueError:
        return False
    return True

if not is_safe_url(url):
    abort(403)
resp = requests.get(url, timeout=5)
```

### 2.7 XXE 防护

**漏洞代码：**
```java
// Java - 默认 XML 解析（启用外部实体）
DocumentBuilderFactory dbf = DocumentBuilderFactory.newInstance();
DocumentBuilder db = dbf.newDocumentBuilder();
Document doc = db.parse(inputStream);  // XXE！可读取系统文件
```

**安全代码：**
```java
// Java - 禁用外部实体
DocumentBuilderFactory dbf = DocumentBuilderFactory.newInstance();
dbf.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
dbf.setFeature("http://xml.org/sax/features/external-general-entities", false);
dbf.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
dbf.setXIncludeAware(false);
dbf.setExpandEntityReferences(false);
DocumentBuilder db = dbf.newDocumentBuilder();
Document doc = db.parse(inputStream);  // 安全
```

```python
# Python - 使用 defusedxml
from defusedxml import ElementTree as ET
tree = ET.parse('file.xml')  # 自动禁用外部实体
```

```php
// PHP - 禁用外部实体加载
libxml_disable_entity_loader(true);
$dom = new DOMDocument();
$dom->loadXML($xml, LIBXML_NOENT | LIBXML_DTDLOAD);
```
