# 认证与授权编码

### 3.1 密码存储安全

**漏洞代码：**
```python
# Python - 明文或弱哈希存储密码
import hashlib
password_hash = hashlib.md5(password.encode()).hexdigest()  # MD5不安全！
```

**安全代码：**
```python
# Python - 使用 bcrypt
import bcrypt

# 注册时哈希
salt = bcrypt.gensalt(rounds=12)
password_hash = bcrypt.hashpw(password.encode('utf-8'), salt)

# 登录时验证
if bcrypt.checkpw(input_password.encode('utf-8'), password_hash):
    print("密码正确")
```

```java
// Java - 使用 Argon2（推荐）
import de.mkammerer.argon2.Argon2;
import de.mkammerer.argon2.Argon2Factory;

Argon2 argon2 = Argon2Factory.create();
// 哈希密码（自动生成盐值）
String hash = argon2.hash(10, 65536, 1, password.toCharArray());
// 验证密码
if (argon2.verify(hash, inputPassword.toCharArray())) {
    System.out.println("密码正确");
}
```

```go
// Go - 使用 scrypt
import "golang.org/x/crypto/scrypt"

func hashPassword(password string) ([]byte, error) {
    salt := generateRandomBytes(32)
    hash, err := scrypt.Key([]byte(password), salt, 32768, 8, 1, 32)
    return append(salt, hash...), err
}
```

```php
// PHP - 使用 password_hash（默认 bcrypt）
$hash = password_hash($password, PASSWORD_BCRYPT, ['cost' => 12]);
if (password_verify($input, $hash)) {
    // 认证成功
}
```

### 3.2 Session 管理安全编码

```python
# Python Flask - 安全 Session 配置
app.config.update(
    SESSION_COOKIE_SECURE=True,      # 仅 HTTPS 传输
    SESSION_COOKIE_HTTPONLY=True,    # 禁止 JavaScript 访问
    SESSION_COOKIE_SAMESITE='Lax',  # 防 CSRF
    PERMANENT_SESSION_LIFETIME=3600, # 1小时过期
    SECRET_KEY=os.urandom(32)       # 强随机密钥
)
```

```java
// Java - Session 安全配置（web.xml）
<session-config>
    <cookie-config>
        <http-only>true</http-only>
        <secure>true</secure>
    </cookie-config>
    <session-timeout>30</session-timeout>
</session-config>
```

### 3.3 CSRF 防护编码

```python
# Python Flask - CSRF Token
from flask_wtf.csrf import CSRFProtect

app.config['SECRET_KEY'] = os.urandom(32)
csrf = CSRFProtect(app)

# 模板中自动渲染 Token
# <form method="POST">
#   <input type="hidden" name="csrf_token" value="{{ csrf_token() }}">
# </form>
```

```javascript
// JavaScript (Express) - csurf 中间件
const csrf = require('csurf');
const csrfProtection = csrf({ cookie: true });
app.get('/form', csrfProtection, (req, res) => {
    res.render('form', { csrfToken: req.csrfToken() });
});
```

### 3.4 JWT 安全使用规范

**漏洞代码：**
```python
# Python - 不安全 JWT 使用
import jwt
token = jwt.encode({'user': username, 'admin': True}, 'secret', algorithm='HS256')
# 问题1：硬编码密钥；问题2：算法不指定为none漏洞
```

**安全代码：**
```python
# Python - 安全 JWT 使用
import jwt
from cryptography.hazmat.primitives.asymmetric import rsa

# 使用 RS256（非对称签名），密钥从安全配置读取
private_key = load_private_key_from_vault()  # 从密钥管理服务获取

token = jwt.encode(
    {'user_id': user.id, 'role': user.role, 'exp': datetime.utcnow() + timedelta(hours=1)},
    private_key,
    algorithm='RS256'  # 禁止 none 算法
)

# 验证时强制指定算法
try:
    payload = jwt.decode(token, public_key, algorithms=['RS256'])  # 明确允许的算法
except jwt.InvalidTokenError:
    abort(401)
```

### 3.5 OAuth 安全集成

```python
# Python - OAuth2 安全集成要点
# 1. 使用 state 参数防 CSRF
# 2. 验证 redirect_uri 白名单
# 3. 使用 PKCE（Proof Key for Code Exchange）

import secrets

# 生成 state 和 PKCE
state = secrets.token_urlsafe(32)
code_verifier = secrets.token_urlsafe(64)
import hashlib, base64
code_challenge = base64.urlsafe_b64encode(
    hashlib.sha256(code_verifier.encode()).digest()
).rstrip(b'=').decode()

# 构造授权 URL
auth_url = f"{auth_server}/authorize?response_type=code&client_id={client_id}" \
           f"&redirect_uri={redirect_uri}&state={state}" \
           f"&code_challenge={code_challenge}&code_challenge_method=S256"
```

### 3.6 权限校验编码模式

**漏洞代码：**
```python
# Python - 缺少权限校验（IDOR）
@app.route('/api/orders/<int:order_id>')
def get_order(order_id):
    order = Order.query.get(order_id)  # 任何用户可查看任意订单！
    return jsonify(order.to_dict())
```

**安全代码：**
```python
# Python - 基于用户身份的权限校验
@app.route('/api/orders/<int:order_id>')
@login_required
def get_order(order_id):
    order = Order.query.filter_by(id=order_id, user_id=current_user.id).first()
    if not order:
        abort(404)  # 不泄露资源是否存在
    return jsonify(order.to_dict())
```

```java
// Java - 注解式权限控制
@PreAuthorize("hasRole('ADMIN') or #order.userId == authentication.principal.id")
@GetMapping("/api/orders/{id}")
public Order getOrder(@PathVariable Long id) {
    return orderService.findById(id);
}
```
