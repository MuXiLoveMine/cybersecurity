# 总结

容器与Kubernetes安全是一个纵深防御体系，核心原则可归纳为：

1. **最小权限**：镜像最小化、Capabilities最小化、RBAC最小化
2. **纵深防御**：镜像扫描 + 运行时加固 + 网络隔离 + 准入控制 + 监控告警多层叠加
3. **供应链可信**：镜像签名验证贯穿构建到部署全流程
4. **持续运营**：CIS基准检查、运行时威胁检测、定期审计形成闭环

**学习路径建议**：

| 阶段 | 工具 | 目标 |
|------|------|------|
| 入门 | Trivy、kube-bench | 掌握镜像扫描与基准检查 |
| 进阶 | Kyverno/OPA、Falco | 掌握准入策略与运行时监控 |
| 高级 | Tetragon、Cosign、Vault | 掌握eBPF安全、供应链签名、Secret管理 |
| 实战 | Python脚本、取证流程 | 掌握自动化审计与事件响应 |

  "tlscacert": "/etc/docker/ca.pem",
  "tlscert": "/etc/docker/server-cert.pem",
  "tlskey": "/etc/docker/server-key.pem",
  "tlsverify": true,
  "hosts": ["tcp://0.0.0.0:2376", "unix:///var/run/docker.sock"]

#### 3. 容器运行时安全

```bash
# 以安全配置启动容器（最佳实践）
docker run -d \
  --name secure-container \
  --read-only                     `# 只读根文件系统` \
  --tmpfs /tmp:rw,noexec,nosuid   `# /tmp 挂载为 tmpfs` \
  --tmpfs /run:rw,noexec,nosuid   \
  --cap-drop ALL                  `# 删除所有能力` \
  --cap-add NET_BIND_SERVICE      `# 只添加必要能力` \
  --security-opt no-new-privileges:true \
  --security-opt seccomp=default  `# 使用默认 seccomp 配置` \
  --security-opt apparmor=docker-default `# AppArmor 配置` \
  --memory 512m                   `# 内存限制` \
  --cpus 1                        `# CPU 限制` \
  --pids-limit 100                `# 进程数限制` \
  --user 1000:1000                `# 非 root 用户运行` \
  --network none                  `# 无网络（按需配置）` \
  nginx:latest

# 检查容器安全配置
docker inspect secure-container | jq '.[0].HostConfig | {
  CapAdd, CapDrop, SecurityOpt, ReadonlyRootfs,
  Memory, CpuShares, PidsLimit, User
}'

# 查看容器中运行的进程（检查是否有异常进程）
docker top secure-container

# 查看容器资源使用情况
docker stats --no-stream

# 检查 Docker Bench for Security（CIS 基准检查）
# 运行 Docker Bench
docker run -it --net host --pid host --userns host --cap-add audit_control \
  -e DOCKER_CONTENT_TRUST=$DOCKER_CONTENT_TRUST \
  -v /etc:/etc:ro \
  -v /usr/bin/docker-containerd:/usr/bin/docker-containerd:ro \
  -v /usr/bin/docker-runc:/usr/bin/docker-runc:ro \
  -v /usr/lib/systemd:/usr/lib/systemd:ro \
  -v /var/lib:/var/lib:ro \
  -v /var/run/docker.sock:/var/run/docker.sock:ro \
  --label docker_bench_security \
  docker/docker-bench-security

# 检查容器逃逸风险
# 检查特权容器
docker ps --quiet --filter status=running | xargs -I{} docker inspect {} \
  --format '{{.Name}}: Privileged={{.HostConfig.Privileged}}'

# 检查挂载了 Docker socket 的容器（高危）
docker ps --quiet --filter status=running | xargs -I{} docker inspect {} \
  | jq -r '.[] | select(.Mounts[]?.Source == "/var/run/docker.sock") | .Name'
```

apiVersion: v1
kind: Pod
metadata:
  name: secure-pod
  labels:
    app: secure-app
spec:
  securityContext:
    runAsNonRoot: true
    runAsUser: 1000
    runAsGroup: 1000
    fsGroup: 1000
    seccompProfile:
      type: RuntimeDefault
  containers:
  - name: app
    image: nginx:stable-alpine
    securityContext:
      allowPrivilegeEscalation: false
      readOnlyRootFilesystem: true
      capabilities:
        drop:
          - ALL
    resources:
      limits:
        memory: "256Mi"
        cpu: "500m"
      requests:
        memory: "64Mi"
        cpu: "100m"

    - name: tmp
      mountPath: /tmp

      sizeLimit: 64Mi
  - name: cache
    emptyDir:
      medium: Memory
      sizeLimit: 64Mi

### 四、容器安全一键检查脚本

```bash
#!/bin/bash
# 容器安全快速检查脚本 docker_security_audit.sh

echo "=== Docker & Kubernetes Security Audit ==="
echo ""

echo "[1] Docker Daemon Security Check"
echo "---"
docker info 2>/dev/null | grep -E "Security Options|Userns|Rootless" || echo "  Docker not available"

echo ""
echo "[2] Privileged Containers Check"
echo "---"
docker ps --quiet --filter status=running 2>/dev/null | while read id; do
  name=$(docker inspect --format '{{.Name}}' "$id")
  priv=$(docker inspect --format '{{.HostConfig.Privileged}}' "$id")
  if [ "$priv" = "true" ]; then echo "  WARNING: $name is PRIVILEGED"; fi
done || echo "  No running containers"

echo ""
echo "[3] Docker Socket Mount Check"
echo "---"
docker ps -q 2>/dev/null | while read id; do
  mounts=$(docker inspect "$id" --format '{{range .Mounts}}{{.Source}} {{end}}' 2>/dev/null)
  if echo "$mounts" | grep -q "docker.sock"; then
    name=$(docker inspect --format '{{.Name}}' "$id")
    echo "  CRITICAL: $name has docker.sock mounted"
  fi
done || echo "  No running containers"

echo ""
echo "[4] Image Vulnerability Scan (Trivy)"
echo "---"
if command -v trivy &> /dev/null; then
  echo "  Scanning nginx:latest..."
  trivy image --severity CRITICAL --quiet nginx:latest 2>&1 | tail -5
else
  echo "  Trivy not installed"
fi

echo ""
echo "[5] Kubernetes Security Check"
echo "---"
if command -v kubectl &> /dev/null; then
  echo "  Checking anonymous access..."
  kubectl auth can-i get pods --as=system:anonymous 2>&1
  echo "  Privileged pods:"
  kubectl get pods -A -o json 2>/dev/null | jq -r '.items[] | select(.spec.containers[].securityContext.privileged == true) | "  WARNING: \(.metadata.namespace)/\(.metadata.name)"' 2>/dev/null || echo "  None"
else
  echo "  kubectl not available"
fi

echo ""
echo "=== Audit Complete ==="
```
