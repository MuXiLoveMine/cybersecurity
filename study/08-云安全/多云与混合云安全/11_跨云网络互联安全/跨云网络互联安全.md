# 跨云网络互联安全

### 11.1 跨云连接方式对比

```text
跨云互联方式对比:

| 互联方式 | 带宽 | 延迟 | 安全性 | 成本 | 适用场景 |
|----------|------|------|--------|------|----------|
| 公网IPSec VPN | 中(1Gbps) | 高 | 中(加密) | 低 | 开发/测试环境 |
| 云专线(Direct Connect) | 高(10Gbps) | 低 | 高(私有) | 高 | 生产环境 |
| SD-WAN | 中高 | 中 | 中高 | 中 | 分支机构多云 |
| 云对等连接 | 高 | 低 | 高 | 中 | 同区域云互联 |
| Transit Gateway | 高 | 低 | 高 | 中 | AWS多VPC |
| 云厂商互联(阿里云-CEN) | 高 | 低 | 高 | 中 | 国内多云 |
```

### 11.2 IPSec VPN跨云配置

```bash
# AWS Site-to-Site VPN配置
# 1. 创建Customer Gateway
aws ec2 create-customer-gateway \
    --type ipsec.1 \
    --public-ip 203.0.113.1 \
    --bgp-asn 65000

# 2. 创建VPN Connection
aws ec2 create-vpn-connection \
    --customer-gateway-id cgw-123456 \
    --vpn-gateway-id vgw-123456 \
    --type ipsec.1

# Azure VPN Gateway配置
az network vpn-gateway create \
    --name azure-vpn-gw \
    --resource-group rg-network \
    --vnet vnet-prod \
    --gateway-type Vpn \
    --sku VpnGw1 \
    --vpn-type RouteBased

# 阿里云IPsec VPN网关
aliyun vpc CreateVpnGateway \
    --Region cn-hangzhou \
    --VpcId vpc-123 \
    --Bandwidth 10 \
    --EnableBgp true

# StrongSwan配置示例(本地网关)
# /etc/ipsec.conf
config setup
    charondebug="all"
    uniqueids=yes

conn aws-vpn
    authby=secret
    left=%defaultroute
    leftid=203.0.113.1
    leftsubnet=10.0.0.0/16
    right=52.0.0.1
    rightsubnet=172.16.0.0/16
    ike=aes256-sha256-modp2048
    esp=aes256-sha256
    keyexchange=ikev2
    auto=start
    keyingtries=%forever
```

### 11.3 SD-WAN跨云安全

```bash
# 使用Cisco Viptela SD-WAN跨云连接
# vBond配置
vbond 10.0.0.1 local system-ip 1.1.1.1 \
    organization "Company" \
    vbond 10.0.0.1 port 12346

# IPSec隧道加密
# 所有SD-WAN隧道默认使用AES-256-GCM加密
# 可配置DTLS作为备用隧道
vpn 0
    interface GigabitEthernet0/0
        tunnel-interface
            encapsulation ipsec
            color biz-internet
            allow-service ssh
            allow-service icmp

# 应用感知路由策略
policy
    app-list critical_apps
        app skype
        app teams
        app sap
    data-policy failover
        action forward
        match
            app-list critical_apps
        set
            tloc-action primary
```

### 11.4 跨云网络分段与微分段

```yaml
# 跨云网络安全组统一策略
# 使用Terraform管理跨云安全组规则

# AWS安全组
resource "aws_security_group" "web_tier" {
  name        = "web-tier-sg"
  description = "Web tier security group"
  vpc_id      = aws_vpc.main.id

  ingress {
    from_port   = 443
    to_port     = 443
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  ingress {
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    security_groups = [aws_security_group.alb.id]
  }

  egress {
    from_port   = 3306
    to_port     = 3306
    protocol    = "tcp"
    security_groups = [aws_security_group.db.id]
  }
}

# Azure NSG
resource "azurerm_network_security_group" "web_tier" {
  name                = "web-tier-nsg"
  location            = azurerm_resource_group.main.location
  resource_group_name = azurerm_resource_group.main.name

  security_rule {
    name                       = "allow-https"
    priority                   = 100
    direction                  = "Inbound"
    access                     = "Allow"
    protocol                   = "Tcp"
    source_port_range          = "*"
    destination_port_range     = "443"
    source_address_prefix      = "*"
    destination_address_prefix = "*"
  }
}

# 阿里云安全组
resource "alicloud_security_group" "web_tier" {
  name        = "web-tier-sg"
  description = "Web tier security group"
  vpc_id      = alicloud_vpc.main.id

  ingress {
    port_range = "443/443"
    ip_protocol = "tcp"
    source_cidr_ip = "0.0.0.0/0"
  }
}
```
