# 多云事件响应与取证

### 14.1 多云IR流程

```text
多云事件响应流程:

Phase 1: 检测 (Detection)
  ├── 各云SIEM告警聚合
  ├── CSPM配置异常检测
  ├── CWPP工作负载异常
  └── 流量分析异常

Phase 2: 分析 (Analysis)
  ├── 确定影响范围(哪些云受影响)
  ├── 识别攻击路径
  ├── 关联跨云日志
  └── 评估数据泄露风险

Phase 3: 遏制 (Containment)
  ├── 隔离受影响实例
  ├── 撤销IAM权限
  ├── 阻断恶意IP
  ├── 切换流量到备用云
  └── 保留取证证据

Phase 4: 根除 (Eradication)
  ├── 删除恶意资源
  ├── 轮换所有密钥
  ├── 修复配置漏洞
  └── 部署补丁

Phase 5: 恢复 (Recovery)
  ├── 从备份恢复数据
  ├── 验证服务可用性
  ├── 监控异常行为
  └── 逐步恢复流量

Phase 6: 总结 (Lessons Learned)
  ├── 编写事件报告
  ├── 更新IR剧本
  └── 改进检测规则
```

### 14.2 跨云取证工具与命令

```bash
# AWS取证
# 获取CloudTrail事件历史
aws cloudtrail lookup-events \
    --start-time 2026-09-01 \
    --end-time 2026-09-15 \
    --lookup-attributes AttributeKey=EventName,AttributeValue=RunInstances

# 获取实例元数据
aws ec2 describe-instances --instance-ids i-1234567890
# 获取EBS快照用于取证
aws ec2 create-snapshot --volume-id vol-1234567890 --description "forensic-snapshot"

# 获取VPC Flow Logs
aws logs filter-log-events \
    --log-group-name VPCFlowLogs \
    --filter-pattern "{ $.sourceIp = \"1.2.3.4\" }"

# Azure取证
# 获取Activity Log
az monitor activity-log list \
    --resource-group rg-prod \
    --status Failed,Started

# 获取NSG Flow Logs
az network watcher flow-log show \
    --resource-group NetworkWatcherRG \
    --nsg-name prod-nsg

# 获取VM取证快照
az snapshot create \
    --resource-group forensic \
    --name forensic-snap \
    --source /subscriptions/xxx/resourceGroups/rg/providers/Microsoft.Compute/disks/disk-name

# 阿里云取证
# 获取ActionTrail日志
aliyun actiontrail LookupEvents \
    --StartTime "2026-09-01T00:00:00Z" \
    --EndTime "2026-09-15T23:59:59Z"

# 创建云盘快照
aliyun ecs CreateSnapshot \
    --DiskId d-123 \
    --Description "forensic-snapshot"
```

### 14.3 云取证自动化脚本

```python
#!/usr/bin/env python3
"""多云取证自动化工具"""

import boto3
import json
from datetime import datetime, timedelta
import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')
logger = logging.getLogger(__name__)

class CloudForensics:
    def __init__(self):
        self.ec2 = boto3.client('ec2', region_name='us-east-1')
        self.cloudtrail = boto3.client('cloudtrail', region_name='us-east-1')
        self.logs = boto3.client('logs', region_name='us-east-1')
        self.s3 = boto3.client('s3')

    def collect_cloudtrail_events(self, start_time, end_time, event_name=None):
        """收集CloudTrail事件"""
        events = []
        next_token = None
        while True:
            params = {
                'StartTime': start_time,
                'EndTime': end_time,
            }
            if event_name:
                params['LookupAttributes'] = [
                    {'AttributeKey': 'EventName', 'AttributeValue': event_name}
                ]
            if next_token:
                params['NextToken'] = next_token
            response = self.cloudtrail.lookup_events(**params)
            events.extend(response.get('Events', []))
            next_token = response.get('NextToken')
            if not next_token:
                break
        return events

    def create_forensic_snapshot(self, instance_id, description="forensic"):
        """创建取证快照"""
        # 获取实例的EBS卷
        instance = self.ec2.describe_instances(InstanceIds=[instance_id])
        volumes = []
        for reservation in instance['Reservations']:
            for inst in reservation['Instances']:
                for block in inst.get('BlockDeviceMappings', []):
                    volumes.append(block['Ebs']['VolumeId'])
        # 创建快照
        snapshots = []
        for vol_id in volumes:
            snap = self.ec2.create_snapshot(
                VolumeId=vol_id,
                Description=f"{description} - {vol_id} - {datetime.now()}"
            )
            snapshots.append(snap['SnapshotId'])
            logger.info(f"已创建快照: {snap['SnapshotId']} for volume {vol_id}")
        return snapshots

    def isolate_instance(self, instance_id, security_group_id):
        """隔离实例"""
        self.ec2.modify_instance_attribute(
            InstanceId=instance_id,
            Groups=[security_group_id]
        )
        logger.info(f"实例 {instance_id} 已隔离到安全组 {security_group_id}")

    def revoke_iam_credentials(self, access_key_id):
        """撤销IAM访问密钥"""
        iam = boto3.client('iam')
        iam.update_access_key(
            AccessKeyId=access_key_id,
            Status='Inactive'
        )
        logger.info(f"IAM访问密钥 {access_key_id} 已禁用")

# 使用示例
if __name__ == "__main__":
    forensics = CloudForensics()
    # 收集事件
    start = datetime(2026, 9, 1)
    end = datetime(2026, 9, 15)
    events = forensics.collect_cloudtrail_events(start, end, "RunInstances")
    logger.info(f"收集到 {len(events)} 个事件")
    # 创建取证快照
    snapshots = forensics.create_forensic_snapshot("i-1234567890")
    # 隔离实例
    forensics.isolate_instance("i-1234567890", "sg-isolated")
```
