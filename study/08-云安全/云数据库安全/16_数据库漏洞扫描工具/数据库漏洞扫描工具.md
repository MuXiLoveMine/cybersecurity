### 15.2 数据库安全基线检查脚本

```python
import pymysql
import psycopg2
import re
from datetime import datetime

class DatabaseSecurityBaseline:
    """数据库安全基线检查"""

    def __init__(self, db_type, host, port, user, password, dbname):
        self.db_type = db_type
        self.host = host
        self.port = port
        self.user = user
        self.password = password
        self.dbname = dbname
        self.conn = None
        self.results = []

    def connect(self):
        if self.db_type == 'mysql':
            self.conn = pymysql.connect(
                host=self.host, port=self.port,
                user=self.user, password=self.password,
                database=self.dbname
            )
        elif self.db_type == 'postgres':
            self.conn = psycopg2.connect(
                host=self.host, port=self.port,
                user=self.user, password=self.password,
                dbname=self.dbname
            )

    def check_mysql_baseline(self):
        cursor = self.conn.cursor()

        checks = [
            ("默认数据库", "SELECT user FROM mysql.user WHERE user IN ('root','admin') AND authentication_string=''", "不应存在无密码root/admin账户"),
            ("匿名用户", "SELECT user FROM mysql.user WHERE user=''", "不应存在匿名用户"),
            ("远程root", "SELECT host FROM mysql.user WHERE user='root' AND host NOT IN ('localhost','127.0.0.1')", "root不应允许远程登录"),
            ("空密码", "SELECT user,host FROM mysql.user WHERE authentication_string=''", "不应有空密码账户"),
            ("SSL连接", "SHOW VARIABLES LIKE 'require_secure_transport'", "应强制SSL连接"),
            ("日志功能", "SHOW VARIABLES LIKE 'general_log'", "应启用查询日志"),
            ("错误日志", "SHOW VARIABLES LIKE 'log_error'", "应配置错误日志"),
            ("审计日志", "SHOW VARIABLES LIKE 'server_audit_logging'", "应启用审计日志"),
            ("最大连接数", "SHOW VARIABLES LIKE 'max_connections'", "应限制最大连接数"),
            ("密码策略", "SHOW VARIABLES LIKE 'validate_password_length'", "应启用密码复杂度策略"),
        ]

        for name, query, recommendation in checks:
            try:
                cursor.execute(query)
                result = cursor.fetchall()
                self._evaluate(name, result, recommendation)
            except Exception as e:
                self.results.append({
                    'check': name,
                    'status': 'ERROR',
                    'detail': str(e),
                    'recommendation': recommendation
                })

        cursor.close()

    def _evaluate(self, name, result, recommendation):
        if name in ("默认数据库", "匿名用户", "远程root", "空密码"):
            if result:
                self.results.append({
                    'check': name,
                    'status': 'FAIL',
                    'detail': f'发现: {result}',
                    'recommendation': recommendation
                })
            else:
                self.results.append({
                    'check': name,
                    'status': 'PASS',
                    'detail': '合规',
                    'recommendation': recommendation
                })
        elif name == "SSL连接":
            val = result[0][1] if result else ''
            status = 'PASS' if val == 'ON' else 'FAIL'
            self.results.append({
                'check': name,
                'status': status,
                'detail': f'当前值: {val}',
                'recommendation': recommendation
            })
        else:
            self.results.append({
                'check': name,
                'status': 'INFO',
                'detail': f'值: {result}',
                'recommendation': recommendation
            })

    def generate_report(self):
        report = f"\n{'='*60}\n"
        report += f"数据库安全基线检查报告\n"
        report += f"数据库类型: {self.db_type}\n"
        report += f"主机: {self.host}:{self.port}\n"
        report += f"检查时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n"
        report += f"{'='*60}\n\n"

        pass_count = sum(1 for r in self.results if r['status'] == 'PASS')
        fail_count = sum(1 for r in self.results if r['status'] == 'FAIL')
        info_count = sum(1 for r in self.results if r['status'] == 'INFO')
        error_count = sum(1 for r in self.results if r['status'] == 'ERROR')

        report += f"通过: {pass_count}  失败: {fail_count}  "
        report += f"信息: {info_count}  错误: {error_count}\n\n"

        for r in self.results:
            status_symbol = {'PASS': '[PASS]', 'FAIL': '[FAIL]', 'INFO': '[INFO]', 'ERROR': '[ERR ]'}
            report += f"{status_symbol.get(r['status'], '[????]')} {r['check']}\n"
            report += f"       详情: {r['detail']}\n"
            report += f"       建议: {r['recommendation']}\n\n"

        return report

    def close(self):
        if self.conn:
            self.conn.close()

# 使用示例
checker = DatabaseSecurityBaseline('mysql', '10.0.1.10', 3306, 'audit_user', 'password', 'myapp')
checker.connect()
checker.check_mysql_baseline()
print(checker.generate_report())
checker.close()
```
