# 审计日志

```bash
# AWS RDS审计日志
# 启用审计日志
aws rds modify-db-parameter-group \
    --db-parameter-group-name my-pg \
    --parameters "ParameterName=server_audit_events,ApplyMethod=immediate,ParameterValue=CONNECT,QUERY,TABLE"

# 导出日志到CloudWatch
aws rds modify-db-instance \
    --db-instance-identifier mydb \
    --cloudwatch-logs-export-configuration '{"EnableLogTypes":["audit","error","general"]}'

# 查看审计日志
aws logs filter-log-events \
    --log-group-name /aws/rds/instance/mydb/audit \
    --filter-pattern "DELETE"

# Azure SQL审计
az sql db audit-policy update \
    --resource-group myGroup \
    --server myServer \
    --database myDB \
    --state Enabled \
    --storage-account mystorageaccount
```

**审计日志分析**

```python
# 审计日志分析脚本示例
import re
from collections import Counter

def analyze_audit_logs(log_file):
    """分析数据库审计日志"""
    patterns = {
        "drop_table": r"DROP\s+TABLE",
        "delete_data": r"DELETE\s+FROM\s+\w+\s+WHERE\s+1=1",
        "grant_all": r"GRANT\s+ALL\s+PRIVILEGES",
        "create_user": r"CREATE\s+USER",
        "sql_injection": r"(?:UNION\s+SELECT|OR\s+1=1|--|/\*|\*/)",
        "bulk_select": r"SELECT\s+\*\s+FROM",
        "after_hours": None,  # 需要时间检查
    }

    alerts = []
    with open(log_file, "r") as f:
        for line_num, line in enumerate(f, 1):
            for name, pattern in patterns.items():
                if pattern and re.search(pattern, line, re.IGNORECASE):
                    alerts.append({
                        "line": line_num,
                        "type": name,
                        "content": line.strip()[:200],
                    })

    # 统计
    type_counts = Counter(a["type"] for a in alerts)
    print(f"审计日志分析结果:")
    for alert_type, count in type_counts.items():
        print(f"  {alert_type}: {count}")

    return alerts
```
