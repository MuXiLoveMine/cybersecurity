# 安全自动化与基础设施即代码

```yaml
# Terraform安全配置示例

# 启用S3加密
resource "aws_s3_bucket_server_side_encryption_configuration" "example" {
  bucket = aws_s3_bucket.example.id
  rule {
    apply_server_side_encryption_by_default {
      sse_algorithm = "aws:kms"
      kms_master_key_id = aws_kms_key.s3_key.arn
    }
  }
}

# 私有子网中的RDS
resource "aws_db_instance" "database" {
  engine               = "mysql"
  instance_class       = "db.r5.large"
  publicly_accessible  = false  # 不公开访问
  storage_encrypted    = true   # 存储加密
  kms_key_id          = aws_kms_key.rds_key.arn
  backup_retention_period = 30  # 备份保留30天
  deletion_protection = true   # 删除保护
  multi_az            = true   # 多可用区

  vpc_security_group_ids = [aws_security_group.db.id]
  db_subnet_group_name   = aws_db_subnet_group.private.name

  parameter_group_name = aws_db_parameter_group.secure.name
}

# 限制性安全组
resource "aws_security_group" "db" {
  name        = "database-sg"
  vpc_id      = aws_vpc.main.id
}

resource "aws_security_group_rule" "db_ingress" {
  type                     = "ingress"
  from_port                = 3306
  to_port                  = 3306
  protocol                 = "tcp"
  source_security_group_id = aws_security_group.app.id  # 只允许应用SG
  security_group_id        = aws_security_group.db.id
}

# 无出站规则，默认拒绝所有出站
resource "aws_security_group_rule" "db_egress" {
  type              = "egress"
  from_port         = 0
  to_port           = 0
  protocol          = "-1"
  cidr_blocks       = ["10.0.0.0/8"]  # 只允许内网出站
  security_group_id = aws_security_group.db.id
}
```
